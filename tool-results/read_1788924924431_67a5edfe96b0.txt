    98→        decidedBy: record.decidedBy,
    99→        decidedAt: record.decidedAt ? new Date(record.decidedAt) : null,
   100→        reason: record.decisionReason,
   101→        expiresAt: record.expiresAt ? new Date(record.expiresAt) : null,
   102→        consumedAt: record.isConsumed ? new Date() : null,
   103→        scope: (record.scope as any) ?? {},
   104→      },
   105→    });

   107→    return this.mapPrismaToApproval(created);

   120→    const db = await requireAuthoritativeDatabase();
   121→    const where: any = {};
   122→    if (filter?.workflowInstanceId) where.workflowInstanceId = filter.workflowInstanceId;
   123→    if (filter?.stepId) where.stepId = filter.stepId;
   124→    if (filter?.employeeRole) where.employeeRole = filter.employeeRole;
   125→    if (filter?.classification) where.classification = filter.classification;
   126→    if (filter?.status) where.decision = filter.status;
   127→
   128→    const records = await db.approvalRecord.findMany({
   129→      where,

   136→  public async findActiveMatching(params: {
   137→    workflowInstanceId: string;
   138→    stepId: string;
   139→    employeeRole?: string;
   140→    actionName?: string;
   141→    classification?: SideEffectClassification;
   142→  }): Promise<FounderApprovalRecord | null> {
   143→    const db = await requireAuthoritativeDatabase();
   144→    const where: any = {
   145→      workflowInstanceId: params.workflowInstanceId,
   146→    };
   147→    if (params.classification) where.classification = params.classification;
   148→    if (params.employeeRole) where.employeeRole = params.employeeRole;
   149→    if (params.actionName) where.actionType = params.actionName;
   150→
   151→    const candidates = await db.approvalRecord.findMany({
   152→      where,
   153→      orderBy: { createdAt: 'desc' },
   154→    });
   155→
   156→    const now = Date.now();
   157→    for (const raw of candidates) {
   158→      const record = this.mapPrismaToApproval(raw);
   159→      if (record.scope.scopeType === 'step' || record.scope.scopeType === 'single_action') {
   160→        if (record.stepId !== params.stepId) continue;
   161→      }
   162→
   163→      // Check TTL expiration
   164→      if (record.expiresAt && new Date(record.expiresAt).getTime() < now) {
   165→        if (record.decision === 'approved') {
   166→          await db.approvalRecord.update({
   167→            where: { id: record.id },
   168→            data: {
   169→              decision: 'expired',
   170→              reason: record.decisionReason || 'Approval expired automatically due to TTL',
   171→            },
   172→          });
   173→          record.decision = 'expired';
   174→          record.decisionReason = 'Approval expired automatically due to TTL';
   175→        }
   176→      }
   177→
   178→      return record;
   179→    }
   180→
   181→    return null;
   182→  }

   184→  public async decide(
   185→    id: string,
   186→    decision: 'approved' | 'rejected',
   187→    decidedBy: string,
   188→    reason?: string,
   189→    expiresAt?: string
   190→  ): Promise<FounderApprovalRecord> {
   191→    const db = await requireAuthoritativeDatabase();
   192→    const decidedAt = new Date();
   193→    const updated = await db.approvalRecord.update({
   194→      where: { id },
   195→      data: {
   196→        decision,
   197→        decidedBy,
   198→        decidedAt,
   199→        reason: reason || null,
   200→        expiresAt: expiresAt ? new Date(expiresAt) : null,
   201→      },
   202→    });
   203→
   204→    return this.mapPrismaToApproval(updated);
   205→  }
   206→
   207→  public async revoke(id: string, revokedBy: string, reason?: string): Promise<FounderApprovalRecord> {
   208→    const db = await requireAuthoritativeDatabase();
   209→    const decidedAt = new Date();
   210→    const updated = await db.approvalRecord.update({
   211→      where: { id },
   212→      data: {
   213→        decision: 'revoked',
   214→        decidedBy: revokedBy,
   215→        decidedAt,
   216→        reason: reason || 'Revoked by Founder',
   217→      },
   218→    });
   219→
   220→    return this.mapPrismaToApproval(updated);
   221→  }
   222→
   223→  public async consume(id: string): Promise<FounderApprovalRecord> {
   224→    const db = await requireAuthoritativeDatabase();
   225→    return await db.$transaction(async (tx) => {
   226→      const existing = await tx.approvalRecord.findUnique({ where: { id } });
   227→      if (!existing) {
   228→        throw new Error(`Approval record not found: ${id}`);
   229→      }
   230→
   231→      const scope = (existing.scope as any) || {};
   232→      const allowedUses = scope.maxUses ?? scope.allowedUses ?? 1;
   233→      const currentUses = scope.usedCount ?? 0;
   234→
   235→      if (existing.consumedAt !== null || currentUses >= allowedUses) {
   236→        throw new ApprovalAlreadyConsumedError(id, existing.consumedAt ?? undefined);
   237→      }
   238→
   239→      scope.usedCount = currentUses + 1;
   240→
   241→      const updated = await tx.approvalRecord.update({
   242→        where: { id },
   243→        data: {
   244→          consumedAt: new Date(),
   245→          scope,
   246→        },
   247→      });
   248→
   249→      return this.mapPrismaToApproval(updated);
   250→    });
   251→  }
   252→
   253→  private mapPrismaToApproval(r: any): FounderApprovalRecord {
   254→    const scope = (r.scope as any) || { allowedUses: 1, usedCount: 0 };
   255→    return {
   256→      id: r.id,
   257→      workflowInstanceId: r.workflowInstanceId || '',
   258→      stepId: r.stepId || '',
   259→      actionName: r.actionType,
   260→      employeeRole: r.employeeRole as any,
   261→      classification: r.classification as any,
   262→      target: {
   263→        targetSystem: r.targetSystem as any,
   264→        metadata: r.payload as any,
   265→      },
   266→      payloadHash: r.payloadHash || undefined,
   267→      decision: r.decision as any,
   268→      decidedBy: r.decidedBy || undefined,
   269→      decidedAt: r.decidedAt ? r.decidedAt.toISOString() : undefined,
   270→      decisionReason: r.reason || undefined,
   271→      expiresAt: r.expiresAt ? r.expiresAt.toISOString() : undefined,
   272→      isConsumed: Boolean(r.consumedAt),
   273→      requestedAt: r.createdAt.toISOString(),
   274→      scope: {
   275→        scopeType: scope.scopeType || 'single_action',
   276→        workflowInstanceId: r.workflowInstanceId || undefined,
   277→        stepId: r.stepId || undefined,
   278→        campaignId: r.campaignId || undefined,
   279→        allowedUses: scope.allowedUses ?? 1,
   280→        usedCount: scope.usedCount ?? (r.consumedAt ? 1 : 0),
   281→      } as any,
   282→    };
   283→  }
   284→}

   306→  public async record(audit: SideEffectAuditRecord): Promise<SideEffectAuditRecord> {
   307→    const db = await requireAuthoritativeDatabase();
   308→    await db.sideEffectAudit.create({
   309→      data: {
   310→        id: audit.id,

   312→        approvalId: audit.approvalId,
   313→        workflowInstanceId: audit.workflowInstanceId,
   314→        stepId: audit.stepId,
   315→        employeeRole: audit.employeeRole,
   316→        classification: audit.actionClassification,
   317→        actionType: audit.actionName || 'UNKNOWN',
   318→        targetSystem: audit.target?.targetSystem || 'internal',
   319→        payload: (audit.target?.metadata as any) ?? {},
   320→        decisionOutcome: audit.decision,
   321→        reason: audit.reason,
   322→        executionReference: audit.executionReference,
   323→        timestamp: new Date(audit.timestamp),
   324→      },
   325→    });
   326→
   327→    return JSON.parse(JSON.stringify(audit));
   328→  }

   330→  public async get(id: string): Promise<SideEffectAuditRecord | null> {
   331→    const db = await requireAuthoritativeDatabase();
   332→    const found = await db.sideEffectAudit.findUnique({
   333→      where: { id },
   334→    });
   335→    if (!found) return null;
   336→    return this.mapPrismaToAudit(found);
   337→  }

   339→  public async list(filter?: AuditFilter): Promise<SideEffectAuditRecord[]> {
   340→    const db = await requireAuthoritativeDatabase();
   341→    const where: any = {};
   342→    if (filter?.workflowInstanceId) where.workflowInstanceId = filter.workflowInstanceId;
   343→    if (filter?.stepId) where.stepId = filter.stepId;
   344→    if (filter?.employeeRole) where.employeeRole = filter.employeeRole;
   345→    if (filter?.classification) where.classification = filter.classification;
   346→    if (filter?.decision) where.decisionOutcome = filter.decision;
   347→
   348→    const records = await db.sideEffectAudit.findMany({
   349→      where,
   350→      orderBy: { timestamp: 'desc' },
   351→    });
   352→
   353→    let results = records.map((r) => this.mapPrismaToAudit(r));
   354→    if (filter?.executedOnly) {
   355→      results = results.filter((a) => a.executed === true);
   356→    }
   357→    return results;
   358→  }
   359→
   360→  private mapPrismaToAudit(r: any): SideEffectAuditRecord {
   361→    return {
   362→      id: r.id,
   363→      requestId: r.requestId || uuidv4(),
   364→      requestedBy: r.employeeRole || 'unknown',
   365→      approvalId: r.approvalId || '',
   366→      workflowInstanceId: r.workflowInstanceId || '',
   367→      stepId: r.stepId || '',
   368→      actionName: r.actionType,
   369→      actionClassification: r.classification as any,
   370→      employeeRole: r.employeeRole as any,
   371→      target: {
   372→        targetSystem: r.targetSystem as any,
   373→        metadata: r.payload as any,
   374→      },
   375→      decision: r.decisionOutcome as any,
   376→      reasonCode: (r.decisionOutcome === 'allowed' ? 'APPROVED_BY_FOUNDER' : 'APPROVAL_REQUIRED_STEP_POLICY') as any,
   377→      reason: r.reason || '',
   378→      executionReference: r.executionReference || undefined,
   379→      executed: Boolean(r.executionReference),
   380→      timestamp: r.timestamp.toISOString(),
   381→    };
   382→  }
   383→}

   407→    try {
   408→      const persisted = DurableFileStore.getInstance().readCollection<FounderApprovalRecord>('approvals');
   409→      for (const [id, record] of Object.entries(persisted)) {

   413→      // fallback
   414→    }

   425→      // fallback

   557→  public async findActiveMatching(params: {
   558→    workflowInstanceId: string;
   559→    stepId: string;
   560→    employeeRole?: string;
   561→    actionName?: string;
   562→    classification?: SideEffectClassification;
   563→  }): Promise<FounderApprovalRecord | null> {
   564→    if (isAuthoritativeMode()) {
   565→      return PostgresApprovalStore.getInstance().findActiveMatching(params);
   566→    }
   567→
   568→    const now = Date.now();
   569→    for (const record of this.approvals.values()) {
   570→      if (record.workflowInstanceId !== params.workflowInstanceId) continue;
   571→
   572→      if (record.scope.scopeType === 'step' || record.scope.scopeType === 'single_action') {
   573→        if (record.stepId !== params.stepId) continue;
   574→      }
   575→
   576→      if (params.classification && record.classification !== params.classification) continue;
   577→      if (params.employeeRole && record.employeeRole !== params.employeeRole) continue;
   578→
   579→      if (record.expiresAt && new Date(record.expiresAt).getTime() < now) {
   580→        if (record.decision === 'approved') {
   581→          record.decision = 'expired';
   582→          record.decisionReason = record.decisionReason || 'Approval expired automatically due to TTL';
   583→          this.approvals.set(record.id, record);
   584→        }
   585→      }
   586→
   587→      return JSON.parse(JSON.stringify(record));
   588→    }
   589→    return null;
   590→  }
   591→
   592→  public async decide(
   593→    id: string,
   594→    decision: 'approved' | 'rejected',
   595→    decidedBy: string,
   596→    reason?: string,
   597→    expiresAt?: string
   598→  ): Promise<FounderApprovalRecord> {
   599→    if (isAuthoritativeMode()) {
   600→      return PostgresApprovalStore.getInstance().decide(id, decision, decidedBy, reason, expiresAt);
   601→    }
   602→
   603→    const record = this.approvals.get(id);
   604→    if (!record) {
   605→      throw new Error(`Approval record not found: ${id}`);
   606→    }
   607→
   608→    record.decision = decision;
   609→    record.decidedAt = new Date().toISOString();
   610→    record.decidedBy = decidedBy;
   611→    if (reason) record.decisionReason = reason;
   612→    if (expiresAt) record.expiresAt = expiresAt;
   613→
   614→    this.approvals.set(id, record);
   615→    try {
   616→      DurableFileStore.getInstance().saveItem('approvals', id, record);
   617→    } catch {}
   618→
   619→    if (process.env.DATABASE_URL) {
   620→      try {
   621→        await prisma.approvalRecord.update({
   622→          where: { id },
   623→          data: {
   624→            decision,
   625→            decidedBy,
   626→            decidedAt: new Date(record.decidedAt),
   627→            reason: record.decisionReason,
   628→            expiresAt: expiresAt ? new Date(expiresAt) : null,
   629→          },
   630→        });
   631→      } catch {
   632→        // Fallback
   633→      }
   634→    }
   635→
   636→    return JSON.parse(JSON.stringify(record));
   637→  }
   638→
   639→  public async revoke(id: string, revokedBy: string, reason?: string): Promise<FounderApprovalRecord> {
   640→    if (isAuthoritativeMode()) {
   641→      return PostgresApprovalStore.getInstance().revoke(id, revokedBy, reason);
   642→    }
   643→
   644→    const record = this.approvals.get(id);
   645→    if (!record) {
   646→      throw new Error(`Approval record not found: ${id}`);
   647→    }
   648→
   649→    record.decision = 'revoked';
   650→    record.decidedAt = new Date().toISOString();
   651→    record.decidedBy = revokedBy;
   652→    record.decisionReason = reason || 'Revoked by Founder';
   653→
   654→    this.approvals.set(id, record);
   655→    try {
   656→      DurableFileStore.getInstance().saveItem('approvals', id, record);
   657→    } catch {}
   658→
   659→    if (process.env.DATABASE_URL) {
   660→      try {
   661→        await prisma.approvalRecord.update({
   662→          where: { id },
   663→          data: {
   664→            decision: 'revoked',
   665→            decidedBy: revokedBy,
   666→            decidedAt: new Date(record.decidedAt),
   667→            reason: record.decisionReason,
   668→          },
   669→        });
   670→      } catch {
   671→        // Fallback
   672→      }
   673→    }
   674→
   675→    return JSON.parse(JSON.stringify(record));
   676→  }
   677→
   678→  public async consume(id: string): Promise<FounderApprovalRecord> {
   679→    if (isAuthoritativeMode()) {
   680→      return PostgresApprovalStore.getInstance().consume(id);
   681→    }
   682→
   683→    const record = this.approvals.get(id);
   684→    if (!record) {
   685→      throw new Error(`Approval record not found: ${id}`);
   686→    }
   687→
   688→    const allowedUses = (record.scope as any)?.maxUses ?? (record.scope as any)?.allowedUses ?? 1;
   689→    const currentUses = record.scope?.usedCount ?? 0;
   690→
   691→    if (record.isConsumed || currentUses >= allowedUses) {
   692→      throw new ApprovalAlreadyConsumedError(id, record.decidedAt);
   693→    }
   694→
   695→    record.isConsumed = true;
   696→    record.scope.usedCount = currentUses + 1;
   697→    this.approvals.set(id, record);
   698→    try {
   699→      DurableFileStore.getInstance().saveItem('approvals', id, record);
   700→    } catch {}
   701→
   702→    if (process.env.DATABASE_URL) {
   703→      try {
   704→        await prisma.approvalRecord.update({
   705→          where: { id },
   706→          data: {
   707→            consumedAt: new Date(),
   708→            scope: record.scope as any,
   709→          },
   710→        });
   711→      } catch {
   712→        // Fallback
   713→      }
   714→    }
   715→
   716→    return JSON.parse(JSON.stringify(record));
   717→  }
   718→}

   720→/**
   721→ * Dual-Mode Audit Trail Store.
   722→ * In Authoritative Mode: delegates directly to PostgresAuditStore (fail-closed).
   723→ * In Test/Local Mode: uses fast in-memory maps with local file persistence.
   724→ */
   725→export class InMemoryAuditStore implements IAuditStore {
   726→  private static instance: InMemoryAuditStore;
   727→  public audits: Map<string, SideEffectAuditRecord> = new Map();
   728→
   729→  private constructor() {
   730→    this.loadFromDurableStorage();
   731→  }
   732→
   733→  public static getInstance(): InMemoryAuditStore {
   734→    if (!InMemoryAuditStore.instance) {
   735→      InMemoryAuditStore.instance = new InMemoryAuditStore();
   736→    }
   737→    return InMemoryAuditStore.instance;
   738→  }
   739→
   740→  public loadFromDurableStorage(): void {
   741→    try {
   742→      const persisted = DurableFileStore.getInstance().readCollection<SideEffectAuditRecord>('audits');
   743→      for (const [id, record] of Object.entries(persisted)) {
   744→        this.audits.set(id, record);
   745→      }
   746→    } catch {
   747→      // fallback
   748→    }
   749→  }
   750→
   751→  public clear(): void {
   752→    if (process.env.NODE_ENV === 'production') {
   753→      throw new Error('Audit trail is strictly append-only and immutable; cannot be cleared in production.');
   754→    }
   755→    this.audits.clear();
   756→    try {
   757→      DurableFileStore.getInstance().clearCollection('audits');
   758→    } catch {
   759→      // fallback
   760→    }
   761→  }
   762→
   763→  public async record(audit: SideEffectAuditRecord): Promise<SideEffectAuditRecord> {
   764→    if (isAuthoritativeMode()) {
   765→      return PostgresAuditStore.getInstance().record(audit);
   766→    }
   767→
   768→    const clone = JSON.parse(JSON.stringify(audit));
   769→    this.audits.set(audit.id, clone);
   770→
   771→    try {
   772→      DurableFileStore.getInstance().saveItem('audits', audit.id, clone);
   773→    } catch (err) {
   774→      console.warn('[AuditStore] Error saving to durable file store:', err);
   775→    }
   776→
   777→    if (process.env.DATABASE_URL) {
   778→      try {
   779→        await prisma.sideEffectAudit.create({
   780→          data: {
   781→            id: audit.id,
   782→            requestId: audit.requestId || null,
   783→            approvalId: audit.approvalId,
   784→            workflowInstanceId: audit.workflowInstanceId,
   785→            stepId: audit.stepId,
   786→            employeeRole: audit.employeeRole,
   787→            classification: audit.actionClassification,
   788→            actionType: audit.actionName || 'UNKNOWN',
   789→            targetSystem: audit.target?.targetSystem || 'internal',
   790→            payload: (audit.target?.metadata as any) ?? {},
   791→            decisionOutcome: audit.decision,
   792→            reason: audit.reason,
   793→            executionReference: audit.executionReference,
   794→            timestamp: new Date(audit.timestamp),
   795→          },
   796→        });
   797→      } catch {
   798→        // Fallback
   799→      }
   800→    }
   801→
   802→    return clone;
   803→  }
   804→
   805→  public async get(id: string): Promise<SideEffectAuditRecord | null> {
   806→    if (isAuthoritativeMode()) {
   807→      return PostgresAuditStore.getInstance().get(id);
   808→    }
   809→
   810→    const record = this.audits.get(id);
   811→    if (!record) return null;
   812→    return JSON.parse(JSON.stringify(record));
   813→  }
   814→
   815→  public async list(filter?: AuditFilter): Promise<SideEffectAuditRecord[]> {
   816→    if (isAuthoritativeMode()) {
   817→      return PostgresAuditStore.getInstance().list(filter);
   818→    }
   819→
   820→    let list = Array.from(this.audits.values());
   821→
   822→    if (filter?.workflowInstanceId) {
   823→      list = list.filter((a) => a.workflowInstanceId === filter.workflowInstanceId);
   824→    }
   825→    if (filter?.stepId) {
   826→      list = list.filter((a) => a.stepId === filter.stepId);
   827→    }
   828→    if (filter?.employeeRole) {
   829→      list = list.filter((a) => a.employeeRole === filter.employeeRole);
   830→    }
   831→    if (filter?.classification) {
   832→      list = list.filter((a) => a.actionClassification === filter.classification);
   833→    }
   834→    if (filter?.decision) {
   835→      list = list.filter((a) => a.decision === filter.decision);
   836→    }
   837→    if (filter?.executedOnly) {
   838→      list = list.filter((a) => a.executed === true);
   839→    }
   840→
   841→    list.sort((a, b) => new Date(b.timestamp).getTime() - new Date(a.timestamp).getTime());
   842→
   843→    return JSON.parse(JSON.stringify(list));
   844→  }
   845→}
