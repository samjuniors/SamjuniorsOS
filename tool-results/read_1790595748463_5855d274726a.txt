    86→export function mapSemanticToPresentationState(
    87→  runtime: GraphRuntimeState,
    88→  governance: GraphGovernanceState,
    89→  epistemic: GraphEpistemicValidity
    90→): GraphPresentationState {
    91→  // 1. Governance halt takes absolute precedence at authority boundary
    92→  if (governance === 'awaiting_founder_approval') {
    93→    return 'waiting'; // Amber authorization seal, NEVER processing
    94→  }
    95→
    96→  // 2. Failure / Invariant rejection takes precedence
    97→  if (runtime === 'failed' || epistemic === 'disputed') {
    98→    return 'error'; // Alert rose boundary
    99→  }
   100→
   101→  // 3. Actively running
   102→  if (runtime === 'running') {
   103→    return 'active'; // Vibrant cyan pulse
   104→  }
   105→
   106→  // 4. Paused or halted (without active approval)
   107→  if (runtime === 'paused' || runtime === 'halted') {
   108→    return 'waiting';
   109→  }
   110→
   111→  // 5. Completed & verified
   112→  if (runtime === 'completed' || epistemic === 'promoted_to_fact') {
   113→    return 'success'; // Emerald halo
   114→  }
   115→
   116→  // 6. Calm idle baseline
   117→  return 'default'; // Quiet obsidian glass
   118→}

   124→function layoutColumn(
   125→  nodes: GraphNodeDTO[],
   126→  centerY: number,
   127→  gap: number
   128→): void {
   129→  if (nodes.length === 0) return;
   130→  const totalH =
   131→    nodes.reduce((acc, n) => acc + n.geometry.height, 0) + (nodes.length - 1) * gap;
   132→  let curY = centerY - totalH / 2;
   133→
   134→  for (const n of nodes) {
   135→    n.geometry.y = Math.round(curY + n.geometry.height / 2);
   136→    curY += n.geometry.height + gap;
   137→  }
   138→}

   140→/**
   141→ * Deterministic identity hash for derived graph entities (workstreams).
   142→ * Exported since Phase 4.4C: the authoritative Activity projection reuses the
   143→ * EXACT same derivation so Activity provenance traces to the same workstream
   144→ * identities the canvas read-model projects — no parallel identity scheme.
   145→ */
   146→export function deterministicIdHash(prefix: string, text: string): string {
   147→  let h = 5381;
   148→  for (let i = 0; i < text.length; i++) {
   149→    h = ((h << 5) + h + text.charCodeAt(i)) | 0;
   150→  }
   151→  return `${prefix}-${(h >>> 0).toString(36)}`;
   152→}

   154→function truncateTitle(text: string, max = 50): string {
   155→  if (text.length <= max) return text;
   156→  return `${text.slice(0, max - 1).trimEnd()}…`;
   157→}

   163→/**
   164→ * Canonical 9-step agent work protocol (src/types/os.ts AgentWorkProtocolStep),
   165→ * in protocol order, with display labels. Used to shape the local workflow
   166→ * revealed when a work object is focused. Steps without runs are honestly
   167→ * reported as 'pending' — the protocol is company truth, outcomes are not
   168→ * fabricated.
   169→ */
   170→const CANONICAL_PROTOCOL_STEPS: ReadonlyArray<{
   171→  step: string;
   172→  label: string;
   173→}> = [
   174→  { step: 'understand', label: 'Understand & Scope' },
   175→  { step: 'research', label: 'Research' },
   176→  { step: 'analyze', label: 'Analyze' },
   177→  { step: 'plan', label: 'Plan' },
   178→  { step: 'build_execute', label: 'Build & Execute' },
   179→  { step: 'test', label: 'Test' },
   180→  { step: 'verify', label: 'Verify' },
   181→  { step: 'review', label: 'Review' },
   182→  { step: 'report', label: 'Report' },
   183→];
   184→
   185→function protocolStepLabel(step: string): string {
   186→  const canonical = CANONICAL_PROTOCOL_STEPS.find((c) => c.step === step);
   187→  if (canonical) return canonical.label;
   188→  return step.replace(/_/g, ' ').replace(/^\w/, (c) => c.toUpperCase());
   189→}

   191→function runStepStatus(runs: AgentRunRecord[]): GraphExecutionStepDTO['status'] {
   192→  if (runs.length === 0) return 'pending';
   193→  if (runs.some((r) => r.status === 'failed' || !!r.error)) return 'failed';
   194→  if (runs.some((r) => r.status === 'running')) return 'current';
   195→  if (runs.some((r) => r.status === 'completed')) return 'done';
   196→  return 'pending';
   197→}

   199→/**
   200→ * Derives the deterministic execution trail of a workstream from its
   201→ * chronological run group. Unknown protocol steps are appended after the
   202→ * canonical sequence in first-occurrence order (never dropped, never invented).
   203→ */
   204→function deriveExecutionStepsFromRuns(sortedRuns: AgentRunRecord[]): GraphExecutionStepDTO[] {
   205→  const byStep = new Map<string, AgentRunRecord[]>();
   206→  for (const r of sortedRuns) {
   207→    const key = r.protocolStep;
   208→    if (!key) continue;
   209→    const list = byStep.get(key) ?? [];
   210→    list.push(r);
   211→    byStep.set(key, list);
   212→  }
   213→
   214→  const unknownSteps = [...byStep.keys()].filter(
   215→    (k) => !CANONICAL_PROTOCOL_STEPS.some((c) => c.step === k)
   216→  );
   217→
   218→  const steps: GraphExecutionStepDTO[] = CANONICAL_PROTOCOL_STEPS.map(({ step, label }) => {
   219→    const runs = byStep.get(step) ?? [];
   220→    const latest = runs[runs.length - 1];
   221→    return {
   222→      step,
   223→      label,
   224→      status: runStepStatus(runs),
   225→      ownerAgentId: latest?.agentId,
   226→      durationMs: latest?.durationMs,
   227→    };
   228→  });
   229→
   230→  for (const step of unknownSteps) {
   231→    const runs = byStep.get(step) ?? [];
   232→    const latest = runs[runs.length - 1];
   233→    steps.push({
   234→      step,
   235→      label: protocolStepLabel(step),
   236→      status: runStepStatus(runs),
   237→      ownerAgentId: latest?.agentId,
   238→      durationMs: latest?.durationMs,
   239→    });
   240→  }
   241→
   242→  return steps;
   243→}

   245→/** Maps durable WorkflowStepState status onto the execution trail status. */
   246→function workflowStepStatus(status: string): GraphExecutionStepDTO['status'] {
   247→  switch (status) {
   248→    case 'running':
   249→    case 'ready':
   250→      return 'current';
   251→    case 'waiting':
   252→    case 'blocked':
   253→    case 'awaiting_approval':
   254→      return 'waiting';
   255→    case 'completed':
   256→      return 'done';
   257→    case 'failed':
   258→    case 'cancelled':
   259→      return 'failed';
   260→    default:
   261→      return 'pending';
   262→  }
   263→}

   265→/**
   266→ * Derives the execution trail of a workstream backed by a durable
   267→ * WorkflowInstance from its authoritative stepStates.
   268→ */
   269→function deriveExecutionStepsFromInstance(wf: WorkflowInstanceState): GraphExecutionStepDTO[] {
   270→  const skillOrder = (skill: string) => {
   271→    const idx = CANONICAL_PROTOCOL_STEPS.findIndex((c) => c.step === skill);
   272→    return idx === -1 ? CANONICAL_PROTOCOL_STEPS.length : idx;
   273→  };
   274→  return Object.values(wf.stepStates ?? {})
   275→    .sort((a, b) => {
   276→      const sa = skillOrder(a.skill);
   277→      const sb = skillOrder(b.skill);
   278→      if (sa !== sb) return sa - sb;
   279→      return a.stepId.localeCompare(b.stepId);
   280→    })
   281→    .map((s) => ({
   282→      step: s.stepId,
   283→      label: protocolStepLabel(s.skill),
   284→      status: workflowStepStatus(s.status),
   285→      ownerAgentId: s.assignedRole,
   286→    }));
   287→}

   309→  const hasOpenApprovals = pendingApprovals.length > 0;
   310→
   311→  // Group durable agent runs by directive (canonical workstreams)
   312→  const directiveGroups = new Map<string, AgentRunRecord[]>();
   313→  for (const r of agentRuns) {
   314→    const key = r.directive?.trim();
   315→    if (!key) continue;
   316→    const list = directiveGroups.get(key) ?? [];
   317→    list.push(r);
   318→    directiveGroups.set(key, list);
   319→  }
   320→
   321→  // Derive workstreams from groups
   322→  interface DerivedWorkstream {
   323→    id: string;
   324→    directive: string;
   325→    title: string;
   326→    latestRun: AgentRunRecord;
   327→    owner: string;
   328→    stepStage: 'discovery' | 'build' | 'review' | 'ship' | 'done';
   329→    runtimeState: GraphRuntimeState;
   330→    hasFailed: boolean;
   331→    hasReport: boolean;
   332→    latestRunTime: number;
   333→    executionSteps: GraphExecutionStepDTO[];
   334→  }
   335→
   336→  const workstreams: DerivedWorkstream[] = [];
   337→  for (const [directive, group] of directiveGroups) {
   338→    const sorted = [...group].sort((a, b) => {
   339→      const ta = (a as any).timestamp || (a as any).createdAt ? new Date((a as any).timestamp || (a as any).createdAt).getTime() : 0;
   340→      const tb = (b as any).timestamp || (b as any).createdAt ? new Date((b as any).timestamp || (b as any).createdAt).getTime() : 0;
   341→      return ta - tb;
   342→    });
   343→    const latest = sorted[sorted.length - 1];
   344→    const hasReport = sorted.some((r) => r.protocolStep === 'report');
   345→    const hasFailed = sorted.some((r) => r.status === 'failed' || !!r.error);
   346→    const hasRunning = sorted.some((r) => r.status === 'running');
   347→
   348→    const specialistRuns = sorted.filter((r) => r.agentId !== 'coo');
   349→    const latestSpecialist = specialistRuns[specialistRuns.length - 1];
   350→    const owner = latestSpecialist ? latestSpecialist.agentId : 'researcher';
   351→
   352→    let stepStage: DerivedWorkstream['stepStage'] = 'discovery';
   353→    if (hasReport) {
   354→      stepStage = 'done';
   355→    } else if (latest.protocolStep === 'review' || latest.protocolStep === 'verify') {
   356→      stepStage = 'review';
   357→    } else if (latest.protocolStep === 'plan' || latest.protocolStep === 'build_execute' || latest.protocolStep === 'test') {
   358→      stepStage = 'build';
   359→    } else if (latest.protocolStep === 'report') {
   360→      stepStage = 'ship';
   361→    }
   362→
   363→    const runtimeState: GraphRuntimeState = hasFailed
   364→      ? 'failed'
   365→      : hasReport
   366→      ? 'completed'
   367→      : hasRunning
   368→      ? 'running'
   369→      : 'paused';
   370→
   371→    workstreams.push({
   372→      id: deterministicIdHash('ws', directive),
   373→      directive,
   374→      title: truncateTitle(directive),
   375→      latestRun: latest,
   376→      owner,
   377→      stepStage,
   378→      runtimeState,
   379→      hasFailed,
   380→      hasReport,
   381→      latestRunTime: (() => {
   382→        const t = (latest as any).timestamp || (latest as any).createdAt;
   383→        return t ? new Date(t).getTime() : 0;
   384→      })(),
   385→      executionSteps: deriveExecutionStepsFromRuns(sorted),
   386→    });
   387→  }
   388→
   389→  // Cross-reference durable WorkflowInstances
   390→  for (const wf of workflows) {
   391→    if (workstreams.some((w) => w.directive === wf.objective)) continue;
   392→    const isDone = wf.status === 'completed';
   393→    const isFailed = wf.status === 'failed';
   394→    const isRunning = wf.status === 'running';
   395→    workstreams.push({
   396→      id: deterministicIdHash('wf', wf.instanceId),
   397→      directive: wf.objective,
   398→      title: truncateTitle(wf.objective),
   399→      latestRun: {
   400→        runId: wf.instanceId,
   401→        agentId: 'coo',
   402→        agentName: 'Sophia Vance',
   403→        protocolStep: 'plan',
   404→        taskTitle: wf.objective,
   405→        directive: wf.objective,
   406→        status: wf.status as any,
   407→        durationMs: 0,
   408→        outputContent: '',
   409→        timestamp: wf.createdAt || new Date().toISOString(),
   410→        provenance: {
   411→          agentId: 'coo',
   412→          agentName: 'Sophia Vance',
   413→          taskId: wf.instanceId,
   414→          protocolStep: 'plan',
   415→          timestamp: wf.createdAt || new Date().toISOString(),
   416→          isVerified: true,
   417→          evidenceBasis: 'calculation',
   418→        },
   419→      },
   420→      owner: 'coo',
   421→      stepStage: isDone ? 'done' : 'build',
   422→      runtimeState: isFailed ? 'failed' : isDone ? 'completed' : isRunning ? 'running' : 'idle',
   423→      hasFailed: isFailed,
   424→      hasReport: isDone,
   425→      latestRunTime: wf.createdAt ? new Date(wf.createdAt).getTime() : 0,

   432→  const completedWorkstreams = workstreams.filter((w) => w.runtimeState === 'completed');

   434→  const hasActiveWork = activeWorkstreams.length > 0;

  1103→    if (a.geometry.x !== b.geometry.x) return a.geometry.x - b.geometry.x;
  1104→    if (a.geometry.y !== b.geometry.y) return a.geometry.y - b.geometry.y;
  1105→    return a.id.localeCompare(b.id);
  1106→  });
  1107→  edges.sort((a, b) => a.id.localeCompare(b.id));
  1108→
  1109→  // Compute deterministic SHA-256 fingerprint over topology
  1110→  const canonicalData = JSON.stringify({
  1111→    nodes: nodes.map((n) => ({
  1112→      id: n.id,
  1113→      r: n.runtimeState,
  1114→      g: n.governanceState,
  1115→      e: n.epistemicValidity,
  1116→      p: n.presentationState,
  1117→      x: n.geometry.x,
  1118→      y: n.geometry.y,
  1119→    })),
  1120→    edges: edges.map((e) => ({
  1121→      id: e.id,
  1122→      s: e.source,
  1123→      t: e.target,
  1124→      r: e.runtimeState,
  1125→      p: e.presentationState,
  1126→    })),
  1127→  });
  1128→
  1129→  const deterministicHash = crypto
  1130→    .createHash('sha256')
  1131→    .update(canonicalData)
  1132→    .digest('hex');
  1133→
  1134→  return {
  1135→    asOf: new Date(inputs.now ?? Date.now()).toISOString(),
  1136→    deterministicHash,
  1137→    topologyVersion: '1.1.0',
  1138→    company,
  1139→    summary: {
  1140→      totalNodes: nodes.length,
  1141→      totalEdges: edges.length,
  1142→      activeWorkstreams: activeWorkstreams.length,
  1143→      pendingApprovals: pendingApprovals.length,
  1144→      blockedItems: failedWorkstreams.length,
  1145→    },
  1146→    nodes,
  1147→    edges,
  1148→    spatialCards,
  1149→    inspectionContext,
  1150→  };
  1151→}
