   136→/**
   137→ * Independent re-derivation of the pre-M4-C policy (same mechanism family as
   138→ * the m4c suite, kept so fallback equality can be asserted without trusting
   139→ * the implementation under test).
   140→ */
   141→function expectedPreM4cOrder(records: EvalMem[]): string[] {
   142→  const byType = new Map<string, EvalMem[]>();
   143→  for (const r of records) {
   144→    byType.set(r.memoryType, [...(byType.get(r.memoryType) ?? []), r]);
   145→  }
   146→  for (const list of byType.values()) {
   147→    list.sort((a, b) => {
   148→      if (b.confidence !== a.confidence) return b.confidence - a.confidence;
   149→      const ta = new Date(a.updatedAt).getTime();
   150→      const tb = new Date(b.updatedAt).getTime();
   151→      if (tb !== ta) return tb - ta;
   152→      return a.id < b.id ? -1 : a.id > b.id ? 1 : 0;
   153→    });
   154→  }
   155→  const order: string[] = [];
   156→  for (let round = 0; order.length < 20; round++) {
   157→    let advanced = false;
   158→    for (const t of SOPHIA_MEMORY_TYPES) {
   159→      if (order.length >= 20) break;
   160→      const list = byType.get(t);
   161→      if (list && list.length > round) {
   162→        order.push(list[round].id);
   163→        advanced = true;
   164→      }
   165→    }
   166→    if (!advanced) break;
   167→  }
   168→  return order;
   169→}

   171→/** Lexical score re-derivation for evaluation (same shared exported tokenizer). */
   172→function lexicalScore(m: EvalMem, query: string): number {
   173→  const queryTokens = new Set(extractTokens(query));
   174→  const contentTokens = new Set(extractTokens(m.content));
   175→  let score = 0;
   176→  for (const t of queryTokens) if (contentTokens.has(t)) score++;
   177→  return score;
   178→}

   180→interface QueryMetrics {
   181→  queryId: string;
   182→  label: string;
   183→  query: string;
   184→  corpusSize: number;
   185→  goldSize: number;
   186→  goldRanks: Array<{ id: string; rank: number }>;
   187→  tier1Size: number;
   188→  recall5: number;
   189→  recall10: number;
   190→  recall20: number;
   191→  precision5: number;
   192→  precision10: number;
   193→  precision20: number;
   194→  firstHitRank: number | null;
   195→}
   196→
   197→function measureQuery(
   198→  queryId: string,
   199→  label: string,
   200→  query: string,
   201→  corpus: EvalMem[],
   202→  gold: string[]
   203→): QueryMetrics {
   204→  const ranked = ids(selectPersonalMindMemories(corpus, query));
   205→  const goldSet = new Set(gold);
   206→  const recallAt = (k: number) =>
   207→    gold.filter((g) => ranked.slice(0, k).includes(g)).length / gold.length;
   208→  const precisionAt = (k: number) =>
   209→    ranked.slice(0, k).filter((id) => goldSet.has(id)).length / Math.min(k, ranked.length);
   210→  const goldRanks = gold.map((g) => ({ id: g, rank: ranked.indexOf(g) + 1 }));
   211→  const tier1Size = corpus.filter((m) => lexicalScore(m, query) > 0).length;
   212→  const firstHit = goldRanks.find((r) => r.rank > 0)?.rank ?? null;
   213→  return {
   214→    queryId,
   215→    label,
   216→    query,
   217→    corpusSize: corpus.length,
   218→    goldSize: gold.length,
   219→    goldRanks,
   220→    tier1Size,
   221→    recall5: recallAt(5),
   222→    recall10: recallAt(10),
   223→    recall20: recallAt(20),
   224→    precision5: precisionAt(5),
   225→    precision10: precisionAt(10),
   226→    precision20: precisionAt(20),
   227→    firstHitRank: firstHit,
   228→  };
   229→}

   438→function mulberry32(seed: number): () => number {
   439→  let a = seed >>> 0;
   440→  return () => {
   441→    a = (a + 0x6d2b79f5) | 0;
   442→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
   443→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
   444→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
   445→  };
   446→}
