   519→  function setCoreState(stateKey, notifyUser = true) {
   520→    if (!CORE_STATES[stateKey]) return;
   521→    currentState = stateKey;
   522→
   523→    Object.keys(CORE_STATES).forEach(s => body.classList.remove('state-' + s));
   524→    body.classList.add('state-' + stateKey);
   525→
   526→    const meta = CORE_STATES[stateKey];
   527→    if (coreStatusLabel) coreStatusLabel.textContent = meta.label;
   528→    if (coreHeadline) coreHeadline.textContent = meta.headline;
   529→    if (coreTagline) coreTagline.textContent = meta.tagline;
   530→    if (headerContextText) headerContextText.textContent = meta.contextHeader;
   531→
   532→    // Update state toolbar
   533→    stateChips.forEach(chip => {
   534→      if (chip.dataset.state === stateKey) {
   535→        chip.classList.add('active');
   536→      } else {
   537→        chip.classList.remove('active');
   538→      }
   539→    });
   540→
   541→    // Update Contextual Beacons visibility/appearance based on state
   542→    if (beaconAttention) {
   543→      if (stateKey === 'waiting_for_founder') {
   544→        beaconAttention.style.display = 'flex';
   545→        beaconAttention.classList.add('highlight-attention');
   546→      } else {
   547→        beaconAttention.classList.remove('highlight-attention');
   548→      }
   549→    }
   550→
   551→    if (notifyUser) {
   552→      showToast(`Core state: ${stateKey.toUpperCase().replace(/_/g, ' ')}`);
   553→    }
   554→  }

   608→  function applySteering(steeringType) {
   609→    closeSteering();
   610→    if (simulationTimer) clearTimeout(simulationTimer);
   611→
   612→    if (steeringType === 'product') {
   613→      activeTaskTitle = "Product Context Deep Dive";
   614→      if (workingTaskName) workingTaskName.textContent = activeTaskTitle;
   615→      if (headerContextText) headerContextText.textContent = "LUMORA · PRODUCT CONTEXT";
   616→      showToast("Steered work: Core focused on Product context.");
   617→    } else if (steeringType === 'research') {
   618→      activeTaskTitle = "Market Research Signal Synthesis";
   619→      if (workingTaskName) workingTaskName.textContent = activeTaskTitle;
   620→      if (headerContextText) headerContextText.textContent = "RESEARCH & SIGNALS";
   621→      showToast("Steered work: Core focused on Market Research.");
   622→    } else if (steeringType === 'findings') {
   623→      setCoreState('completed');
   624→      showToast("Core paused: Showing findings accumulated so far.");
   625→      return;
   626→    } else if (steeringType === 'new_directive') {
   627→      setCoreState('ready');
   628→      commandInput.focus();
   629→      showToast("Core reset: Ready for new instruction.");
   630→      return;
   631→    }
   632→
   633→    // Continue working with updated focus
   634→    setCoreState('working', false);
   635→    simulationTimer = setTimeout(() => {
   636→      setCoreState('waiting_for_founder', false);
   637→      showToast("Steered task reached Decision Gate.");
   638→    }, 2400);
   639→  }

   648→  // --- 10. COMMAND EXECUTION LIFECYCLE SIMULATION ---
   649→  function runCommandSimulation(commandText) {
   650→    if (!commandText || commandText.trim() === '') return;
   651→    const cleanCmd = commandText.trim();
   652→    commandInput.value = '';
   653→
   654→    if (!creditsAvailable) {
   655→      showToast("Cannot run Jarvis command: AI Credits depleted. Use Manual mode.");
   656→      return;
   657→    }
   658→
   659→    if (simulationTimer) clearTimeout(simulationTimer);
   660→
   661→    // Contextual direct navigation check
   662→    const lower = cleanCmd.toLowerCase();
   663→    if (lower.includes('status') || lower.includes('company')) {
   664→      toggleCompanyState(true);
   665→      return;
   666→    }
   667→    if (lower.includes('workforce') || lower.includes('sophia') || lower.includes('thorne')) {
   668→      toggleWorkforce(true);
   669→      return;
   670→    }
   671→
   672→    activeTaskTitle = cleanCmd;
   673→    if (intentQuoteDisplay) intentQuoteDisplay.textContent = `"${cleanCmd}"`;
   674→    if (workingTaskName) workingTaskName.textContent = cleanCmd;
   675→
   676→    showToast(`Intent captured: "${cleanCmd}"`);
   677→    setCoreState('understanding', false);
   678→
   679→    // 1. Understanding -> Working (1.4s)
   680→    simulationTimer = setTimeout(() => {
   681→      setCoreState('working', false);
   682→      showToast("Core working: Evaluating company context & signals...");
   683→
   684→      // 2. Working -> Waiting for Founder (3.0s)
   685→      simulationTimer = setTimeout(() => {
   686→        setCoreState('waiting_for_founder', false);
   687→        showToast("Consequential Decision Gate reached: Founder Attention Required.");
   688→      }, 3000);
   689→
   690→    }, 1400);
   691→  }

   760→  // --- 12. REAL-TIME CLOCK ---
   761→  function updateClock() {
   762→    if (!clockDisplay) return;
   763→    const now = new Date();
   764→    const dateOpts = { weekday: 'short', month: 'short', day: 'numeric' };
   765→    const dateStr = now.toLocaleDateString('en-US', dateOpts);
   766→    const timeStr = now.toTimeString().substring(0, 5);
   767→
   768→    const dateSpan = clockDisplay.querySelector('.clock-date');
   769→    const timeSpan = clockDisplay.querySelector('.clock-time');
   770→    if (dateSpan) dateSpan.textContent = dateStr;
   771→    if (timeSpan) timeSpan.textContent = timeStr;
   772→  }
