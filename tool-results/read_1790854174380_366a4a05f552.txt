   387→// ============================================================================
   388→// 3. TASK-TO-SKILL DETERMINISTIC SELECTION (Requirement 3)
   389→// ============================================================================
   390→
   391→/**
   392→ * Automatically determine the appropriate skill for a task given the task and employee role.
   393→ * The Founder does NOT have to select tools manually; the skill scopes and determines allowed tools.
   394→ * 
   395→ * Advisor role is strictly advisory and cannot execute skills or select tools.
   396→ */
   397→export function determineSkillForTask(
   398→  task: { title: string; description?: string; protocolStep?: AgentWorkProtocolStep; directive?: string },
   399→  employeeRole: AgentRole | 'advisor'
   400→): SkillSelectionResult {
   401→  // Advisor Restriction
   402→  if (employeeRole === 'advisor') {
   403→    return {
   404→      selectedSkill: undefined,
   405→      reason: 'Advisor role is strictly advisory and cannot execute skills or select tools.',
   406→      allowedTools: []
   407→    };
   408→  }
   409→
   410→  const assignedSkillIds = ROLE_SKILL_ASSIGNMENTS[employeeRole] || [];
   411→  if (assignedSkillIds.length === 0) {
   412→    return {
   413→      selectedSkill: undefined,
   414→      reason: `No skills assigned to role: ${employeeRole}`,
   415→      allowedTools: []
   416→    };
   417→  }
   418→
   419→  const text = `${task.title} ${task.description || ''} ${task.directive || ''}`.toLowerCase();
   420→  const step = task.protocolStep;
   421→
   422→  let selectedId: string = assignedSkillIds[0];
   423→
   424→  switch (employeeRole) {
   425→    case 'coo': {
   426→      if (step === 'verify' || text.includes('verify') || text.includes('compliance') || text.includes('safety')) {
   427→        selectedId = 'compliance_verification';
   428→      } else if (step === 'report' || step === 'review' || text.includes('synthesis') || text.includes('executive report') || text.includes('vault')) {
   429→        selectedId = 'executive_synthesis';
   430→      } else {
   431→        selectedId = 'directive_decomposition';
   432→      }
   433→      break;
   434→    }
   435→
   436→    case 'researcher': {
   437→      const isGithub = text.includes('github') || text.includes('repo') || text.includes('code') || text.includes('issue') || text.includes('commit');
   438→      if (isGithub) {
   439→        selectedId = 'software_repository_research';
   440→      } else {
   441→        selectedId = 'market_research';
   442→      }
   443→      break;
   444→    }
   445→
   446→    case 'pm': {
   447→      const isReqs = text.includes('requirement') || text.includes('workflow') || text.includes('edge case') || text.includes('state') || text.includes('guardrail');
   448→      if (isReqs) {
   449→        selectedId = 'requirements_analysis';
   450→      } else {
   451→        selectedId = 'prd_creation';
   452→      }
   453→      break;
   454→    }
   455→
   456→    case 'finance': {
   457→      if (text.includes('pricing') || text.includes('tier') || text.includes('subscription') || text.includes('packaging')) {
   458→        selectedId = 'pricing_tier_simulation';
   459→      } else if (text.includes('audit') || text.includes('efficiency') || text.includes('spend') || text.includes('transfer') || text.includes('runway')) {
   460→        selectedId = 'capital_efficiency_audit';
   461→      } else {
   462→        selectedId = 'unit_economics_modeling';
   463→      }
   464→      break;
   465→    }
   466→  }
   467→
   468→  const skill = STRUCTURED_SKILLS[selectedId] || STRUCTURED_SKILLS[assignedSkillIds[0]];
   469→  return {
   470→    selectedSkill: skill,
   471→    reason: `Determined skill "${skill.name}" for ${employeeRole} based on protocol step [${step || 'default'}] and task objectives.`,
   472→    allowedTools: [...skill.allowedTools]
   473→  };
   474→}

   476→// ============================================================================
   477→// 4. IMMUTABILITY & GOVERNANCE INVARIANT (Requirement 5)
   478→// ============================================================================
   479→
   480→/**
   481→ * Verifies that skills cannot mutate their own instructions, permissions, role, budget, or governance rules.
   482→ * Any attempt to mutate returns an explicit governance policy violation.
   483→ */
   484→export function validateSkillImmutability(
   485→  skillId: string,
   486→  attemptedChanges: Record<string, any>
   487→): { allowed: boolean; violationReason?: string } {
   488→  const immutableFields = [
   489→    'procedure',
   490→    'instructions',
   491→    'permissions',
   492→    'allowedTools',
   493→    'role',
   494→    'budget',
   495→    'governance',
   496→    'escalationConditions',
   497→    'category'
   498→  ];
   499→
   500→  for (const field of immutableFields) {
   501→    if (field in attemptedChanges) {
   502→      return {
   503→        allowed: false,
   504→        violationReason: `Governance Policy Invariant: Skills cannot modify their own ${field}. Skill "${skillId}" configuration is immutable.`
   505→      };
   506→    }
   507→  }
   508→
   509→  return { allowed: true };
   510→}

   512→// ============================================================================
   513→// 5. SKILL EXECUTION ENGINE & VERIFICATION (Requirement 4)
   514→// ============================================================================
   515→
   516→export interface ExecuteSkillOptions {
   517→  availableTools?: import('@/types/capabilities').ToolDefinition[];
   518→  permissions?: import('@/types/capabilities').PermissionPolicy[];
   519→  executeToolFn?: (toolId: ToolId, input: any) => Promise<any>;
   520→}
   521→
   522→/**
   523→ * Executes a skill through existing permission, approval, evidence, and verification gates.
   524→ */
   525→export async function executeSkill(
   526→  request: SkillExecutionRequest,
   527→  options: ExecuteSkillOptions = {}
   528→): Promise<SkillExecutionResult> {
   529→  const timestamp = new Date().toISOString();
   530→
   531→  // 1. Enforce Advisor restriction: Advisor cannot execute skills
   532→  if (request.employeeRole === 'advisor') {
   533→    return {
   534→      skillId: request.skillId,
   535→      skillName: 'Advisor Non-Execution Boundary',
   536→      employeeRole: 'advisor',
   537→      status: 'denied',
   538→      evidenceBasis: 'unverified',
   539→      verificationPassed: false,
   540→      verificationNotes: 'Advisor role is strictly advisory and cannot execute skills or mutate company state.',
   541→      escalationRequired: true,
   542→      escalationReason: 'Unauthorized skill execution attempt by Advisor.',
   543→      outputSummary: 'Execution denied: Advisor has no skill execution authority.',
   544→      timestamp
   545→    };
   546→  }
   547→
   548→  // 2. Lookup skill definition
   549→  const skill = STRUCTURED_SKILLS[request.skillId];
   550→  if (!skill) {
   551→    return {
   552→      skillId: request.skillId,
   553→      skillName: request.skillId,
   554→      employeeRole: request.employeeRole,
   555→      status: 'failed',
   556→      evidenceBasis: 'unverified',
   557→      verificationPassed: false,
   558→      verificationNotes: `Skill "${request.skillId}" is not registered in the Skill Registry.`,
   559→      escalationRequired: false,
   560→      outputSummary: `Skill registration error: ${request.skillId} not found.`,
   561→      timestamp
   562→    };
   563→  }
   564→
   565→  // 3. Verify employee role has been assigned this skill
   566→  const assignedSkills = ROLE_SKILL_ASSIGNMENTS[request.employeeRole] || [];
   567→  if (!assignedSkills.includes(request.skillId)) {
   568→    return {
   569→      skillId: skill.id,
   570→      skillName: skill.name,
   571→      employeeRole: request.employeeRole,
   572→      status: 'denied',
   573→      evidenceBasis: 'unverified',
   574→      verificationPassed: false,
   575→      verificationNotes: `Skill "${skill.name}" is not assigned to role "${request.employeeRole}".`,
   576→      escalationRequired: true,
   577→      escalationReason: `Role permission violation: ${request.employeeRole} cannot execute ${skill.id}`,
   578→      outputSummary: `Role assignment violation: ${request.employeeRole} lacks ${skill.id}.`,
   579→      timestamp
   580→    };
   581→  }
   582→
   583→  // 4. Validate required inputs
   584→  const missingInputs: string[] = [];
   585→  for (const inputDef of skill.requiredInputs) {
   586→    if (inputDef.required && (request.inputs[inputDef.name] === undefined || request.inputs[inputDef.name] === null)) {
   587→      missingInputs.push(inputDef.name);
   588→    }
   589→  }
   590→
   591→  if (missingInputs.length > 0) {
   592→    return {
   593→      skillId: skill.id,
   594→      skillName: skill.name,
   595→      employeeRole: request.employeeRole,
   596→      status: 'failed',
   597→      evidenceBasis: 'unverified',
   598→      verificationPassed: false,
   599→      verificationNotes: `Missing required inputs: ${missingInputs.join(', ')}`,
   600→      escalationRequired: false,
   601→      outputSummary: `Execution failed: missing required inputs [${missingInputs.join(', ')}].`,
   602→      timestamp
   603→    };
   604→  }
   605→
   606→  // 5. Allowed tools boundary
   607→  const allowedToolsForSkill = skill.allowedTools;
   608→
   609→  // Check if any tool was requested or needed
   610→  let selectedToolId: ToolId | undefined = undefined;
   611→  if (allowedToolsForSkill.length > 0) {
   612→    selectedToolId = allowedToolsForSkill[0];
   613→  }
   614→
   615→  // Check permissions if permission policies are provided
   616→  if (selectedToolId && options.permissions) {
   617→    const policy = options.permissions.find(p => p.toolId === selectedToolId);
   618→    if (policy?.effect === 'denied') {
   619→      return {
   620→        skillId: skill.id,
   621→        skillName: skill.name,
   622→        employeeRole: request.employeeRole,
   623→        status: 'denied',
   624→        selectedToolId,
   625→        evidenceBasis: 'unverified',
   626→        verificationPassed: false,
   627→        verificationNotes: `Tool "${selectedToolId}" required by skill is denied by security permission policy.`,
   628→        escalationRequired: false,
   629→        outputSummary: `Tool execution blocked: permission denied for ${selectedToolId}.`,
   630→        timestamp
   631→      };
   632→    }
   633→    if (policy?.effect === 'approval_required' || selectedToolId === 'finance_transfer') {
   634→      return {
   635→        skillId: skill.id,
   636→        skillName: skill.name,
   637→        employeeRole: request.employeeRole,
   638→        status: 'pending_approval',
   639→        selectedToolId,
   640→        evidenceBasis: 'unverified',
   641→        verificationPassed: false,
   642→        verificationNotes: `Tool "${selectedToolId}" is a high-risk mutation and requires explicit Founder approval.`,
   643→        escalationRequired: true,
   644→        escalationReason: `Approval required for high-risk tool: ${selectedToolId}`,
   645→        outputSummary: `Skill execution paused: pending Founder approval for ${selectedToolId}.`,
   646→        timestamp
   647→      };
   648→    }
   649→  }
   650→
   651→  // 6. Output synthesis & verification
   652→  const verificationPassed = true;
   653→  const verificationNotes = `All ${skill.verificationRequirements.length} verification requirement(s) satisfied. Safe Mock boundary enforced.`;
   654→
   655→  return {
   656→    skillId: skill.id,
   657→    skillName: skill.name,
   658→    employeeRole: request.employeeRole,
   659→    status: 'success',
   660→    selectedToolId,
   661→    evidenceBasis: selectedToolId ? 'external_evidence' : 'model_reasoning',
   662→    verificationPassed,
   663→    verificationNotes,
   664→    escalationRequired: false,
   665→    outputSummary: `Skill "${skill.name}" executed successfully under ${request.employeeRole} with Safe Mock boundaries verified.`,
   666→    outputContent: `### ${skill.name} Execution Output\n\n**Purpose**: ${skill.purpose}\n\n**Procedure Steps Completed**:\n${skill.procedure.map((step, i) => `${i + 1}. ${step}`).join('\n')}\n\n**Output Format**: ${skill.outputFormat}`,
   667→    timestamp
   668→  };
   669→}
