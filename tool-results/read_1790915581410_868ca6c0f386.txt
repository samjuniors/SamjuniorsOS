   125→function getDevAuthHeaders(): Record<string, string> {
   126→  // R0 secrets hygiene: NO hardcoded fallback secret. When
   127→  // NEXT_PUBLIC_SAMJUNIORS_DEV_SECRET is not configured at build time the
   128→  // client attaches no founder credentials — development mode stays open
   129→  // by design (single-tenant sandbox), and production fails closed on the
   130→  // server until an explicit secret is provisioned.
   131→  const secret = process.env.NEXT_PUBLIC_SAMJUNIORS_DEV_SECRET;
   132→  if (!secret) {
   133→    return {};
   134→  }
   135→  if (typeof window !== "undefined") {
   136→    try {
   137→      document.cookie = `samjuniors-dev-as=founder; path=/; SameSite=Lax`;
   138→      document.cookie = `samjuniors-dev-secret=${secret}; path=/; SameSite=Lax`;
   139→    } catch {
   140→      /* ignore */
   141→    }
   142→  }
   143→  return {
   144→    "x-samjuniors-dev-as": "founder",
   145→    "x-samjuniors-dev-secret": secret,
   146→  };
   147→}
   148→
   149→async function jsonFetch<T>(url: string, init?: RequestInit): Promise<T> {
   150→  const devHeaders = getDevAuthHeaders();
   151→  const res = await fetch(url, {
   152→    ...init,
   153→    headers: {
   154→      ...devHeaders,
   155→      ...init?.headers,
   156→    },
   157→  });
   158→  let body: any = null;
   159→  try { body = await res.json(); } catch { /* non-JSON error body */ }
   160→  if (!res.ok) {
   161→    throw new Error(body?.error || `Request failed (${res.status} ${url})`);
   162→  }
   163→  return body as T;
   164→}

   497→export function looksLikeDirective(text: string): boolean {

   501→    /^(research|investigate|analyze|evaluate|audit|model|draft|design|create|build|prepare|synthesize|simulate)\s+(the|our|a|an|all)\s+/i,
   502→    /\b(give me a recommendation|give me a plan|create a prd|draft a prd|model the unit economics|model our pricing|audit compute burn|conduct a research|run an analysis|synthesize a proposal)\b/i,
   503→    /^(execute|orchestrate|launch task|run task|start initiative)\b/i,
   504→    /^(prepare|research|analyze|build|create|model|draft|evaluate|audit|generate)\b/i,
   505→    /^(work|start)\b/i,
   506→  ];
   507→  return patterns.some((p) => p.test(clean));

   519→/** Dispatch a founder directive to the real orchestration path. While the council
   520→ *  executes (~minutes), polls the durable agent-run store so the OS surfaces and
   521→ *  the operating graph visualize REAL step-by-step progress — never simulated. */
   522→export async function dispatchDirective(directive: string, opts?: { agents?: string[] }): Promise<OrchestrationRun> {
   523→  const poll = setInterval(() => {
   524→    syncFromServer({ quiet: true }).catch(() => { /* transient poll failure — next tick retries */ });
   525→  }, 3000);
   526→  try {
   527→    const run = await orchestrate(directive, opts?.agents);
   528→    clearInterval(poll);
   529→    applyOrchestrationRun(run);
   530→    await syncFromServer();
   531→    if (typeof window !== "undefined") {
   532→      window.dispatchEvent(new CustomEvent("samjuniors:graph-refresh"));
   533→    }
   534→    return run;
   535→  } catch (err) {
   536→    clearInterval(poll);
   537→    // Sync whatever steps actually persisted server-side before the failure,
   538→    // then surface the honest error — no fake local success.
   539→    await syncFromServer().catch(() => undefined);
   540→    if (typeof window !== "undefined") {
   541→      window.dispatchEvent(new CustomEvent("samjuniors:graph-refresh"));
   542→    }
   543→    throw err instanceof Error ? err : new Error(String(err));
   544→  }
   545→}

   547→/** One-line real summary of a completed run (for Sophia's reply). */
   548→export function summarizeRun(run: OrchestrationRun): string {
   549→  const verified = run.verificationResult?.isCompliant;
   550→  const outcome = run.executiveResult?.executionOutcome ?? run.status;
   551→  const parts: string[] = [];
   552→  parts.push(run.liveAi === false ? "Council executed in safe unconfigured mode." : `Directive executed: ${run.title}.`);
   553→  if (verified === true) parts.push("Constitutional verification passed.");
   554→  if (verified === false) parts.push(`Verification rejected: ${(run.verificationResult?.checksFailed ?? ["unknown"]).join("; ")}.`);
   555→  if (run.executionSummary) parts.push(`${run.executionSummary.totalTasksExecuted} protocol steps · ${run.executionSummary.agentsInvoked.length} agents · ${run.deliverables.length} deliverables.`);
   556→  if (run.executiveResult?.founderDecision?.required) parts.push(`A decision is waiting for you: ${run.executiveResult.founderDecision.title}.`);
   557→  parts.push(`Outcome: ${outcome}.`);
   558→  return parts.join(" ");
   559→}

   563→function hashId(prefix: string, text: string): string {
   564→  let h = 5381;
   565→  for (let i = 0; i < text.length; i++) h = ((h << 5) + h + text.charCodeAt(i)) | 0;
   566→  return `${prefix}-${(h >>> 0).toString(36)}`;
   567→}

   569→const titleOf = (directive: string) => (directive.length > 60 ? `${directive.slice(0, 58)}…` : directive);

   571→/** Groups durable agent-run records by directive and derives honest workstreams:
   572→ *  stage from the furthest real protocol step, owner from the latest specialist
   573→ *  that actually executed, and state from real recency. */
   574→export function workstreamsFromRuns(runs: AgentRunRecord[], dismissed: string[] = []): Workstream[] {
   575→  const groups = new Map<string, AgentRunRecord[]>();
   576→  for (const r of runs) {
   577→    const key = r.directive?.trim();
   578→    if (!key) continue;
   579→    (groups.get(key) ?? groups.set(key, []).get(key)!).push(r);
   580→  }
   581→
   582→  const work: Workstream[] = [];
   583→  for (const [directive, group] of groups) {
   584→    const sorted = [...group].sort((a, b) => a.timestamp.localeCompare(b.timestamp));
   585→    const latest = sorted[sorted.length - 1];
   586→    const id = hashId("srv", directive);
   587→    if (dismissed.includes(id)) continue;
   588→
   589→    const specialistRuns = sorted.filter((r) => r.agentId !== "coo");
   590→    const latestSpecialist = specialistRuns[specialistRuns.length - 1];
   591→    const owner = latestSpecialist ? toUiAgentId(latestSpecialist.agentId) : "sophia";
   592→
   593→    const hasReport = sorted.some((r) => r.protocolStep === "report");
   594→    const lastStep = latest.protocolStep;
   595→    const age = Date.now() - new Date(latest.timestamp).getTime();
   596→    const inFlight = !hasReport && age < IN_FLIGHT_WINDOW_MS;
   597→
   598→    const stage: Stage = hasReport ? "done" : STEP_STAGE[lastStep] ?? "discovery";
   599→    const state: Workstream["state"] = hasReport ? "done" : inFlight ? "active" : "paused";
   600→
   601→    work.push({
   602→      id,
   603→      title: titleOf(directive),
   604→      owner,
   605→      stage,
   606→      state,
   607→      note: `${sorted.length} real protocol steps · last: ${latest.taskTitle}`,
   608→      at: new Date(latest.timestamp).getTime(),
   609→      origin: "server",
   610→      directive,
   611→    });
   612→    if (work.length >= 5) break; // most recent directives only — groups arrive in run-recency order
   613→  }
   614→  return work;
   615→}

   617→/** Real agent runtime state from durable run records. Only agents currently
   618→ *  participating in an in-flight directive show "working"; everything else
   619→ *  decays to ready. Founder's local offline choices are preserved by osStore. */
   620→export function agentStatesFromRuns(runs: AgentRunRecord[], work: Workstream[]): Record<string, { state: Agent["state"]; current?: string }> {
   621→  const out: Record<string, { state: Agent["state"]; current?: string }> = {};
   622→  const active = work.some((w) => w.state === "active");
   623→  if (!active) return out;
   624→
   625→  for (const r of runs) {
   626→    const uiId = toUiAgentId(r.agentId);
   627→    if (out[uiId]?.state === "working") continue;
   628→    const age = Date.now() - new Date(r.timestamp).getTime();
   629→    if (age < IN_FLIGHT_WINDOW_MS) {
   630→      out[uiId] = { state: "working", current: r.taskTitle };
   631→    }
   632→  }
   633→  return out;
   634→}

   638→let syncing: Promise<void> | null = null;
   639→
   640→/** Pulls roster + runs + approvals + the company Activity projection from the
   641→ *  server and feeds the OS store. Server-origin state REPLACES local
   642→ *  projections; local founder records (notes, focus, offline choices) are
   643→ *  preserved. The Activity projection (Phase 4.4C) is the AUTHORITATIVE
   644→ *  company Activity source — the client os.log remains only an ambient
   645→ *  browser-session supplement. */
   646→export async function syncFromServer(opts?: { quiet?: boolean }): Promise<void> {
   647→  if (syncing) return syncing;
   648→  syncing = (async () => {
   649→    try {
   650→      const [roster, runs, approvals, activity, epistemic] = await Promise.all([
   651→        fetchRoster(),
   652→        fetchRuns(),
   653→        fetchApprovals(),
   654→        fetchActivity().catch((err) => {
   655→          // Activity is additive to the core sync: an activity-specific failure
   656→          // must not take down roster/runs/approvals sync. Honest signal only.
   657→          os.log(`Company activity unavailable: ${err instanceof Error ? err.message : String(err)}`);
   658→          return null;
   659→        }),
   660→        fetchEpistemicBoard().catch((err) => {
   661→          // Phase 4.4E: the epistemic board is likewise additive — a
   662→          // board-specific failure degrades honestly without breaking the
   663→          // core sync. Governed epistemic state is never faked locally.
   664→          os.log(`Epistemic board unavailable: ${err instanceof Error ? err.message : String(err)}`);
   665→          return null;
   666→        }),
   667→      ]);
   668→      applyServerReadModel({ roster, runs, approvals });
   669→      if (activity) os.setServerActivity(activity.map(serverActivityToEvent));
   670→      if (epistemic) os.setServerEpistemic(epistemic);
   671→    } catch (err) {
   672→      if (!opts?.quiet) {
   673→        // Honest signal — never fake a successful sync.
   674→        os.log(`Server sync unavailable: ${err instanceof Error ? err.message : String(err)}`);
   675→      }
   676→      throw err;
   677→    } finally {
   678→      syncing = null;
   679→    }
   680→  })();
   681→  return syncing;
   682→}

   684→function applyServerReadModel(input: { roster: ServerAgentDef[]; runs: AgentRunRecord[]; approvals: ApprovalRecord[] }) {
   685→  const { roster, runs, approvals } = input;
   686→  const dismissed = os.getDismissed();
   687→  const work = workstreamsFromRuns(runs, dismissed);
   688→  const agentStates = agentStatesFromRuns(runs, work);
   689→
   690→  // Agents: authoritative identity (existence/name/role) from the server roster,
   691→  // presentation copy from the UI's own constants, runtime state from real runs.
   692→  const agents: Agent[] = roster.map((def) => {
   693→    const uiId = toUiAgentId(def.id);
   694→    const presentation = presentationFor(uiId);
   695→    const rt = agentStates[uiId];
   696→    return {
   697→      ...presentation,
   698→      id: uiId,
   699→      name: def.name,
   700→      role: def.role,
   701→      state: rt?.state ?? "ready",
   702→      current: rt?.current,
   703→    };
   704→  });
   705→
   706→  // Decisions + attention from REAL pending approval gate records.
   707→  const decisions: Decision[] = approvals.map((a) => ({
   708→    id: a.id,
   709→    title: a.actionName,
   710→    context: a.target?.summary || a.notes || `${a.classification} · ${a.workflowInstanceId}`,
   711→    options: ["Approve", "Reject"],
   712→    raisedBy: SERVER_TO_UI[a.employeeRole] ?? a.employeeRole,
   713→    at: new Date(a.requestedAt).getTime(),
   714→    status: "open",
   715→    approvalId: a.id,
   716→  }));
   717→
   718→  const attention: AttentionItem[] = approvals.map((a) => ({
   719→    id: `att-${a.id}`,
   720→    kind: "decision",
   721→    title: a.actionName,
   722→    detail: a.target?.summary || a.notes,
   723→    from: SERVER_TO_UI[a.employeeRole] ?? a.employeeRole,
   724→    at: new Date(a.requestedAt).getTime(),
   725→    decisionId: a.id,
   726→    server: true,
   727→  }));
   728→
   729→  os.applyServerState({ agents, work, decisions, attention });
   730→}
