    98→    98→function samplePath(raw: [number, number][], radius = 22, step = 4): P[] {
    99→    99→  const pts = raw.map(([x, y]) => ({ x, y }));
   100→   100→  if (pts.length < 2) return pts;
   101→   101→  const out: P[] = [];
   102→   102→  const pushLine = (a: P, b: P) => {
   103→   103→    const len = Math.hypot(b.x - a.x, b.y - a.y);
   104→   104→    const n = Math.max(1, Math.ceil(len / step));
   105→   105→    for (let i = 0; i < n; i++) out.push({ x: a.x + (b.x - a.x) * (i / n), y: a.y + (b.y - a.y) * (i / n) });
   106→   106→  };
   107→   107→  const pushQuad = (a: P, c: P, b: P) => {
   108→   108→    const N = 10;
   109→   109→    for (let i = 0; i < N; i++) {
   110→   110→      const t = i / N, u = 1 - t;
   111→   111→      out.push({ x: u * u * a.x + 2 * u * t * c.x + t * t * b.x, y: u * u * a.y + 2 * u * t * c.y + t * t * b.y });
   112→   112→    }
   113→   113→  };
   114→   114→  let cur = pts[0];
   115→   115→  for (let i = 1; i < pts.length - 1; i++) {
   116→   116→    const p = pts[i], prev = pts[i - 1], next = pts[i + 1];
   117→   117→    const lin = Math.hypot(p.x - prev.x, p.y - prev.y);
   118→   118→    const lout = Math.hypot(next.x - p.x, next.y - p.y);
   119→   119→    if (lin < 1e-3 || lout < 1e-3) continue;
   120→   120→    const r = Math.min(radius, lin / 2, lout / 2);
   121→   121→    const din = { x: (p.x - prev.x) / lin, y: (p.y - prev.y) / lin };
   122→   122→    const dout = { x: (next.x - p.x) / lout, y: (next.y - p.y) / lout };
   123→   123→    const pin = { x: p.x - din.x * r, y: p.y - din.y * r };
   124→   124→    const pout = { x: p.x + dout.x * r, y: p.y + dout.y * r };
   125→   125→    pushLine(cur, pin);
   126→   126→    pushQuad(pin, p, pout);
   127→   127→    cur = pout;
   128→   128→  }
   129→   129→  pushLine(cur, pts[pts.length - 1]);
   130→   130→  out.push(pts[pts.length - 1]);
   131→   131→  return out;
   132→   132→}

   151→   151→/**
   152→   152→ * Collision-aware vertical column layout engine.
   153→   153→ * Computes balanced Y coordinates for nodes in a column respecting their bounding box heights,
   154→   154→ * applying relaxation passes to eliminate any overlap.
   155→   155→ */
   156→   156→function layoutColumn(nodesInCol: FlowNode[], centerY = 440, minGap = 24) {
   157→   157→  if (nodesInCol.length === 0) return;
   158→   158→  if (nodesInCol.length === 1) {
   159→   159→    nodesInCol[0].y = centerY;
   160→   160→    return;
   161→   161→  }
   162→   162→  const totalH = nodesInCol.reduce((sum, n) => sum + n.h, 0) + (nodesInCol.length - 1) * minGap;
   163→   163→  let curY = centerY - totalH / 2;
   164→   164→  for (const n of nodesInCol) {
   165→   165→    n.y = Math.round(curY + n.h / 2);
   166→   166→    curY += n.h + minGap;
   167→   167→  }
   168→   168→  // Boundary safeguards & relaxation passes
   169→   169→  for (let iter = 0; iter < 4; iter++) {
   170→   170→    for (let i = 0; i < nodesInCol.length - 1; i++) {
   171→   171→      const a = nodesInCol[i];
   172→   172→      const b = nodesInCol[i + 1];
   173→   173→      const reqDist = (a.h + b.h) / 2 + minGap;
   174→   174→      const actualDist = b.y - a.y;
   175→   175→      if (actualDist < reqDist) {
   176→   176→        const overlap = (reqDist - actualDist) / 2;
   177→   177→        a.y = Math.max(120, Math.round(a.y - overlap));
   178→   178→        b.y = Math.min(780, Math.round(b.y + overlap));
   179→   179→      }
   180→   180→    }
   181→   181→  }
   182→   182→}

   187→   187→function connectNodes(
   188→   188→  fromNode: FlowNode,
   189→   189→  toNode: FlowNode,
   190→   190→  id: string,

   192→   192→  relationship: GraphRelationship,
   193→   193→  state: GraphNodeState,
   194→   194→  activity?: string

   196→   196→  const x1 = fromNode.x + fromNode.w / 2;
   197→   197→  const y1 = fromNode.y;
   198→   198→  const x2 = toNode.x - toNode.w / 2;
   199→   199→  const y2 = toNode.y;
   200→   200→
   201→   201→  let pts: [number, number][];
   202→   202→  if (Math.abs(y1 - y2) < 4) {
   203→   203→    pts = [[x1, y1], [x2, y2]];
   204→   204→  } else {
   205→   205→    const xMid = Math.round((x1 + x2) / 2);
   206→   206→    pts = [[x1, y1], [xMid, y1], [xMid, y2], [x2, y2]];
   207→   207→  }
   208→   208→
   209→   209→  return {
   210→   210→    id,
   211→   211→    from: fromNode.id,
   212→   212→    to: toNode.id,
   213→   213→    pts,
   214→   214→    style,
   215→   215→    arrow: true,
   216→   216→    relationship,
   217→   217→    state,
   218→   218→    activity,
   219→   219→  };
   220→   220→}

   820→   820→      }

   850→   850→  pointAt(p: Sampled, d: number): P {
   851→   851→    const cum = p.cum;
   852→   852→    if (d <= 0) return p.pts[0];
   853→   853→    if (d >= p.len) return p.pts[p.pts.length - 1];
   854→   854→    let lo = 0,
   855→   855→      hi = cum.length - 1;
   856→   856→    while (hi - lo > 1) {
   857→   857→      const mid = (lo + hi) >> 1;
   858→   858→      if (cum[mid] <= d) lo = mid;
   859→   859→      else hi = mid;
   860→   860→    }
   861→   861→    const seg = cum[hi] - cum[lo] || 1;
   862→   862→    const t = (d - cum[lo]) / seg;
   863→   863→    return { x: p.pts[lo].x + (p.pts[hi].x - p.pts[lo].x) * t, y: p.pts[lo].y + (p.pts[hi].y - p.pts[lo].y) * t };

   877→   877→        : p.edge.style === "rose"
   878→   878→        ? "rose"
   879→   879→        : p.edge.style === "amber"
   880→   880→        ? "amber"
   881→   881→        : p.edge.style === "emerald"
   882→   882→        ? "emerald"
   883→   883→        : "cyan";
   884→   884→
   885→   885→    // Source-node ignition: dispatch visibly energizes the emitting node's
   886→   886→    // perimeter (matches reference frame 2 — C-arc border light-up).
   887→   887→    const src = this.energy.get(p.edge.from) || 0;
   888→   888→    this.energy.set(p.edge.from, Math.max(src, 0.72));
   889→   889→
   890→   890→    this.packets.push({
