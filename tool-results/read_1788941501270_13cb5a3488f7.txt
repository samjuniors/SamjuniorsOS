   197→  /**
   198→   * Atomic step claim with genuine compare-and-swap (Phase 2.6 hardening).
   199→   *
   200→   * The previous implementation validated stateVersion at READ time only and then issued
   201→   * an unguarded UPDATE — under READ COMMITTED two workers could both validate the same
   202→   * version and both write (lost update / double claim). The corrected write carries the
   203→   * validated stateVersion in the UPDATE WHERE clause: exactly one writer can transition
   204→   * v→v+1; every other writer observes 0 rows and fails with ConcurrencyConflictError.
   205→   */
   206→  async claimStepAtomic(
   207→    instanceId: string,
   208→    stepId: string,
   209→    workerId: string,
   210→    expectedVersion?: number
   211→  ): Promise<{ instance: WorkflowInstanceState; step: WorkflowStepState }> {
   212→    const db = await requireAuthoritativeDatabase();
   213→
   214→    const dbInst = await db.workflowInstance.findUnique({
   215→      where: { id: instanceId },
   216→    });
   217→    if (!dbInst) {
   218→      throw new Error(`Workflow instance not found: ${instanceId}`);
   219→    }
   220→
   221→    // Version validated for this attempt — the CAS below is anchored to THIS value.
   222→    const validatedVersion = dbInst.stateVersion ?? 0;
   223→    if (expectedVersion !== undefined && validatedVersion !== expectedVersion) {
   224→      throw new ConcurrencyConflictError(instanceId, expectedVersion, validatedVersion);
   225→    }
   226→
   227→    const stepStates: Record<string, WorkflowStepState> = (dbInst.stepStates as any) || {};
   228→    const step = stepStates[stepId];
   229→    if (!step) {
   230→      throw new Error(`Step not found in workflow instance: ${stepId}`);
   231→    }
   232→
   233→    if (step.status !== 'ready') {
   234→      throw new StepClaimError(instanceId, stepId, step.status, step.claimedBy);
   235→    }
   236→
   237→    if (step.claimedBy && step.claimedBy !== workerId) {
   238→      throw new StepClaimError(instanceId, stepId, step.status, step.claimedBy);
   239→    }
   240→
   241→    validateStepTransition(step.status, 'running');
   242→
   243→    const nowIso = new Date().toISOString();
   244→    step.status = 'running';
   245→    step.claimedBy = workerId;
   246→    step.claimedAt = nowIso;
   247→    if (!step.startedAt) {
   248→      step.startedAt = nowIso;
   249→    }
   250→    stepStates[stepId] = step;
   251→
   252→    let instanceStatus = dbInst.status;
   253→    if (instanceStatus === 'pending' || instanceStatus === 'waiting') {
   254→      validateInstanceTransition(instanceStatus as any, 'running');
   255→      instanceStatus = 'running';
   256→    }
   257→
   258→    const nextVersion = validatedVersion + 1;
   259→
   260→    // Compare-and-swap: the WHERE clause carries the validated stateVersion.
   261→    const cas = await db.workflowInstance.updateMany({
   262→      where: {
   263→        id: instanceId,
   264→        stateVersion: validatedVersion,
   265→      },
   266→      data: {
   267→        status: instanceStatus,
   268→        stepStates: stepStates as any,
   269→        stateVersion: nextVersion,
   270→        claimedBy: workerId,
   271→        claimedAt: new Date(),
   272→        updatedAt: new Date(),
   273→      },
   274→    });
   275→
   276→    if (cas.count === 0) {
   277→      // Lost the race: another writer transitioned the instance between read and write.
   278→      const fresh = await db.workflowInstance.findUnique({ where: { id: instanceId } });
   279→      throw new ConcurrencyConflictError(instanceId, validatedVersion, fresh?.stateVersion ?? -1);
   280→    }
   281→
   282→    const updated = await db.workflowInstance.findUnique({ where: { id: instanceId } });
   283→    return {
   284→      instance: this.mapPrismaToInstance(updated!),
   285→      step,
   286→    };
   287→  }
   288→
   289→  /**
   290→   * Atomic step transition with genuine compare-and-swap (Phase 2.6 hardening).
   291→   *
   292→   * Same lost-update defect as claimStepAtomic: the UPDATE WHERE clause now carries the
   293→   * validated stateVersion, so exactly one writer wins and every loser fails with
   294→   * ConcurrencyConflictError instead of silently overwriting the winner.
   295→   *
   296→   * Crash-recovery hardening: transitioning a step to 'ready' now clears the stale claim
   297→   * identity (claimedBy/claimedAt). Previously a crashed worker's step could never be
   298→   * re-claimed because claimStepAtomic rejects any step whose claimedBy is set to a
   299→   * different (dead) worker — a permanent orphan.
   300→   */
   301→  async transitionStepAtomic(
   302→    instanceId: string,
   303→    stepId: string,
   304→    targetStatus: WorkflowStepStatus,
   305→    expectedVersion?: number,
   306→    patch?: Partial<WorkflowStepState>,
   307→    instancePatch?: Partial<WorkflowInstanceState>
   308→  ): Promise<WorkflowInstanceState> {
   309→    const db = await requireAuthoritativeDatabase();
   310→
   311→    const dbInst = await db.workflowInstance.findUnique({
   312→      where: { id: instanceId },
   313→    });
   314→    if (!dbInst) {
   315→      throw new Error(`Workflow instance not found: ${instanceId}`);
   316→    }
   317→
   318→    // Version validated for this attempt — the CAS below is anchored to THIS value.
   319→    const validatedVersion = dbInst.stateVersion ?? 0;
   320→    if (expectedVersion !== undefined && validatedVersion !== expectedVersion) {
   321→      throw new ConcurrencyConflictError(instanceId, expectedVersion, validatedVersion);
   322→    }
   323→
   324→    const stepStates: Record<string, WorkflowStepState> = (dbInst.stepStates as any) || {};
   325→    const step = stepStates[stepId];
   326→    if (!step) {
   327→      throw new Error(`Step not found in workflow instance: ${stepId}`);
   328→    }
   329→
   330→    validateStepTransition(step.status, targetStatus);
   331→
   332→    const nowIso = new Date().toISOString();
   333→    step.status = targetStatus;
   334→    if (targetStatus === 'completed' || targetStatus === 'failed' || targetStatus === 'cancelled') {
   335→      step.completedAt = nowIso;
   336→      step.claimedBy = undefined;
   337→      step.claimedAt = undefined;
   338→    }
   339→    if (targetStatus === 'ready') {
   340→      // Crash recovery / recurring reset: clear stale claim identity so the step can be
   341→      // re-claimed by ANY live worker (otherwise the orphaned claimedBy blocks forever).
   342→      step.claimedBy = undefined;
   343→      step.claimedAt = undefined;
   344→      step.completedAt = undefined;
   345→    }
   346→    if (patch) {
   347→      Object.assign(step, patch);
   348→    }
   349→    stepStates[stepId] = step;
   350→
   351→    let nextInstanceStatus = instancePatch?.status ?? dbInst.status;
   352→    if (instancePatch?.status && instancePatch.status !== dbInst.status) {
   353→      validateInstanceTransition(dbInst.status as any, instancePatch.status);
   354→    }
   355→
   356→    const nextVersion = validatedVersion + 1;
   357→
   358→    // Compare-and-swap: the WHERE clause carries the validated stateVersion.
   359→    const cas = await db.workflowInstance.updateMany({
   360→      where: {
   361→        id: instanceId,
   362→        stateVersion: validatedVersion,
   363→      },
   364→      data: {
   365→        status: nextInstanceStatus,
   366→        stepStates: stepStates as any,
   367→        stateVersion: nextVersion,
   368→        error: instancePatch?.failureReason ?? (targetStatus === 'failed' ? (step.error || 'Step failed') : dbInst.error),
   369→        completedAt: (nextInstanceStatus === 'completed' || targetStatus === 'completed') ? new Date() : dbInst.completedAt,
   370→        updatedAt: new Date(),
   371→      },
   372→    });
   373→
   374→    if (cas.count === 0) {
   375→      // Lost the race: another writer transitioned the instance between read and write.
   376→      const fresh = await db.workflowInstance.findUnique({ where: { id: instanceId } });
   377→      throw new ConcurrencyConflictError(instanceId, validatedVersion, fresh?.stateVersion ?? -1);
   378→    }
   379→
   380→    const updated = await db.workflowInstance.findUnique({ where: { id: instanceId } });
   381→    return this.mapPrismaToInstance(updated!);
   382→  }
   383→
   384→  async rehydrateActiveInstances(): Promise<WorkflowInstanceState[]> {
   385→    const db = await requireAuthoritativeDatabase();
   386→    const activeStatuses = ['pending', 'running', 'waiting', 'blocked', 'awaiting_approval'];
   387→    const dbInsts = await db.workflowInstance.findMany({
   388→      where: {
   389→        status: { in: activeStatuses },
   390→      },
   391→      orderBy: { updatedAt: 'asc' },
   392→    });
   393→    return dbInsts.map((inst) => this.mapPrismaToInstance(inst));
   394→  }
   395→
   396→  private mapPrismaToInstance(dbInst: any): WorkflowInstanceState {
   397→    return {
   398→      instanceId: dbInst.id,
   399→      workflowId: dbInst.definitionId,
   400→      version: dbInst.version,
   401→      stateVersion: dbInst.stateVersion ?? 0,
   402→      claimedBy: dbInst.claimedBy || undefined,
   403→      claimedAt: dbInst.claimedAt ? dbInst.claimedAt.toISOString() : undefined,
   404→      status: dbInst.status as any,
   405→      objective: dbInst.objective,
   406→      stepStates: (dbInst.stepStates as any) || {},
   407→      createdAt: dbInst.createdAt.toISOString(),
   408→      updatedAt: dbInst.updatedAt.toISOString(),
   409→      outputs: {},
   410→      evidenceReferences: [],
   411→      failureReason: dbInst.error || undefined,
   412→    };
   413→  }
   414→
   415→  clear(): void {
   416→    if (process.env.NODE_ENV === 'production') {
   417→      throw new Error('Workflow records cannot be cleared in production.');
   418→    }
   419→  }

   444→  public loadFromDurableStorage(): void {
   445→    try {
   446→      const persistedDefs = DurableFileStore.getInstance().readCollection<WorkflowDefinition>('workflow_definitions');
   447→      for (const [key, def] of Object.entries(persistedDefs)) {
   448→        this.definitions.set(key, def);
   449→      }
   450→      const persistedInsts = DurableFileStore.getInstance().readCollection<WorkflowInstanceState>('workflow_instances');
   451→      for (const [id, inst] of Object.entries(persistedInsts)) {
   452→        this.instances.set(id, inst);
   453→      }
   454→    } catch {
   455→      // fallback
   456→    }
   457→  }

   463→  async saveDefinition(definition: WorkflowDefinition): Promise<void> {
   464→    if (isAuthoritativeMode()) {
   465→      return PostgresWorkflowStore.getInstance().saveDefinition(definition);
   466→    }
   467→
   468→    const key = `${definition.id}@${definition.version}`;
   469→    this.definitions.set(key, definition);
   470→    this.definitions.set(definition.id, definition);
   471→
   472→    try {
   473→      DurableFileStore.getInstance().saveItem('workflow_definitions', key, definition);
   474→      DurableFileStore.getInstance().saveItem('workflow_definitions', definition.id, definition);
   475→    } catch {}
   476→
   477→    if (process.env.DATABASE_URL) {
   478→      try {
   479→        await prisma.workflowDefinition.upsert({
   480→          where: {
   481→            id_version: {
   482→              id: definition.id,
   483→              version: definition.version || '1.0.0',
   484→            },
   485→          },
   486→          create: {
   487→            id: definition.id,

   489→            name: definition.name,
   490→            description: definition.description,
   491→            steps: (definition.steps as any) ?? [],

   494→            name: definition.name,
   495→            description: definition.description,
   496→            steps: (definition.steps as any) ?? [],
   497→          },
   498→        });
   499→      } catch {
   500→        // Fallback safely for offline test environments
   501→      }
   502→    }
   503→  }

   505→  async getDefinition(id: string, version?: string): Promise<WorkflowDefinition | null> {
   506→    if (isAuthoritativeMode()) {
   507→      return PostgresWorkflowStore.getInstance().getDefinition(id, version);
   508→    }
   509→
   510→    const key = version ? `${id}@${version}` : id;
   511→    const cached = this.definitions.get(key);
   512→    if (cached) return cached;
   513→
   514→    if (process.env.DATABASE_URL) {
   515→      try {
   516→        const dbDef = await prisma.workflowDefinition.findFirst({
   517→          where: {
   518→            id,
   519→            ...(version ? { version } : {}),
   520→          },
   521→          orderBy: { updatedAt: 'desc' },
   522→        });
   523→        if (dbDef) {
   524→          const mapped: WorkflowDefinition = {
   525→            id: dbDef.id,
   526→            version: dbDef.version,
   527→            name: dbDef.name,
   528→            description: dbDef.description || '',
   529→            objective: '',
   530→            steps: (dbDef.steps as any) || [],
   531→            dependencies: [],
   532→            requiredApprovals: 0,
   533→            allowedRoles: [],
   534→            allowedSkills: [],
   535→            expectedOutputs: [],
   536→          };
   537→          this.definitions.set(key, mapped);
   538→          return mapped;
   539→        }
   540→      } catch {
   541→        // Fallback
   542→      }
   543→    }
   544→
   545→    return null;
   546→  }
   547→
   548→  async listDefinitions(): Promise<WorkflowDefinition[]> {
   549→    if (isAuthoritativeMode()) {
   550→      return PostgresWorkflowStore.getInstance().listDefinitions();
   551→    }
   552→
   553→    const result = new Map<string, WorkflowDefinition>();
   554→    for (const [key, def] of this.definitions.entries()) {
   555→      if (!key.includes('@')) {
   556→        result.set(def.id, def);
   557→      }
   558→    }
   559→    return Array.from(result.values());
   560→  }

   614→  async getInstance(instanceId: string): Promise<WorkflowInstanceState | null> {
   615→    if (isAuthoritativeMode()) {
   616→      return PostgresWorkflowStore.getInstance().getInstance(instanceId);
   617→    }
   618→
   619→    const instance = this.instances.get(instanceId);
   620→    if (instance) return JSON.parse(JSON.stringify(instance));
   621→
   622→    if (process.env.DATABASE_URL) {
   623→      try {
   624→        const dbInst = await prisma.workflowInstance.findUnique({
   625→          where: { id: instanceId },
   626→        });
   627→        if (dbInst) {
   628→          const mapped: WorkflowInstanceState = {
   629→            instanceId: dbInst.id,
   630→            workflowId: dbInst.definitionId,
   631→            version: dbInst.version,
   632→            stateVersion: dbInst.stateVersion ?? 0,
   633→            claimedBy: dbInst.claimedBy || undefined,
   634→            claimedAt: dbInst.claimedAt ? dbInst.claimedAt.toISOString() : undefined,
   635→            status: dbInst.status as any,
   636→            objective: dbInst.objective,
   637→            stepStates: (dbInst.stepStates as any) || {},
   638→            createdAt: dbInst.createdAt.toISOString(),
   639→            updatedAt: dbInst.updatedAt.toISOString(),
   640→            outputs: {},
   641→            evidenceReferences: [],
   642→            failureReason: dbInst.error || undefined,
   643→          };
   644→          this.instances.set(mapped.instanceId, mapped);
   645→          return mapped;
   646→        }
   647→      } catch {
   648→        // Fallback
   649→      }
   650→    }
   651→
   652→    return null;
   653→  }

   655→  async listInstances(status?: string): Promise<WorkflowInstanceState[]> {
   656→    if (isAuthoritativeMode()) {

   667→  async claimStepAtomic(
   668→    instanceId: string,
   669→    stepId: string,
   670→    workerId: string,
   671→    expectedVersion?: number
   672→  ): Promise<{ instance: WorkflowInstanceState; step: WorkflowStepState }> {
   673→    if (isAuthoritativeMode()) {
   674→      return PostgresWorkflowStore.getInstance().claimStepAtomic(instanceId, stepId, workerId, expectedVersion);
   675→    }
   676→
   677→    const instance = this.instances.get(instanceId);
   678→    if (!instance) {
   679→      throw new Error(`Workflow instance not found: ${instanceId}`);
   680→    }
   681→
   682→    const currentVersion = instance.stateVersion ?? 0;
   683→    if (expectedVersion !== undefined && currentVersion !== expectedVersion) {
   684→      throw new ConcurrencyConflictError(instanceId, expectedVersion, currentVersion);
   685→    }
   686→
   687→    const step = instance.stepStates[stepId];
   688→    if (!step) {
   689→      throw new Error(`Step not found in workflow instance: ${stepId}`);
   690→    }
   691→
   692→    if (step.status !== 'ready') {
   693→      throw new StepClaimError(instanceId, stepId, step.status, step.claimedBy);
   694→    }
   695→
   696→    if (step.claimedBy && step.claimedBy !== workerId) {
   697→      throw new StepClaimError(instanceId, stepId, step.status, step.claimedBy);
   698→    }
   699→
   700→    validateStepTransition(step.status, 'running');
   701→
   702→    const nowIso = new Date().toISOString();
   703→    step.status = 'running';
   704→    step.claimedBy = workerId;
   705→    step.claimedAt = nowIso;
   706→    if (!step.startedAt) {
   707→      step.startedAt = nowIso;
   708→    }
   709→
   710→    if (instance.status === 'pending' || instance.status === 'waiting') {
   711→      validateInstanceTransition(instance.status, 'running');
   712→      instance.status = 'running';
   713→    }
   714→
   715→    instance.stateVersion = currentVersion + 1;
   716→    instance.claimedBy = workerId;
   717→    instance.claimedAt = nowIso;
   718→    instance.updatedAt = nowIso;
   719→
   720→    const clone = JSON.parse(JSON.stringify(instance));
   721→    this.instances.set(instanceId, clone);
   722→
   723→    try {
   724→      DurableFileStore.getInstance().saveItem('workflow_instances', instanceId, clone);
   725→    } catch {}
   726→
   727→    return {
   728→      instance: clone,
   729→      step: JSON.parse(JSON.stringify(step)),
   730→    };
   731→  }

   733→  async transitionStepAtomic(
   734→    instanceId: string,
   735→    stepId: string,
   736→    targetStatus: WorkflowStepStatus,
   737→    expectedVersion?: number,
   738→    patch?: Partial<WorkflowStepState>,
   739→    instancePatch?: Partial<WorkflowInstanceState>
   740→  ): Promise<WorkflowInstanceState> {
   741→    if (isAuthoritativeMode()) {
   742→      return PostgresWorkflowStore.getInstance().transitionStepAtomic(
   743→        instanceId,
   744→        stepId,
   745→        targetStatus,
   746→        expectedVersion,
   747→        patch,
   748→        instancePatch
   749→      );
   750→    }
   751→
   752→    const instance = this.instances.get(instanceId);
   753→    if (!instance) {
   754→      throw new Error(`Workflow instance not found: ${instanceId}`);
   755→    }
   756→
   757→    const currentVersion = instance.stateVersion ?? 0;
   758→    if (expectedVersion !== undefined && currentVersion !== expectedVersion) {
   759→      throw new ConcurrencyConflictError(instanceId, expectedVersion, currentVersion);
   760→    }
   761→
   762→    const step = instance.stepStates[stepId];
   763→    if (!step) {
   764→      throw new Error(`Step not found in workflow instance: ${stepId}`);
   765→    }
   766→
   767→    validateStepTransition(step.status, targetStatus);
   768→
   769→    const nowIso = new Date().toISOString();
   770→    step.status = targetStatus;
   771→    if (targetStatus === 'completed' || targetStatus === 'failed' || targetStatus === 'cancelled') {
   772→      step.completedAt = nowIso;
   773→      step.claimedBy = undefined;
   774→      step.claimedAt = undefined;
   775→    }
   776→    if (targetStatus === 'ready') {
   777→      // Crash recovery / recurring reset: clear stale claim identity so any live worker
   778→      // can re-claim (mirrors PostgresWorkflowStore Phase 2.6 hardening).
   779→      step.claimedBy = undefined;
   780→      step.claimedAt = undefined;
   781→      step.completedAt = undefined;
   782→    }
   783→    if (patch) {
   784→      Object.assign(step, patch);
   785→    }
   786→
   787→    if (instancePatch?.status && instancePatch.status !== instance.status) {
   788→      validateInstanceTransition(instance.status, instancePatch.status);
   789→      instance.status = instancePatch.status;
   790→    }
   791→    if (instancePatch?.failureReason) {
   792→      instance.failureReason = instancePatch.failureReason;
   793→    }
   794→
   795→    instance.stateVersion = currentVersion + 1;
   796→    instance.updatedAt = nowIso;
   797→
   798→    const clone = JSON.parse(JSON.stringify(instance));
   799→    this.instances.set(instanceId, clone);
   800→
   801→    try {
   802→      DurableFileStore.getInstance().saveItem('workflow_instances', instanceId, clone);
   803→    } catch {}
   804→
   805→    return clone;
   806→  }

   808→  async rehydrateActiveInstances(): Promise<WorkflowInstanceState[]> {
   809→    if (isAuthoritativeMode()) {
   810→      return PostgresWorkflowStore.getInstance().rehydrateActiveInstances();
   811→    }
   812→
   813→    const activeStatuses: Set<string> = new Set(['pending', 'running', 'waiting', 'blocked', 'awaiting_approval']);
   814→    const active = Array.from(this.instances.values())
   815→      .filter((i) => activeStatuses.has(i.status))
   816→      .sort((a, b) => new Date(a.updatedAt).getTime() - new Date(b.updatedAt).getTime());
   817→
   818→    return JSON.parse(JSON.stringify(active));
   819→  }

   822→    this.definitions.clear();
   823→    this.instances.clear();
   824→    try {
   825→      DurableFileStore.getInstance().clearCollection('workflow_definitions');
   826→      DurableFileStore.getInstance().clearCollection('workflow_instances');
   827→    } catch {}
   828→  }
