    33→/**
    34→ * Normalizes and cleans the COMPOSIO_API_KEY.
    35→ * Handles prefix wrappers, accidental export strings, quotes, and missing leading prefix.
    36→ * Strictly never logs or leaks the key.
    37→ */
    38→export function cleanComposioApiKey(rawKey?: string): string | undefined {
    39→  if (!rawKey || typeof rawKey !== 'string') return undefined;
    40→  let cleaned = rawKey.trim();
    41→  if (cleaned.startsWith('export COMPOSIO_API_KEY=')) {
    42→    cleaned = cleaned.replace(/^export COMPOSIO_API_KEY=/, '').trim();
    43→  }
    44→  if (cleaned.startsWith('COMPOSIO_API_KEY=')) {
    45→    cleaned = cleaned.replace(/^COMPOSIO_API_KEY=/, '').trim();
    46→  }
    47→  if ((cleaned.startsWith('"') && cleaned.endsWith('"')) || (cleaned.startsWith("'") && cleaned.endsWith("'"))) {
    48→    cleaned = cleaned.slice(1, -1).trim();
    49→  }
    50→  if (cleaned.startsWith('k_') && cleaned.length > 15) {
    51→    cleaned = 'a' + cleaned;
    52→  }
    53→  return cleaned.length > 0 ? cleaned : undefined;
    54→}

    56→/**
    57→ * Evaluates whether the Composio integration is configured with an API key.
    58→ * Strictly non-throwing: returns explicit configuration state.
    59→ */
    60→export function getComposioConfigStatus(): ProviderConfigStatus {
    61→  const apiKey = cleanComposioApiKey(process.env.COMPOSIO_API_KEY);
    62→  if (!apiKey) {
    63→    return 'unconfigured';
    64→  }
    65→  return 'configured';
    66→}

    68→/**
    69→ * Lazy client accessor.
    70→ * Never attempts instantiation if COMPOSIO_API_KEY is missing or empty.
    71→ */
    72→let cachedComposioClient: Composio | null = null;
    73→let cachedApiKey: string | null = null;
    74→
    75→export async function getComposioClientAsync(): Promise<Composio | null> {
    76→  const status = getComposioConfigStatus();
    77→  if (status !== 'configured') {
    78→    return null;
    79→  }
    80→
    81→  const cleanedKey = cleanComposioApiKey(process.env.COMPOSIO_API_KEY)!;
    82→  if (!cachedComposioClient || cachedApiKey !== cleanedKey) {
    83→    try {
    84→      // Dynamic import: safely load optional package without triggering bundler resolution errors
    85→      const dynamicImport = new Function('pkg', 'return import(pkg)');
    86→      const mod: any = await dynamicImport('@composio/core').catch(() => null);
    87→      if (!mod || typeof mod.Composio !== 'function') {
    88→        console.error('[ComposioProvider] @composio/core is not installed in this deployment. External tool execution stays disabled.');
    89→        return null;
    90→      }
    91→      cachedComposioClient = new mod.Composio({
    92→        apiKey: cleanedKey,
    93→      }) as Composio;
    94→      cachedApiKey = cleanedKey;
    95→    } catch (error) {
    96→      console.error('[ComposioProvider] Initialization error:', error instanceof Error ? error.message : 'Unknown error');
    97→      return null;
    98→    }
    99→  }
   100→
   101→  return cachedComposioClient;
   102→}

   108→export function getComposioClient(): Composio | null {

   111→    return null;

   183→/**
   184→ * Sanitizes untrusted external text (descriptions, issue bodies, readmes).
   185→ * Quarantines potential prompt injection instructions.
   186→ */
   187→export function sanitizeUntrustedExternalText(text?: string | null): string {
   188→  if (!text || typeof text !== 'string') return '';
   189→  return text
   190→    .replace(/<script[\s\S]*?>[\s\S]*?<\/script>/gi, '')
   191→    .replace(
   192→      /(?:system\s*prompt\s*override|ignore\s*all\s*previous\s*instructions|ignore\s*previous\s*instructions|system\s*prompt|system\s*message|system\s*:|assistant\s*:|instruction\s*:|human\s*:|override\s*permission|grant\s*permission|grant\s*admin|elevate\s*role|approve\s*governance|approve\s*decision|alter\s*policy|bypass\s*sandbox|bypass\s*guardrails)/gi,
   193→      ' [quarantined external text] '
   194→    )
   195→    .trim();
   196→}
   197→
   198→/**
   199→ * Validates GitHub repository read input.
   200→ * Prevents arbitrary injection and malformed inputs.
   201→ */
   202→export function validateGitHubInput(input: any): { valid: boolean; error?: string; owner?: string; repo?: string } {
   203→  if (!input || typeof input !== 'object') {
   204→    return { valid: false, error: 'Input must be an object containing "owner" and "repo".' };
   205→  }
   206→  const owner = typeof input.owner === 'string' ? input.owner.trim() : '';
   207→  const repo = typeof input.repo === 'string' ? input.repo.trim() : '';
   208→
   209→  if (!owner || !repo) {
   210→    return { valid: false, error: 'Both "owner" and "repo" parameters are required.' };
   211→  }
   212→
   213→  // Enforce standard alphanumeric and dash/underscore/dot GitHub identifier format
   214→  if (!/^[a-zA-Z0-9_.-]+$/.test(owner) || !/^[a-zA-Z0-9_.-]+$/.test(repo)) {
   215→    return { valid: false, error: 'Invalid characters in repository owner or repository name.' };
   216→  }
   217→
   218→  return { valid: true, owner, repo };
   219→}

   225→export interface ResolvedConnection {
   226→  connectedAccountId: string;
   227→  userId: string;
   228→}

   230→/**
   231→ * Resolves an active connected account and its corresponding user_id/entity_id in Composio.
   232→ * Never leaks access tokens or auth secrets.
   233→ */
   234→export async function resolveActiveConnectedAccount(
   235→  client: Composio,
   236→  toolkit: string,
   237→  explicitAccountId?: string
   238→): Promise<{ resolved?: ResolvedConnection; error?: string }> {
   239→  try {
   240→    const listRes = await client.connectedAccounts.list();
   241→    const items = (listRes.items || []) as any[];
   242→
   243→    // Match explicit ID if provided, otherwise find first ACTIVE account for toolkit
   244→    let target = explicitAccountId 
   245→      ? items.find(acc => acc.id === explicitAccountId)
   246→      : items.find(acc => acc.toolkit?.slug === toolkit && acc.status === 'ACTIVE');
   247→
   248→    if (!target && !explicitAccountId) {
   249→      // Fallback: check any ACTIVE account
   250→      target = items.find(acc => acc.status === 'ACTIVE');
   251→    }
   252→
   253→    if (!target) {
   254→      return { error: `No active connected account found for toolkit "${toolkit}".` };
   255→    }
   256→
   257→    // Retrieve full connected account details to obtain user_id / entity_id
   258→    const fullAccount = await (client as any).client?.connectedAccounts?.retrieve?.(target.id) as any;
   259→    const userId = fullAccount?.user_id || target.userId || (target as any).user_id || 'default';
   260→
   261→    return {
   262→      resolved: {
   263→        connectedAccountId: target.id,
   264→        userId,
   265→      },
   266→    };
   267→  } catch (err: any) {
   268→    return { error: `Failed to resolve connected account: ${err.message || 'Unknown connection error'}` };
   269→  }
   270→}

   301→// ----------------------------------------------------------------------------
   302→// 6. PERMISSION BOUNDARY ENFORCEMENT
   303→// ----------------------------------------------------------------------------
   304→
   305→/**
   306→ * Validates that the requested tool is explicitly permitted by SamJuniors.
   307→ * Composio availability NEVER implies SamJuniors authorization.
   308→ */
   309→export function validateComposioPermission(
   310→  toolId: ToolId,
   311→  sessionScope: ComposioSessionScope
   312→): { authorized: boolean; reason?: string } {
   313→  if (!sessionScope.permittedToolIds.includes(toolId)) {
   314→    return {
   315→      authorized: false,
   316→      reason: `Tool "${toolId}" is not in the permitted tools list for this session scope. Composio cannot bypass SamJuniors permissions.`,
   317→    };
   318→  }
   319→
   320→  // Advisor role cannot execute mutation or non-research external actions
   321→  if (sessionScope.employeeRole === 'advisor') {
   322→    return {
   323→      authorized: false,
   324→      reason: `Advisor role is strictly advisory and cannot execute external tools.`,
   325→    };
   326→  }
   327→
   328→  return { authorized: true };
   329→}

   331→// ----------------------------------------------------------------------------
   332→// 7. EXTERNAL TOOL PROVIDER IMPLEMENTATION
   333→// ----------------------------------------------------------------------------
   334→
   335→export class ComposioProvider implements ExternalToolProvider {
   336→  public readonly providerId = 'composio';
   337→  public readonly name = 'Composio External Tool Infrastructure';
   338→
   339→  public getStatus(): ProviderConfigStatus {
   340→    return getComposioConfigStatus();
   341→  }
   342→
   343→  public isAvailable(): boolean {
   344→    return this.getStatus() === 'configured';
   345→  }
   346→
   347→  public async createSession(scope: ComposioSessionScope): Promise<ComposioSession> {
   348→    return createComposioSession(scope);
   349→  }
   350→
   351→  /**
   352→   * Executes external tool through Composio with strict SamJuniors governance.
   353→   */
   354→  public async executeTool(request: ExternalExecutionRequest): Promise<ToolExecutionEvidence> {
   355→    const timestamp = new Date().toISOString();
   356→
   357→    // 1. Enforce SamJuniors Permission Boundary First
   358→    const authCheck = validateComposioPermission(request.toolId, request.sessionScope);
   359→    if (!authCheck.authorized) {
   360→      return {
   361→        toolId: request.toolId,
   362→        toolName: `Composio:${request.toolId}`,
   363→        status: 'denied',
   364→        timestamp,
   365→        inputSummary: `Execution attempted for ${request.toolId}`,
   366→        outputSummary: 'Permission denied: Tool not permitted in current session scope.',
   367→        provenance: request.provenance,
   368→        verificationState: 'unverified',
   369→        executionSafetyState: 'verified_safe',
   370→        errorMessage: authCheck.reason,
   371→        limitations: ['Execution blocked by SamJuniors authorization boundary.'],
   372→      };
   373→    }
   374→
   375→    // 2. Check Configuration Status
   376→    const status = this.getStatus();
   377→    if (status !== 'configured') {
   378→      return {
   379→        toolId: request.toolId,
   380→        toolName: `Composio:${request.toolId}`,
   381→        status: 'not_executed',
   382→        timestamp,
   383→        inputSummary: `Execution requested for ${request.toolId}`,
   384→        outputSummary: 'Composio provider is unconfigured (COMPOSIO_API_KEY is absent). No external request attempted.',
   385→        provenance: request.provenance,
   386→        verificationState: 'unverified',
   387→        executionSafetyState: 'verified_safe',
   388→        limitations: [
   389→          'COMPOSIO_API_KEY is not configured in the environment.',
   390→          'No external network call was made.',
   391→          'No synthetic execution evidence generated.',
   392→        ],
   393→      };
   394→    }
   395→
   396→    // 3. Verify Mapping Exists in Allowlist
   397→    const mapping = getComposioToolMapping(request.toolId);
   398→    if (!mapping) {
   399→      return {
   400→        toolId: request.toolId,
   401→        toolName: `Composio:${request.toolId}`,
   402→        status: 'not_executed',
   403→        timestamp,
   404→        inputSummary: `Execution requested for unmapped tool: ${request.toolId}`,
   405→        outputSummary: 'No Composio action mapping registered for this tool.',
   406→        provenance: request.provenance,
   407→        verificationState: 'unverified',
   408→        executionSafetyState: 'verified_safe',
   409→        errorMessage: `Unmapped toolId: ${request.toolId}`,
   410→        limitations: ['Tool has no registered server-side mapping.'],
   411→      };
   412→    }
   413→
   414→    // Enforce Read-Only Allowlist Guard
   415→    if (!ALLOWED_COMPOSIO_READONLY_ACTIONS.has(mapping.action)) {
   416→      return {
   417→        toolId: request.toolId,
   418→        toolName: `${mapping.toolkit}:${mapping.action}`,
   419→        status: 'denied',
   420→        timestamp,
   421→        inputSummary: `Action: ${mapping.action}`,
   422→        outputSummary: 'Execution denied: Action is not in the read-only allowlist.',
   423→        provenance: request.provenance,
   424→        verificationState: 'unverified',
   425→        executionSafetyState: 'verified_safe',
   426→        errorMessage: `Action ${mapping.action} is forbidden in read-only phase.`,
   427→        limitations: ['Only allowlisted read-only actions may be executed.'],
   428→      };
   429→    }
   430→
   431→    // 4. Validate Input
   432→    if (mapping.toolkit === 'github') {
   433→      const validation = validateGitHubInput(request.input);
   434→      if (!validation.valid) {
   435→        return {
   436→          toolId: request.toolId,
   437→          toolName: `${mapping.toolkit}:${mapping.action}`,
   438→          status: 'failed',
   439→          timestamp,
   440→          inputSummary: `Input validation failed for ${request.toolId}`,
   441→          outputSummary: `Validation error: ${validation.error}`,
   442→          provenance: request.provenance,
   443→          verificationState: 'unverified',
   444→          executionSafetyState: 'verified_safe',
   445→          errorMessage: validation.error,
   446→          limitations: ['Input rejected before reaching external provider.'],
   447→        };
   448→      }
   449→    }
   450→
   451→    // 5. Retrieve Client
   452→    const client = await getComposioClientAsync();
   453→    if (!client) {
   454→      return {
   455→        toolId: request.toolId,
   456→        toolName: `${mapping.toolkit}:${mapping.action}`,
   457→        status: 'failed',
   458→        timestamp,
   459→        inputSummary: `Client retrieval failed for ${mapping.action}`,
   460→        outputSummary: 'Failed to initialize Composio client.',
   461→        provenance: request.provenance,
   462→        verificationState: 'unverified',
   463→        executionSafetyState: 'verified_safe',
   464→        errorMessage: 'Composio client initialization failed.',
   465→      };
   466→    }
   467→
   468→    // 6. Resolve Active Connected Account
   469→    const connectionResult = await resolveActiveConnectedAccount(
   470→      client,
   471→      mapping.toolkit,
   472→      request.sessionScope.connectedAccountId
   473→    );
   474→
   475→    if (connectionResult.error || !connectionResult.resolved) {
   476→      return {
   477→        toolId: request.toolId,
   478→        toolName: `${mapping.toolkit}:${mapping.action}`,
   479→        status: 'failed',
   480→        timestamp,
   481→        inputSummary: `Connecting to ${mapping.toolkit} for ${mapping.action}`,
   482→        outputSummary: connectionResult.error || 'No active connection available.',
   483→        provenance: request.provenance,
   484→        verificationState: 'unverified',
   485→        executionSafetyState: 'verified_safe',
   486→        errorMessage: connectionResult.error || 'Connection unavailable',
   487→        limitations: [
   488→          `No active connected account found for ${mapping.toolkit} in Composio.`,
   489→          'Connect the account in the Composio dashboard to enable live execution.',
   490→        ],
   491→      };
   492→    }
   493→
   494→    const { connectedAccountId, userId } = connectionResult.resolved;
   495→
   496→    // 7. Perform Real Execution via Composio
   497→    try {
   498→      const execResult = await client.tools.execute(mapping.action as any, {
   499→        userId,
   500→        connectedAccountId,
   501→        arguments: request.input,
   502→        dangerouslySkipVersionCheck: true,
   503→      });
   504→
   505→      if (!execResult.successful) {
   506→        const rawData = execResult.data as any;
   507→        const errorText = typeof execResult.error === 'string' ? execResult.error : JSON.stringify(execResult.error || '');
   508→        const is404 = rawData?.status_code === 404 || 
   509→                      rawData?.http_error?.includes('404') || 
   510→                      errorText.includes('Not Found') || 
   511→                      errorText.includes('404');
   512→
   513→        const outputSummary = is404
   514→          ? `Repository "${request.input?.owner}/${request.input?.repo}" not found on GitHub.`
   515→          : (execResult.error || 'External action returned unsuccessful result.');
   516→
   517→        const errorMessage = is404
   518→          ? `Repository "${request.input?.owner}/${request.input?.repo}" not found.`
   519→          : (execResult.error || 'Composio tool execution reported failure.');
   520→
   521→        return {
   522→          toolId: request.toolId,
   523→          toolName: `${mapping.toolkit}:${mapping.action}`,
   524→          status: 'failed',
   525→          timestamp,
   526→          inputSummary: `Repository: ${request.input?.owner}/${request.input?.repo}`,
   527→          outputSummary,
   528→          provenance: request.provenance,
   529→          verificationState: 'unverified',
   530→          executionSafetyState: 'verified_safe',
   531→          errorMessage,
   532→          limitations: [
   533→            is404 
   534→              ? 'Resource does not exist on GitHub or account lacks access.' 
   535→              : 'External provider reported an error during execution.'
   536→          ],
   537→        };
   538→      }
   539→
   540→      // 8. Normalize Result into ToolExecutionEvidence
   541→      const rawData = execResult.data as any;
   542→
   543→      // Handle not-found or API error response payloads returned with 200/success wrapper
   544→      if (rawData?.message === 'Not Found' || rawData?.response_data?.message === 'Not Found') {
   545→        return {
   546→          toolId: request.toolId,
   547→          toolName: `${mapping.toolkit}:${mapping.action}`,
   548→          status: 'failed',
   549→          timestamp,
   550→          inputSummary: `Repository: ${request.input.owner}/${request.input.repo}`,
   551→          outputSummary: `Repository "${request.input.owner}/${request.input.repo}" not found on GitHub.`,
   552→          provenance: request.provenance,
   553→          verificationState: 'unverified',
   554→          executionSafetyState: 'verified_safe',
   555→          errorMessage: 'Repository not found.',
   556→          limitations: ['Resource does not exist on GitHub or account lacks access.'],
   557→        };
   558→      }
   559→
   560→      if (mapping.action === 'GITHUB_GET_A_REPOSITORY') {
   561→        const repoData = rawData?.response_data || rawData;
   562→        const name = repoData.name || request.input.repo;
   563→        const fullName = repoData.full_name || `${request.input.owner}/${request.input.repo}`;
   564→        const description = sanitizeUntrustedExternalText(repoData.description || 'No description provided.');
   565→        const htmlUrl = repoData.html_url || `https://github.com/${fullName}`;
   566→        const stars = repoData.stargazers_count ?? 0;
   567→        const forks = repoData.forks_count ?? 0;
   568→        const openIssues = repoData.open_issues_count ?? 0;
   569→        const defaultBranch = repoData.default_branch || 'main';
   570→        const visibility = repoData.visibility || (repoData.private ? 'private' : 'public');
   571→
   572→        const inputSummary = `Repository: ${fullName}`;
   573→        const outputSummary = `Retrieved GitHub repository "${fullName}": ${description} (Stars: ${stars}, Forks: ${forks}, Open Issues: ${openIssues}, Branch: ${defaultBranch}, Visibility: ${visibility})`;
   574→
   575→        return {
   576→          toolId: request.toolId,
   577→          toolName: 'GitHub Repository Research',
   578→          status: 'success',
