    77→/**
    78→ * Central Side-Effect Authorization Gate.
    79→ * Enforces company safety, Founder approval boundaries, audit trail generation,
    80→ * and single server-side authorization enforcement before executing side effects.
    81→ */
    82→export class SideEffectAuthorizationGate {
    83→  private static instance: SideEffectAuthorizationGate;
    84→  private approvalStore: InMemoryApprovalStore;
    85→  private auditStore: InMemoryAuditStore;
    86→  private evaluator: SideEffectPolicyEvaluator;
    87→
    88→  constructor(
    89→    approvalStore?: InMemoryApprovalStore,
    90→    auditStore?: InMemoryAuditStore,
    91→    evaluator?: SideEffectPolicyEvaluator
    92→  ) {
    93→    this.approvalStore = approvalStore || InMemoryApprovalStore.getInstance();
    94→    this.auditStore = auditStore || InMemoryAuditStore.getInstance();
    95→    this.evaluator = evaluator || new SideEffectPolicyEvaluator(this.approvalStore);
    96→  }
    97→
    98→  public static getInstance(): SideEffectAuthorizationGate {
    99→    if (!SideEffectAuthorizationGate.instance) {
   100→      SideEffectAuthorizationGate.instance = new SideEffectAuthorizationGate();
   101→    }
   102→    return SideEffectAuthorizationGate.instance;
   103→  }
   104→
   105→  /**
   106→   * Evaluate whether an intended action is allowed, requires approval, or is denied.
   107→   */
   108→  public async evaluateAuthorization(
   109→    request: AuthorizationEvaluationRequest
   110→  ): Promise<AuthorizationDecision> {
   111→    return this.evaluator.evaluate(request);
   112→  }
   113→
   114→  /**
   115→   * Request a new Founder Approval for an intended side effect.
   116→   * When params.scope carries an occurrenceId (scheduled per-occurrence
   117→   * approval, Phase 4.4B.1), the dedupe lookup is occurrence-bound: a prior
   118→   * occurrence's approval never satisfies this occurrence's request.
   119→   */
   120→  public async requestApproval(params: RequestApprovalParams): Promise<FounderApprovalRecord> {
   121→    const existing = await this.approvalStore.findActiveMatching({
   122→      workflowInstanceId: params.workflowInstanceId,
   123→      stepId: params.stepId,
   124→      employeeRole: params.employeeRole,
   125→      actionName: params.actionName,
   126→      classification: params.classification,
   127→      occurrenceId: params.scope?.occurrenceId,
   128→    });
   129→
   130→    // If there is already a pending or approved approval, return it
   131→    if (existing && (existing.decision === 'pending' || existing.decision === 'approved')) {
   132→      return existing;
   133→    }
   134→
   135→    const id = `appr-${uuidv4()}`;
   136→    const now = new Date().toISOString();
   137→
   138→    const scope: ApprovalScope = {
   139→      scopeType: params.scope?.scopeType || 'step',
   140→      workflowInstanceId: params.workflowInstanceId,
   141→      stepId: params.stepId,
   142→      campaignId: params.scope?.campaignId,
   143→      targetSystem: params.target?.targetSystem || params.scope?.targetSystem,
   144→      operationPattern: params.scope?.operationPattern,
   145→      occurrenceId: params.scope?.occurrenceId,
   146→      maxUses: params.scope?.maxUses || 1,
   147→      usedCount: 0,
   148→    };
   149→
   150→    const payloadHash = computeApprovalPayloadHash(
   151→      params.actionName,
   152→      params.target,
   153→      params.payload
   154→    );
   155→
   156→    const record: FounderApprovalRecord = {
   157→      id,
   158→      decision: 'pending',
   159→      actionName: params.actionName,
   160→      classification: params.classification,
   161→      workflowInstanceId: params.workflowInstanceId,
   162→      stepId: params.stepId,
   163→      employeeRole: params.employeeRole,
   164→      scope,
   165→      requestedAt: now,
   166→      notes: params.notes,
   167→      target: params.target,
   168→      payload: params.payload,
   169→      payloadHash,
   170→      isConsumed: false,
   171→    };
   172→
   173→    return this.approvalStore.save(record);
   174→  }
   175→
   176→  /**
   177→   * Founder Decision Handler (Approve / Reject).
   178→   * Strictly enforces that ONLY the Founder can approve or reject requests.
   179→   * AI Employees (researcher, pm, coo, finance, advisor) cannot self-approve or approve others.
   180→   */
   181→  public async decideApproval(params: DecideApprovalParams): Promise<FounderApprovalRecord> {
   182→    const ALLOWED_FOUNDER_IDENTITIES = [
   183→      'founder',
   184→      'founder-001',
   185→      'founder-local-session',
   186→      'executive_founder',
   187→      'executive-founder',
   188→    ];
   189→    const actor = params.decidedBy?.toLowerCase()?.trim() || '';
   190→    const isAllowedFounder =
   191→      ALLOWED_FOUNDER_IDENTITIES.includes(actor) ||
   192→      (actor.startsWith('founder-') && !actor.includes('agent')) ||
   193→      params.userContext?.role === 'FOUNDER';
   194→
   195→    if (!isAllowedFounder) {
   196→      throw new Error(
   197→        `Permission denied: Identity "${params.decidedBy}" is not an authorized Founder. Only verified Founder identities can approve or reject side-effect requests.`
   198→      );
   199→    }
   200→
   201→    return this.approvalStore.decide(
   202→      params.approvalId,
   203→      params.decision,
   204→      params.decidedBy,
   205→      params.reason,
   206→      params.expiresAt
   207→    );
   208→  }
   209→
   210→  /**
   211→   * Founder Revocation Handler.
   212→   * Immediately revokes a previously granted approval.
   213→   */
   214→  public async revokeApproval(params: RevokeApprovalParams): Promise<FounderApprovalRecord> {
   215→    const prohibitedRoles = ['researcher', 'pm', 'coo', 'finance', 'advisor', 'agent', 'system', 'developer', 'guest'];
   216→    const actor = params.revokedBy?.toLowerCase()?.trim() || '';
   217→    if (!actor || prohibitedRoles.includes(actor)) {
   218→      throw new Error(
   219→        `Permission denied: Only the Founder can revoke approvals. Attempted by "${params.revokedBy}".`
   220→      );
   221→    }
   222→
   223→    return this.approvalStore.revoke(params.approvalId, params.revokedBy, params.reason);
   224→  }
   225→
   226→  /**
   227→   * Execute an operation behind the centralized Side-Effect Authorization Gate.
   228→   * 1. Evaluates policy.
   229→   * 2. If approval is required or denied, refuses execution and returns the decision.
   230→   * 3. If allowed, verifies cryptographic payload binding and consumes single-use approval (if applicable).
   231→   * 4. Executes executeFn.
   232→   * 5. Emits a deterministic, unforgeable audit record.
   233→   */
   234→  public async executeWithGate<T>(params: ExecuteWithGateParams<T>): Promise<ExecuteWithGateResult<T>> {
   235→    const { request, executeFn, executionRef } = params;
   236→    const auditId = `audit-${uuidv4()}`;
   237→    const timestamp = new Date().toISOString();
   238→    const requestId = `req-${uuidv4()}`;
   239→
   240→    // 0. Pre-Execution Idempotency Check (completed replay, in-progress conflict, ambiguous recovery, payload tampering)
   241→    const shouldEnforceIdempotency = Boolean(
   242→      params.idempotency ||
   243→      request.classification === 'external_communication' ||
   244→      request.classification === 'financial_action'
   245→    );
   246→
   247→    let idempotencyKey: string | undefined;
   248→    let payloadHash: string | undefined;
   249→
   250→    if (shouldEnforceIdempotency) {
   251→      idempotencyKey = params.idempotency?.key || generateLogicalIdempotencyKey({
   252→        actionName: request.actionName,
   253→        targetSystem: params.idempotency?.targetSystem || request.target?.targetSystem || 'default',
   254→        logicalOpId: params.idempotency?.logicalOpId || (request.workflowContext?.workflowInstanceId ? `${request.workflowContext.workflowInstanceId}-${request.workflowContext.stepId}` : (executionRef || requestId)),
   255→      });
   256→
   257→      const effectivePayload = params.idempotency?.payload !== undefined ? params.idempotency.payload : request.payload;
   258→      payloadHash = computeApprovalPayloadHash(request.actionName, request.target, effectivePayload ?? {});
   259→
   260→      const existing = await getIdempotencyStore().get(idempotencyKey);
   261→      if (existing) {
   262→        if (payloadHash && existing.payloadHash && existing.payloadHash !== payloadHash) {
   263→          const mismatchErr = new IdempotencyPayloadMismatchError(
   264→            idempotencyKey,
   265→            existing.payloadHash,
   266→            payloadHash
   267→          );
   268→          const auditRecord: SideEffectAuditRecord = {
   269→            id: auditId,
   270→            timestamp,
   271→            requestId,
   272→            employeeRole: request.employeeRole,
   273→            requestedBy: request.requestedBy || request.employeeRole,
   274→            skillId: typeof request.skillId === 'string' ? request.skillId : undefined,
   275→            workflowInstanceId: request.workflowContext?.workflowInstanceId,
   276→            stepId: request.workflowContext?.stepId,
   277→            actionClassification: request.classification,
   278→            actionName: request.actionName,
   279→            target: request.target,
   280→            decision: 'denied',
   281→            reasonCode: 'IDEMPOTENCY_PAYLOAD_MISMATCH',
   282→            reason: mismatchErr.message,
   283→            approvalId: request.approvalId,
   284→            executionReference: executionRef,
   285→            executed: false,
   286→          };
   287→          await this.auditStore.record(auditRecord);
   288→          throw mismatchErr;
   289→        }
   290→
   291→        if (existing.status === 'completed') {
   292→          const auditRecord: SideEffectAuditRecord = {
   293→            id: auditId,
   294→            timestamp,
   295→            requestId,
   296→            employeeRole: request.employeeRole,
   297→            requestedBy: request.requestedBy || request.employeeRole,
   298→            skillId: typeof request.skillId === 'string' ? request.skillId : undefined,
   299→            workflowInstanceId: request.workflowContext?.workflowInstanceId,
   300→            stepId: request.workflowContext?.stepId,
   301→            actionClassification: request.classification,
   302→            actionName: request.actionName,
   303→            target: request.target,
   304→            decision: 'allowed',
   305→            reasonCode: 'IDEMPOTENT_REPLAY',
   306→            reason: `Operation already executed successfully with idempotency key ${idempotencyKey}. Replaying cached response without re-invoking side effect.`,
   307→            approvalId: request.approvalId,
   308→            executionReference: existing.executionRef || executionRef,
   309→            executed: false,
   310→          };
   311→          await this.auditStore.record(auditRecord);
   312→
   313→          return {
   314→            allowed: true,
   315→            executed: false,
   316→            decision: {
   317→              effect: 'allowed',
   318→              reasonCode: 'IDEMPOTENT_REPLAY',
   319→              reason: `Operation already executed successfully with idempotency key ${idempotencyKey}. Replaying cached response without re-invoking side effect.`,
   320→              approvalId: request.approvalId,
   321→              evaluatedAt: timestamp,
   322→              evaluator: 'central_side_effect_gate',
   323→            },
   324→            result: existing.response as T,
   325→            auditId,
   326→            idempotencyKey,
   327→            idempotentReplay: true,
   328→          };
   329→        }
   330→
   331→        if (existing.status === 'in_progress') {
   332→          const inProgressErr = new OperationInProgressError(idempotencyKey, existing.executionRef || undefined, existing.createdAt);
   333→          const auditRecord: SideEffectAuditRecord = {
   334→            id: auditId,
   335→            timestamp,
   336→            requestId,
   337→            employeeRole: request.employeeRole,
   338→            requestedBy: request.requestedBy || request.employeeRole,
   339→            skillId: typeof request.skillId === 'string' ? request.skillId : undefined,
   340→            workflowInstanceId: request.workflowContext?.workflowInstanceId,
   341→            stepId: request.workflowContext?.stepId,
   342→            actionClassification: request.classification,
   343→            actionName: request.actionName,
   344→            target: request.target,
   345→            decision: 'denied',
   346→            reasonCode: 'OPERATION_IN_PROGRESS',
   347→            reason: inProgressErr.message,
   348→            approvalId: request.approvalId,
   349→            executionReference: executionRef,
   350→            executed: false,
   351→          };
   352→          await this.auditStore.record(auditRecord);
   353→          throw inProgressErr;
   354→        }
   355→
   356→        if (existing.status === 'unknown') {
   357→          const unknownErr = new UnknownExternalResultError(
   358→            idempotencyKey,
   359→            existing.executionRef || undefined,
   360→            'Operation has ambiguous prior execution status (crash/timeout recovery state). Manual or reconciliatory inspection required before retrying.'
   361→          );
   362→          const auditRecord: SideEffectAuditRecord = {
   363→            id: auditId,
   364→            timestamp,
   365→            requestId,
   366→            employeeRole: request.employeeRole,
   367→            requestedBy: request.requestedBy || request.employeeRole,
   368→            skillId: typeof request.skillId === 'string' ? request.skillId : undefined,
   369→            workflowInstanceId: request.workflowContext?.workflowInstanceId,
   370→            stepId: request.workflowContext?.stepId,
   371→            actionClassification: request.classification,
   372→            actionName: request.actionName,
   373→            target: request.target,
   374→            decision: 'denied',
   375→            reasonCode: 'UNKNOWN_EXTERNAL_RESULT',
   376→            reason: unknownErr.message,
   377→            approvalId: request.approvalId,
   378→            executionReference: executionRef,
   379→            executed: false,
   380→          };
   381→          await this.auditStore.record(auditRecord);
   382→          throw unknownErr;
   383→        }
   384→      }
   385→    }
   386→
   387→    // 1. Evaluate authorization
   388→    const decision = await this.evaluateAuthorization(request);
   389→
   390→    // 2. If not allowed, record audit and return blocked result
   391→    if (decision.effect !== 'allowed') {
   392→      const auditRecord: SideEffectAuditRecord = {
   393→        id: auditId,
   394→        timestamp,
   395→        requestId,
   396→        employeeRole: request.employeeRole,
   397→        requestedBy: request.requestedBy || request.employeeRole,
   398→        skillId: typeof request.skillId === 'string' ? request.skillId : undefined,
   399→        workflowInstanceId: request.workflowContext?.workflowInstanceId,
   400→        stepId: request.workflowContext?.stepId,
   401→        actionClassification: request.classification,
   402→        actionName: request.actionName,
   403→        target: request.target,
   404→        decision: decision.effect,
   405→        reasonCode: decision.reasonCode,
   406→        reason: decision.reason,
   407→        approvalId: decision.approvalId,
   408→        executionReference: undefined,
   409→        executed: false,
   410→      };
   411→
   412→      await this.auditStore.record(auditRecord);
   413→
   414→      return {
   415→        allowed: false,
   416→        executed: false,
   417→        decision,
   418→        auditId,
   419→        error: decision.reason,
   420→      };
   421→    }
   422→
   423→    // 3. If allowed, verify cryptographic approval-payload binding
   424→    let approval: FounderApprovalRecord | null = null;
   425→    if (decision.approvalId) {
   426→      approval = await this.approvalStore.get(decision.approvalId);
   427→      if (!approval) {
   428→        const missingApprovalReason = `Approval record (${decision.approvalId}) not found in store. Execution blocked.`;
   429→        const auditRecord: SideEffectAuditRecord = {
   430→          id: auditId,
   431→          timestamp,
   432→          requestId,
   433→          employeeRole: request.employeeRole,
   434→          requestedBy: request.requestedBy || request.employeeRole,
   435→          skillId: typeof request.skillId === 'string' ? request.skillId : undefined,
   436→          workflowInstanceId: request.workflowContext?.workflowInstanceId,
   437→          stepId: request.workflowContext?.stepId,
   438→          actionClassification: request.classification,
   439→          actionName: request.actionName,
   440→          target: request.target,
   441→          decision: 'denied',
   442→          reasonCode: 'APPROVAL_NOT_FOUND',
   443→          reason: missingApprovalReason,
   444→          approvalId: decision.approvalId,
   445→          executionReference: executionRef,
   446→          executed: false,
   447→        };
   448→
   449→        await this.auditStore.record(auditRecord);
   450→
   451→        return {
   452→          allowed: false,
   453→          executed: false,
   454→          decision: {
   455→            effect: 'denied',
   456→            reasonCode: 'APPROVAL_NOT_FOUND',
   457→            reason: missingApprovalReason,
   458→            approvalId: decision.approvalId,
   459→            evaluatedAt: timestamp,
   460→            evaluator: 'central_side_effect_gate',
   461→          },
   462→          auditId,
   463→          error: missingApprovalReason,
   464→        };
   465→      }
   466→
   467→      // Enforce mandatory cryptographic payload binding:
   468→      // Any approval-backed consequential execution MUST have a payload hash.
   469→      if (!approval.payloadHash) {
   470→        const missingHashReason = `Cryptographic payload binding missing: approval record (${approval.id}) does not contain a mandatory payloadHash. Consequential execution denied.`;
   471→        const auditRecord: SideEffectAuditRecord = {
   472→          id: auditId,
   473→          timestamp,
   474→          requestId,
   475→          employeeRole: request.employeeRole,
   476→          requestedBy: request.requestedBy || request.employeeRole,
   477→          skillId: typeof request.skillId === 'string' ? request.skillId : undefined,
   478→          workflowInstanceId: request.workflowContext?.workflowInstanceId,
   479→          stepId: request.workflowContext?.stepId,
   480→          actionClassification: request.classification,
   481→          actionName: request.actionName,
   482→          target: request.target,
   483→          decision: 'denied',
   484→          reasonCode: 'APPROVAL_PAYLOAD_HASH_MISSING',
   485→          reason: missingHashReason,
   486→          approvalId: decision.approvalId,
   487→          executionReference: executionRef,
   488→          executed: false,
   489→        };
   490→
   491→        await this.auditStore.record(auditRecord);
   492→
   493→        return {
   494→          allowed: false,
   495→          executed: false,
   496→          decision: {
   497→            effect: 'denied',
   498→            reasonCode: 'APPROVAL_PAYLOAD_HASH_MISSING',
   499→            reason: missingHashReason,
   500→            approvalId: decision.approvalId,
   501→            evaluatedAt: timestamp,
   502→            evaluator: 'central_side_effect_gate',
   503→          },
   504→          auditId,
   505→          error: missingHashReason,
   506→        };
   507→      }
   508→
   509→      // Enforce deterministic payload binding verification
   510→      const bindingCheck = verifyApprovalPayloadBinding(approval, {
   511→        actionName: request.actionName,
   512→        target: request.target,
   513→        payload: request.payload,
   514→      });
   515→
   516→      if (!bindingCheck.isMatch) {
   517→        const mismatchReason = `Cryptographic payload binding mismatch: payload or target was altered after Founder approval was granted (expected: ${bindingCheck.expectedHash}, got: ${bindingCheck.actualHash}). Execution blocked.`;
   518→        const auditRecord: SideEffectAuditRecord = {
   519→          id: auditId,
   520→          timestamp,
   521→          requestId,
   522→          employeeRole: request.employeeRole,
   523→          requestedBy: request.requestedBy || request.employeeRole,
   524→          skillId: typeof request.skillId === 'string' ? request.skillId : undefined,
   525→          workflowInstanceId: request.workflowContext?.workflowInstanceId,
   526→          stepId: request.workflowContext?.stepId,
   527→          actionClassification: request.classification,
   528→          actionName: request.actionName,
   529→          target: request.target,
   530→          decision: 'denied',
   531→          reasonCode: 'APPROVAL_PAYLOAD_HASH_MISMATCH',
   532→          reason: mismatchReason,
   533→          approvalId: decision.approvalId,
   534→          executionReference: executionRef,
   535→          executed: false,
   536→        };
   537→
   538→        await this.auditStore.record(auditRecord);
   539→
   540→        return {
   541→          allowed: false,
   542→          executed: false,
   543→          decision: {
   544→            effect: 'denied',
   545→            reasonCode: 'APPROVAL_PAYLOAD_HASH_MISMATCH',
   546→            reason: mismatchReason,
   547→            approvalId: decision.approvalId,
   548→            evaluatedAt: timestamp,
   549→            evaluator: 'central_side_effect_gate',
   550→          },
   551→          auditId,
   552→          error: mismatchReason,
   553→        };
   554→      }
   555→    }
   556→
   557→    // 4. Atomic Idempotency Claiming
   558→    if (shouldEnforceIdempotency && idempotencyKey && payloadHash) {
   559→      try {
   560→        await getIdempotencyStore().claim({
   561→          key: idempotencyKey,
   562→          actionName: request.actionName,
   563→          payloadHash,
   564→          executionRef: executionRef || auditId,
   565→          ttlMs: params.idempotency?.ttlMs,
   566→        });
   567→      } catch (claimErr: any) {
   568→        if (claimErr instanceof IdempotencyPayloadMismatchError) {
   569→          const auditRecord: SideEffectAuditRecord = {
   570→            id: auditId,
   571→            timestamp,
   572→            requestId,
   573→            employeeRole: request.employeeRole,
   574→            requestedBy: request.requestedBy || request.employeeRole,
   575→            skillId: typeof request.skillId === 'string' ? request.skillId : undefined,
   576→            workflowInstanceId: request.workflowContext?.workflowInstanceId,
   577→            stepId: request.workflowContext?.stepId,
   578→            actionClassification: request.classification,
   579→            actionName: request.actionName,
   580→            target: request.target,
   581→            decision: 'denied',
   582→            reasonCode: 'IDEMPOTENCY_PAYLOAD_MISMATCH',
   583→            reason: claimErr.message,
   584→            approvalId: decision.approvalId,
   585→            executionReference: executionRef,
   586→            executed: false,
   587→          };
   588→          await this.auditStore.record(auditRecord);
   589→        } else if (claimErr instanceof OperationInProgressError) {
   590→          const auditRecord: SideEffectAuditRecord = {
   591→            id: auditId,
   592→            timestamp,
   593→            requestId,
   594→            employeeRole: request.employeeRole,
   595→            requestedBy: request.requestedBy || request.employeeRole,
   596→            skillId: typeof request.skillId === 'string' ? request.skillId : undefined,
   597→            workflowInstanceId: request.workflowContext?.workflowInstanceId,
   598→            stepId: request.workflowContext?.stepId,
   599→            actionClassification: request.classification,
   600→            actionName: request.actionName,
   601→            target: request.target,
   602→            decision: 'denied',
   603→            reasonCode: 'OPERATION_IN_PROGRESS',
   604→            reason: claimErr.message,
   605→            approvalId: decision.approvalId,
   606→            executionReference: executionRef,
   607→            executed: false,
   608→          };
   609→          await this.auditStore.record(auditRecord);
   610→        } else if (claimErr instanceof UnknownExternalResultError) {
   611→          const auditRecord: SideEffectAuditRecord = {
   612→            id: auditId,
   613→            timestamp,
   614→            requestId,
   615→            employeeRole: request.employeeRole,
   616→            requestedBy: request.requestedBy || request.employeeRole,
   617→            skillId: typeof request.skillId === 'string' ? request.skillId : undefined,
   618→            workflowInstanceId: request.workflowContext?.workflowInstanceId,
   619→            stepId: request.workflowContext?.stepId,
   620→            actionClassification: request.classification,
   621→            actionName: request.actionName,
   622→            target: request.target,
   623→            decision: 'denied',
   624→            reasonCode: 'UNKNOWN_EXTERNAL_RESULT',
   625→            reason: claimErr.message,
   626→            approvalId: decision.approvalId,
   627→            executionReference: executionRef,
   628→            executed: false,
   629→          };
   630→          await this.auditStore.record(auditRecord);
   631→        }
   632→        throw claimErr;
   633→      }
   634→    }
   635→
   636→    // 5. Consume single-use approval only after authorization + idempotency claim pass.
   637→    // Phase 4.4B.1: occurrence-bound approvals are single-use BY DEFINITION (one
   638→    // founder authorization per scheduled occurrence) — consume them at
   639→    // execution regardless of the recorded scopeType, in addition to the
   640→    // existing single_action rule.
   641→    if (
   642→      decision.approvalId &&
   643→      approval &&
   644→      (approval.scope.scopeType === 'single_action' || Boolean(approval.scope.occurrenceId))
   645→    ) {
   646→      await this.approvalStore.consume(decision.approvalId);
   647→    }
   648→
   649→    // 6. Execute the authorized operation
   650→    let result: T;
   651→    try {
   652→      result = await executeFn();
   653→    } catch (err: any) {
   654→      const isAmbiguousTimeout =
   655→        err?.name === 'TimeoutError' ||
   656→        err?.code === 'ETIMEDOUT' ||
   657→        err?.code === 'ECONNRESET' ||
   658→        err?.isAmbiguous === true ||
   659→        /timeout|timed out|abort/i.test(err?.message || '');
   660→
   661→      if (shouldEnforceIdempotency && idempotencyKey) {
   662→        if (isAmbiguousTimeout) {
   663→          await getIdempotencyStore().markUnknown(
   664→            idempotencyKey,
   665→            err?.message || 'Execution timed out or aborted with ambiguous provider state'
   666→          );
   667→        } else {
   668→          await getIdempotencyStore().fail(
   669→            idempotencyKey,
   670→            err?.message || 'Execution failed'
   671→          );
   672→        }
   673→      }
   674→
   675→      const auditRecord: SideEffectAuditRecord = {
   676→        id: auditId,
   677→        timestamp,
   678→        requestId,
   679→        employeeRole: request.employeeRole,
   680→        requestedBy: request.requestedBy || request.employeeRole,
   681→        skillId: typeof request.skillId === 'string' ? request.skillId : undefined,
   682→        workflowInstanceId: request.workflowContext?.workflowInstanceId,
   683→        stepId: request.workflowContext?.stepId,
   684→        actionClassification: request.classification,
   685→        actionName: request.actionName,
   686→        target: request.target,
   687→        decision: 'allowed',
   688→        reasonCode: decision.reasonCode,
   689→        reason: decision.reason,
   690→        approvalId: decision.approvalId,
   691→        executionReference: executionRef,
   692→        executed: false, // Execution threw error
   693→      };
   694→
   695→      await this.auditStore.record(auditRecord);
   696→
   697→      if (isAmbiguousTimeout) {
   698→        throw new UnknownExternalResultError(idempotencyKey || 'unknown', executionRef, err?.message || 'Operation timed out; external state is unknown');
   699→      }
   700→      throw err;
   701→    }
   702→
   703→    // 7. Record completion in idempotency store
   704→    if (shouldEnforceIdempotency && idempotencyKey) {
   705→      await getIdempotencyStore().complete(idempotencyKey, result, executionRef || auditId);
   706→    }
   707→
   708→    // 8. Record successful execution in audit trail
   709→    const auditRecord: SideEffectAuditRecord = {
   710→      id: auditId,
   711→      timestamp,
   712→      requestId,
   713→      employeeRole: request.employeeRole,
   714→      requestedBy: request.requestedBy || request.employeeRole,
   715→      skillId: typeof request.skillId === 'string' ? request.skillId : undefined,
   716→      workflowInstanceId: request.workflowContext?.workflowInstanceId,
   717→      stepId: request.workflowContext?.stepId,
   718→      actionClassification: request.classification,
   719→      actionName: request.actionName,
   720→      target: request.target,
   721→      decision: 'allowed',
   722→      reasonCode: decision.reasonCode,
   723→      reason: decision.reason,
   724→      approvalId: decision.approvalId,
   725→      executionReference: executionRef || `exec-${Date.now()}`,
   726→      executed: true,
   727→    };
   728→
   729→    await this.auditStore.record(auditRecord);
   730→
   731→    return {
   732→      allowed: true,
   733→      executed: true,
   734→      decision,
   735→      result,
   736→      auditId,
   737→      idempotencyKey,
   738→      idempotentReplay: false,
   739→    };
   740→  }
