     1→     1→/**
     2→     2→ * The resilience layer — every provider chain in one place.
     3→     3→ *
     4→     4→ * The assistant is a conversation, and a conversation dies the moment one
     5→     5→ * sentence goes unanswered. Every external service it depends on — the model
     6→     6→ * that thinks, the transcriber that hears, the voice that speaks — has a
     7→     7→ * moment where it rate-limits, times out, or simply goes down. So none of
     8→     8→ * them is load-bearing alone: each is one link in a chain, and the chain is
     9→     9→ * walked on failure, automatically, mid-turn.
    10→    10→ *
    11→    11→ *   LLM   z-ai (built-in, no key) → Gemini → any local OpenAI-compatible
    12→    12→ *         server (Ollama, LM Studio, vLLM, llama.cpp)
    13→    13→ *   STT   Deepgram → ElevenLabs Scribe → z-ai ASR (no key) → a local
    14→    14→ *         OpenAI-compatible transcription server (faster-whisper, LocalAI)
    15→    15→ *   TTS   ElevenLabs → z-ai neural → a local OpenAI-compatible speech server
    16→    16→ *         (kokoro-fastapi, LocalAI, openedai-speech) → the browser's own voice
    17→    17→ *
    18→    18→ * Three ideas make the chains safe rather than just long:
    19→    19→ *
    20→    20→ *   Circuit breakers. A provider that fails twice in a row (LLM) or three
    21→    21→ *   times (STT) is skipped for a cooldown — 90s / 60s — so a dead provider
    22→    22→ *   costs one failed attempt, not one per sentence. When every link is
    23→    23→ *   cooling the chain is walked anyway, in order: a retry is better than a
    24→    24→ *   shrug. Any success heals the provider immediately.
    25→    25→ *
    26→    26→ *   Pins. SOFIA_LLM_PROVIDER / SOFIA_STT_PROVIDER / SOFIA_TTS_PROVIDER
    27→    27→ *   collapse a chain to one link on purpose — for testing, or because a
    28→    28→ *   person has decided they know better than the ladder. (The JARVIS_*
    29→    29→ *   spellings from the bridge era still work as aliases.)
    30→    30→ *
    31→    31→ *   Honesty. llmInfo() / sttInfo() are the single source of truth for
    32→    32→ *   /api/sofia/health, which is what the settings panel shows the user:
    33→    33→ *   which links exist, which are configured, which one is answering right
    34→    34→ *   now. A fallback you can't see is a fallback you can't trust.
    35→    35→ *
    36→    36→ * Everything speaks the OpenAI wire format — the request bodies and the
    37→    37→ * SSE event stream — which is why one parser (in brain.ts) serves every
    38→    38→ * link: the z-ai SDK, Gemini's OpenAI-compat endpoint and a local server
    39→    39→ * all say `data: {"choices":[{"delta":{...}}]}`.
    40→    40→ *
    41→    41→ * Ported from bridge/providers.mjs (Vite era) — behaviour identical; the env
    42→    42→ * vars gained SOFIA_* aliases and Next.js loads .env.local itself.
    43→    43→ */

    83→    83→// Circuit breaker — the one abstraction every chain shares.

    85→    85→
    86→    86→type GuardState = { fails: number; until: number }
    87→    87→
    88→    88→function makeGuard(threshold: number, cooldownMs: number) {
    89→    89→  const state = new Map<string, GuardState>()
    90→    90→
    91→    91→  const trip = (id: string) => {
    92→    92→    const s = state.get(id) ?? { fails: 0, until: 0 }
    93→    93→    s.fails += 1
    94→    94→    // Only a run of failures opens the breaker, and opening it resets the
    95→    95→    // count, so the cooldown that follows is a fixed window, not a ratchet.
    96→    96→    if (s.fails >= threshold) {
    97→    97→      s.until = Date.now() + cooldownMs
    98→    98→      s.fails = 0
    99→    99→    }
   100→   100→    state.set(id, s)
   101→   101→  }
   102→   102→  return {
   103→   103→    trip,
   104→   104→    heal: (id: string) => state.set(id, { fails: 0, until: 0 }),
   105→   105→    cooling: (id: string) => {
   106→   106→      const s = state.get(id)
   107→   107→      return Boolean(s && s.until > Date.now())
   108→   108→    },
   109→   109→  }
   110→   110→}

   121→   121→  return /\b(429|rate|too many|502|503|504|timeout|network|fetch failed|ECONN)\b/i.test(
   122→   122→    errText(err),
   123→   123→  )
   124→   124→}

   126→   126→/** A provider that refused our tool schema (small local models often do)
   127→   127→ *  gets one retry without tools — a plain answer beats a failed turn. A
   128→   128→ *  missing endpoint (404) is NOT a schema refusal, or every typo would
   129→   129→ *  silently strip the tools for good. Latched so the retry is paid once per
   130→   130→ *  provider, not once per turn. */
   131→   131→function looksLikeToolRejection(err: unknown): boolean {
   132→   132→  const t = errText(err)
   133→   133→  return /(\b400\b|\b422\b|unsupported|not support|tool|function)/i.test(t)
   134→   134→}

   190→   190→export type LlmBody = {
   191→   191→  messages: Array<Record<string, unknown>>

   197→   197→/** One SSE reader from any OpenAI-compatible chat endpoint. The shape —
   198→   198→ *  `data: {...}\n\n` lines ending in `data: [DONE]` — is exactly what
   199→   199→ *  brain.ts already parses for the z-ai SDK's streams. */
   200→   200→async function openAiCompatReader(
   201→   201→  url: string,
   202→   202→  key: string,
   203→   203→  body: LlmBody,
   204→   204→): Promise<ReadableStreamDefaultReader<Uint8Array>> {
   205→   205→  const res = await fetch(url, {
   206→   206→    method: 'POST',
   207→   207→    headers: {
   208→   208→      'content-type': 'application/json',
   209→   209→      ...(key ? { authorization: `Bearer ${key}` } : {}),
   210→   210→    },
   211→   211→    body: JSON.stringify(body),
   212→   212→    // No abort signal: it would kill the body mid-stream, and local models
   213→   213→    // legitimately take their time. The brain's own stall guard owns hangs.
   214→   214→  })
   215→   215→  if (!res.ok) {
   216→   216→    const detail = await res.text().catch(() => '')
   217→   217→    throw new Error(`${res.status} ${res.statusText} ${String(detail).slice(0, 200)}`)
   218→   218→  }
   219→   219→  const reader = res.body?.getReader()
   220→   220→  if (!reader) throw new Error('the provider returned no stream')
   221→   221→  return reader
   222→   222→}

   233→   233→/** Each link adapts the shared body to its own dialect. */
   234→   234→function buildLlmBody(p: { id: string; model?: string }, body: LlmBody, withTools: boolean): LlmBody {
   235→   235→  const tools =
   236→   236→    withTools && Array.isArray(body.tools) && body.tools.length ? body.tools : undefined
   237→   237→  if (p.id === 'zai') {
   238→   238→    // `thinking` is a z-ai extension; harmless nowhere else but not theirs
   239→   239→    // to carry, so it stays on the one link that reads it.
   240→   240→    return { ...body, tools, thinking: { type: 'disabled' } }
   241→   241→  }
   242→   242→  const out: LlmBody = { ...body, tools, model: p.model }
   243→   243→  delete out.thinking
   244→   244→  return out
   245→   245→}

   328→   328→/**
   329→   329→ * Walk the chain and return the first live SSE reader.
   330→   330→ *
   331→   331→ * `announce(provider)` fires exactly when the active link *changes* — the
   332→   332→ * browser turns that into the BRAIN rail line, so a person can see the
   333→   333→ * moment Gemini takes over. Failures are announced the loud way instead: a
   334→   334→ * warning on the server log naming the link and the error.
   335→   335→ */
   336→   336→export async function openLlmStream(
   337→   337→  body: LlmBody,
   338→   338→  announce?: (p: LlmProvider) => void,
   339→   339→): Promise<{ reader: ReadableStreamDefaultReader<Uint8Array>; provider: LlmProvider }> {
   340→   340→  const configured = LLM_CHAIN.filter((p) => p.available())
   341→   341→  if (!configured.length) throw new Error('no llm provider is configured')
   342→   342→
   343→   343→  const pinId = llmPinId()
   344→   344→  const pin = pinId ? configured.find((p) => p.id === pinId) : null
   345→   345→  if (pinId && !pin) {
   346→   346→    console.warn(`[sofia] SOFIA_LLM_PROVIDER="${pinId}" matches nothing — using the chain`)
   347→   347→  }
   348→   348→  const ordered = pin ? [pin] : configured
   349→   349→
   350→   350→  let awake = ordered.filter((p) => !LLM_GUARD.cooling(p.id))
   351→   351→  if (!awake.length) {
   352→   352→    // Every link cooling. Try them in order anyway — a chain that has given
   353→   353→    // up is worse than one that keeps failing loudly.
   354→   354→    awake = ordered
   355→   355→  }
   356→   356→
   357→   357→  let lastErr: unknown
   358→   358→  for (const p of awake) {
   359→   359→    for (let attempt = 0; attempt < 2; attempt++) {
   360→   360→      const withTools = !noTools.has(p.id)
   361→   361→      try {
   362→   362→        const reader = await p.start(buildLlmBody(p, body, withTools))
   363→   363→        if (activeLlm !== p.id) {
   364→   364→          activeLlm = p.id
   365→   365→          announce?.(p)
   366→   366→        }
   367→   367→        llmHealth.set(p.id, 'ok')
   368→   368→        llmErrors.delete(p.id)
   369→   369→        LLM_GUARD.heal(p.id)
   370→   370→        return { reader, provider: p }
   371→   371→      } catch (err) {
   372→   372→        lastErr = err
   373→   373→        // First failure with tools + a rejection-shaped error + never latched:
   374→   374→        // retry the same link once with the tools stripped.
   375→   375→        if (attempt === 0 && withTools && !noTools.has(p.id) && looksLikeToolRejection(err)) {
   376→   376→          noTools.add(p.id)
   377→   377→          console.warn(`[sofia] llm "${p.id}" refused the tool schema — continuing without tools`)
   378→   378→          continue
   379→   379→        }
   380→   380→        if (!isTransient(err) || attempt === 1) break
   381→   381→        await sleep(600)
   382→   382→      }
   383→   383→    }
   384→   384→    LLM_GUARD.trip(p.id)
   385→   385→    llmHealth.set(p.id, 'err')
   386→   386→    llmErrors.set(p.id, errText(lastErr))
   387→   387→    console.warn(`[sofia] llm provider "${p.id}" failed: ${errText(lastErr)}`)
   388→   388→  }
   389→   389→  throw lastErr ?? new Error('every llm provider failed')
   390→   390→}

   449→   449→ * z-ai ASR is keyless, which makes it the difference between "server-side
   450→   450→ * transcription needs a key" and "it works for everyone" — but on a machine
   451→   451→ * without the sandbox's z-ai credentials it cannot work at all. So it is
   452→   452→ * probed once at startup with half a second of silence: ok means live,
   453→   453→ * rate-limited means live (a quota window closes on its own), and an auth
   454→   454→ * error means honestly OFF.
   455→   455→ */
   456→   456→const asrProbe = { state: 'pending' as 'pending' | 'ok' | 'no', detail: '' }

   458→   458→function silenceWav(): Buffer {
   459→   459→  const samples = 8000 // half a second of 16kHz zeros
   460→   460→  const buf = Buffer.alloc(44 + samples * 2)
   461→   461→  buf.write('RIFF', 0)
   462→   462→  buf.writeUInt32LE(36 + samples * 2, 4)
   463→   463→  buf.write('WAVE', 8)
   464→   464→  buf.write('fmt ', 12)
   465→   465→  buf.writeUInt32LE(16, 16)
   466→   466→  buf.writeUInt16LE(1, 20) // PCM
   467→   467→  buf.writeUInt16LE(1, 22) // mono
   468→   468→  buf.writeUInt32LE(16000, 24)
   469→   469→  buf.writeUInt32LE(32000, 28)
   470→   470→  buf.writeUInt16LE(2, 32)
   471→   471→  buf.writeUInt16LE(16, 34)
   472→   472→  buf.write('data', 36)
   473→   473→  buf.writeUInt32LE(samples * 2, 40)
   474→   474→  return buf
   475→   475→}

   477→   477→export async function probeZaiAsr(): Promise<'ok' | 'no' | 'pending'> {
   478→   478→  try {
   479→   479→    const api = await zaiClient()
   480→   480→    const res = await (
   481→   481→      api.audio.asr.create as (b: unknown) => Promise<{ text?: unknown }>
   482→   482→    ).call(api.audio.asr, { file_base64: silenceWav().toString('base64') })
   483→   483→    asrProbe.state = 'ok'
   484→   484→    asrProbe.detail = typeof res?.text === 'string' ? '' : 'unexpected shape'
   485→   485→  } catch (err) {
   486→   486→    const text = errText(err)
   487→   487→    if (/\b(429|rate|too many|502|503|504)\b/i.test(text)) {
   488→   488→      // A quota window — the same one that trips the LLM chain — not an
   489→   489→      // outage. Available, and the breaker will absorb the failures.
   490→   490→      asrProbe.state = 'ok'
   491→   491→      asrProbe.detail = 'rate-limited at probe time'
   492→   492→    } else {
   493→   493→      asrProbe.state = 'no'
   494→   494→      asrProbe.detail = text
   495→   495→    }
   496→   496→  }
   497→   497→  console.log(
   498→   498→    `[sofia] z-ai asr: ${asrProbe.state}${asrProbe.detail ? ` (${asrProbe.detail})` : ''}`,
   499→   499→  )
   500→   500→  return asrProbe.state
   501→   501→}

   611→   611→/**
   612→   612→ * Transcribe one segment through the first link that answers. Per-request
   613→   613→ * fallthrough (not just breaker-driven): a segment is a few hundred
   614→   614→ * milliseconds of speech, and the words are only useful now.
   615→   615→ */
   616→   616→export async function transcribeAnywhere(
   617→   617→  buffer: Buffer,
   618→   618→  type: string | undefined,
   619→   619→): Promise<{ text: string; provider: string }> {
   620→   620→  const configured = STT_CHAIN.filter((p) => p.available())
   621→   621→  if (!configured.length) throw new Error('no transcription provider is configured')
   622→   622→
   623→   623→  const pinId = sttPinId()
   624→   624→  const pin = pinId ? configured.find((p) => p.id === pinId) : null
   625→   625→  if (pinId && !pin) {
   626→   626→    console.warn(`[sofia] SOFIA_STT_PROVIDER="${pinId}" matches nothing — using the chain`)
   627→   627→  }
   628→   628→  const ordered = pin ? [pin] : configured
   629→   629→
   630→   630→  let awake = ordered.filter((p) => !STT_GUARD.cooling(p.id))
   631→   631→  if (!awake.length) awake = ordered
   632→   632→
   633→   633→  let lastErr: unknown
   634→   634→  for (const p of awake) {
   635→   635→    try {
   636→   636→      const text = await p.transcribe(buffer, type)
   637→   637→      sttHealth.set(p.id, 'ok')
   638→   638→      sttErrors.delete(p.id)
   639→   639→      STT_GUARD.heal(p.id)
   640→   640→      activeStt = p.id
   641→   641→      return { text, provider: p.id }
   642→   642→    } catch (err) {
   643→   643→      lastErr = err
   644→   644→      STT_GUARD.trip(p.id)
   645→   645→      sttHealth.set(p.id, 'err')
   646→   646→      sttErrors.set(p.id, errText(err))
   647→   647→      console.warn(`[sofia] stt provider "${p.id}" failed: ${errText(err)}`)
   648→   648→    }
   649→   649→  }
   650→   650→  throw lastErr ?? new Error('every transcription provider failed')
   651→   651→}

   681→   681→      label: p.label,
   682→   682→      note: p.note,
   683→   683→      configured: p.available(),
   684→   684→      state,
   685→   685→      healthy: state !== 'err',
   686→   686→      cooling: STT_GUARD.cooling(p.id),
   687→   687→      error: sttErrors.get(p.id) ?? null,
   688→   688→      detail: p.id === 'zai' ? asrProbe.detail : '',
   689→   689→    }
   690→   690→  })
   691→   691→  return {
   692→   692→    pin: pinId || null,
   693→   693→    active: activeStt || providers.find((p) => p.configured)?.id || '',

   695→   695→  }
