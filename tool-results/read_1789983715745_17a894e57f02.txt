    38→    38→/**
    39→    39→ * The conversation.
    40→    40→ *
    41→    41→ * This used to be a sequential loop — greet, await a capture, await an answer,
    42→    42→ * repeat — with the microphone opened and closed around each step. That shape
    43→    43→ * cannot be interrupted: while it is awaiting the answer, nothing is listening,
    44→    44→ * so there is no way for the user to get a word in.
    45→    45→ *
    46→    46→ * It is an event machine now. The voice loop runs continuously and pushes
    47→    47→ * events at us; every one of them is legal in every phase. Saying anything at
    48→    48→ * all stops him talking, and whatever you say next becomes the new turn.
    49→    49→ */
    50→    50→
    51→    51→/** How long to wait for someone to start speaking after he wakes. Generous:
    52→    52→ *  people say his name and *then* think about what they wanted. */
    53→    53→const AWAIT_SPEECH_MS = 14000
    54→    54→
    55→    55→/** After an answer, how long the mic stays open for a follow-up before he
    56→    56→ *  drops back to standby. Long enough that you don't have to say the name
    57→    57→ *  again to continue a thought. */
    58→    58→const FOLLOW_UP_MS = 11000
    59→    59→
    60→    60→/** crypto.randomUUID needs a secure context, which a LAN address over plain
    61→    61→ *  http is not. Not worth failing a whole turn over an id. */
    62→    62→const newId = () =>
    63→    63→  globalThis.crypto?.randomUUID?.() ??
    64→    64→  `id${Date.now().toString(36)}${Math.random().toString(36).slice(2, 10)}`
    65→    65→
    66→    66→/** The same mishearings voice.ts accepts for the wake word — otherwise a turn
    67→    67→ *  that woke the machine as “travis” gets that word sent on to the model as a
    68→    68→ *  question. Read fresh from the persona each time: the character can change
    69→    69→ *  under a long-lived page, and a stale name would strip the wrong word. */
    70→    70→const namePattern = () => personaById(useStore.getState().persona).heardAs
    71→    71→/** A bare vocative — “Sofia”, “hey sofia” — with nothing asked. */
    72→    72→const bareName = () =>
    73→    73→  new RegExp(`^(?:hey|hi|ok|okay|yo)?\\s*(?:${namePattern()})[\\s,.!?]*$`, 'i')
    74→    74→/** A leading vocative on a real command: “Sofia, what's the weather”. */
    75→    75→const leadingName = () =>
    76→    76→  new RegExp(`^(?:hey|hi|ok|okay|yo)?\\s*(?:${namePattern()})\\b[\\s,.:!?-]*`, 'i')

   149→   149→  // -- one turn -------------------------------------------------------------
   150→   150→
   151→   151→  const respond = async (said: string): Promise<void> => {
   152→   152→    const mine = ++turn.current
   153→   153→    const stale = () => mine !== turn.current
   154→   154→
   155→   155→    clearIdle()
   156→   156→    const s = store.getState()
   157→   157→    // Last turn's panels and blades go now, before the new answer starts
   158→   158→    // putting its own up. Anything the model marked sticky survives.
   159→   159→    s.clearPanels()
   160→   160→    s.clearBlades()
   161→   161→    s.setCaption('')
   162→   162→    s.pushTurn({ id: newId(), role: 'user', text: said })
   163→   163→    s.setPhase('thinking')
   164→   164→
   165→   165→    const spk = createSpeaker()
   166→   166→    speaker.current = spk
   167→   167→    sfx.duck(true)
   168→   168→    music.duck(true)
   169→   169→
   170→   170→    const turnId = newId()
   171→   171→    let started = false
   172→   172→    let filled = false
   173→   173→
   174→   174→    try {
   175→   175→      const { text } = await ask(said, history.current, {
   176→   176→        onText: (delta) => {
   177→   177→          if (stale()) return
   178→   178→          if (!started) {
   179→   179→            started = true
   180→   180→            store.getState().setPhase('speaking')
   181→   181→            // The answer arriving is what ends the tool phase — a timer would
   182→   182→            // clear the readout while a slow tool was still running.
   183→   183→            store.getState().setActiveTool(null)
   184→   184→            music.working(false)
   185→   185→            store.getState().pushTurn({ id: turnId, role: 'jarvis', text: '' })
   186→   186→          }
   187→   187→          store.getState().appendToLastTurn(delta)
   188→   188→          spk.push(delta)
   189→   189→        },
   190→   190→        onTool: (name) => {
   191→   191→          if (stale()) return
   192→   192→          // Only claim the tooling phase while he has nothing to say yet.
   193→   193→          // Setting it unconditionally pinned the machine in 'tooling' for the
   194→   194→          // rest of any answer that called a tool after it started talking,
   195→   195→          // which also broke the reactor's lip-sync for the remainder.
   196→   196→          if (!started) store.getState().setPhase('tooling')
   197→   197→          store.getState().setActiveTool(name)
   198→   198→          sfx.play('tool')
   199→   199→          music.working(true)
   200→   200→          // Say something the moment work starts — a tool can take ten seconds
   201→   201→          // and silence that long reads as a crash. Once per turn only; a
   202→   202→          // chain of five tools shouldn't produce five apologies.
   203→   203→          if (!filled && !started) {
   204→   204→            filled = true
   205→   205→            spk.say(forTool(name))
   206→   206→          }
   207→   207→        },
   208→   208→        // The character the brain should answer as — re-read per turn so a
   209→   209→        // mid-session persona switch takes effect on the very next question.
   210→   210→      }, store.getState().persona)
   211→   211→
   212→   212→      if (stale()) return
   213→   213→
   214→   214→      // The bridge keeps conversation state in its own session, so history is
   215→   215→      // only threaded through on the direct path.
   216→   216→      if (!usingBridge) {
   217→   217→        history.current.push({ role: 'user', content: said })
   218→   218→        history.current.push({ role: 'assistant', content: text || '…' })
   219→   219→        if (history.current.length > 16) {
   220→   220→          history.current = history.current.slice(-16)
   221→   221→        }
   222→   222→      }
   223→   223→
   224→   224→      await spk.end()
   225→   225→      if (stale()) return
   226→   226→      sfx.play('done')
   227→   227→    } catch (err) {
   228→   228→      if (stale()) return
   229→   229→      console.error(err)
   230→   230→      sfx.play('error')
   231→   231→      store
   232→   232→        .getState()
   233→   233→        .setError(err instanceof Error ? err.message : 'Something went wrong.')
   234→   234→    } finally {
   235→   235→      if (!stale()) {
   236→   236→        speaker.current = null
   237→   237→        sfx.duck(false)
   238→   238→        music.duck(false)
   239→   239→        store.getState().setActiveTool(null)
   240→   240→        music.working(false)
   241→   241→        // Stay open. Having to say his name again to add one more sentence is
   242→   242→        // the difference between a conversation and a vending machine.
   243→   243→        listen(FOLLOW_UP_MS)
   244→   244→      }
   245→   245→    }
   246→   246→  }
   247→   247→
   248→   248→  // -- voice events ---------------------------------------------------------
   249→   249→
   250→   250→  /** What the voice loop should do with what it hears, derived from phase. */

   254→   254→      case 'boot':
   255→   255→        return 'deaf'

   258→   258→      case 'waking':
   259→   259→      case 'listening':

   262→   262→        return 'guard' // thinking, tooling, speaking
   263→   263→    }
   264→   264→  }

   266→   266→  const onWake = (trailing: string) => {
   267→   267→    const phase = store.getState().phase
   268→   268→    if (phase === 'offline' || phase === 'boot') return
   269→   269→
   270→   270→    store.getState().setError(null)
   271→   271→    sfx.play('wake')
   272→   272→
   273→   273→    // "Jarvis, what's happening in AI this week" in one breath. Waiting for a
   274→   274→    // greeting he didn't need is the most common way an assistant wastes time.
   275→   275→    if (trailing) {
   276→   276→      void respond(trailing)
   277→   277→      return
   278→   278→    }
   279→   279→
   280→   280→    store.getState().setPhase('waking')
   281→   281→
   282→   282→    // Answer to his name. Deliberately NOT awaited any more: the microphone is
   283→   283→    // already open and the echo filter knows his voice, so the user can talk
   284→   284→    // straight over the greeting instead of waiting it out.
   285→   285→    const greeting = createSpeaker()
   286→   286→    speaker.current = greeting
   287→   287→    greeting.say(attention())
   288→   288→    void greeting.end()
   289→   289→
   290→   290→    listen(AWAIT_SPEECH_MS)
   291→   291→  }
   292→   292→
   293→   293→  /**
   294→   294→   * Someone started talking. This is the whole point of the rewrite: he stops,
   295→   295→   * immediately, whatever he was doing.
   296→   296→   */
   297→   297→  const onSpeechStart = () => {
   298→   298→    clearIdle()
   299→   299→    const phase = store.getState().phase
   300→   300→    if (phase === 'offline' || phase === 'boot' || phase === 'dormant') return
   301→   301→
   302→   302→    const wasBusy =
   303→   303→      phase === 'thinking' || phase === 'tooling' || phase === 'speaking'
   304→   304→
   305→   305→    silence()
   306→   306→    if (wasBusy) {
   307→   307→      // Abandon the answer in flight. The turn counter moves in respond()'s
   308→   308→      // replacement; bumping it here covers the case where nothing replaces it.
   309→   309→      turn.current++
   310→   310→      interrupt()
   311→   311→      store.getState().setActiveTool(null)
   312→   312→      music.working(false)
   313→   313→      sfx.duck(false)
   314→   314→      music.duck(false)
   315→   315→    }
   316→   316→    store.getState().setPhase('listening')
   317→   317→  }
   318→   318→
   319→   319→  const onUtterance = (text: string) => {
   320→   320→    const phase = store.getState().phase
   321→   321→    if (phase === 'offline' || phase === 'boot' || phase === 'dormant') return
   322→   322→
   323→   323→    // People keep using the name as a vocative once they're already talking to
   324→   324→    // the machine. Strip it rather than sending "sofia" to the model as a
   325→   325→    // question.
   326→   326→    if (bareName().test(text)) {
   327→   327→      listen(AWAIT_SPEECH_MS)
   328→   328→      return
   329→   329→    }
   330→   330→    const said = text.replace(leadingName(), '').trim()
   331→   331→    if (!said) {
   332→   332→      listen(AWAIT_SPEECH_MS)
   333→   333→      return
   334→   334→    }
   335→   335→
   336→   336→    void respond(said)
   337→   337→  }
   338→   338→
   339→   339→  /**
   340→   340→   * The typed command line. A `jarvis:say` event is the keyboard's version of
   341→   341→   * a recognised utterance: same road, same rules. Idle, it starts a turn;
   342→   342→   * busy, it barge-ins exactly like speaking over him does — the in-flight
   343→   343→   * answer is interrupted, its turn goes stale, and this one replaces it.
   344→   344→   */
   345→   345→  const onSay = (e: Event) => {
   346→   346→    const said = String((e as CustomEvent<string>).detail ?? '').trim()
   347→   347→    if (!said) return
   348→   348→
   349→   349→    const phase = store.getState().phase
   350→   350→    if (phase === 'boot') return
   351→   351→
   352→   352→    if (phase === 'offline') {
   353→   353→      // The ignition screen is still up; boot first, then ask the moment the
   354→   354→      // interface lands.
   355→   355→      void powerOn().then(() => respond(said))
   356→   356→      return
   357→   357→    }
   358→   358→
   359→   359→    if (phase === 'thinking' || phase === 'tooling' || phase === 'speaking') {
   360→   360→      silence()
   361→   361→      interrupt()
   362→   362→    }
   363→   363→    void respond(said)
   364→   364→  }
   365→   365→
   366→   366→  const onPartial = (text: string) => {
   367→   367→    store.getState().setCaption(text)
   368→   368→  }
   369→   369→
   370→   370→  const onVoiceError = (message: string) => {
   371→   371→    store.getState().setError(message)
   372→   372→    // The two mic failures are the only ones that mean voice is off for good
   373→   373→    // — transient recogniser hiccups fire this too, and must not flip the
   374→   374→    // standby hint to “type a command”. The ref survives startVoice returning
   375→   375→    // its dead object, so the optimistic setVoiceLive(true) after it cannot
   376→   376→    // overwrite what happened here.
   377→   377→    if (/microphone/i.test(message)) {
   378→   378→      micDead.current = true
   379→   379→      store.getState().setVoiceLive(false)
   380→   380→    }
   381→   381→  }
   382→   382→
   383→   383→  // -- power on -------------------------------------------------------------
   384→   384→
   385→   385→  const powerOn = async () => {
   386→   386→    // The ignition button and the space bar can both land here, and the phase
   387→   387→    // only moves after the first await — so without this a double press boots
   388→   388→    // twice, arming two voice loops and two download polls.
   389→   389→    if (booting.current) return
   390→   390→    booting.current = true
   391→   391→
   392→   392→    try {
   393→   393→      await ignite()
   394→   394→    } catch (err) {
   395→   395→      // The guard must not outlive a failed boot. Audio unlock can be refused,
   396→   396→      // the microphone prompt dismissed, the bridge unreachable at the wrong
   397→   397→      // moment — and with the flag still latched the ignition button was dead
   398→   398→      // for the rest of the page, recoverable only by reloading. Reset it and
   399→   399→      // put the button back so the user can simply press it again.
   400→   400→      booting.current = false
   401→   401→      console.error('[jarvis] power-up failed:', err)
   402→   402→      store.getState().setPhase('offline')
   403→   403→      store
   404→   404→        .getState()
   405→   405→        .setError(
   406→   406→          err instanceof Error
   407→   407→            ? `Power-up failed: ${err.message}`
   408→   408→            : 'Power-up failed. Click to try again.',
   409→   409→        )
   410→   410→    }
   411→   411→  }
   412→   412→
   413→   413→  const ignite = async () => {
   414→   414→    const s = store.getState()
   415→   415→
   416→   416→    // Must happen inside the click handler — browsers won't start an
   417→   417→    // AudioContext or speech synthesis without a user gesture.
   418→   418→    await sfx.unlockAudio()
   419→   419→    sfx.play('boot')
   420→   420→    // The score. Must be started from inside this click handler for the same
   421→   421→    // reason as the rest of the audio.
   422→   422→    music.enable()
   423→   423→    music.playBoot()
   424→   424→    music.startAmbient()
   425→   425→
   426→   426→    s.setPhase('boot')
   427→   427→
   428→   428→    watchServers((servers) => store.getState().setConnected(servers))
   429→   429→    // Which link of the chain is answering — announced on connect and on
   430→   430→    // every mid-session failover, so the rail's BRAIN line is always the
   431→   431→    // brain that is actually speaking.
   432→   432→    watchBrain((b) => store.getState().setBrain(b))
   433→   433→    watchPanels((panel) => store.getState().pushPanel(panel))
   434→   434→    watchBlades((blade) => store.getState().pushBlade(blade))
   435→   435→
   436→   436→    /**
   437→   437→     * JARVIS asking to see something.
   438→   438→     *
   439→   439→     * Announced on screen for as long as it takes, with whatever he said he was
   440→   440→     * looking for. The camera's own light is on too, but a hardware light that
   441→   441→     * appears with no explanation is exactly the thing that makes people
   442→   442→     * distrust an assistant — so the interface says it before they have to ask.
   443→   443→     */
   444→   444→    watchCapture(async (req) => {
   445→   445→      const note =
   446→   446→        req.mode === 'watch'
   447→   447→          ? req.when === 'past'
   448→   448→            ? req.reason || 'reviewing the last few seconds'
   449→   449→            : `${req.reason || 'watching'} · ${req.seconds}s`
   450→   450→          : req.reason || 'taking a look'
   451→   451→      store.getState().setLooking(note)
   452→   452→
   453→   453→      // The past is only available if something has been remembering it, and
   454→   454→      // that only happens while the camera is on screen. Answering plainly
   455→   455→      // beats opening the camera and recording the next few seconds instead,
   456→   456→      // which is a different question from the one that was asked.
   457→   457→      if (req.mode === 'watch' && req.when === 'past' && camera.bufferedSeconds() < 1) {
   458→   458→        store.getState().setLooking(null)
   459→   459→        return {
   460→   460→          error:
   461→   461→            'There is no recent footage — the camera has to be open on screen ' +
   462→   462→            'for me to remember what just happened. Ask me to open the camera, ' +
   463→   463→            'and I can watch from then on.',
   464→   464→        }
   465→   465→      }
   466→   466→
   467→   467→      // Held for the whole capture. Without this the stream can be torn down by
   468→   468→      // whoever else was using it half way through a six-second watch.
   469→   469→      let held = false
   470→   470→      try {
   471→   471→        await camera.holdCamera()
   472→   472→        held = true
   473→   473→        if (req.mode === 'look') return camera.grabFrame()
   474→   474→        if (req.when === 'past') {
   475→   475→          const grid = camera.recentGrid(req.seconds, 9)
   476→   476→          return grid ?? { error: 'There is not enough recent footage to review.' }
   477→   477→        }
   478→   478→        return await camera.watchAhead(req.seconds, 9)
   479→   479→      } catch (err) {
   480→   480→        return {
   481→   481→          error:
   482→   482→            (err as DOMException)?.name === 'NotAllowedError'
   483→   483→              ? 'The camera is not permitted, so I cannot see anything.'
   484→   484→              : `The camera could not be read: ${(err as Error)?.message ?? err}`,
   485→   485→        }
   486→   486→      } finally {
   487→   487→        if (held) camera.releaseCamera()
   488→   488→        store.getState().setLooking(null)
   489→   489→      }
   490→   490→    })
   491→   491→
   492→   492→    // The interface is JARVIS's to drive. These arrive out of band, pushed
   493→   493→    // mid-turn the way panels are, so a command can retint the reactor or put
   494→   494→    // something into orbit while he is still speaking the sentence about it.
   495→   495→    watchUi((op, args) => {
   496→   496→      const s = store.getState()
   497→   497→      const a = (args ?? {}) as Record<string, never>
   498→   498→      switch (op) {
   499→   499→        case 'patch':
   500→   500→          s.applyUi(args)
   501→   501→          break
   502→   502→        case 'orbit':
   503→   503→          if (a.action === 'add') s.addOrbit(args)
   504→   504→          else if (a.action === 'remove') s.removeOrbit(String(a.id))
   505→   505→          else s.clearOrbits()
   506→   506→          break
   507→   507→        case 'effect':
   508→   508→          s.fireEffect(a.kind)
   509→   509→          break
   510→   510→        case 'reset':
   511→   511→          s.resetUi()
   512→   512→          break
   513→   513→        case 'screen':
   514→   514→          s.clearScreen(a.what ?? 'all')
   515→   515→          break
   516→   516→        default:
   517→   517→          console.warn('[jarvis] unknown ui op:', op, args)
   518→   518→      }
   519→   519→    })
   520→   520→    // In bridge mode the conversation lives in the agent session, which is tied
   521→   521→    // to the socket — so a drop silently wipes his memory while the transcript
   522→   522→    // on screen still shows it. Better to say so than to let him quietly forget.
   523→   523→    watchConnection((state) => {
   524→   524→      if (state === 'lost') {
   525→   525→        store.getState().setError('Bridge connection lost — reconnecting.')
   526→   526→      } else if (state === 'reconnected') {
   527→   527→        store
   528→   528→          .getState()
   529→   529→          .setError('Bridge reconnected. The previous conversation was not kept.')
   530→   530→      }
   531→   531→    })
   532→   532→    const warming = warm().catch((err: Error) => s.setError(err.message))
   533→   533→
   534→   534→    if (!usingBridge && !env.anthropicKey) {
   535→   535→      s.setError(
   536→   536→        'No Anthropic API key — copy .env.example to .env.local and set VITE_ANTHROPIC_API_KEY.',
   537→   537→      )
   538→   538→    }
   539→   539→
   540→   540→    // Pull the neural voice down during the boot sequence so the first
   541→   541→    // "Hey Jarvis" isn't waiting on an 86MB download. Deliberately not awaited
   542→   542→    // — if it's slow, JARVIS comes up on the system voice and swaps over the
   543→   543→    // moment the model is ready.
   544→   544→    if (TTS_ENGINE === 'kokoro') {
   545→   545→      void kokoro.load()
   546→   546→      voicePoll.current = setInterval(() => {
   547→   547→        const p = kokoro.loadProgress()
   548→   548→        if (kokoro.isReady() || kokoro.isUnavailable()) {
   549→   549→          store.getState().setBootNote('')
   550→   550→          if (voicePoll.current) clearInterval(voicePoll.current)
   551→   551→          voicePoll.current = null
   552→   552→        } else if (p > 0 && p < 1) {
   553→   553→          store.getState().setBootNote(`voice ${Math.round(p * 100)}%`)
   554→   554→        }
   555→   555→      }, 200)
   556→   556→    }
   557→   557→
   558→   558→    // No boot screen any more — the initializing screen was the last gate, and
   559→   559→    // the interface lands the moment the click lands. Everything below is the
   560→   560→    // live machine coming up: the warm-up, the mic analyser, the speech probe
   561→   561→    // and the voice loop, in that order because each is cheap enough not to be
   562→   562→    // worth a screen of its own.
   563→   563→    await warming
   564→   564→    store.getState().setConnected(connectedLabels())
   565→   565→
   566→   566→    // The analyser is what makes the reactor pulse with your voice. It needs a
   567→   567→    // getUserMedia stream; speech recognition does not, and gets its own. So a
   568→   568→    // failure here costs the animation and nothing else — saying "voice input
   569→   569→    // is unavailable" was both alarming and untrue.
   570→   570→    try {
   571→   571→      await startAnalyser()
   572→   572→    } catch {
   573→   573→      console.warn(
   574→   574→        '[jarvis] no microphone stream — the reactor will not pulse with your ' +
   575→   575→          'voice. Speech recognition is unaffected.',
   576→   576→      )
   577→   577→    }
   578→   578→
   579→   579→    // Ask the bridge which speech engines exist before the loop starts, so the
   580→   580→    // first turn already uses ElevenLabs when a key is present and the browser
   581→   581→    // fallback when it is not — no flag, no reload.
   582→   582→    await probeCapabilities()
   583→   583→
   584→   584→    // Only now can the HUD name the voice honestly: the label depends on the
   585→   585→    // probe (neural catalogue vs system inventory). Reading it before the
   586→   586→    // probe landed showed "default" for the whole session.
   587→   587→    store.getState().setVoice(currentVoiceName())
   588→   588→
   589→   589→    // One voice loop, started once, running until the page closes.
   590→   590→    voice.current = await startVoice({
   591→   591→      mode,
   592→   592→      onWake,
   593→   593→      onSpeechStart,
   594→   594→      onPartial,
   595→   595→      onUtterance,
   596→   596→      onError: onVoiceError,
   597→   597→    })
   598→   598→
   599→   599→    // The loop is live unless the mic failed on the way up — a denied mic
   600→   600→    // reports itself through onError before this line runs, and that verdict
   601→   601→    // is the one that must stick.
   602→   602→    if (!micDead.current) store.getState().setVoiceLive(true)
   603→   603→
   604→   604→    store.getState().setPhase('dormant')
   605→   605→  }
   606→   606→
   607→   607→  // -- clap to wake ------------------------------------------------------------
   608→   608→
   609→   609→  /**
   610→   610→   * A clap brings her up — from the ignition screen, and from standby.
   611→   611→   *
   612→   612→   * Two places it can land, and both are "the machine is idle and dark": while
   613→   613→   * the initializing screen is showing (the click alternative), and once the
   614→   614→   * HUD is standing by (the "hey sofia" alternative). Any busier than that —
   615→   615→   * waking, listening, thinking, speaking — and a clap is either redundant or
   616→   616→   * a stray interruption, so the listener is stood down instead.
   617→   617→   *
   618→   618→   * The microphone stream is shared with the voice loop, so arming here can
   619→   619→   * never steal the assistant's hearing; it is torn down on every phase change
   620→   620→   * and re-armed when the machine goes idle again.
   621→   621→   *
   622→   622→   * clapLive tracks whether the detector actually exists — a pane that blocks
   623→   623→   * getUserMedia gets an honest hint instead of a promise a clap can't keep.
   624→   624→   */
   625→   625→  useEffect(() => {
   626→   626→    if (phase !== 'offline' && phase !== 'dormant') return
   627→   627→    let live: { stop: () => void } | null = null
   628→   628→    let gone = false
   629→   629→    void listenForClap(() => {
   630→   630→      if (gone) return
   631→   631→      const now = store.getState().phase
   632→   632→      if (now === 'offline') {
   633→   633→        void powerOn()
   634→   634→      } else if (now === 'dormant') {
   635→   635→        onWake('')
   636→   636→      }
   637→   637→    }).then((l) => {
   638→   638→      if (gone) {
   639→   639→        l?.stop()
   640→   640→      } else {
   641→   641→        live = l
   642→   642→        if (l) store.getState().setClapLive(true)
   643→   643→      }
   644→   644→    })
   645→   645→    return () => {
   646→   646→      gone = true
   647→   647→      live?.stop()
   648→   648→      store.getState().setClapLive(false)
   649→   649→    }
   650→   650→    // eslint-disable-next-line react-hooks/exhaustive-deps
   651→   651→  }, [phase])
   652→   652→
   653→   653→  // -- level pump + keys ----------------------------------------------------
   654→   654→
   655→   655→  useEffect(() => {
   656→   656→    let raf = 0
   657→   657→
   658→   658→    const pump = () => {
   659→   659→      const st = store.getState()
   660→   660→      // While speaking, follow JARVIS's own output rather than the mic, so the
   661→   661→      // orb lip-syncs instead of reacting to room noise.
   662→   662→      const lvl =
   663→   663→        st.phase === 'speaking' && speaker.current
   664→   664→          ? speaker.current.level()
   665→   665→          : micLevel()
   666→   666→      st.setLevel(lvl)
   667→   667→      raf = requestAnimationFrame(pump)
   668→   668→    }
   669→   669→    pump()
   670→   670→
   671→   671→    const onKey = (e: KeyboardEvent) => {
   672→   672→      const tag = (e.target as HTMLElement)?.tagName
   673→   673→      if (tag === 'INPUT' || tag === 'TEXTAREA') return
   674→   674→
   675→   675→      // V auditions the next British voice installed on this machine. Which
   676→   676→      // ones exist varies per Mac, so hearing them beats trusting a ranking.
   677→   677→      // Bare V only — ⌘V and ⌃V are paste, and swallowing those was rude.
   678→   678→      if (
   679→   679→        e.key === 'v' &&
   680→   680→        !e.repeat &&
   681→   681→        !e.metaKey &&
   682→   682→        !e.ctrlKey &&
   683→   683→        !e.altKey
   684→   684→      ) {
   685→   685→        e.preventDefault()
   686→   686→        const name = cycleVoice()
   687→   687→        store.getState().setVoice(name)
   688→   688→        silence()
   689→   689→        const demo = createSpeaker()
   690→   690→        speaker.current = demo
   691→   691→        demo.say(personaById(store.getState().persona).fillers.voiceSet)
   692→   692→        void demo.end()
   693→   693→        return
   694→   694→      }
   695→   695→
   696→   696→      // G puts the camera on and starts tracking hands. Off by default and
   697→   697→      // never implicit: a webcam that turns itself on because an interface
   698→   698→      // thought it might be useful is not a trade anyone agreed to.
   699→   699→      if (e.key === 'g' && !e.repeat && !e.metaKey && !e.ctrlKey && !e.altKey) {
   700→   700→        e.preventDefault()
   701→   701→        const on = store.getState().gestures
   702→   702→        if (on) {
   703→   703→          hands.disableHands()
   704→   704→          store.getState().setGestures(false)
   705→   705→        } else {
   706→   706→          store.getState().setError(null)
   707→   707→          void hands
   708→   708→            .enableHands()
   709→   709→            .then(() => store.getState().setGestures(true))
   710→   710→            .catch((err: Error) => {
   711→   711→              store.getState().setGestures(false)
   712→   712→              store
   713→   713→                .getState()
   714→   714→                .setError(
   715→   715→                  err?.name === 'NotAllowedError'
   716→   716→                    ? 'Camera access denied — gesture control is unavailable.'
   717→   717→                    : `Gesture control failed to start: ${err?.message ?? err}`,
   718→   718→                )
   719→   719→            })
   720→   720→        }
   721→   721→        return
   722→   722→      }
   723→   723→
   724→   724→      // T speaks a fixed line, bypassing the wake word, the recogniser and the
   725→   725→      // model entirely. When "I can't hear him" is the report, this is the one
   726→   726→      // keypress that separates a broken voice engine from a broken voice loop
   727→   727→      // — and it prints the verdict rather than making you infer it.
   728→   728→      if (e.key === 't' && !e.repeat && !e.metaKey && !e.ctrlKey && !e.altKey) {
   729→   729→        e.preventDefault()
   730→   730→        silence()
   731→   731→        const t = createSpeaker()
   732→   732→        speaker.current = t
   733→   733→        t.say('Audio test. If you can hear this, speech output is working, sir.')
   734→   734→        void t.end().then(() => {
   735→   735→          const d = (window as unknown as Record<string, Record<string, unknown>>).__tts
   736→   736→          console.info('[jarvis] audio test →', d)
   737→   737→          if (d && d.started === 0 && d.rescued === 0) {
   738→   738→            store.getState().setError(
   739→   739→              `No sound produced. engine=${d.engine} voice=${d.voice} error=${d.lastError || 'none'}`,
   740→   740→            )
   741→   741→          }
   742→   742→        })
   743→   743→        return
   744→   744→      }
   745→   745→
   746→   746→      // Escape stands the whole thing down — the one thing the old build had
   747→   747→      // no key for at all.
   748→   748→      if (e.key === 'Escape') {
   749→   749→        e.preventDefault()
   750→   750→        if (store.getState().phase !== 'offline') goDormant()
   751→   751→        return
   752→   752→      }
   753→   753→
   754→   754→      // Space starts a turn without the wake word. Worth using while filming so
   755→   755→      // a missed wake word doesn't cost a take.
   756→   756→      if (e.code !== 'Space' || e.repeat) return
   757→   757→      e.preventDefault()
   758→   758→
   759→   759→      const phase = store.getState().phase
   760→   760→      if (phase === 'offline') {
   761→   761→        void powerOn()
   762→   762→      } else if (phase === 'boot') {
   763→   763→        /* ignore — the boot sequence owns the phase until it finishes */
   764→   764→      } else if (
   765→   765→        phase === 'thinking' ||
   766→   766→        phase === 'tooling' ||
   767→   767→        phase === 'speaking'
   768→   768→      ) {
   769→   769→        onSpeechStart()
   770→   770→        listen(AWAIT_SPEECH_MS)
   771→   771→      } else {
   772→   772→        onWake('')
   773→   773→      }
   774→   774→    }
   775→   775→    window.addEventListener('keydown', onKey)
   776→   776→    window.addEventListener('jarvis:say', onSay as EventListener)
   777→   777→
   778→   778→    return () => {
   779→   779→      cancelAnimationFrame(raf)
   780→   780→      window.removeEventListener('keydown', onKey)
   781→   781→      window.removeEventListener('jarvis:say', onSay as EventListener)
   782→   782→      clearIdle()
   783→   783→      if (voicePoll.current) clearInterval(voicePoll.current)
   784→   784→      voice.current?.stop()
   785→   785→      speaker.current?.cancel()
   786→   786→      // The camera must not outlive the page that turned it on.
   787→   787→      hands.disableHands()
   788→   788→    }
   789→   789→    // eslint-disable-next-line react-hooks/exhaustive-deps
   790→   790→  }, [])
   791→   791→
   792→   792→  return (
   793→   793→    <>
   794→   794→      <Scene />
   795→   795→      <Hud />
   796→   796→      <Diagnostics />
   797→   797→      <CommandLine />
   798→   798→      <Ignition onStart={() => void powerOn()} />
   799→   799→    </>
   800→   800→  )
   801→   801→}
