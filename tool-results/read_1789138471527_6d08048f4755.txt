    98→function samplePath(raw: [number, number][], radius = 22, step = 4): P[] {
    99→  const pts = raw.map(([x, y]) => ({ x, y }));
   100→  if (pts.length < 2) return pts;
   101→  const out: P[] = [];
   102→  const pushLine = (a: P, b: P) => {
   103→    const len = Math.hypot(b.x - a.x, b.y - a.y);
   104→    const n = Math.max(1, Math.ceil(len / step));
   105→    for (let i = 0; i < n; i++) out.push({ x: a.x + (b.x - a.x) * (i / n), y: a.y + (b.y - a.y) * (i / n) });
   106→  };
   107→  const pushQuad = (a: P, c: P, b: P) => {
   108→    const N = 10;
   109→    for (let i = 0; i < N; i++) {
   110→      const t = i / N, u = 1 - t;
   111→      out.push({ x: u * u * a.x + 2 * u * t * c.x + t * t * b.x, y: u * u * a.y + 2 * u * t * c.y + t * t * b.y });
   112→    }
   113→  };
   114→  let cur = pts[0];
   115→  for (let i = 1; i < pts.length - 1; i++) {
   116→    const p = pts[i], prev = pts[i - 1], next = pts[i + 1];
   117→    const lin = Math.hypot(p.x - prev.x, p.y - prev.y);
   118→    const lout = Math.hypot(next.x - p.x, next.y - p.y);
   119→    if (lin < 1e-3 || lout < 1e-3) continue;
   120→    const r = Math.min(radius, lin / 2, lout / 2);
   121→    const din = { x: (p.x - prev.x) / lin, y: (p.y - prev.y) / lin };
   122→    const dout = { x: (next.x - p.x) / lout, y: (next.y - p.y) / lout };
   123→    const pin = { x: p.x - din.x * r, y: p.y - din.y * r };
   124→    const pout = { x: p.x + dout.x * r, y: p.y + dout.y * r };
   125→    pushLine(cur, pin);
   126→    pushQuad(pin, p, pout);
   127→    cur = pout;
   128→  }
   129→  pushLine(cur, pts[pts.length - 1]);
   130→  out.push(pts[pts.length - 1]);
   131→  return out;
   132→}

   134→function sprite(r: number, g: number, b: number, size = 64): HTMLCanvasElement {
   135→  const c = document.createElement("canvas");
   136→  c.width = c.height = size;
   137→  const x = c.getContext("2d")!;
   138→  const h = size / 2;
   139→  const grd = x.createRadialGradient(h, h, 0, h, h, h);
   140→  grd.addColorStop(0, "rgba(255,255,255,1)");
   141→  grd.addColorStop(0.18, `rgba(${r},${g},${b},0.95)`);
   142→  grd.addColorStop(0.45, `rgba(${r},${g},${b},0.35)`);
   143→  grd.addColorStop(1, `rgba(${r},${g},${b},0)`);
   144→  x.fillStyle = grd;
   145→  x.fillRect(0, 0, size, size);
   146→  return c;
   147→}

   151→/**
   152→ * Collision-aware vertical column layout engine.
   153→ * Computes balanced Y coordinates for nodes in a column respecting their bounding box heights,
   154→ * applying relaxation passes to eliminate any overlap.
   155→ */
   156→function layoutColumn(nodesInCol: FlowNode[], centerY = 440, minGap = 24) {
   157→  if (nodesInCol.length === 0) return;
   158→  if (nodesInCol.length === 1) {
   159→    nodesInCol[0].y = centerY;
   160→    return;
   161→  }
   162→  const totalH = nodesInCol.reduce((sum, n) => sum + n.h, 0) + (nodesInCol.length - 1) * minGap;
   163→  let curY = centerY - totalH / 2;
   164→  for (const n of nodesInCol) {
   165→    n.y = Math.round(curY + n.h / 2);
   166→    curY += n.h + minGap;
   167→  }
   168→  // Boundary safeguards & relaxation passes
   169→  for (let iter = 0; iter < 4; iter++) {
   170→    for (let i = 0; i < nodesInCol.length - 1; i++) {
   171→      const a = nodesInCol[i];
   172→      const b = nodesInCol[i + 1];
   173→      const reqDist = (a.h + b.h) / 2 + minGap;
   174→      const actualDist = b.y - a.y;
   175→      if (actualDist < reqDist) {
   176→        const overlap = (reqDist - actualDist) / 2;
   177→        a.y = Math.max(120, Math.round(a.y - overlap));
   178→        b.y = Math.min(780, Math.round(b.y + overlap));
   179→      }
   180→    }
   181→  }
   182→}

   187→function connectNodes(
   188→  fromNode: FlowNode,
   189→  toNode: FlowNode,
   190→  id: string,
   191→  style: EdgeStyle,
   192→  relationship: GraphRelationship,
   193→  state: GraphNodeState,
   194→  activity?: string
   195→): FlowEdge {
   196→  const x1 = fromNode.x + fromNode.w / 2;
   197→  const y1 = fromNode.y;
   198→  const x2 = toNode.x - toNode.w / 2;
   199→  const y2 = toNode.y;
   200→
   201→  let pts: [number, number][];
   202→  if (Math.abs(y1 - y2) < 4) {
   203→    pts = [[x1, y1], [x2, y2]];
   204→  } else {
   205→    const xMid = Math.round((x1 + x2) / 2);
   206→    pts = [[x1, y1], [xMid, y1], [xMid, y2], [x2, y2]];
   207→  }
   208→
   209→  return {
   210→    id,
   211→    from: fromNode.id,
   212→    to: toNode.id,
   213→    pts,
   214→    style,
   215→    arrow: true,
   216→    relationship,
   217→    state,
   218→    activity,
   219→  };
   220→}

   850→  pointAt(p: Sampled, d: number): P {

   853→    if (d >= p.len) return p.pts[p.pts.length - 1];

   855→      hi = cum.length - 1;

   861→    const seg = cum[hi] - cum[lo] || 1;
   862→    const t = (d - cum[lo]) / seg;
   863→    return { x: p.pts[lo].x + (p.pts[hi].x - p.pts[lo].x) * t, y: p.pts[lo].y + (p.pts[hi].y - p.pts[lo].y) * t };

  1028→              ? "#fca5a5"
  1029→              : pk.tone === "emerald"

  1035→      if (pk.d < path.len) continue;
  1036→

  1038→      const end = path.pts[path.pts.length - 1];

  1042→      const activeNext = next?.filter((idx) => this.paths[idx]?.edge.state === "active" || this.paths[idx]?.edge.state === "blocked");

  1051→          nextEdge.style === "fire"
  1052→            ? "fire"
  1053→            : nextEdge.style === "rose"
  1054→            ? "rose"
  1055→            : nextEdge.style === "amber"
  1056→            ? "amber"
  1057→            : nextEdge.style === "emerald"
  1058→            ? "emerald"
  1059→            : "cyan";
  1060→      } else {
  1061→        this.packets.splice(i, 1);
  1062→      }
  1063→    }
