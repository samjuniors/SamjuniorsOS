   136→   136→/**
   137→   137→ * Independent re-derivation of the pre-M4-C policy (same mechanism family as
   138→   138→ * the m4c suite, kept so fallback equality can be asserted without trusting
   139→   139→ * the implementation under test).
   140→   140→ */
   141→   141→function expectedPreM4cOrder(records: EvalMem[]): string[] {
   142→   142→  const byType = new Map<string, EvalMem[]>();
   143→   143→  for (const r of records) {
   144→   144→    byType.set(r.memoryType, [...(byType.get(r.memoryType) ?? []), r]);
   145→   145→  }
   146→   146→  for (const list of byType.values()) {
   147→   147→    list.sort((a, b) => {
   148→   148→      if (b.confidence !== a.confidence) return b.confidence - a.confidence;
   149→   149→      const ta = new Date(a.updatedAt).getTime();
   150→   150→      const tb = new Date(b.updatedAt).getTime();
   151→   151→      if (tb !== ta) return tb - ta;
   152→   152→      return a.id < b.id ? -1 : a.id > b.id ? 1 : 0;
   153→   153→    });
   154→   154→  }
   155→   155→  const order: string[] = [];
   156→   156→  for (let round = 0; order.length < 20; round++) {
   157→   157→    let advanced = false;
   158→   158→    for (const t of SOPHIA_MEMORY_TYPES) {
   159→   159→      if (order.length >= 20) break;
   160→   160→      const list = byType.get(t);
   161→   161→      if (list && list.length > round) {
   162→   162→        order.push(list[round].id);
   163→   163→        advanced = true;
   164→   164→      }
   165→   165→    }
   166→   166→    if (!advanced) break;
   167→   167→  }
   168→   168→  return order;
   169→   169→}

   172→   172→function lexicalScore(m: EvalMem, query: string): number {
   173→   173→  const queryTokens = new Set(extractTokens(query));
   174→   174→  const contentTokens = new Set(extractTokens(m.content));
   175→   175→  let score = 0;
   176→   176→  for (const t of queryTokens) if (contentTokens.has(t)) score++;
   177→   177→  return score;
   178→   178→}

   197→   197→function measureQuery(

   199→   199→  label: string,

   207→   207→    gold.filter((g) => ranked.slice(0, k).includes(g)).length / gold.length;

   214→   214→    queryId,
   215→   215→    label,
   216→   216→    query,

   438→   438→function mulberry32(seed: number): () => number {
   439→   439→  let a = seed >>> 0;
   440→   440→  return () => {
   441→   441→    a = (a + 0x6d2b79f5) | 0;
   442→   442→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
   443→   443→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
   444→   444→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
   445→   445→  };
   446→   446→}
