   125→function getDevAuthHeaders(): Record<string, string> {
   126→  const secret = process.env.NEXT_PUBLIC_SAMJUNIORS_DEV_SECRET || "samjuniors_dev_secret_local";
   127→  if (typeof window !== "undefined") {
   128→    try {
   129→      document.cookie = `samjuniors-dev-as=founder; path=/; SameSite=Lax`;
   130→      document.cookie = `samjuniors-dev-secret=${secret}; path=/; SameSite=Lax`;
   131→    } catch {
   132→      /* ignore */
   133→    }
   134→  }
   135→  return {
   136→    "x-samjuniors-dev-as": "founder",
   137→    "x-samjuniors-dev-secret": secret,
   138→  };
   139→}

   141→async function jsonFetch<T>(url: string, init?: RequestInit): Promise<T> {
   142→  const devHeaders = getDevAuthHeaders();
   143→  const res = await fetch(url, {
   144→    ...init,
   145→    headers: {
   146→      ...devHeaders,
   147→      ...init?.headers,
   148→    },
   149→  });
   150→  let body: any = null;
   151→  try { body = await res.json(); } catch { /* non-JSON error body */ }
   152→  if (!res.ok) {
   153→    throw new Error(body?.error || `Request failed (${res.status} ${url})`);
   154→  }
   155→  return body as T;
   156→}

   489→export function looksLikeDirective(text: string): boolean {
   490→  const clean = text.trim().toLowerCase();
   491→  const patterns = [
   492→    /\b(research|investigate|analyze|evaluate|audit|model|draft|design|create|build|prepare|synthesize|simulate|spec|spec out)\b.+\b(and (give|provide|recommend|write|report|present|model)|recommendation|proposal|prd|spec|plan|deliverable|architecture|breakdown|forecast|strategy)\b/i,
   493→    /^(research|investigate|analyze|evaluate|audit|model|draft|design|create|build|prepare|synthesize|simulate)\s+(the|our|a|an|all)\s+/i,
   494→    /\b(give me a recommendation|give me a plan|create a prd|draft a prd|model the unit economics|model our pricing|audit compute burn|conduct a research|run an analysis|synthesize a proposal)\b/i,
   495→    /^(execute|orchestrate|launch task|run task|start initiative)\b/i,
   496→    /^(prepare|research|analyze|build|create|model|draft|evaluate|audit|generate)\b/i,
   497→    /^(work|start)\b/i,
   498→  ];
   499→  return patterns.some((p) => p.test(clean));

   511→/** Dispatch a founder directive to the real orchestration path. While the council
   512→ *  executes (~minutes), polls the durable agent-run store so the OS surfaces and
   513→ *  the operating graph visualize REAL step-by-step progress — never simulated. */
   514→export async function dispatchDirective(directive: string, opts?: { agents?: string[] }): Promise<OrchestrationRun> {
   515→  const poll = setInterval(() => {
   516→    syncFromServer({ quiet: true }).catch(() => { /* transient poll failure — next tick retries */ });
   517→  }, 3000);
   518→  try {
   519→    const run = await orchestrate(directive, opts?.agents);
   520→    clearInterval(poll);
   521→    applyOrchestrationRun(run);
   522→    await syncFromServer();
   523→    if (typeof window !== "undefined") {
   524→      window.dispatchEvent(new CustomEvent("samjuniors:graph-refresh"));
   525→    }
   526→    return run;
   527→  } catch (err) {
   528→    clearInterval(poll);
   529→    // Sync whatever steps actually persisted server-side before the failure,
   530→    // then surface the honest error — no fake local success.
   531→    await syncFromServer().catch(() => undefined);
   532→    if (typeof window !== "undefined") {
   533→      window.dispatchEvent(new CustomEvent("samjuniors:graph-refresh"));
   534→    }
   535→    throw err instanceof Error ? err : new Error(String(err));
   536→  }
   537→}

   539→/** One-line real summary of a completed run (for Sophia's reply). */
   540→export function summarizeRun(run: OrchestrationRun): string {
   541→  const verified = run.verificationResult?.isCompliant;
   542→  const outcome = run.executiveResult?.executionOutcome ?? run.status;
   543→  const parts: string[] = [];
   544→  parts.push(run.liveAi === false ? "Council executed in safe unconfigured mode." : `Directive executed: ${run.title}.`);
   545→  if (verified === true) parts.push("Constitutional verification passed.");
   546→  if (verified === false) parts.push(`Verification rejected: ${(run.verificationResult?.checksFailed ?? ["unknown"]).join("; ")}.`);
   547→  if (run.executionSummary) parts.push(`${run.executionSummary.totalTasksExecuted} protocol steps · ${run.executionSummary.agentsInvoked.length} agents · ${run.deliverables.length} deliverables.`);
   548→  if (run.executiveResult?.founderDecision?.required) parts.push(`A decision is waiting for you: ${run.executiveResult.founderDecision.title}.`);
   549→  parts.push(`Outcome: ${outcome}.`);
   550→  return parts.join(" ");
   551→}

   555→function hashId(prefix: string, text: string): string {
   556→  let h = 5381;
   557→  for (let i = 0; i < text.length; i++) h = ((h << 5) + h + text.charCodeAt(i)) | 0;
   558→  return `${prefix}-${(h >>> 0).toString(36)}`;
   559→}

   561→const titleOf = (directive: string) => (directive.length > 60 ? `${directive.slice(0, 58)}…` : directive);
   562→
   563→/** Groups durable agent-run records by directive and derives honest workstreams:
   564→ *  stage from the furthest real protocol step, owner from the latest specialist
   565→ *  that actually executed, and state from real recency. */
   566→export function workstreamsFromRuns(runs: AgentRunRecord[], dismissed: string[] = []): Workstream[] {
   567→  const groups = new Map<string, AgentRunRecord[]>();
   568→  for (const r of runs) {
   569→    const key = r.directive?.trim();
   570→    if (!key) continue;
   571→    (groups.get(key) ?? groups.set(key, []).get(key)!).push(r);
   572→  }
   573→
   574→  const work: Workstream[] = [];
   575→  for (const [directive, group] of groups) {
   576→    const sorted = [...group].sort((a, b) => a.timestamp.localeCompare(b.timestamp));
   577→    const latest = sorted[sorted.length - 1];
   578→    const id = hashId("srv", directive);
   579→    if (dismissed.includes(id)) continue;
   580→
   581→    const specialistRuns = sorted.filter((r) => r.agentId !== "coo");
   582→    const latestSpecialist = specialistRuns[specialistRuns.length - 1];
   583→    const owner = latestSpecialist ? toUiAgentId(latestSpecialist.agentId) : "sophia";
   584→
   585→    const hasReport = sorted.some((r) => r.protocolStep === "report");
   586→    const lastStep = latest.protocolStep;
   587→    const age = Date.now() - new Date(latest.timestamp).getTime();
   588→    const inFlight = !hasReport && age < IN_FLIGHT_WINDOW_MS;
   589→
   590→    const stage: Stage = hasReport ? "done" : STEP_STAGE[lastStep] ?? "discovery";
   591→    const state: Workstream["state"] = hasReport ? "done" : inFlight ? "active" : "paused";
   592→
   593→    work.push({
   594→      id,
   595→      title: titleOf(directive),
   596→      owner,
   597→      stage,
   598→      state,
   599→      note: `${sorted.length} real protocol steps · last: ${latest.taskTitle}`,
   600→      at: new Date(latest.timestamp).getTime(),
   601→      origin: "server",
   602→      directive,
   603→    });
   604→    if (work.length >= 5) break; // most recent directives only — groups arrive in run-recency order
   605→  }
   606→  return work;
   607→}

   609→/** Real agent runtime state from durable run records. Only agents currently
   610→ *  participating in an in-flight directive show "working"; everything else
   611→ *  decays to ready. Founder's local offline choices are preserved by osStore. */
   612→export function agentStatesFromRuns(runs: AgentRunRecord[], work: Workstream[]): Record<string, { state: Agent["state"]; current?: string }> {
   613→  const out: Record<string, { state: Agent["state"]; current?: string }> = {};
   614→  const active = work.some((w) => w.state === "active");
   615→  if (!active) return out;
   616→
   617→  for (const r of runs) {
   618→    const uiId = toUiAgentId(r.agentId);
   619→    if (out[uiId]?.state === "working") continue;
   620→    const age = Date.now() - new Date(r.timestamp).getTime();
   621→    if (age < IN_FLIGHT_WINDOW_MS) {
   622→      out[uiId] = { state: "working", current: r.taskTitle };
   623→    }
   624→  }
   625→  return out;
   626→}

   630→let syncing: Promise<void> | null = null;

   632→/** Pulls roster + runs + approvals + the company Activity projection from the
   633→ *  server and feeds the OS store. Server-origin state REPLACES local
   634→ *  projections; local founder records (notes, focus, offline choices) are
   635→ *  preserved. The Activity projection (Phase 4.4C) is the AUTHORITATIVE
   636→ *  company Activity source — the client os.log remains only an ambient
   637→ *  browser-session supplement. */
   638→export async function syncFromServer(opts?: { quiet?: boolean }): Promise<void> {
   639→  if (syncing) return syncing;
   640→  syncing = (async () => {
   641→    try {
   642→      const [roster, runs, approvals, activity, epistemic] = await Promise.all([
   643→        fetchRoster(),
   644→        fetchRuns(),
   645→        fetchApprovals(),
   646→        fetchActivity().catch((err) => {
   647→          // Activity is additive to the core sync: an activity-specific failure
   648→          // must not take down roster/runs/approvals sync. Honest signal only.
   649→          os.log(`Company activity unavailable: ${err instanceof Error ? err.message : String(err)}`);
   650→          return null;
   651→        }),
   652→        fetchEpistemicBoard().catch((err) => {
   653→          // Phase 4.4E: the epistemic board is likewise additive — a
   654→          // board-specific failure degrades honestly without breaking the
   655→          // core sync. Governed epistemic state is never faked locally.
   656→          os.log(`Epistemic board unavailable: ${err instanceof Error ? err.message : String(err)}`);
   657→          return null;
   658→        }),
   659→      ]);
   660→      applyServerReadModel({ roster, runs, approvals });
   661→      if (activity) os.setServerActivity(activity.map(serverActivityToEvent));
   662→      if (epistemic) os.setServerEpistemic(epistemic);
   663→    } catch (err) {
   664→      if (!opts?.quiet) {
   665→        // Honest signal — never fake a successful sync.
   666→        os.log(`Server sync unavailable: ${err instanceof Error ? err.message : String(err)}`);
   667→      }
   668→      throw err;
   669→    } finally {
   670→      syncing = null;
   671→    }
   672→  })();
   673→  return syncing;
   674→}

   676→function applyServerReadModel(input: { roster: ServerAgentDef[]; runs: AgentRunRecord[]; approvals: ApprovalRecord[] }) {
   677→  const { roster, runs, approvals } = input;
   678→  const dismissed = os.getDismissed();
   679→  const work = workstreamsFromRuns(runs, dismissed);
   680→  const agentStates = agentStatesFromRuns(runs, work);
   681→
   682→  // Agents: authoritative identity (existence/name/role) from the server roster,
   683→  // presentation copy from the UI's own constants, runtime state from real runs.
   684→  const agents: Agent[] = roster.map((def) => {
   685→    const uiId = toUiAgentId(def.id);
   686→    const presentation = presentationFor(uiId);
   687→    const rt = agentStates[uiId];
   688→    return {
   689→      ...presentation,
   690→      id: uiId,
   691→      name: def.name,
   692→      role: def.role,
   693→      state: rt?.state ?? "ready",
   694→      current: rt?.current,
   695→    };
   696→  });
   697→
   698→  // Decisions + attention from REAL pending approval gate records.
   699→  const decisions: Decision[] = approvals.map((a) => ({
   700→    id: a.id,
   701→    title: a.actionName,
   702→    context: a.target?.summary || a.notes || `${a.classification} · ${a.workflowInstanceId}`,
   703→    options: ["Approve", "Reject"],
   704→    raisedBy: SERVER_TO_UI[a.employeeRole] ?? a.employeeRole,
   705→    at: new Date(a.requestedAt).getTime(),
   706→    status: "open",
   707→    approvalId: a.id,
   708→  }));
   709→
   710→  const attention: AttentionItem[] = approvals.map((a) => ({
   711→    id: `att-${a.id}`,
   712→    kind: "decision",
   713→    title: a.actionName,
   714→    detail: a.target?.summary || a.notes,
   715→    from: SERVER_TO_UI[a.employeeRole] ?? a.employeeRole,
   716→    at: new Date(a.requestedAt).getTime(),
   717→    decisionId: a.id,
   718→    server: true,
   719→  }));
   720→
   721→  os.applyServerState({ agents, work, decisions, attention });
   722→}
