    67→/** Consume the SSE response into parsed frames (mirrors the browser client). */
    68→async function readSseFrames(res: Response): Promise<any[]> {
    69→  const frames: any[] = [];
    70→  if (!res.body) return frames;
    71→  const reader = (res.body as ReadableStream<Uint8Array>).getReader();
    72→  const decoder = new TextDecoder();
    73→  let buf = '';
    74→  for (;;) {
    75→    const { done, value } = await reader.read();
    76→    if (done) break;
    77→    buf += decoder.decode(value, { stream: true });
    78→    let nl: number;
    79→    while ((nl = buf.indexOf('\n')) !== -1) {
    80→      const line = buf.slice(0, nl).replace(/\r$/, '');
    81→      buf = buf.slice(nl + 1);
    82→      if (!line.startsWith('data:')) continue;
    83→      const data = line.slice(5).trim();
    84→      if (!data) continue;
    85→      try {
    86→        frames.push(JSON.parse(data));
    87→      } catch {
    88→        /* torn line — ignore */
    89→      }
    90→    }
    91→  }
    92→  return frames;
    93→}

    95→async function runAsk(body: Record<string, unknown>, founderId: string) {
    96→  const res = await sofiaAskPostHandler(askRequest(body, founderId));
    97→  const frames = await readSseFrames(res);
    98→  return {
    99→    status: res.status,
   100→    frames,
   101→    ready: frames.find((f) => f.type === 'ready'),
   102→    done: frames.find((f) => f.type === 'done'),
   103→    error: frames.find((f) => f.type === 'error'),
   104→  };
   105→}

   107→function runChild(args: string[], env: Record<string, string> = {}): Promise<any> {
   108→  return new Promise((resolve, reject) => {
   109→    // @ts-ignore — Bun global exists when the suite runs under bun
   110→    const proc = Bun.spawn(['bun', 'tests/sophia/m3-conversation-child.ts', ...args], {
   111→      stdout: 'pipe',
   112→      stderr: 'pipe',
   113→      env: { ...process.env, ...env },
   114→    });
   115→    new Response(proc.stdout)
   116→      .text()
   117→      .then((out) =>
   118→        proc.exited.then((code) => {
   119→          if (code !== 0) {
   120→            return new Response(proc.stderr)
   121→              .text()
   122→              .then((errText) =>
   123→                reject(
   124→                  new Error(
   125→                    `child (${args.join(' ')}) exited ${code}; stderr: ${errText.slice(0, 500)}`
   126→                  )
   127→                )
   128→            );
   129→          }
   130→          try {
   131→            // The child's stdout is a single JSON line by contract; be
   132→            // tolerant of any stray library output by taking the LAST line
   133→            // that parses as JSON.
   134→            const lines = out.trim().split('\n').filter(Boolean);
   135→            let parsed: any = null;
   136→            for (const line of lines) {
   137→              try {
   138→                const candidate = JSON.parse(line);
   139→                if (candidate && typeof candidate === 'object') parsed = candidate;
   140→              } catch {
   141→                /* not a JSON line — skip */
   142→              }
   143→            }
   144→            if (!parsed) throw new Error('no JSON line found');
   145→            resolve(parsed);
   146→          } catch {
   147→            reject(
   148→              new Error(
   149→                `child (${args.join(' ')}) emitted non-JSON output: ${out.trim().slice(0, 300)}`
   150→              )
   151→            );
   152→          }
   153→        })
   154→      )
   155→      .catch(reject);
   156→  });
   157→}
