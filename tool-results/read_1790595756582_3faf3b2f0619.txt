   212→   212→/**
   213→   213→ * Computes the SHA-256 content hash stored in the `CompanyKnowledge.hash`
   214→   214→ * column (M2: the ingestion path records it on every explicit write).
   215→   215→ *
   216→   216→ * Exact guarantee: it fingerprints ONLY the `content` field.
   217→   217→ *  - Identical content ⇒ identical hash (deterministic, byte-for-byte).
   218→   218→ *  - Changed content ⇒ different hash (SHA-256 collision resistance).
   219→   219→ *  - Title/category/version/metadata changes alone do NOT change the hash.
   220→   220→ *
   221→   221→ * See isKnowledgeContentUnchanged() for the comparison contract.
   222→   222→ */
   223→   223→export function computeKnowledgeContentHash(content: string): string {
   224→   224→  return createHash('sha256').update(content || '', 'utf-8').digest('hex');
   225→   225→}
   226→   226→
   227→   227→/**
   228→   228→ * Minimal change-detection contract (no vectors/embeddings/FTS): compares a
   229→   229→ * persisted hash against the hash of candidate content.
   230→   230→ *  - equal  ⇒ content unchanged since the hash was recorded
   231→   231→ *  - different ⇒ content changed
   232→   232→ * A missing/blank stored hash never claims "unchanged".
   233→   233→ */
   234→   234→export function isKnowledgeContentUnchanged(
   235→   235→  storedHash: string | null | undefined,
   236→   236→  content: string
   237→   237→): boolean {
   238→   238→  if (!storedHash) return false;
   239→   239→  return storedHash === computeKnowledgeContentHash(content);
   240→   240→}

   242→   242→/**
   243→   243→ * ============================================================================
   244→   244→ * COMPANY KNOWLEDGE STORE — M2: CANONICAL PERSISTENCE ACTIVATED
   245→   245→ * ============================================================================
   246→   246→ *
   247→   247→ * Implements ICompanyKnowledgeStore with the SAME dual-mode persistence
   248→   248→ * pattern as its sibling stores (memory-store, conversation store, etc.):
   249→   249→ *
   250→   250→ *  - LOCAL MODE: in-memory cache (constructor-hydrated from DurableFileStore)
   251→   251→ *    + .data/company_knowledge.json durability + best-effort Prisma dual-write.
   252→   252→ *  - AUTHORITATIVE MODE: Prisma fail-closed (requireAuthoritativeDatabase);
   253→   253→ *    reads and writes go to the CompanyKnowledge table (target: PostgreSQL;
   254→   254→ *    currently the SQLite sandbox port — see the M0 naming note).
   255→   255→ *
   256→   256→ * Seed migration (one-time, bootstrap-only): the 8 canonical documents that
   257→   257→ * previously existed ONLY as code constants are persisted to DurableFileStore
   258→   258→ * on first boot (empty durable collection), and bootstrapped into the Prisma
   259→   259→ * table lazily on first store usage — CREATE-ONLY: a canonical document is
   260→   260→ * seeded only when no row exists for its id; an existing persisted row is
   261→   261→ * NEVER overwritten by the code constants on restart (Founder directive,
   262→   262→ * pre-M3 correction pass). Explicit writes (addKnowledge / setKnowledge)
   263→   263→ * remain the only mutation paths after boot.
   264→   264→ *
   265→   265→ * The `hash` column (SHA-256 of the `content` field ONLY) provides change
   266→   266→ * detection: comparing a stored hash against
   267→   267→ * computeKnowledgeContentHash(candidateContent) distinguishes unchanged
   268→   268→ * content (equal) from changed content (different) — see
   269→   269→ * isKnowledgeContentUnchanged(). It does NOT fingerprint title/category/
   270→   270→ * metadata. A different id with byte-identical content collides on the
   271→   271→ * unique hash index and is treated as a duplicate (seed skipped; explicit
   272→   272→ * writes still dual-write best-effort in local mode).
   273→   273→ *
   274→   274→ * Designed so the target PostgreSQL / Supabase deployment can replace the
   275→   275→ * persistence internals without altering retrieval or business logic.
   276→   276→ */
   277→   277→export class CompanyKnowledgeStore implements ICompanyKnowledgeStore {
   278→   278→  private static instance: CompanyKnowledgeStore | null = null;
   279→   279→
   280→   280→  private knowledgeItems: CompanyKnowledgeItem[] = [...CANONICAL_COMPANY_KNOWLEDGE];
   281→   281→
   282→   282→  /** One-shot guard for the lazy Prisma seed mirror (M2 seed migration). */
   283→   283→  private static prismaSeedPromise: Promise<void> | null = null;
   284→   284→
   285→   285→  private constructor() {
   286→   286→    this.loadFromDurableStorage();
   287→   287→  }
   288→   288→
   289→   289→  public static getInstance(): CompanyKnowledgeStore {
   290→   290→    if (!CompanyKnowledgeStore.instance) {
   291→   291→      CompanyKnowledgeStore.instance = new CompanyKnowledgeStore();
   292→   292→    }
   293→   293→    return CompanyKnowledgeStore.instance;
   294→   294→  }
   295→   295→
   296→   296→  /**
   297→   297→   * Hydrates the in-memory cache from DurableFileStore, merging over the
   298→   298→   * canonical seed documents. On first boot (no durable collection), the
   299→   299→   * canonical seeds are themselves persisted — the M2 one-time seed
   300→   300→   * migration of the 8 canonical documents.
   301→   301→   */
   302→   302→  private loadFromDurableStorage(): void {
   303→   303→    try {
   304→   304→      const persisted = DurableFileStore.getInstance().readCollection<CompanyKnowledgeItem>('company_knowledge');
   305→   305→      const persistedItems = Object.values(persisted);
   306→   306→      if (persistedItems.length === 0) {
   307→   307→        // First boot: persist the canonical seed set (one-time migration).
   308→   308→        const seedRecord: Record<string, CompanyKnowledgeItem> = {};
   309→   309→        for (const item of CANONICAL_COMPANY_KNOWLEDGE) {
   310→   310→          seedRecord[item.id] = item;
   311→   311→        }
   312→   312→        DurableFileStore.getInstance().writeCollection('company_knowledge', seedRecord);
   313→   313→        this.knowledgeItems = [...CANONICAL_COMPANY_KNOWLEDGE];
   314→   314→        return;
   315→   315→      }
   316→   316→      // Merge: canonical seeds first (stable order), persisted items overlay
   317→   317→      // by id (edits win), extras appended.
   318→   318→      const byId = new Map<string, CompanyKnowledgeItem>();
   319→   319→      for (const item of CANONICAL_COMPANY_KNOWLEDGE) byId.set(item.id, item);
   320→   320→      for (const item of persistedItems) byId.set(item.id, item);
   321→   321→      this.knowledgeItems = [...byId.values()];
   322→   322→    } catch {
   323→   323→      // fallback to canonical seeds
   324→   324→    }
   325→   325→  }
   326→   326→
   327→   327→  /**
   328→   328→   * Lazily bootstraps the canonical seed documents into the Prisma table
   329→   329→   * once per process. BOOTSTRAP-ONLY (Founder directive, pre-M3 correction):
   330→   330→   * a canonical document is created ONLY when no row exists for its id — an
   331→   331→   * existing persisted row is preserved as-is, so a restart can never
   332→   332→   * overwrite authoritative persisted knowledge with the code constants.
   333→   333→   * Explicit writes (addKnowledge / setKnowledge) remain the only mutation
   334→   334→   * paths after boot. A unique-hash collision on create means byte-identical
   335→   335→   * content is already persisted under another id — the seed is skipped (the
   336→   336→   * hash column is the duplicate-content guard); any other error propagates
   337→   337→   * (fail-closed in authoritative mode). Local mode: best-effort.
   338→   338→   */
   339→   339→  private ensurePrismaSeeded(): Promise<void> {
   340→   340→    if (!CompanyKnowledgeStore.prismaSeedPromise) {
   341→   341→      CompanyKnowledgeStore.prismaSeedPromise = (async () => {
   342→   342→        try {
   343→   343→          const db = isAuthoritativeMode()
   344→   344→            ? await requireAuthoritativeDatabase()
   345→   345→            : prisma;
   346→   346→          for (const item of CANONICAL_COMPANY_KNOWLEDGE) {
   347→   347→            const existing = await db.companyKnowledge.findUnique({
   348→   348→              where: { id: item.id },
   349→   349→            });
   350→   350→            // Persisted document wins over the code constant — never overwrite.
   351→   351→            if (existing) continue;
   352→   352→            try {
   353→   353→              await db.companyKnowledge.create({
   354→   354→                data: {
   355→   355→                  id: item.id,
   356→   356→                  category: item.category,
   357→   357→                  title: item.title,
   358→   358→                  content: item.content,
   359→   359→                  hash: computeKnowledgeContentHash(item.content),
   360→   360→                  metadata: {
   361→   361→                    documentId: item.documentId,
   362→   362→                    version: item.version,
   363→   363→                    summary: item.summary,
   364→   364→                    tags: item.tags,
   365→   365→                    applicableDepartments: item.applicableDepartments,
   366→   366→                    authorAuthority: item.authorAuthority,
   367→   367→                    lastVerifiedDate: item.lastVerifiedDate,
   368→   368→                    isDurableReference: item.isDurableReference,
   369→   369→                  },
   370→   370→                },
   371→   371→              });
   372→   372→            } catch (createErr: any) {
   373→   373→              const msg = String(createErr?.message || createErr);
   374→   374→              const isUniqueCollision =
   375→   375→                createErr?.code === 'P2002' || /unique constraint/i.test(msg);
   376→   376→              if (!isUniqueCollision) throw createErr;
   377→   377→              // Byte-identical content already persisted under another id:
   378→   378→              // skip seeding a duplicate (unique hash = duplicate guard).
   379→   379→            }
   380→   380→          }
   381→   381→        } catch (err: any) {
   382→   382→          if (isAuthoritativeMode()) throw err;
   383→   383→          // Local mode: best-effort only — the durable file store remains the
   384→   384→          // local-mode primary per ADR 0002.
   385→   385→        }
   386→   386→      })().catch(() => {
   387→   387→        // Do not cache a failed best-effort seed; retry on next usage.

   538→   538→  /**
   539→   539→   * Deterministically queries durable reference information.
   540→   540→   * Attaches epistemic label: 'durable_reference' and provenance metadata.

   542→   542→   * KNOWN AUTHORITY GAP (M3 hardening review — recorded as the NEXT
   543→   543→   * authority-audit item; deliberately NOT fixed in this pass):
   544→   544→   *   This method always scores the in-process `knowledgeItems` cache
   545→   545→   *   (hydrated from DurableFileStore + code seeds at construction, coherent
   546→   546→   *   only for writes made within THIS process). Unlike getAllKnowledge() /
   547→   547→   *   getKnowledgeById(), it does NOT read the fail-closed authoritative
   548→   548→   *   Prisma table in authoritative mode — so knowledge edited or added in
   549→   549→   *   another process (or before a restart) is invisible to RETRIEVAL while
   550→   550→   *   being visible to getAllKnowledge(). Callers are context-assembly hot
   551→   551→   *   paths (SophiaContextAssembler, context-retrieval, context-assembly),
   552→   552→   *   so the smallest safe correction (routing authoritative mode through
   553→   553→   *   getAllKnowledge()) is NOT fully isolated: it adds a per-call DB
   554→   554→   *   dependency and fail-closed 503 propagation to every Sophia turn and
   555→   555→   *   needs its own authoritative-mode verification matrix. Fix requires a

   557→   557→   *   (DATABASE_MODE=local) the gap is latent, not active.
   558→   558→   */
   559→   559→  public async queryKnowledge(params: KnowledgeQueryParams): Promise<RetrievedKnowledgeItem[]> {
   560→   560→    const queryText = (params.queryText || '').toLowerCase();
   561→   561→    const queryTokens = new Set([
   562→   562→      ...extractTokens(queryText),
   563→   563→      ...(params.keywords ? params.keywords.flatMap(extractTokens) : []),
   564→   564→      ...(params.tags ? params.tags.map((t) => t.toLowerCase()) : []),
   565→   565→    ]);
   566→   566→
   567→   567→    const targetRole = params.role;
   568→   568→    const nowIso = new Date().toISOString();
   569→   569→    const results: RetrievedKnowledgeItem[] = [];
   570→   570→
   571→   571→    for (const item of this.knowledgeItems) {
   572→   572→      // Role match check
   573→   573→      const isRoleApplicable = !targetRole || targetRole === 'orchestrator' || targetRole === 'council' || targetRole === 'advisor'
   574→   574→        || item.applicableDepartments.includes(targetRole)
   575→   575→        || item.applicableDepartments.includes('council');
   576→   576→
   577→   577→      const itemTokens = [
   578→   578→        ...extractTokens(item.title),
   579→   579→        ...extractTokens(item.summary),
   580→   580→        ...extractTokens(item.content),
   581→   581→        ...item.tags.flatMap(extractTokens),
   582→   582→        item.documentId.toLowerCase(),
   583→   583→      ];
   584→   584→
   585→   585→      const matchedTerms = itemTokens.filter((t) => queryTokens.has(t));
   586→   586→      const hasTermMatch = matchedTerms.length > 0;
   587→   587→
   588→   588→      // Check category match if provided
   589→   589→      const categoryMatch = params.category && item.category.toLowerCase().includes(params.category.toLowerCase());
   590→   590→
   591→   591→      if (hasTermMatch || categoryMatch) {
   592→   592→        const relevanceScore = (hasTermMatch ? matchedTerms.length * 2 : 0)
   593→   593→          + (categoryMatch ? 3 : 0)
   594→   594→          + (isRoleApplicable ? 1 : 0);
   595→   595→
   596→   596→        results.push({
   597→   597→          knowledgeId: item.id,
   598→   598→          documentId: item.documentId,
   599→   599→          title: item.title,
   600→   600→          category: item.category,
   601→   601→          version: item.version,
   602→   602→          summary: item.summary,
   603→   603→          contentSnippet: item.summary,
   604→   604→          fullContent: item.content,
   605→   605→          applicableDepartments: item.applicableDepartments,
   606→   606→          relevanceScore,
   607→   607→          matchReason: matchedTerms.length > 0
   608→   608→            ? `Matched knowledge terms: ${[...new Set(matchedTerms)].join(', ')}`
   609→   609→            : `Category match (${item.category})`,
   610→   610→          provenance: {
   611→   611→            sourceSystem: 'company_knowledge',
   612→   612→            sourceId: `knowledge:${item.documentId}`,
   613→   613→            sourceTitle: item.title,
   614→   614→            epistemicType: 'durable_reference',
   615→   615→            authority: item.authorAuthority,
   616→   616→            timestamp: item.lastVerifiedDate || nowIso,
   617→   617→            confidence: 'reference_standard',
   618→   618→            notes: `Durable company reference documentation (v${item.version})`,
   619→   619→          },
   620→   620→        });
   621→   621→      }
   622→   622→    }
   623→   623→
   624→   624→    // Sort by relevance score descending
   625→   625→    results.sort((a, b) => b.relevanceScore - a.relevanceScore);
   626→   626→    const limit = params.limit || 5;
   627→   627→    return results.slice(0, limit);
   628→   628→  }
