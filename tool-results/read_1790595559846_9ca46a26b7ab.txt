    16→/**
    17→ * ============================================================================
    18→ * SOPHIA PERSONAL MEMORY STORE (M3 K-2 — Personal Mind)
    19→ * ============================================================================
    20→ * Server-authoritative, founder-scoped personal/interaction memory store for
    21→ * the Sophia Personal Mind (SOPHIA_MEMORY_ARCHITECTURE.md §1).
    22→ *
    23→ * TWO BRAINS, ONE GATE:
    24→ *   This store is the Personal Mind's ONLY persistence boundary. It holds
    25→ *   founder-specific interaction context (preferences, patterns, personal
    26→ *   contextual notes). It is NOT the Company Brain:
    27→ *     - it never stores company facts / strategy / financial truth
    28→ *     - it never feeds CompanyKnowledge, CompanyMemory, or the epistemic
    29→ *       claim/fact pipeline
    30→ *     - it is NEVER an authorization source — personal memories cannot
    31→ *       approve, bypass, or govern anything (the SideEffectAuthorizationGate
    32→ *       and SophiaServerGateway read approval/policy state exclusively from
    33→ *       their own governed stores)
    34→ *     - there is NO promotion path out of this store; a personal memory
    35→ *       stays personal/unverified context forever unless a Founder moves the
    36→ *       underlying fact through the governed epistemic pipeline by hand.
    37→ *
    38→ * AUTHORITY SEMANTICS (four layers — keep these distinct in any claim):
    39→ *
    40→ *   [canonical abstraction]  This class IS the canonical Personal Mind
    41→ *                            memory abstraction. Callers never read/write
    42→ *                            `.data/sophia_memories.json` or the Prisma
    43→ *                            SophiaMemory table directly.
    44→ *
    45→ *   [current local persistence — AUTHORITATIVE TODAY]
    46→ *                            The DurableFileStore collection
    47→ *                            `.data/sophia_memories.json` is the primary
    48→ *                            write target and the authoritative read source
    49→ *                            for every method (mirrors ConversationStore):
    50→ *                              - listMemories()/findByIdempotencyKey() read
    51→ *                                the file ONLY
    52→ *                              - getMemory()/updateMemory()/deleteMemory()
    53→ *                                resolve through the file ONLY
    54→ *
    55→ *   [current Prisma dual-write — opportunistic mirror, NOT authoritative]
    56→ *                            createMemory()/updateMemory() upsert to the
    57→ *                            Prisma SophiaMemory table best-effort AFTER
    58→ *                            the durable file write; errors are swallowed
    59→ *                            (never blocks the interaction).
    60→ *                            deleteMemory() best-effort deletes the mirror
    61→ *                            row. There is NO Prisma read fallback — Prisma
    62→ *                            SophiaMemory rows are write-only shadows
    63→ *                            (ADR 0002 §7 layering, applied to K-2).
    64→ *
    65→ *   [authoritative mode]     THIS STORE HAS NONE — deliberately, exactly
    66→ *                            like ConversationStore. The same local-style
    67→ *                            semantics (file primary, Prisma best-effort,
    68→ *                            swallowed dual-write errors) apply in every
    69→ *                            DATABASE_MODE, including production. The
    70→ *                            M6 authoritative (PostgreSQL) migration moves
    71→ *                            conversations and personal memories together;
    72→ *                            until then Prisma must NOT be claimed as
    73→ *                            authoritative for personal memory.
    74→ *
    75→ * INVARIANTS:
    76→ * 1. Founder Ownership: every read/write/delete is bound to the caller's
    77→ *    founderId. Cross-founder access FAILS CLOSED (403) — never a silent
    78→ *    empty result when the record exists (existence is leaked only through
    79→ *    the explicit 403; the victim's content is never returned).
    80→ * 2. Process-Restart Durability: atomic DurableFileStore writes + best-effort
    81→ *    Prisma dual-write.
    82→ * 3. Deterministic Retrieval: listMemories is founder-scoped, filter-bounded,
    83→ *    sorted (updatedAt DESC, id ASC tie-break), hard-capped — no vector
    84→ *    search, no embeddings, no LLM-generated retrieval decisions.
    85→ * 4. Idempotent Writes (optional): a founder-scoped idempotencyKey dedupes
    86→ *    createMemory exactly like ChatMessage's idempotency convention
    87→ *    (no key → no dedupe).
    88→ * 5. No Fabrication: an empty store returns empty results — never seeded,
    89→ *    never invented (bootstrap seeds are a CompanyKnowledge concept and do
    90→ *    NOT apply to personal memory).
    91→ */

   286→const SOPHIA_MEMORIES_COLLECTION = 'sophia_memories';
   287→
   288→export class SophiaMemoryStore {
   289→  private static instance: SophiaMemoryStore;
   290→  private fileStore: DurableFileStore;
   291→
   292→  private constructor() {
   293→    this.fileStore = DurableFileStore.getInstance();
   294→  }
   295→
   296→  public static getInstance(): SophiaMemoryStore {
   297→    if (!SophiaMemoryStore.instance) {
   298→      SophiaMemoryStore.instance = new SophiaMemoryStore();
   299→    }
   300→    return SophiaMemoryStore.instance;
   301→  }
   302→
   303→  // --------------------------------------------------------------------------
   304→  // Validation helpers (fail-closed; never silently coerce or truncate)
   305→  // --------------------------------------------------------------------------
   306→
   307→  private requireFounderId(founderId: string): string {
   308→    if (typeof founderId !== 'string' || !founderId.trim()) {
   309→      throw new SophiaMemorySecurityError(
   310→        'Authenticated founderId is required for every personal memory operation.'
   311→      );
   312→    }
   313→    return founderId.trim();
   314→  }
   315→
   316→  private validateMemoryType(memoryType: string): SophiaMemoryType {
   317→    if (
   318→      typeof memoryType !== 'string' ||
   319→      !(SOPHIA_MEMORY_TYPES as readonly string[]).includes(memoryType)
   320→    ) {
   321→      throw new SophiaMemoryValidationError(
   322→        `memoryType must be one of [${SOPHIA_MEMORY_TYPES.join(', ')}] (got: ${String(memoryType)}).`
   323→      );
   324→    }
   325→    return memoryType as SophiaMemoryType;
   326→  }
   327→
   328→  private validateContent(content: string): string {
   329→    if (typeof content !== 'string' || !content.trim()) {
   330→      throw new SophiaMemoryValidationError('content must be a non-empty string.');
   331→    }
   332→    if (content.length > SOPHIA_MEMORY_CONTENT_MAX_CHARS) {
   333→      throw new SophiaMemoryValidationError(
   334→        `content exceeds the ${SOPHIA_MEMORY_CONTENT_MAX_CHARS}-character bound (got ${content.length}).`
   335→      );
   336→    }
   337→    return content.trim();
   338→  }
   339→
   340→  private validateProvenance(provenance: string): string {
   341→    if (typeof provenance !== 'string' || !provenance.trim()) {
   342→      throw new SophiaMemoryValidationError('provenance must be a non-empty source reference.');
   343→    }
   344→    if (provenance.length > SOPHIA_MEMORY_PROVENANCE_MAX_CHARS) {
   345→      throw new SophiaMemoryValidationError(
   346→        `provenance exceeds the ${SOPHIA_MEMORY_PROVENANCE_MAX_CHARS}-character bound.`
   347→      );
   348→    }
   349→    return provenance.trim();
   350→  }
   351→
   352→  private validateConfidence(confidence: number): number {
   353→    if (typeof confidence !== 'number' || !Number.isFinite(confidence) || confidence < 0 || confidence > 1) {
   354→      throw new SophiaMemoryValidationError('confidence must be a finite number in [0, 1].');
   355→    }
   356→    return confidence;
   357→  }
   358→
   359→  /**
   360→   * M4-A HARDENING: deterministic fail-closed check on the AUTHORITY /
   361→   * PRIVILEGE / CONTROL / GOVERNANCE semantic category (see
   362→   * authority-content-guard.ts). Applied on create AND on content updates
   363→   * so neither ingress path — capture or founder-direct authoring — can
   364→   * persist authorization-bearing personal memory.
   365→   */
   366→  private validateNotAuthorityContent(content: string): string {
   367→    const evaluation = evaluateAuthorityContent(content);
   368→    if (evaluation.blocked) {
   369→      throw new SophiaMemoryAuthorityError();
   370→    }
   371→    return content;
   372→  }
   373→
   374→  /**
   375→   * Resolves a memory from the durable file store (the ONLY read source today)
   376→   * and enforces founder ownership STRICTLY. Existence + owner mismatch fails
   377→   * closed (403); the victim's content is never returned to the caller.
   378→   */
   379→  private getOwnedMemory(founderId: string, memoryId: string): SophiaMemoryRecord {
   380→    const record = this.fileStore.getItem<SophiaMemoryRecord>(SOPHIA_MEMORIES_COLLECTION, memoryId);
   381→    if (!record) {
   382→      throw new SophiaMemoryNotFoundError(
   383→        `Personal memory "${memoryId}" not found for Founder "${founderId}".`
   384→      );
   385→    }
   386→    if (record.founderId !== founderId) {
   387→      throw new SophiaMemorySecurityError(
   388→        `Principal "${founderId}" is not authorized to access personal memory "${memoryId}".`
   389→      );
   390→    }
   391→    return record;
   392→  }
   393→
   394→  // --------------------------------------------------------------------------
   395→  // Writes
   396→  // --------------------------------------------------------------------------
   397→
   398→  /**
   399→   * Creates a personal memory bound to the authenticated founder.
   400→   *
   401→   * Idempotency (ChatMessage convention): when a non-empty idempotencyKey is
   402→   * supplied and a memory with the SAME key already exists for the SAME
   403→   * founder, the existing record is returned and nothing is written. Without
   404→   * a key there is NO dedupe. The key is founder-scoped — Founder B may use
   405→   * Founder A's key without collision.
   406→   *
   407→   * Lifecycle (M4-B.1): lifecycleState is the authority; `active` is the
   408→   * derived mirror (active === (lifecycleState === 'ACTIVE')). Created records
   409→   * are ACTIVE by default (founder-direct). The M4-A capture path creates
   410→   * PENDING_REVIEW candidates that stay out of every context render until a
   411→   * Founder explicitly activates them. Only birth states are accepted.
   412→   */
   413→  public async createMemory(params: CreateSophiaMemoryParams): Promise<SophiaMemoryRecord> {
   414→    const founderId = this.requireFounderId(params.founderId);
   415→    const memoryType = this.validateMemoryType(params.memoryType);
   416→    const content = this.validateNotAuthorityContent(this.validateContent(params.content));
   417→    const provenance = this.validateProvenance(params.provenance || 'founder_direct');
   418→    const confidence = this.validateConfidence(params.confidence ?? 0.8);
   419→
   420→    // --- Lifecycle resolution (M4-B.1): explicit birth state wins; the
   421→    // legacy active flag derives one; both together must be consistent. ---
   422→    let lifecycleState: SophiaMemoryLifecycleState;
   423→    if (params.lifecycleState !== undefined) {
   424→      if (!isSophiaMemoryLifecycleState(params.lifecycleState)) {
   425→        throw new SophiaMemoryValidationError(
   426→          `lifecycleState must be one of [${SOPHIA_MEMORY_LIFECYCLE_STATES.join(', ')}].`
   427→        );
   428→      }
   429→      if (!(SOPHIA_MEMORY_LIFECYCLE_BIRTH_STATES as readonly string[]).includes(params.lifecycleState)) {
   430→        throw new SophiaMemoryLifecycleError(
   431→          `A new personal memory can only be created as PENDING_REVIEW or ACTIVE (got: ${params.lifecycleState}); ` +
   432→            `SUPERSEDED/ARCHIVED/REJECTED are outcomes of Founder-executed transitions on an existing record.`
   433→        );
   434→      }
   435→      lifecycleState = params.lifecycleState;
   436→      if (params.active !== undefined && params.active !== (lifecycleState === 'ACTIVE')) {
   437→        throw new SophiaMemoryValidationError(
   438→          'Inconsistent creation parameters: active flag contradicts lifecycleState.'
   439→        );
   440→      }
   441→    } else {
   442→      lifecycleState = (params.active ?? true) ? 'ACTIVE' : 'PENDING_REVIEW';
   443→    }
   444→    const active = lifecycleState === 'ACTIVE';
   445→
   446→    const cleanKey =
   447→      typeof params.idempotencyKey === 'string' && params.idempotencyKey.trim()
   448→        ? params.idempotencyKey.trim()
   449→        : undefined;
   450→
   451→    const now = new Date().toISOString();
   452→    const record: SophiaMemoryRecord = {
   453→      id: `smem-${randomUUID()}`,
   454→      founderId,
   455→      memoryType,
   456→      content,
   457→      provenance,
   458→      confidence,
   459→      active,
   460→      lifecycleState,
   461→      idempotencyKey: cleanKey,
   462→      createdAt: now,
   463→      updatedAt: now,
   464→      metadata: params.metadata && typeof params.metadata === 'object' ? { ...params.metadata } : {},
   465→    };
   466→
   467→    // 1. Atomic durable file write (authoritative today). The idempotency
   468→    //    check and the save are ONE serialized cross-process read-check-write
   469→    //    unit (M4-A hardening): two concurrent creators of the same key can
   470→    //    no longer interleave the check and both persist, and concurrent
   471→    //    writers can no longer silently erase each other's records. The lock
   472→    //    is re-entrant, so the saveItem inside runs without re-acquiring.
   473→    const written = this.fileStore.withCollectionLock(SOPHIA_MEMORIES_COLLECTION, () => {
   474→      if (cleanKey) {
   475→        const existing = this.findByIdempotencyKeySync(founderId, cleanKey);
   476→        if (existing) {
   477→          return existing;
   478→        }
   479→      }
   480→      this.fileStore.saveItemStrict(SOPHIA_MEMORIES_COLLECTION, record.id, record);
   481→      return record;
   482→    });
   483→
   484→    // Idempotent replay: the existing record is returned and nothing else
   485→    // (including the Prisma mirror) is touched.
   486→    if (written !== record) {
   487→      return written;
   488→    }
   489→
   490→    // 2. Best-effort Prisma dual-write (opportunistic mirror, never authoritative)
   491→    try {
   492→      if ((await isDatabaseAvailable()) && (prisma as any).sophiaMemory) {
   493→        await (prisma as any).sophiaMemory.upsert({
   494→          where: { id: record.id },
   495→          create: {
   496→            id: record.id,
   497→            founderId: record.founderId,
   498→            memoryType: record.memoryType,
   499→            content: record.content,
   500→            provenance: record.provenance,
   501→            confidence: record.confidence,
   502→            active: record.active,
   503→            lifecycleState: record.lifecycleState,
   504→            idempotencyKey: record.idempotencyKey ?? null,
   505→            metadata: record.metadata,
   506→            createdAt: new Date(record.createdAt),
   507→            updatedAt: new Date(record.updatedAt),
   508→          },
   509→          update: {
   510→            content: record.content,
   511→            confidence: record.confidence,
   512→            active: record.active,
   513→            lifecycleState: record.lifecycleState,
   514→            metadata: record.metadata,
   515→            updatedAt: new Date(record.updatedAt),
   516→          },
   517→        });
   518→      }
   519→    } catch {
   520→      // Non-fatal if DB is offline (local durable mode)
   521→    }
   522→
   523→    return record;
   524→  }
   525→
   526→  /**
   527→   * Updates a personal memory's mutable fields (content / confidence /
   528→   * lifecycle). Ownership fails closed (403); unknown id → 404.
   529→   *
   530→   * LIFECYCLE (M4-B.1): this method is the SINGLE choke point for every
   531→   * belief-bearing lifecycle transition. The requested target state is
   532→   * resolved from the patch (explicit lifecycleState, or the legacy active
   533→   * flag), validated against the deterministic transition table
   534→   * (memory-lifecycle.ts) — fail-closed (400) on any illegal transition —
   535→   * and every executed transition is recorded in an append-only audit trail
   536→   * (metadata.lifecycle.transitions[]: from / to / at / by) with the
   537→   * authenticated Founder as the actor. `active` stays synchronized as the
   538→   * derived mirror of the new state.
   539→   *
   540→   * Legacy compat mapping (behavior-preserving for every K-2/M4-A pinned
   541→   * toggle): active:true → ensure ACTIVE (illegal only from REJECTED);
   542→   * active:false → ACTIVE becomes ARCHIVED, any other state is an
   543→   * idempotent no-op (a pending candidate stays PENDING_REVIEW, exactly as
   544→   * before M4-B.1).
   545→   */
   546→  public async updateMemory(
   547→    founderId: string,
   548→    memoryId: string,
   549→    patch: UpdateSophiaMemoryParams
   550→  ): Promise<SophiaMemoryRecord> {
   551→    const owner = this.requireFounderId(founderId);
   552→    if (typeof memoryId !== 'string' || !memoryId.trim()) {
   553→      throw new SophiaMemoryValidationError('memoryId is required.');
   554→    }
   555→    // Fail-closed validation BEFORE the lock (pure checks, no store state):
   556→    // a content patch may not introduce authorization/privilege semantics
   557→    // into an existing record (M4-A hardening — PATCH must not become the
   558→    // laundering path around the create-time guard).
   559→    if (patch.content !== undefined) {
   560→      this.validateNotAuthorityContent(this.validateContent(patch.content));
   561→    }
   562→    if (patch.confidence !== undefined) {
   563→      this.validateConfidence(patch.confidence);
   564→    }
   565→    if (patch.active !== undefined && typeof patch.active !== 'boolean') {
   566→      throw new SophiaMemoryValidationError('active must be a boolean.');
   567→    }
   568→
   569→    // --- M4-B.1 lifecycle patch validation (pure, pre-lock) ---
   570→    if (patch.lifecycleState !== undefined) {
   571→      if (!isSophiaMemoryLifecycleState(patch.lifecycleState)) {
   572→        throw new SophiaMemoryValidationError(
   573→          `lifecycleState must be one of [${SOPHIA_MEMORY_LIFECYCLE_STATES.join(', ')}] (got: ${String(patch.lifecycleState)}).`
   574→        );
   575→      }
   576→      if (patch.active !== undefined) {
   577→        throw new SophiaMemoryValidationError(
   578→          'Ambiguous lifecycle patch: provide either `active` (legacy) or `lifecycleState`, not both.'
   579→        );
   580→      }
   581→      if (patch.supersededByMemoryId !== undefined && patch.lifecycleState !== 'SUPERSEDED') {
   582→        throw new SophiaMemoryValidationError(
   583→          'supersededByMemoryId is only valid together with lifecycleState: "SUPERSEDED".'
   584→        );
   585→      }
   586→    }
   587→    if (patch.supersededByMemoryId !== undefined) {
   588→      if (typeof patch.supersededByMemoryId !== 'string' || !patch.supersededByMemoryId.trim()) {
   589→        throw new SophiaMemoryValidationError('supersededByMemoryId must be a non-empty string.');
   590→      }
   591→    }
   592→
   593→    // Read-merge-write serialized on the collection's cross-process lock
   594→    // (M4-A hardening): a concurrent update or create can no longer be
   595→    // silently erased by this writer, and vice versa.
   596→    const next = this.fileStore.withCollectionLock(SOPHIA_MEMORIES_COLLECTION, () => {
   597→      // Resolves the CURRENT state (legacy rows derived) — the record read
   598→      // here carries the authoritative lifecycleState for this transition.
   599→      const record = withResolvedLifecycle(this.getOwnedMemory(owner, memoryId.trim()));
   600→
   601→      const updated: SophiaMemoryRecord = { ...record };
   602→
   603→      if (patch.content !== undefined) {
   604→        updated.content = this.validateContent(patch.content);
   605→      }
   606→      if (patch.confidence !== undefined) {
   607→        updated.confidence = this.validateConfidence(patch.confidence);
   608→      }
   609→
   610→      // --- Lifecycle transition resolution (M4-B.1) ---
   611→      const currentState = record.lifecycleState;
   612→      let targetState: SophiaMemoryLifecycleState | undefined;
   613→
   614→      if (patch.lifecycleState !== undefined) {
   615→        targetState = patch.lifecycleState;
   616→      } else if (patch.active === true) {
   617→        // Legacy activation: ensure ACTIVE (the ONLY activation path for
   618→        // captured candidates — M4-A contract unchanged).
   619→        targetState = 'ACTIVE';
   620→      } else if (patch.active === false) {
   621→        // Legacy deactivation: behavior-preserving mapping — ACTIVE becomes
   622→        // ARCHIVED; a non-ACTIVE record is an idempotent no-op (a pending
   623→        // candidate stays PENDING_REVIEW, exactly as before M4-B.1).
   624→        targetState = currentState === 'ACTIVE' ? 'ARCHIVED' : currentState;
   625→      }
   626→
   627→      if (targetState !== undefined && targetState !== currentState) {
   628→        if (targetState === 'PENDING_REVIEW') {
   629→          throw new SophiaMemoryLifecycleError(
   630→            `Illegal lifecycle transition: a memory can never return to PENDING_REVIEW (current: ${currentState}).`
   631→          );
   632→        }
   633→        if (!canTransitionSophiaMemory(currentState, targetState)) {
   634→          throw new SophiaMemoryLifecycleError(
   635→            `Illegal lifecycle transition: ${currentState} → ${targetState} is not allowed ` +
   636→              `(allowed from ${currentState}: [${SOPHIA_MEMORY_LIFECYCLE_TRANSITIONS[currentState].join(', ') || 'none'}]).`
   637→          );
   638→        }
   639→
   640→        // Supersession provenance pointer (create-B + mark-A pattern): the
   641→        // successor must exist, belong to the SAME founder, and not be the
   642→        // record itself. The successor is never auto-mutated.
   643→        let supersededBy: Record<string, unknown> | undefined;
   644→        if (targetState === 'SUPERSEDED') {
   645→          const successorId = patch.supersededByMemoryId?.trim();
   646→          if (successorId) {
   647→            if (successorId === record.id) {
   648→              throw new SophiaMemoryValidationError('A memory cannot supersede itself.');
   649→            }
   650→            const successor = withResolvedLifecycle(
   651→              this.getOwnedMemory(owner, successorId)
   652→            );
   653→            supersededBy = {
   654→              memoryId: successor.id,
   655→              content: successor.content, // audit snapshot at supersession time
   656→              at: new Date().toISOString(),
   657→            };
   658→          }
   659→        }
   660→
   661→        // M4-A founder-review confirmation stamp (unchanged semantics):
   662→        // activating a pending captured candidate records WHEN — and now
   663→        // also BY WHOM (M4-B.1 attribution) — the Founder confirmed it.
   664→        // Deterministic lifecycle metadata (server-side), not new authority.
   665→        if (targetState === 'ACTIVE' && updated.metadata?.captureStatus === 'pending') {
   666→          updated.metadata = {
   667→            ...updated.metadata,
   668→            captureStatus: 'confirmed',
   669→            confirmedAt: new Date().toISOString(),
   670→            confirmedBy: owner,
   671→          };
   672→        }
   673→        if (targetState === 'SUPERSEDED' && supersededBy) {
   674→          updated.metadata = { ...updated.metadata, supersededBy };
   675→        }
   676→
   677→        // Append-only audit trail — every executed transition carries the
   678→        // authenticated Founder as actor (M4-B.1 auditability; provenance is
   679→        // never silently deleted, only extended).
   680→        const lifecycleMeta = (updated.metadata?.lifecycle as Record<string, unknown>) ?? {};
   681→        const transitions = Array.isArray(lifecycleMeta.transitions)
   682→          ? [...(lifecycleMeta.transitions as unknown[])]
   683→          : [];
   684→        transitions.push({
   685→          from: currentState,
   686→          to: targetState,
   687→          at: new Date().toISOString(),
   688→          by: owner,
   689→        });
   690→        updated.metadata = {
   691→          ...updated.metadata,
   692→          lifecycle: { ...lifecycleMeta, transitions },
   693→        };
   694→
   695→        // lifecycleState is the authority; the boolean is the derived mirror.
   696→        updated.lifecycleState = targetState;
   697→        updated.active = isEligibleForPersonalMindContext(targetState);
   698→      } else if (targetState !== undefined) {
   699→        // Idempotent same-state patch: no transition entry, but keep the
   700→        // derived boolean synchronized (defensive — it is already consistent).
   701→        updated.active = isEligibleForPersonalMindContext(currentState);
   702→      }
   703→
   704→      updated.updatedAt = new Date().toISOString();
   705→
   706→      // 1. Atomic durable file write (STRICT — M4-A hardening: a failed
   707→      //    authoritative write THROWS instead of returning a phantom update;
   708→      //    M4-B.1: the write also persists the resolved lifecycleState, so
   709→      //    legacy rows are lazily migrated on their first update)
   710→      this.fileStore.saveItemStrict(SOPHIA_MEMORIES_COLLECTION, updated.id, updated);
   711→      return updated;
   712→    });
   713→
   714→    // 2. Best-effort Prisma dual-write
   715→    try {
   716→      if ((await isDatabaseAvailable()) && (prisma as any).sophiaMemory) {
   717→        await (prisma as any).sophiaMemory.update({
   718→          where: { id: next.id },
   719→          data: {
   720→            content: next.content,
   721→            confidence: next.confidence,
   722→            active: next.active,
   723→            lifecycleState: next.lifecycleState,
   724→            metadata: next.metadata as any,
   725→            updatedAt: new Date(next.updatedAt),
   726→          },
   727→        });
   728→      }
   729→    } catch {
   730→      // Non-fatal (mirror only)
   731→    }
   732→
   733→    return next;
   734→  }
   735→
   736→  /**
   737→   * Deletes a personal memory. Ownership fails closed (403); unknown id →
   738→   * returns false (idempotent delete for the OWNER — a second delete of a
   739→   * memory the caller legitimately owned is a no-op, never an error).
   740→   */
   741→  public async deleteMemory(founderId: string, memoryId: string): Promise<boolean> {
   742→    const owner = this.requireFounderId(founderId);
   743→    if (typeof memoryId !== 'string' || !memoryId.trim()) {
   744→      throw new SophiaMemoryValidationError('memoryId is required.');
   745→    }
   746→    const id = memoryId.trim();
   747→
   748→    // Read-check-delete serialized on the collection's cross-process lock
   749→    // (M4-A hardening — same race class as create/update).
   750→    const deleted = this.fileStore.withCollectionLock(SOPHIA_MEMORIES_COLLECTION, () => {
   751→      const record = this.fileStore.getItem<SophiaMemoryRecord>(SOPHIA_MEMORIES_COLLECTION, id);
   752→      if (!record) {
   753→        return false;
   754→      }
   755→      if (record.founderId !== owner) {
   756→        throw new SophiaMemorySecurityError(
   757→          `Principal "${owner}" is not authorized to delete personal memory "${id}".`
   758→        );
   759→      }
   760→      // 1. Authoritative durable file delete (STRICT — a failed delete-write
   761→      //    throws instead of reporting a deletion that never reached disk)
   762→      return this.fileStore.deleteItemStrict(SOPHIA_MEMORIES_COLLECTION, id);
   763→    });
   764→
   765→    if (!deleted) {
   766→      return false;
   767→    }
   768→
   769→    // 2. Best-effort Prisma mirror delete
   770→    try {
   771→      if ((await isDatabaseAvailable()) && (prisma as any).sophiaMemory) {
   772→        await (prisma as any).sophiaMemory.delete({ where: { id } }).catch(() => {});
   773→      }
   774→    } catch {
   775→      // Non-fatal (mirror only)
   776→    }
   777→
   778→    return true;
   779→  }

   785→  /**
   786→   * Retrieves one personal memory, strictly validating ownership.
   787→   * Unknown id → null; owner mismatch → SophiaMemorySecurityError (403).
   788→   * The returned record always carries a resolved lifecycleState (legacy

   791→  public async getMemory(founderId: string, memoryId: string): Promise<SophiaMemoryRecord | null> {
   792→    const owner = this.requireFounderId(founderId);
   793→    if (typeof memoryId !== 'string' || !memoryId.trim()) {
   794→      return null;
   795→    }
   796→    const record = this.fileStore.getItem<SophiaMemoryRecord>(SOPHIA_MEMORIES_COLLECTION, memoryId.trim());
   797→    if (!record) {
   798→      return null;
   799→    }
   800→    if (record.founderId !== owner) {
   801→      throw new SophiaMemorySecurityError(
   802→        `Principal "${owner}" is not authorized to access personal memory "${memoryId.trim()}".`
   803→      );
   804→    }
   805→    return withResolvedLifecycle({ ...record });
   806→  }
   807→
   808→  /**
   809→   * Deterministic, founder-scoped, bounded retrieval.
   810→   *
   811→   * Ordering: updatedAt DESC, then id ASC (total, stable, explainable order).
   812→   * Filters: memoryType (exact allow-list match), active flag. Limit defaults
   813→   * to 20 and is hard-capped at 50. No scoring, no vectors, no LLM decisions.
   814→   *
   815→   * PAGINATION (P2 follow-up — queue >50 visibility): `offset` skips the
   816→   * first N records of the SAME deterministic order (applied after sorting,
   817→   * before the limit slice), so page k is `listMemories(f, { offset: k*50,
   818→   * limit: 50 })`. Without it, the hard cap made the oldest pending review
   819→   * candidates permanently invisible to both the API and the UI once the
   820→   * founder-scoped set exceeded 50 records. Offset is clamped to
   821→   * [0, SOPHIA_MEMORY_LIST_MAX_OFFSET] and a too-large offset yields an empty
   822→   * page (a legitimate "past the end" result, not an error).
   823→   */
   824→  public async listMemories(
   825→    founderId: string,
   826→    opts?: {
   827→      memoryType?: SophiaMemoryType;
   828→      active?: boolean;
   829→      /**
   830→       * Exact lifecycle-state filter (M4-B.1) — the review queue reads
   831→       * PENDING_REVIEW; context assembly reads eligibility. Mutually
   832→       * exclusive with `active` (fail-closed: passing both is a 400).
   833→       */
   834→      lifecycleState?: SophiaMemoryLifecycleState;
   835→      limit?: number;
   836→      offset?: number;
   837→    }
   838→  ): Promise<SophiaMemoryRecord[]> {
   839→    const owner = this.requireFounderId(founderId);
   840→
   841→    const items = this.sortedFounderMemories(owner, opts);
   842→
   843→    const requested = opts?.limit ?? SOPHIA_MEMORY_LIST_DEFAULT_LIMIT;
   844→    const limit = Math.max(1, Math.min(SOPHIA_MEMORY_LIST_MAX_LIMIT, Math.floor(requested)));
   845→    const offset = Math.max(0, Math.min(SOPHIA_MEMORY_LIST_MAX_OFFSET, Math.floor(opts?.offset ?? 0)));
   846→    return items.slice(offset, offset + limit).map((m) => ({ ...m }));
   847→  }
   848→
   849→  /**
   850→   * TOTAL record count for the same founder-scoped filter listMemories uses
   851→   * (P2 follow-up): callers paginating the review queue need the authoritative
   852→   * total to compute hasMore / badge counts — the page length alone cannot
   853→   * distinguish "exactly one page" from "one visible page of many".
   854→   */
   855→  public async countMemories(
   856→    founderId: string,
   857→    opts?: {
   858→      memoryType?: SophiaMemoryType;
   859→      active?: boolean;
   860→      lifecycleState?: SophiaMemoryLifecycleState;
   861→    }
   862→  ): Promise<number> {
   863→    const owner = this.requireFounderId(founderId);
   864→    return this.sortedFounderMemories(owner, opts).length;
   865→  }
   866→
   867→  /**
   868→   * AUTHORITATIVE COLLECTION READ (P2 follow-up — dedupe must not read the
   869→   * visible page): UNBOUNDED founder-scoped read used ONLY by internal
   870→   * deterministic consumers that must compare against the whole collection:
   871→   *   - the capture stage's duplicate detection (a duplicate of a record
   872→   *     outside the newest-50 window must still be rejected — the phase-2
   873→   *     observation proved the paginated read re-persisted such duplicates);
   874→   *   - the review-queue annotation pool (duplicateOf/similarTo/contradicts
   875→   *     must consider the founder's full set, active + pending).
   876→   * This is the SAME full-collection read every store mutation already
   877→   * performs internally; it is not a new search surface (no filters beyond
   878→   * founder scope / active / type, same total order, no vectors). Route-level
   879→   * pagination (limit/offset) governs what a CLIENT sees; dedupe correctness
   880→   * is a store-internal concern and must never depend on page visibility.
   881→   */
   882→  public async listAllMemories(
   883→    founderId: string,
   884→    opts?: {
   885→      memoryType?: SophiaMemoryType;
   886→      active?: boolean;
   887→      lifecycleState?: SophiaMemoryLifecycleState;
   888→    }
   889→  ): Promise<SophiaMemoryRecord[]> {
   890→    const owner = this.requireFounderId(founderId);
   891→    return this.sortedFounderMemories(owner, opts).map((m) => ({ ...m }));
   892→  }
   893→
   894→  /**
   895→   * Shared deterministic read: full founder-scoped collection, optional exact
   896→   * type/lifecycle filters, total order (updatedAt DESC, id ASC).
   897→   *
   898→   * M4-B.1: every record is lifecycle-resolved BEFORE filtering, so (a)
   899→   * legacy rows behave identically to post-M4-B.1 rows, and (b) the `active`
   900→   * filter is the DERIVED eligibility filter (active === true ⇔ resolved
   901→   * lifecycleState === 'ACTIVE'; active === false ⇔ any other state).
   902→   */
   903→  private sortedFounderMemories(
   904→    owner: string,
   905→    opts?: {
   906→      memoryType?: SophiaMemoryType;
   907→      active?: boolean;
   908→      lifecycleState?: SophiaMemoryLifecycleState;
   909→    }
   910→  ): SophiaMemoryRecord[] {
   911→    if (opts?.active !== undefined && opts?.lifecycleState !== undefined) {
   912→      throw new SophiaMemoryValidationError(
   913→        'Ambiguous filter: provide either `active` (derived) or `lifecycleState`, not both.'
   914→      );
   915→    }
   916→    const all = this.fileStore.readCollection<SophiaMemoryRecord>(SOPHIA_MEMORIES_COLLECTION);
   917→    let items = Object.values(all)
   918→      .filter((m) => m && m.founderId === owner)
   919→      .map((m) => withResolvedLifecycle(m));
   920→
   921→    if (opts?.memoryType !== undefined) {
   922→      const type = this.validateMemoryType(opts.memoryType);
   923→      items = items.filter((m) => m.memoryType === type);
   924→    }
   925→    if (opts?.lifecycleState !== undefined) {
   926→      if (!isSophiaMemoryLifecycleState(opts.lifecycleState)) {
   927→        throw new SophiaMemoryValidationError(
   928→          `lifecycleState filter must be one of [${SOPHIA_MEMORY_LIFECYCLE_STATES.join(', ')}].`
   929→        );
   930→      }
   931→      items = items.filter((m) => m.lifecycleState === opts.lifecycleState);
   932→    } else if (opts?.active !== undefined) {
   933→      // Derived eligibility filter — lifecycleState is the authority.
   934→      items = items.filter((m) => isEligibleForPersonalMindContext(m.lifecycleState) === opts.active);
   935→    }
   936→
   937→    items.sort((a, b) => {
   938→      const ta = new Date(a.updatedAt).getTime();
   939→      const tb = new Date(b.updatedAt).getTime();
   940→      if (tb !== ta) return tb - ta;
   941→      return a.id < b.id ? -1 : a.id > b.id ? 1 : 0;
   942→    });
   943→    return items;
   944→  }

   956→  private findByIdempotencyKeySync(founderId: string, idempotencyKey: string): SophiaMemoryRecord | null {
   957→    const owner = this.requireFounderId(founderId);
   958→    if (typeof idempotencyKey !== 'string' || !idempotencyKey.trim()) {
   959→      return null;
   960→    }
   961→    const key = idempotencyKey.trim();
   962→    const all = this.fileStore.readCollection<SophiaMemoryRecord>(SOPHIA_MEMORIES_COLLECTION);
   963→    for (const m of Object.values(all)) {
   964→      if (m && m.founderId === owner && m.idempotencyKey === key) {
   965→        return withResolvedLifecycle(m);
   966→      }
   967→    }
   968→    return null;

   971→  /**
   972→   * M4-A capture-replay guard: reports whether this founder already has any
   973→   * memory whose idempotencyKey starts with the given prefix. The capture
   974→   * stage derives deterministic keys ("m4cap:<conversationId>:<turnId>:<n>")
   975→   * from authoritative turn identity; a REPLAYED turn must never capture
   976→   * again — the prefix check short-circuits the whole capture stage before
   977→   * any extraction or persistence happens. Unbounded by list limits (reads
   978→   * the whole founder-scoped collection, which is the same read the store
   979→   * already performs for dedupe).

   981→  public async hasIdempotencyKeyPrefix(founderId: string, prefix: string): Promise<boolean> {
   982→    const owner = this.requireFounderId(founderId);
   983→    if (typeof prefix !== 'string' || !prefix.trim()) {
   984→      return false;
   985→    }
   986→    const all = this.fileStore.readCollection<SophiaMemoryRecord>(SOPHIA_MEMORIES_COLLECTION);
   987→    return Object.values(all).some(
   988→      (m) =>
   989→        m &&
   990→        m.founderId === owner &&
   991→        typeof m.idempotencyKey === 'string' &&
   992→        m.idempotencyKey.startsWith(prefix)
   993→    );
   994→  }
