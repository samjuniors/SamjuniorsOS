    37→/**
    38→ * The conversation.
    39→ *
    40→ * This used to be a sequential loop — greet, await a capture, await an answer,
    41→ * repeat — with the microphone opened and closed around each step. That shape
    42→ * cannot be interrupted: while it is awaiting the answer, nothing is listening,
    43→ * so there is no way for the user to get a word in.
    44→ *
    45→ * It is an event machine now. The voice loop runs continuously and pushes
    46→ * events at us; every one of them is legal in every phase. Saying anything at
    47→ * all stops him talking, and whatever you say next becomes the new turn.
    48→ */
    49→
    50→/** How long to wait for someone to start speaking after he wakes. Generous:
    51→ *  people say his name and *then* think about what they wanted. */
    52→const AWAIT_SPEECH_MS = 14000
    53→
    54→/** After an answer, how long the mic stays open for a follow-up before he
    55→ *  drops back to standby. Long enough that you don't have to say the name
    56→ *  again to continue a thought. */
    57→const FOLLOW_UP_MS = 11000
    58→
    59→/** crypto.randomUUID needs a secure context, which a LAN address over plain
    60→ *  http is not. Not worth failing a whole turn over an id. */
    61→const newId = () =>
    62→  globalThis.crypto?.randomUUID?.() ??
    63→  `id${Date.now().toString(36)}${Math.random().toString(36).slice(2, 10)}`
    64→
    65→/** The same mishearings voice.ts accepts for the wake word — otherwise a turn
    66→ *  that woke him as "travis" gets that word sent on to the model as a question. */
    67→const NAME = '(?:jarvis|jarvys|jervis|travis|jarviss|java\'s|jarv)'
    68→/** A bare vocative — "Jarvis", "hey jarvis" — with nothing asked. */
    69→const BARE_NAME = new RegExp(`^(?:hey|hi|ok|okay|yo)?\\s*${NAME}[\\s,.!?]*$`, 'i')
    70→/** A leading vocative on a real command: "Jarvis, what's the weather". */
    71→const LEADING_NAME = new RegExp(`^(?:hey|hi|ok|okay|yo)?\\s*${NAME}\\b[\\s,.:!?-]*`, 'i')
    72→
    73→export default function App() {
    74→  const store = useStore
    75→  const phase = useStore((s) => s.phase)
    76→  const history = useRef<Msg[]>([])
    77→  const speaker = useRef<ReturnType<typeof createSpeaker> | null>(null)
    78→  const voice = useRef<Voice | null>(null)
    79→
    80→  /**
    81→   * Monotonic turn counter. Every await in a turn checks it on the way out:
    82→   * if it has moved, that turn was superseded by a barge-in and must not touch
    83→   * the phase, the speaker, or the busy state on its way to the floor.
    84→   */
    85→  const turn = useRef(0)
    86→  const booting = useRef(false)
    87→  const idleTimer = useRef<ReturnType<typeof setTimeout> | null>(null)
    88→  const voicePoll = useRef<ReturnType<typeof setInterval> | null>(null)
    89→
    90→  // -- helpers --------------------------------------------------------------
    91→
    92→  const clearIdle = () => {
    93→    if (idleTimer.current) clearTimeout(idleTimer.current)
    94→    idleTimer.current = null
    95→  }
    96→
    97→  const silence = () => {
    98→    speaker.current?.cancel()
    99→    speaker.current = null
   100→  }
   101→
   102→  const goDormant = () => {
   103→    clearIdle()
   104→    silence()
   105→    turn.current++
   106→    const s = store.getState()
   107→    s.setCaption('')
   108→    s.setActiveTool(null)
   109→    music.working(false)
   110→    music.duck(false)
   111→    sfx.duck(false)
   112→    s.setPhase('dormant')
   113→  }
   114→
   115→  /** Open the mic and wait. `window` is how long before he gives up. */
   116→  const listen = (window: number) => {
   117→    clearIdle()
   118→    const s = store.getState()
   119→    s.setCaption('')
   120→    s.setPhase('listening')
   121→    sfx.play('listen')
   122→    idleTimer.current = setTimeout(goDormant, window)
   123→  }
   124→
   125→  // -- one turn -------------------------------------------------------------
   126→
   127→  const respond = async (said: string): Promise<void> => {
   128→    const mine = ++turn.current
   129→    const stale = () => mine !== turn.current
   130→
   131→    clearIdle()
   132→    const s = store.getState()
   133→    // Last turn's panels and blades go now, before the new answer starts
   134→    // putting its own up. Anything the model marked sticky survives.
   135→    s.clearPanels()
   136→    s.clearBlades()
   137→    s.setCaption('')
   138→    s.pushTurn({ id: newId(), role: 'user', text: said })
   139→    s.setPhase('thinking')
   140→
   141→    const spk = createSpeaker()
   142→    speaker.current = spk
   143→    sfx.duck(true)
   144→    music.duck(true)
   145→
   146→    const turnId = newId()
   147→    let started = false
   148→    let filled = false
   149→
   150→    try {
   151→      const { text } = await ask(said, history.current, {
   152→        onText: (delta) => {
   153→          if (stale()) return
   154→          if (!started) {
   155→            started = true
   156→            store.getState().setPhase('speaking')
   157→            // The answer arriving is what ends the tool phase — a timer would
   158→            // clear the readout while a slow tool was still running.
   159→            store.getState().setActiveTool(null)
   160→            music.working(false)
   161→            store.getState().pushTurn({ id: turnId, role: 'jarvis', text: '' })
   162→          }
   163→          store.getState().appendToLastTurn(delta)
   164→          spk.push(delta)
   165→        },
   166→        onTool: (name) => {
   167→          if (stale()) return
   168→          // Only claim the tooling phase while he has nothing to say yet.
   169→          // Setting it unconditionally pinned the machine in 'tooling' for the
   170→          // rest of any answer that called a tool after it started talking,
   171→          // which also broke the reactor's lip-sync for the remainder.
   172→          if (!started) store.getState().setPhase('tooling')
   173→          store.getState().setActiveTool(name)
   174→          sfx.play('tool')
   175→          music.working(true)
   176→          // Say something the moment work starts — a tool can take ten seconds
   177→          // and silence that long reads as a crash. Once per turn only; a
   178→          // chain of five tools shouldn't produce five apologies.
   179→          if (!filled && !started) {
   180→            filled = true
   181→            spk.say(forTool(name))
   182→          }
   183→        },
   184→      })
   185→
   186→      if (stale()) return
   187→
   188→      // The bridge keeps conversation state in its own session, so history is
   189→      // only threaded through on the direct path.
   190→      if (!usingBridge) {
   191→        history.current.push({ role: 'user', content: said })
   192→        history.current.push({ role: 'assistant', content: text || '…' })
   193→        if (history.current.length > 16) {
   194→          history.current = history.current.slice(-16)
   195→        }
   196→      }
   197→
   198→      await spk.end()
   199→      if (stale()) return
   200→      sfx.play('done')
   201→    } catch (err) {
   202→      if (stale()) return
   203→      console.error(err)
   204→      sfx.play('error')
   205→      store
   206→        .getState()
   207→        .setError(err instanceof Error ? err.message : 'Something went wrong.')
   208→    } finally {
   209→      if (!stale()) {
   210→        speaker.current = null
   211→        sfx.duck(false)
   212→        music.duck(false)
   213→        store.getState().setActiveTool(null)
   214→        music.working(false)
   215→        // Stay open. Having to say his name again to add one more sentence is
   216→        // the difference between a conversation and a vending machine.
   217→        listen(FOLLOW_UP_MS)
   218→      }
   219→    }
   220→  }
   221→
   222→  // -- voice events ---------------------------------------------------------
   223→
   224→  /** What the voice loop should do with what it hears, derived from phase. */
   225→  const mode = (): VoiceMode => {
   226→    switch (store.getState().phase) {
   227→      case 'offline':
   228→      case 'boot':
   229→        return 'deaf'
   230→      case 'dormant':
   231→        return 'wake'
   232→      case 'waking':
   233→      case 'listening':
   234→        return 'command'
   235→      default:
   236→        return 'guard' // thinking, tooling, speaking
   237→    }
   238→  }
   239→
   240→  const onWake = (trailing: string) => {
   241→    const phase = store.getState().phase
   242→    if (phase === 'offline' || phase === 'boot') return
   243→
   244→    store.getState().setError(null)
   245→    sfx.play('wake')
   246→
   247→    // "Jarvis, what's happening in AI this week" in one breath. Waiting for a
   248→    // greeting he didn't need is the most common way an assistant wastes time.
   249→    if (trailing) {
   250→      void respond(trailing)
   251→      return
   252→    }
   253→
   254→    store.getState().setPhase('waking')
   255→
   256→    // Answer to his name. Deliberately NOT awaited any more: the microphone is
   257→    // already open and the echo filter knows his voice, so the user can talk
   258→    // straight over the greeting instead of waiting it out.
   259→    const greeting = createSpeaker()
   260→    speaker.current = greeting
   261→    greeting.say(attention())
   262→    void greeting.end()
   263→
   264→    listen(AWAIT_SPEECH_MS)
   265→  }
   266→
   267→  /**
   268→   * Someone started talking. This is the whole point of the rewrite: he stops,
   269→   * immediately, whatever he was doing.
   270→   */
   271→  const onSpeechStart = () => {
   272→    clearIdle()
   273→    const phase = store.getState().phase
   274→    if (phase === 'offline' || phase === 'boot' || phase === 'dormant') return
   275→
   276→    const wasBusy =
   277→      phase === 'thinking' || phase === 'tooling' || phase === 'speaking'
   278→
   279→    silence()
   280→    if (wasBusy) {
   281→      // Abandon the answer in flight. The turn counter moves in respond()'s
   282→      // replacement; bumping it here covers the case where nothing replaces it.
   283→      turn.current++
   284→      interrupt()
   285→      store.getState().setActiveTool(null)
   286→      music.working(false)
   287→      sfx.duck(false)
   288→      music.duck(false)
   289→    }
   290→    store.getState().setPhase('listening')
   291→  }
   292→
   293→  const onUtterance = (text: string) => {
   294→    const phase = store.getState().phase
   295→    if (phase === 'offline' || phase === 'boot' || phase === 'dormant') return
   296→
   297→    // People keep using his name as a vocative once they're already talking to
   298→    // him. Strip it rather than sending "jarvis" to the model as a question.
   299→    if (BARE_NAME.test(text)) {
   300→      listen(AWAIT_SPEECH_MS)
   301→      return
   302→    }
   303→    const said = text.replace(LEADING_NAME, '').trim()
   304→    if (!said) {
   305→      listen(AWAIT_SPEECH_MS)
   306→      return
   307→    }
   308→
   309→    void respond(said)
   310→  }
   311→
   312→  /**
   313→   * The typed command line. A `jarvis:say` event is the keyboard's version of
   314→   * a recognised utterance: same road, same rules. Idle, it starts a turn;
   315→   * busy, it barge-ins exactly like speaking over him does — the in-flight
   316→   * answer is interrupted, its turn goes stale, and this one replaces it.
   317→   */
   318→  const onSay = (e: Event) => {
   319→    const said = String((e as CustomEvent<string>).detail ?? '').trim()
   320→    if (!said) return
   321→
   322→    const phase = store.getState().phase
   323→    if (phase === 'boot') return
   324→
   325→    if (phase === 'offline') {
   326→      // The ignition screen is still up; boot first, then ask the moment the
   327→      // interface lands.
   328→      void powerOn().then(() => respond(said))
   329→      return
   330→    }
   331→
   332→    if (phase === 'thinking' || phase === 'tooling' || phase === 'speaking') {
   333→      silence()
   334→      interrupt()
   335→    }
   336→    void respond(said)
   337→  }
   338→
   339→  const onPartial = (text: string) => {
   340→    store.getState().setCaption(text)
   341→  }
   342→
   343→  const onVoiceError = (message: string) => {
   344→    store.getState().setError(message)
   345→  }
   346→
   347→  // -- power on -------------------------------------------------------------
   348→
   349→  const powerOn = async () => {
   350→    // The ignition button and the space bar can both land here, and the phase
   351→    // only moves after the first await — so without this a double press boots
   352→    // twice, arming two voice loops and two download polls.
   353→    if (booting.current) return
   354→    booting.current = true
   355→
   356→    try {
   357→      await ignite()
   358→    } catch (err) {
   359→      // The guard must not outlive a failed boot. Audio unlock can be refused,
   360→      // the microphone prompt dismissed, the bridge unreachable at the wrong
   361→      // moment — and with the flag still latched the ignition button was dead
   362→      // for the rest of the page, recoverable only by reloading. Reset it and
   363→      // put the button back so the user can simply press it again.
   364→      booting.current = false
   365→      console.error('[jarvis] power-up failed:', err)
   366→      store.getState().setPhase('offline')
   367→      store
   368→        .getState()
   369→        .setError(
   370→          err instanceof Error
   371→            ? `Power-up failed: ${err.message}`
   372→            : 'Power-up failed. Click to try again.',
   373→        )
   374→    }
   375→  }
   376→
   377→  const ignite = async () => {
   378→    const s = store.getState()
   379→
   380→    // Must happen inside the click handler — browsers won't start an
   381→    // AudioContext or speech synthesis without a user gesture.
   382→    await sfx.unlockAudio()
   383→    sfx.play('boot')
   384→    // The score. Must be started from inside this click handler for the same
   385→    // reason as the rest of the audio.
   386→    music.enable()
   387→    music.playBoot()
   388→    music.startAmbient()
   389→
   390→    s.setPhase('boot')
   391→
   392→    watchServers((servers) => store.getState().setConnected(servers))
   393→    watchPanels((panel) => store.getState().pushPanel(panel))
   394→    watchBlades((blade) => store.getState().pushBlade(blade))
   395→
   396→    /**
   397→     * JARVIS asking to see something.
   398→     *
   399→     * Announced on screen for as long as it takes, with whatever he said he was
   400→     * looking for. The camera's own light is on too, but a hardware light that
   401→     * appears with no explanation is exactly the thing that makes people
   402→     * distrust an assistant — so the interface says it before they have to ask.
   403→     */
   404→    watchCapture(async (req) => {
   405→      const note =
   406→        req.mode === 'watch'
   407→          ? req.when === 'past'
   408→            ? req.reason || 'reviewing the last few seconds'
   409→            : `${req.reason || 'watching'} · ${req.seconds}s`
   410→          : req.reason || 'taking a look'
   411→      store.getState().setLooking(note)
   412→
   413→      // The past is only available if something has been remembering it, and
   414→      // that only happens while the camera is on screen. Answering plainly
   415→      // beats opening the camera and recording the next few seconds instead,
   416→      // which is a different question from the one that was asked.
   417→      if (req.mode === 'watch' && req.when === 'past' && camera.bufferedSeconds() < 1) {
   418→        store.getState().setLooking(null)
   419→        return {
   420→          error:
   421→            'There is no recent footage — the camera has to be open on screen ' +
   422→            'for me to remember what just happened. Ask me to open the camera, ' +
   423→            'and I can watch from then on.',
   424→        }
   425→      }
   426→
   427→      // Held for the whole capture. Without this the stream can be torn down by
   428→      // whoever else was using it half way through a six-second watch.
   429→      let held = false
   430→      try {
   431→        await camera.holdCamera()
   432→        held = true
   433→        if (req.mode === 'look') return camera.grabFrame()
   434→        if (req.when === 'past') {
   435→          const grid = camera.recentGrid(req.seconds, 9)
   436→          return grid ?? { error: 'There is not enough recent footage to review.' }
   437→        }
   438→        return await camera.watchAhead(req.seconds, 9)
   439→      } catch (err) {
   440→        return {
   441→          error:
   442→            (err as DOMException)?.name === 'NotAllowedError'
   443→              ? 'The camera is not permitted, so I cannot see anything.'
   444→              : `The camera could not be read: ${(err as Error)?.message ?? err}`,
   445→        }
   446→      } finally {
   447→        if (held) camera.releaseCamera()
   448→        store.getState().setLooking(null)
   449→      }
   450→    })
   451→
   452→    // The interface is JARVIS's to drive. These arrive out of band, pushed
   453→    // mid-turn the way panels are, so a command can retint the reactor or put
   454→    // something into orbit while he is still speaking the sentence about it.
   455→    watchUi((op, args) => {
   456→      const s = store.getState()
   457→      const a = (args ?? {}) as Record<string, never>
   458→      switch (op) {
   459→        case 'patch':
   460→          s.applyUi(args)
   461→          break
   462→        case 'orbit':
   463→          if (a.action === 'add') s.addOrbit(args)
   464→          else if (a.action === 'remove') s.removeOrbit(String(a.id))
   465→          else s.clearOrbits()
   466→          break
   467→        case 'effect':
   468→          s.fireEffect(a.kind)
   469→          break
   470→        case 'reset':
   471→          s.resetUi()
   472→          break
   473→        case 'screen':
   474→          s.clearScreen(a.what ?? 'all')
   475→          break
   476→        default:
   477→          console.warn('[jarvis] unknown ui op:', op, args)
   478→      }
   479→    })
   480→    // In bridge mode the conversation lives in the agent session, which is tied
   481→    // to the socket — so a drop silently wipes his memory while the transcript
   482→    // on screen still shows it. Better to say so than to let him quietly forget.
   483→    watchConnection((state) => {
   484→      if (state === 'lost') {
   485→        store.getState().setError('Bridge connection lost — reconnecting.')
   486→      } else if (state === 'reconnected') {
   487→        store
   488→          .getState()
   489→          .setError('Bridge reconnected. The previous conversation was not kept.')
   490→      }
   491→    })
   492→    const warming = warm().catch((err: Error) => s.setError(err.message))
   493→
   494→    if (!usingBridge && !env.anthropicKey) {
   495→      s.setError(
   496→        'No Anthropic API key — copy .env.example to .env.local and set VITE_ANTHROPIC_API_KEY.',
   497→      )
   498→    }
   499→
   500→    // Pull the neural voice down during the boot sequence so the first
   501→    // "Hey Jarvis" isn't waiting on an 86MB download. Deliberately not awaited
   502→    // — if it's slow, JARVIS comes up on the system voice and swaps over the
   503→    // moment the model is ready.
   504→    if (TTS_ENGINE === 'kokoro') {
   505→      void kokoro.load()
   506→      voicePoll.current = setInterval(() => {
   507→        const p = kokoro.loadProgress()
   508→        if (kokoro.isReady() || kokoro.isUnavailable()) {
   509→          store.getState().setBootNote('')
   510→          if (voicePoll.current) clearInterval(voicePoll.current)
   511→          voicePoll.current = null
   512→        } else if (p > 0 && p < 1) {
   513→          store.getState().setBootNote(`voice ${Math.round(p * 100)}%`)
   514→        }
   515→      }, 200)
   516→    }
   517→
   518→    // Long enough for the four-beat start-up sequence in Boot.tsx to play —
   519→    // status bar, rings, suit schematic, reactor power-up — before the live
   520→    // interface takes over. Kept a touch under the boot cue so the music is
   521→    // still rising as the reactor lands.
   522→    await new Promise((r) => setTimeout(r, 9200)) // boot sequence
   523→    await warming
   524→    store.getState().setConnected(connectedLabels())
   525→    store.getState().setVoice(currentVoiceName())
   526→
   527→    // The analyser is what makes the reactor pulse with your voice. It needs a
   528→    // getUserMedia stream; speech recognition does not, and gets its own. So a
   529→    // failure here costs the animation and nothing else — saying "voice input
   530→    // is unavailable" was both alarming and untrue.
   531→    try {
   532→      await startAnalyser()
   533→    } catch {
   534→      console.warn(
   535→        '[jarvis] no microphone stream — the reactor will not pulse with your ' +
   536→          'voice. Speech recognition is unaffected.',
   537→      )
   538→    }
   539→
   540→    // Ask the bridge which speech engines exist before the loop starts, so the
   541→    // first turn already uses ElevenLabs when a key is present and the browser
   542→    // fallback when it is not — no flag, no reload.
   543→    await probeCapabilities()
   544→
   545→    // One voice loop, started once, running until the page closes.
   546→    voice.current = await startVoice({
   547→      mode,
   548→      onWake,
   549→      onSpeechStart,
   550→      onPartial,
   551→      onUtterance,
   552→      onError: onVoiceError,
   553→    })
   554→
   555→    store.getState().setPhase('dormant')
   556→  }
   557→
   558→  // -- clap to start --------------------------------------------------------
   559→
   560→  /**
   561→   * A clap brings him up, as an alternative to the button.
   562→   *
   563→   * Only while the ignition screen is showing, and torn down the moment he
   564→   * boots — the microphone is about to belong to the voice loop, and two
   565→   * analysers arguing over the same stream is how you get an assistant that
   566→   * hears half of what you say.
   567→   *
   568→   * Deliberately silent about failure. If the microphone is refused, or has not
   569→   * been granted yet, the button is still right there; announcing an error
   570→   * about a feature nobody asked for would be worse than quietly doing without.
   571→   */
   572→  useEffect(() => {
   573→    if (phase !== 'offline') return
   574→    let live: { stop: () => void } | null = null
   575→    let gone = false
   576→    void listenForClap(() => {
   577→      if (!gone) void powerOn()
   578→    }).then((l) => {
   579→      if (gone) l.stop()
   580→      else live = l
   581→    })
   582→    return () => {
   583→      gone = true
   584→      live?.stop()
   585→    }
   586→    // eslint-disable-next-line react-hooks/exhaustive-deps
   587→  }, [phase])
   588→
   589→  // -- level pump + keys ----------------------------------------------------
   590→
   591→  useEffect(() => {
   592→    let raf = 0
   593→
   594→    const pump = () => {
   595→      const st = store.getState()
   596→      // While speaking, follow JARVIS's own output rather than the mic, so the
   597→      // orb lip-syncs instead of reacting to room noise.
   598→      const lvl =
   599→        st.phase === 'speaking' && speaker.current
   600→          ? speaker.current.level()
   601→          : micLevel()
   602→      st.setLevel(lvl)
   603→      raf = requestAnimationFrame(pump)
   604→    }
   605→    pump()
   606→
   607→    const onKey = (e: KeyboardEvent) => {
   608→      const tag = (e.target as HTMLElement)?.tagName
   609→      if (tag === 'INPUT' || tag === 'TEXTAREA') return
   610→
   611→      // V auditions the next British voice installed on this machine. Which
   612→      // ones exist varies per Mac, so hearing them beats trusting a ranking.
   613→      // Bare V only — ⌘V and ⌃V are paste, and swallowing those was rude.
   614→      if (
   615→        e.key === 'v' &&
   616→        !e.repeat &&
   617→        !e.metaKey &&
   618→        !e.ctrlKey &&
   619→        !e.altKey
   620→      ) {
   621→        e.preventDefault()
   622→        const name = cycleVoice()
   623→        store.getState().setVoice(name)
   624→        silence()
   625→        const demo = createSpeaker()
   626→        speaker.current = demo
   627→        demo.say(`Voice set to ${name.replace(/\(.*?\)/g, '').trim()}. At your service, sir.`)
   628→        void demo.end()
   629→        return
   630→      }
   631→
   632→      // G puts the camera on and starts tracking hands. Off by default and
   633→      // never implicit: a webcam that turns itself on because an interface
   634→      // thought it might be useful is not a trade anyone agreed to.
   635→      if (e.key === 'g' && !e.repeat && !e.metaKey && !e.ctrlKey && !e.altKey) {
   636→        e.preventDefault()
   637→        const on = store.getState().gestures
   638→        if (on) {
   639→          hands.disableHands()
   640→          store.getState().setGestures(false)
   641→        } else {
   642→          store.getState().setError(null)
   643→          void hands
   644→            .enableHands()
   645→            .then(() => store.getState().setGestures(true))
   646→            .catch((err: Error) => {
   647→              store.getState().setGestures(false)
   648→              store
   649→                .getState()
   650→                .setError(
   651→                  err?.name === 'NotAllowedError'
   652→                    ? 'Camera access denied — gesture control is unavailable.'
   653→                    : `Gesture control failed to start: ${err?.message ?? err}`,
   654→                )
   655→            })
   656→        }
   657→        return
   658→      }
   659→
   660→      // T speaks a fixed line, bypassing the wake word, the recogniser and the
   661→      // model entirely. When "I can't hear him" is the report, this is the one
   662→      // keypress that separates a broken voice engine from a broken voice loop
   663→      // — and it prints the verdict rather than making you infer it.
   664→      if (e.key === 't' && !e.repeat && !e.metaKey && !e.ctrlKey && !e.altKey) {
   665→        e.preventDefault()
   666→        silence()
   667→        const t = createSpeaker()
   668→        speaker.current = t
   669→        t.say('Audio test. If you can hear this, speech output is working, sir.')
   670→        void t.end().then(() => {
   671→          const d = (window as unknown as Record<string, Record<string, unknown>>).__tts
   672→          console.info('[jarvis] audio test →', d)
   673→          if (d && d.started === 0 && d.rescued === 0) {
   674→            store.getState().setError(
   675→              `No sound produced. engine=${d.engine} voice=${d.voice} error=${d.lastError || 'none'}`,
   676→            )
   677→          }
   678→        })
   679→        return
   680→      }
   681→
   682→      // Escape stands the whole thing down — the one thing the old build had
   683→      // no key for at all.
   684→      if (e.key === 'Escape') {
   685→        e.preventDefault()
   686→        if (store.getState().phase !== 'offline') goDormant()
   687→        return
   688→      }
   689→
   690→      // Space starts a turn without the wake word. Worth using while filming so
   691→      // a missed wake word doesn't cost a take.
   692→      if (e.code !== 'Space' || e.repeat) return
   693→      e.preventDefault()
   694→
   695→      const phase = store.getState().phase
   696→      if (phase === 'offline') {
   697→        void powerOn()
   698→      } else if (phase === 'boot') {
   699→        /* ignore — the boot sequence owns the phase until it finishes */
   700→      } else if (
   701→        phase === 'thinking' ||
   702→        phase === 'tooling' ||
   703→        phase === 'speaking'
   704→      ) {
   705→        onSpeechStart()
   706→        listen(AWAIT_SPEECH_MS)
   707→      } else {
   708→        onWake('')
   709→      }
   710→    }
   711→    window.addEventListener('keydown', onKey)
   712→    window.addEventListener('jarvis:say', onSay as EventListener)
   713→
   714→    return () => {
   715→      cancelAnimationFrame(raf)
   716→      window.removeEventListener('keydown', onKey)
   717→      window.removeEventListener('jarvis:say', onSay as EventListener)
   718→      clearIdle()
   719→      if (voicePoll.current) clearInterval(voicePoll.current)
   720→      voice.current?.stop()
   721→      speaker.current?.cancel()
   722→      // The camera must not outlive the page that turned it on.
   723→      hands.disableHands()
   724→    }
   725→    // eslint-disable-next-line react-hooks/exhaustive-deps
   726→  }, [])
   727→
   728→  return (
   729→    <>
   730→      <Scene />
   731→      <Hud />
   732→      <Boot />
   733→      <Diagnostics />
   734→      <CommandLine />
   735→      <Ignition onStart={() => void powerOn()} />
   736→    </>
   737→  )
   738→}
