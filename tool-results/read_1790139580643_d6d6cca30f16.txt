    24→type MessageIntent = "conversation" | "information_request" | "directive" | "ambiguous" | "approval_action";
    25→
    26→interface IntentClassification {
    27→  intent: MessageIntent;
    28→  confidence: number;
    29→  directiveTitle?: string;
    30→  reason: string;
    31→  suggestedScope?: string;
    32→  approvalAction?: 'approve' | 'reject' | 'request_revision';
    33→  approvalNote?: string;
    34→}
    35→
    36→function classifyMessageIntent(message: string): IntentClassification {
    37→  const clean = message.trim().toLowerCase();
    38→
    39→  // 0. Explicit Founder Approval / Governance Action Check
    40→  // e.g. "Approve", "Authorize this", "I approve", "Reject", "Reject the proposal", "Request revision", "Needs revision", "Send it back"
    41→  const approvePatterns = [
    42→    /^(i )?(approve|approved|ratify|ratified|authorize|authorized|sign off|signed off|looks good, approve|approved, proceed|proceed with this|i agree, approve)$/i,
    43→    /^(approve|authorize|ratify)\s+(the\s+)?(decision|proposal|plan|request|recommendation|item|initiative)?$/i,
    44→  ];
    45→
    46→  const rejectPatterns = [
    47→    /^(i )?(reject|rejected|decline|declined|disapprove|veto|vetoed|cancel this|do not proceed|block this)$/i,
    48→    /^(reject|decline|disapprove|veto)\s+(the\s+)?(decision|proposal|plan|request|recommendation|item|initiative)?/i,
    49→  ];
    50→
    51→  const revisionPatterns = [
    52→    /^(i )?(request revision|needs revision|revise this|send back for revision|request changes|change this|needs work|revision required)$/i,
    53→    /^(request revision|revise|request changes)\s+(on|for|the)?/i,
    54→  ];
    55→
    56→  if (approvePatterns.some((p) => p.test(clean))) {
    57→    return {
    58→      intent: "approval_action",
    59→      confidence: 0.98,
    60→      approvalAction: "approve",
    61→      reason: "Explicit Founder approval command targeting governance / pending decisions.",
    62→    };
    63→  }
    64→
    65→  if (rejectPatterns.some((p) => p.test(clean))) {
    66→    return {
    67→      intent: "approval_action",
    68→      confidence: 0.98,
    69→      approvalAction: "reject",
    70→      approvalNote: message,
    71→      reason: "Explicit Founder rejection command targeting governance / pending decisions.",
    72→    };
    73→  }
    74→
    75→  if (revisionPatterns.some((p) => p.test(clean))) {
    76→    return {
    77→      intent: "approval_action",
    78→      confidence: 0.96,
    79→      approvalAction: "request_revision",
    80→      approvalNote: message,
    81→      reason: "Explicit Founder revision request targeting governance / pending decisions.",
    82→    };
    83→  }
    84→
    85→  // 1. Ambiguous pattern check (e.g., "Can you look into this?", "Look into pricing", "Check this out")
    86→  const ambiguousPatterns = [
    87→    /^(can you |could you |please |hey )?(look into|check into|check out|see about|explore|look at)\s+(this|that|it|things|something)(\?)?$/i,
    88→    /^(can you |could you )?(look into|check out|explore)\s+[a-z0-9\s]{1,25}\?*$/i,
    89→    /^(what should we do about|any thoughts on|what about|should we do something with)\s+[a-z0-9\s]{1,30}\?*$/i,
    90→    /^(can you help me with this|help with this|take a look)\?*$/i,
    91→  ];
    92→
    93→  if (ambiguousPatterns.some((p) => p.test(clean))) {
    94→    return {
    95→      intent: "ambiguous",
    96→      confidence: 0.88,
    97→      directiveTitle: message.length > 50 ? message.slice(0, 48) + "..." : message,
    98→      suggestedScope: "Clarification needed: determine whether to perform an informal review or initiate a formal multi-agent task.",
    99→      reason: "Ambiguous query requesting investigation without defined deliverables, scope, or clear action boundaries.",
   100→    };
   101→  }
   102→
   103→  // 2. Explicit directive patterns (Commands to perform work, create PRDs, conduct research, model economics)
   104→  const explicitDirectivePatterns = [
   105→    /\b(research|investigate|analyze|evaluate|audit|model|draft|design|create|build|prepare|synthesize|simulate|spec|spec out)\b.+\b(and (give|provide|recommend|write|report|present|model)|recommendation|proposal|prd|spec|plan|deliverable|architecture|breakdown|forecast|strategy)\b/i,
   106→    /^(research|investigate|analyze|evaluate|audit|model|draft|design|create|build|prepare|synthesize|simulate)\s+(the|our|a|an|all)\s+/i,
   107→    /\b(give me a recommendation|give me a plan|create a prd|draft a prd|model the unit economics|model our pricing|audit compute burn|conduct a research|run an analysis|synthesize a proposal)\b/i,
   108→    /^(execute|orchestrate|launch task|run task|start initiative)\b/i,
   109→  ];
   110→
   111→  if (explicitDirectivePatterns.some((p) => p.test(clean))) {
   112→    return {
   113→      intent: "directive",
   114→      confidence: 0.94,
   115→      directiveTitle: message.length > 60 ? message.slice(0, 58) + "..." : message,
   116→      suggestedScope: "Formal multi-agent task execution across Research, Product, and Finance with verified deliverables.",
   117→      reason: "Explicit work directive requesting research, specification, financial modeling, or strategic recommendations.",
   118→    };
   119→  }
   120→
   121→  // 3. Information request patterns (Queries about existing facts, MRR, initiatives, past findings)
   122→  const infoPatterns = [
   123→    /^(what (is|are|did|was|were)|how (much|many|is|are)|who (is|are)|where (is|are)|when (is|are)|tell me about|show me|status of|update on|do we have|is there|summary of)\b/i,
   124→    /\b(what did you find|what are our numbers|what is the mrr|what is the burn rate|what initiatives are active)\b/i,
   125→  ];
   126→
   127→  if (infoPatterns.some((p) => p.test(clean))) {
   128→    return {
   129→      intent: "information_request",
   130→      confidence: 0.9,
   131→      reason: "Direct factual query referencing existing company state, deliverables, or role-scoped context.",
   132→    };
   133→  }
   134→
   135→  // 4. Standard conversational patterns (Greetings, remarks, casual check-ins)
   136→  const conversationPatterns = [
   137→    /^(hi|hello|hey|good morning|good afternoon|good evening|how are you|how's it going|how are things|how is everything|thanks|thank you|great work|sounds good|ok|okay|cool|nice|who are you)\b/i,
   138→  ];
   139→
   140→  if (conversationPatterns.some((p) => p.test(clean))) {
   141→    return {
   142→      intent: "conversation",
   143→      confidence: 0.92,
   144→      reason: "Casual dialogue, greeting, or informal check-in.",
   145→    };
   146→  }
   147→
   148→  // Fallback heuristic: check for imperative verb starts
   149→  if (/^(prepare|research|analyze|build|create|model|draft|evaluate|audit|generate)\b/i.test(clean)) {
   150→    return {
   151→      intent: "directive",
   152→      confidence: 0.85,
   153→      directiveTitle: message.length > 60 ? message.slice(0, 58) + "..." : message,
   154→      suggestedScope: "Imperative work command requesting analytical or strategic outputs.",
   155→      reason: "Imperative directive verb detected.",
   156→    };
   157→  }
   158→
   159→  return {
   160→    intent: "conversation",
   161→    confidence: 0.75,
   162→    reason: "Standard conversational exchange.",
   163→  };
   164→}

   188→export async function POST(req: NextRequest) {
   189→  console.log('[AGENT-CHAT ENTER]');
   190→  try {
   191→    const session = await getAuthenticatedFounder(req);
   192→    if (!session) {
   193→      return NextResponse.json(
   194→        { error: 'Unauthorized: Session required to communicate with executive agents' },
   195→        { status: 401 }
   196→      );
   197→    }
   198→
   199→    const {
   200→      agentId,
   201→      message,
   202→      history,
   203→      contextSnapshot,
   204→      executeDirective,
   205→      personaConfig,
   206→      conversationId,
   207→      idempotencyKey,
   208→    } = await req.json();
   209→
   210→    if (!agentId || !message) {
   211→      return NextResponse.json({ error: "Agent ID and message are required" }, { status: 400 });
   212→    }
   213→
   214→    if (executeDirective && session.role !== 'FOUNDER') {
   215→      return NextResponse.json(
   216→        { error: 'Forbidden: Only verified Founder can trigger autonomous directive execution' },
   217→        { status: 403 }
   218→      );
   219→    }
   220→
   221→    const tone = (personaConfig?.tone || 'professional') as 'professional' | 'casual' | 'flirty';
   222→
   223→    // =========================================================================
   224→    // SOPHIA CONVERSATIONAL EXECUTIVE PIPELINE (PHASE 1 + PHASE 3 PERSISTENCE)
   225→    // =========================================================================
   226→    const isSophia = agentId === "coo" || agentId === "sophia";
   227→    if (isSophia) {
   228→      const convStore = ConversationStore.getInstance();
   229→
   230→      // 1. Resolve or establish durable conversation identity bound to authenticated Founder
   231→      let conversation;
   232→      try {
   233→        conversation = await convStore.getOrCreateConversation({
   234→          founderId: session.userId,
   235→          conversationId: typeof conversationId === 'string' && conversationId.trim() ? conversationId.trim() : undefined,
   236→          agentId: 'sophia',
   237→        });
   238→      } catch (err: any) {
   239→        if (err instanceof ConversationSecurityError || err.name === 'ConversationSecurityError') {
   240→          return NextResponse.json({ error: `Forbidden: ${err.message}` }, { status: 403 });
   241→        }
   242→        if (err instanceof ConversationNotFoundError || err.name === 'ConversationNotFoundError') {
   243→          return NextResponse.json({ error: `Not Found: ${err.message}` }, { status: 404 });
   244→        }
   245→        return NextResponse.json({ error: err.message || 'Conversation resolution failed' }, { status: 500 });
   246→      }
   247→
   248→      // Check if this turn was already completed (idempotent replay)
   249→      const cleanIdempotencyKey = typeof idempotencyKey === 'string' && idempotencyKey.trim().length > 0
   250→        ? idempotencyKey.trim()
   251→        : undefined;
   252→
   253→      if (cleanIdempotencyKey) {
   254→        const existingAssistantMessage = await convStore.findMessageByIdempotencyKey(
   255→          conversation.id,
   256→          `${cleanIdempotencyKey}:assistant`
   257→        );
   258→        if (existingAssistantMessage) {
   259→          return NextResponse.json({
   260→            success: true,
   261→            agentId: SERVER_AGENTS.coo.id,
   262→            name: SERVER_AGENTS.coo.name,
   263→            role: SERVER_AGENTS.coo.role,
   264→            conversationId: conversation.id,
   265→            messageId: existingAssistantMessage.id,
   266→            intent: existingAssistantMessage.intent || 'conversation',
   267→            classification: {
   268→              intent: existingAssistantMessage.intent || 'conversation',
   269→              confidence: existingAssistantMessage.confidence ?? 1.0,
   270→              reason: 'Idempotent turn replay',
   271→            },
   272→            reply: existingAssistantMessage.content,
   273→            liveAi: existingAssistantMessage.metadata?.liveAi ?? false,
   274→            metrics: {
   275→              contextAssemblyMs: 0,
   276→              modelMs: 0,
   277→              gatewayValidationMs: 0,
   278→              totalTurnMs: 0,
   279→              estimatedTokens: existingAssistantMessage.metadata?.tokens || { input: 0, output: 0 },
   280→            },
   281→            directiveExecuted: existingAssistantMessage.metadata?.directiveExecuted ?? false,
   282→            idempotentReplay: true,
   283→          });
   284→        }
   285→      }
   286→
   287→      // Concurrency lock for duplicate in-flight turns
   288→      const turnKey = cleanIdempotencyKey ? `${session.userId}:${conversation.id}:${cleanIdempotencyKey}` : null;
   289→      if (turnKey && inFlightTurns.has(turnKey)) {
   290→        const cached = await inFlightTurns.get(turnKey)!;
   291→        return NextResponse.json(cached.data, { status: cached.status || 200 });
   292→      }
   293→
   294→      const executeTurn = async (): Promise<{ status?: number; data: any }> => {
   295→        // 2. Persist incoming Founder turn (with deduplication / idempotencyKey)
   296→        let founderMessageRecord;
   297→        try {
   298→          founderMessageRecord = await convStore.saveMessage(
   299→            {
   300→              conversationId: conversation.id,
   301→              sender: 'founder',
   302→              role: 'user',
   303→              content: message,
   304→              idempotencyKey: cleanIdempotencyKey,
   305→            },
   306→            session.userId
   307→          );
   308→        } catch (err: any) {
   309→          if (err instanceof ConversationSecurityError || err.name === 'ConversationSecurityError') {
   310→            return { status: 403, data: { error: `Forbidden: ${err.message}` } };
   311→          }
   312→          return { status: 500, data: { error: err.message || 'Failed to persist turn' } };
   313→        }
   314→
   315→        // 3. Server-authoritative history retrieval: fetch recent bounded dialogue history
   316→        const serverHistory = await convStore.getRecentHistory(conversation.id, session.userId, 10);
   317→        const priorHistory = serverHistory.filter((h) => h.id !== founderMessageRecord.id);
   318→
   319→        const hasExplicitConversation = typeof conversationId === 'string' && conversationId.trim().length > 0;
   320→
   321→        const historyItems = priorHistory.length > 0
   322→          ? priorHistory.map((h) => ({
   323→              sender: h.sender,
   324→              text: h.text,
   325→            }))
   326→          : !hasExplicitConversation && Array.isArray(history)
   327→            ? history
   328→                .filter((h: any) => h && (typeof h.text === "string" || typeof h.content === "string"))
   329→                .map((h: any) => ({
   330→                  sender: h.sender === "founder" || h.role === "user" ? ("founder" as const) : ("assistant" as const),
   331→                  text: (h.text || h.content || "") as string,
   332→                }))
   333→            : [];
   334→
   335→        // 4. Context Assembly: deterministic, multi-source, authority-classified
   336→        //    (M3 K-2: threads the authenticated session principal so the
   337→        //    founder-scoped PERSONAL_MIND_MEMORY slice renders for THIS
   338→        //    founder only — personal context, never company authority.)
   339→        const assembledContext = await SophiaContextAssembler.assemble({
   340→          message,
   341→          history: historyItems,
   342→          founderId: session.userId,
   343→        });
   344→
   345→        // 5. Cognitive Ingress: Contextual semantic intent classification with structural trust boundary
   346→        const classificationResult = await SophiaIntentClassifier.classify({
   347→          message,
   348→          context: assembledContext,
   349→          history: historyItems,
   350→          tone,
   351→          customPrompt: personaConfig?.customPrompt,
   352→        });
   353→
   354→        // 6. Server Trust Boundary Gateway: Verify principal, evaluate policy, enforce invariants, dispatch
   355→        const executionResult = await SophiaServerGateway.process({
   356→          proposal: classificationResult.proposal,
   357→          session,
   358→          message,
   359→          context: assembledContext,
   360→          executeDirective: !!executeDirective,
   361→        });
   362→
   363→        if (!executionResult.success && executionResult.error?.includes('Forbidden')) {
   364→          return { status: 403, data: { error: executionResult.error } };
   365→        }
   366→
   367→        const mapKindToIntent = (kind: string): MessageIntent => {
   368→          switch (kind) {
   369→            case 'conversation': return 'conversation';
   370→            case 'informational_query': return 'information_request';
   371→            case 'directive_proposal':
   372→            case 'steering_proposal':
   373→            case 'operational_inspection': return 'directive';
   374→            case 'approval_proposal': return 'approval_action';
   375→            case 'clarification_prompt': return 'ambiguous';
   376→            default: return 'conversation';
   377→          }
   378→        };
   379→
   380→        const resolvedIntent = executionResult.validatedCommand.type === 'PRESENT_CLARIFICATION'
   381→          ? 'ambiguous'
   382→          : mapKindToIntent(executionResult.proposal.kind);
   383→
   384→        // 7. Persist assistant turn to durable storage with assistantIdempotencyKey
   385→        //    (M4-A note: explicit ChatMessageRecord | null typing — the capture
   386→        //    stage below reads assistantMessageRecord?.id, so the historic
   387→        //    implicit-null narrowing quirk is closed here with a pure
   388→        //    annotation; zero runtime change.)
   389→        let assistantMessageRecord: ChatMessageRecord | null = null;
   390→        if (executionResult.reply) {
   391→          try {
   392→            const assistantIdempotencyKey = cleanIdempotencyKey
   393→              ? `${cleanIdempotencyKey}:assistant`
   394→              : undefined;
   395→
   396→            assistantMessageRecord = await convStore.saveMessage(
   397→              {
   398→                conversationId: conversation.id,
   399→                sender: 'assistant',
   400→                role: 'assistant',
   401→                content: executionResult.reply,
   402→                intent: resolvedIntent,
   403→                confidence: executionResult.proposal.confidence,
   404→                commandType: executionResult.validatedCommand.type,
   405→                idempotencyKey: assistantIdempotencyKey,
   406→                metadata: {
   407→                  directiveExecuted: executionResult.directiveExecuted,
   408→                  liveAi: executionResult.liveAi,
   409→                  tokens: executionResult.metrics?.estimatedTokens,
   410→                  replyToIdempotencyKey: cleanIdempotencyKey,
   411→                },
   412→              },
   413→              session.userId
   414→            );
   415→          } catch (err) {
   416→            console.error('[agent-chat] Failed to persist assistant turn:', err);
   417→          }
   418→        }
   419→
   420→        // 8. M4-A Personal Mind capture (fire-and-forget — NEVER a
   421→        //     conversational dependency): after the assistant turn is
   422→        //     durably persisted, the capture stage proposes personal-memory
   423→        //     candidates through the deterministic MemoryGate. Only
   424→        //     NEEDS_REVIEW candidates persist, INACTIVE, pending explicit
   425→        //     Founder confirmation via the governed /api/sofia/memory PATCH.
   426→        //     Capture failures are contained inside the stage and can never
   427→        //     fail this turn; a replayed turn returns at the idempotency
   428→        //     check above before ever reaching here. (This mirrors the
   429→        //     executeSophiaTurn integration — the two Sophia paths are
   430→        //     deliberately NOT converged in M4-A.)
   431→        if (executionResult.success && executionResult.reply && founderMessageRecord) {
   432→          scheduleSophiaMemoryCapture({
   433→            founderId: session.userId,
   434→            conversationId: conversation.id,
   435→            founderMessageId: founderMessageRecord.id,
   436→            assistantMessageId: assistantMessageRecord?.id,
   437→            turnId: cleanIdempotencyKey,
   438→            founderMessage: message,
   439→            assistantReply: executionResult.reply,
   440→            ingress: 'agent_chat',
   441→          });
   442→        }
   443→
   444→        return {
   445→          status: 200,
   446→          data: {
   447→            success: executionResult.success,
   448→            agentId: SERVER_AGENTS.coo.id,
   449→            name: SERVER_AGENTS.coo.name,
   450→            role: SERVER_AGENTS.coo.role,
   451→            conversationId: conversation.id,
   452→            messageId: assistantMessageRecord?.id || founderMessageRecord.id,
   453→            intent: resolvedIntent,
   454→            classification: {
   455→              intent: resolvedIntent,
   456→              confidence: executionResult.proposal.confidence,
   457→              reason: (executionResult.validatedCommand as any).ambiguityReason || (executionResult.proposal as any).reason || (executionResult.proposal as any).ambiguityReason || 'Contextually classified',
   458→              directiveTitle: (executionResult.proposal as any).title,
   459→              suggestedScope: (executionResult.validatedCommand as any).suggestedScope || (executionResult.proposal as any).suggestedScope,
   460→              structuredOptions: (executionResult.validatedCommand as any).structuredOptions,
   461→              approvalAction: executionResult.validatedCommand.type === 'RESOLVE_APPROVAL'
   462→                ? ((executionResult.proposal as any).decision === 'approved' ? 'approve' : (executionResult.proposal as any).decision === 'rejected' ? 'reject' : undefined)
   463→                : undefined,
   464→              approvalNote: (executionResult.proposal as any).note,
   465→            },
   466→            reply: executionResult.reply,
   467→            liveAi: executionResult.liveAi,
   468→            metrics: executionResult.metrics || {
   469→              contextAssemblyMs: 5,
   470→              modelMs: 20,
   471→              gatewayValidationMs: 2,
   472→              totalTurnMs: 27,
   473→              estimatedTokens: { input: assembledContext.estimatedTokens, output: 80 },
   474→            },
   475→            directiveExecuted: executionResult.directiveExecuted,
   476→            orchestrationRun: executionResult.orchestrationRun,
   477→            authoritativeData: executionResult.authoritativeData,
   478→          },
   479→        };
   480→      };
   481→
   482→      if (turnKey) {
   483→        const turnPromise = executeTurn();
   484→        inFlightTurns.set(turnKey, turnPromise);
   485→        try {
   486→          const res = await turnPromise;
   487→          return NextResponse.json(res.data, { status: res.status || 200 });
   488→        } finally {
   489→          inFlightTurns.delete(turnKey);
   490→        }
   491→      }
   492→
   493→      const res = await executeTurn();
   494→      return NextResponse.json(res.data, { status: res.status || 200 });
   495→    }
   496→
   497→    const classification = classifyMessageIntent(message);
   498→    const isAdvisor = agentId === "advisor";
   499→    const persona = isAdvisor
   500→      ? ADVISOR_PERSONA
   501→      : (agentId in SERVER_AGENTS ? SERVER_AGENTS[agentId as AgentRole] : SERVER_AGENTS.coo);
   502→
   503→
   504→    let toneGuidance = "";
   505→    if (tone === "casual") {
   506→      toneGuidance = `
   507→=== DEMEANOR & TONE ARCHETYPE: [CASUAL & CANDID] ===
   508→- You speak with the Founder as a trusted, high-energy startup collaborator and conversational peer.
   509→- Demeanor: Relaxed, candid, energetic, approachable, friendly, and zero corporate fluff or bureaucratic jargon.

   713→export async function GET(req: NextRequest) {
   714→  try {
   715→    const session = await getAuthenticatedFounder(req);
   716→    if (!session) {
   717→      return NextResponse.json(
   718→        { error: 'Unauthorized: Session required to retrieve conversations' },
   719→        { status: 401 }
   720→      );
   721→    }
   722→
   723→    const { searchParams } = new URL(req.url);
   724→    const conversationId = searchParams.get('conversationId');
   725→    const limitParam = searchParams.get('limit');
   726→    const limit = limitParam ? Math.min(Math.max(parseInt(limitParam, 10) || 20, 1), 100) : 20;
   727→
   728→    const convStore = ConversationStore.getInstance();
   729→
   730→    if (conversationId && conversationId.trim().length > 0) {
   731→      try {
   732→        const conversation = await convStore.getConversation(session.userId, conversationId.trim());
   733→        if (!conversation) {
   734→          return NextResponse.json({ error: `Not Found: Conversation "${conversationId}" not found` }, { status: 404 });
   735→        }
   736→        const messages = await convStore.getMessages(conversation.id, session.userId, limit);
   737→        return NextResponse.json({
   738→          success: true,
   739→          conversation,
   740→          messages,
   741→        });
   742→      } catch (err: any) {
   743→        if (err instanceof ConversationSecurityError || err.name === 'ConversationSecurityError') {
   744→          return NextResponse.json({ error: `Forbidden: ${err.message}` }, { status: 403 });
   745→        }
   746→        if (err instanceof ConversationNotFoundError || err.name === 'ConversationNotFoundError') {
   747→          return NextResponse.json({ error: `Not Found: ${err.message}` }, { status: 404 });
   748→        }
   749→        return NextResponse.json({ error: err.message || 'Failed to retrieve conversation' }, { status: 500 });
   750→      }
   751→    }
   752→
   753→    // List recent conversations for the authenticated Founder
   754→    const conversations = await convStore.listConversations(session.userId, limit);
   755→    return NextResponse.json({
   756→      success: true,
   757→      conversations,
   758→    });
   759→  } catch (error: any) {
   760→    return NextResponse.json({ error: error.message || 'Failed to list conversations' }, { status: 500 });
   761→  }
   762→}
