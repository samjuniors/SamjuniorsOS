   197→  async claimStepAtomic(
   198→    instanceId: string,
   199→    stepId: string,
   200→    workerId: string,
   201→    expectedVersion?: number
   202→  ): Promise<{ instance: WorkflowInstanceState; step: WorkflowStepState }> {
   203→    const db = await requireAuthoritativeDatabase();
   204→    return await db.$transaction(async (tx) => {
   205→      const dbInst = await tx.workflowInstance.findUnique({
   206→        where: { id: instanceId },
   207→      });
   208→      if (!dbInst) {
   209→        throw new Error(`Workflow instance not found: ${instanceId}`);
   210→      }
   211→
   212→      if (expectedVersion !== undefined && dbInst.stateVersion !== expectedVersion) {
   213→        throw new ConcurrencyConflictError(instanceId, expectedVersion, dbInst.stateVersion);
   214→      }
   215→
   216→      const stepStates: Record<string, WorkflowStepState> = (dbInst.stepStates as any) || {};
   217→      const step = stepStates[stepId];
   218→      if (!step) {
   219→        throw new Error(`Step not found in workflow instance: ${stepId}`);
   220→      }
   221→
   222→      if (step.status !== 'ready') {
   223→        throw new StepClaimError(instanceId, stepId, step.status, step.claimedBy);
   224→      }
   225→
   226→      if (step.claimedBy && step.claimedBy !== workerId) {
   227→        throw new StepClaimError(instanceId, stepId, step.status, step.claimedBy);
   228→      }
   229→
   230→      validateStepTransition(step.status, 'running');
   231→
   232→      const nowIso = new Date().toISOString();
   233→      step.status = 'running';
   234→      step.claimedBy = workerId;
   235→      step.claimedAt = nowIso;
   236→      if (!step.startedAt) {
   237→        step.startedAt = nowIso;
   238→      }
   239→      stepStates[stepId] = step;
   240→
   241→      let instanceStatus = dbInst.status;
   242→      if (instanceStatus === 'pending' || instanceStatus === 'waiting') {
   243→        validateInstanceTransition(instanceStatus as any, 'running');
   244→        instanceStatus = 'running';
   245→      }
   246→
   247→      const nextVersion = (dbInst.stateVersion || 0) + 1;
   248→
   249→      const updated = await tx.workflowInstance.update({
   250→        where: { id: instanceId },
   251→        data: {
   252→          status: instanceStatus,
   253→          stepStates: stepStates as any,
   254→          stateVersion: nextVersion,
   255→          claimedBy: workerId,
   256→          claimedAt: new Date(),
   257→          updatedAt: new Date(),
   258→        },
   259→      });
   260→
   261→      return {
   262→        instance: this.mapPrismaToInstance(updated),
   263→        step,
   264→      };
   265→    });
   266→  }

   268→  async transitionStepAtomic(
   269→    instanceId: string,
   270→    stepId: string,
   271→    targetStatus: WorkflowStepStatus,
   272→    expectedVersion?: number,
   273→    patch?: Partial<WorkflowStepState>,
   274→    instancePatch?: Partial<WorkflowInstanceState>
   275→  ): Promise<WorkflowInstanceState> {
   276→    const db = await requireAuthoritativeDatabase();
   277→    return await db.$transaction(async (tx) => {
   278→      const dbInst = await tx.workflowInstance.findUnique({
   279→        where: { id: instanceId },
   280→      });
   281→      if (!dbInst) {
   282→        throw new Error(`Workflow instance not found: ${instanceId}`);
   283→      }
   284→
   285→      if (expectedVersion !== undefined && dbInst.stateVersion !== expectedVersion) {
   286→        throw new ConcurrencyConflictError(instanceId, expectedVersion, dbInst.stateVersion);
   287→      }
   288→
   289→      const stepStates: Record<string, WorkflowStepState> = (dbInst.stepStates as any) || {};
   290→      const step = stepStates[stepId];
   291→      if (!step) {
   292→        throw new Error(`Step not found in workflow instance: ${stepId}`);
   293→      }
   294→
   295→      validateStepTransition(step.status, targetStatus);
   296→
   297→      const nowIso = new Date().toISOString();
   298→      step.status = targetStatus;
   299→      if (targetStatus === 'completed' || targetStatus === 'failed' || targetStatus === 'cancelled') {
   300→        step.completedAt = nowIso;
   301→        step.claimedBy = undefined;
   302→      }
   303→      if (patch) {
   304→        Object.assign(step, patch);
   305→      }
   306→      stepStates[stepId] = step;
   307→
   308→      let nextInstanceStatus = instancePatch?.status ?? dbInst.status;
   309→      if (instancePatch?.status && instancePatch.status !== dbInst.status) {
   310→        validateInstanceTransition(dbInst.status as any, instancePatch.status);
   311→      }
   312→
   313→      const nextVersion = (dbInst.stateVersion || 0) + 1;
   314→
   315→      const updated = await tx.workflowInstance.update({
   316→        where: { id: instanceId },
   317→        data: {
   318→          status: nextInstanceStatus,
   319→          stepStates: stepStates as any,
   320→          stateVersion: nextVersion,
   321→          error: instancePatch?.failureReason ?? (targetStatus === 'failed' ? (step.error || 'Step failed') : dbInst.error),
   322→          completedAt: (nextInstanceStatus === 'completed' || targetStatus === 'completed') ? new Date() : dbInst.completedAt,
   323→          updatedAt: new Date(),
   324→        },
   325→      });
   326→
   327→      return this.mapPrismaToInstance(updated);
   328→    });
   329→  }
   330→
   331→  async rehydrateActiveInstances(): Promise<WorkflowInstanceState[]> {
   332→    const db = await requireAuthoritativeDatabase();
   333→    const activeStatuses = ['pending', 'running', 'waiting', 'blocked', 'awaiting_approval'];
   334→    const dbInsts = await db.workflowInstance.findMany({
   335→      where: {
   336→        status: { in: activeStatuses },
   337→      },
   338→      orderBy: { updatedAt: 'asc' },
   339→    });
   340→    return dbInsts.map((inst) => this.mapPrismaToInstance(inst));
   341→  }
   342→
   343→  private mapPrismaToInstance(dbInst: any): WorkflowInstanceState {
   344→    return {
   345→      instanceId: dbInst.id,
   346→      workflowId: dbInst.definitionId,
   347→      version: dbInst.version,
   348→      stateVersion: dbInst.stateVersion ?? 0,
   349→      claimedBy: dbInst.claimedBy || undefined,
   350→      claimedAt: dbInst.claimedAt ? dbInst.claimedAt.toISOString() : undefined,
   351→      status: dbInst.status as any,
   352→      objective: dbInst.objective,
   353→      stepStates: (dbInst.stepStates as any) || {},
   354→      createdAt: dbInst.createdAt.toISOString(),
   355→      updatedAt: dbInst.updatedAt.toISOString(),
   356→      outputs: {},
   357→      evidenceReferences: [],
   358→      failureReason: dbInst.error || undefined,
   359→    };
   360→  }
   361→
   362→  clear(): void {
   363→    if (process.env.NODE_ENV === 'production') {
   364→      throw new Error('Workflow records cannot be cleared in production.');
   365→    }
   366→  }

   374→export class InMemoryWorkflowStore implements WorkflowDefinitionStore, WorkflowInstanceStore {

   391→  public loadFromDurableStorage(): void {
   392→    try {
   393→      const persistedDefs = DurableFileStore.getInstance().readCollection<WorkflowDefinition>('workflow_definitions');
   394→      for (const [key, def] of Object.entries(persistedDefs)) {
   395→        this.definitions.set(key, def);
   396→      }
   397→      const persistedInsts = DurableFileStore.getInstance().readCollection<WorkflowInstanceState>('workflow_instances');
   398→      for (const [id, inst] of Object.entries(persistedInsts)) {
   399→        this.instances.set(id, inst);
   400→      }
   401→    } catch {
   402→      // fallback
   403→    }
   404→  }

   410→  async saveDefinition(definition: WorkflowDefinition): Promise<void> {
   411→    if (isAuthoritativeMode()) {
   412→      return PostgresWorkflowStore.getInstance().saveDefinition(definition);
   413→    }
   414→
   415→    const key = `${definition.id}@${definition.version}`;

   419→    try {
   420→      DurableFileStore.getInstance().saveItem('workflow_definitions', key, definition);
   421→      DurableFileStore.getInstance().saveItem('workflow_definitions', definition.id, definition);
   422→    } catch {}
   423→
   424→    if (process.env.DATABASE_URL) {
   425→      try {
   426→        await prisma.workflowDefinition.upsert({
   427→          where: {
   428→            id_version: {
   429→              id: definition.id,
   430→              version: definition.version || '1.0.0',
   431→            },
   432→          },
   433→          create: {
   434→            id: definition.id,

   436→            name: definition.name,
   437→            description: definition.description,
   438→            steps: (definition.steps as any) ?? [],
   439→          },

   441→            name: definition.name,
   442→            description: definition.description,

   465→            id,
   466→            ...(version ? { version } : {}),
   467→          },

   614→  async claimStepAtomic(
   615→    instanceId: string,
   616→    stepId: string,
   617→    workerId: string,
   618→    expectedVersion?: number
   619→  ): Promise<{ instance: WorkflowInstanceState; step: WorkflowStepState }> {
   620→    if (isAuthoritativeMode()) {
   621→      return PostgresWorkflowStore.getInstance().claimStepAtomic(instanceId, stepId, workerId, expectedVersion);
   622→    }
   623→
   624→    const instance = this.instances.get(instanceId);
   625→    if (!instance) {
   626→      throw new Error(`Workflow instance not found: ${instanceId}`);
   627→    }
   628→
   629→    const currentVersion = instance.stateVersion ?? 0;
   630→    if (expectedVersion !== undefined && currentVersion !== expectedVersion) {
   631→      throw new ConcurrencyConflictError(instanceId, expectedVersion, currentVersion);
   632→    }
   633→
   634→    const step = instance.stepStates[stepId];
   635→    if (!step) {
   636→      throw new Error(`Step not found in workflow instance: ${stepId}`);
   637→    }
   638→
   639→    if (step.status !== 'ready') {
   640→      throw new StepClaimError(instanceId, stepId, step.status, step.claimedBy);
   641→    }
   642→
   643→    if (step.claimedBy && step.claimedBy !== workerId) {
   644→      throw new StepClaimError(instanceId, stepId, step.status, step.claimedBy);
   645→    }
   646→
   647→    validateStepTransition(step.status, 'running');
   648→
   649→    const nowIso = new Date().toISOString();
   650→    step.status = 'running';
   651→    step.claimedBy = workerId;
   652→    step.claimedAt = nowIso;
   653→    if (!step.startedAt) {
   654→      step.startedAt = nowIso;
   655→    }
   656→
   657→    if (instance.status === 'pending' || instance.status === 'waiting') {
   658→      validateInstanceTransition(instance.status, 'running');
   659→      instance.status = 'running';
   660→    }
   661→
   662→    instance.stateVersion = currentVersion + 1;
   663→    instance.claimedBy = workerId;
   664→    instance.claimedAt = nowIso;
   665→    instance.updatedAt = nowIso;
   666→
   667→    const clone = JSON.parse(JSON.stringify(instance));
   668→    this.instances.set(instanceId, clone);
   669→
   670→    try {
   671→      DurableFileStore.getInstance().saveItem('workflow_instances', instanceId, clone);
   672→    } catch {}
   673→
   674→    return {
   675→      instance: clone,
   676→      step: JSON.parse(JSON.stringify(step)),
   677→    };
   678→  }
   679→
   680→  async transitionStepAtomic(
   681→    instanceId: string,
   682→    stepId: string,
   683→    targetStatus: WorkflowStepStatus,
   684→    expectedVersion?: number,
   685→    patch?: Partial<WorkflowStepState>,
   686→    instancePatch?: Partial<WorkflowInstanceState>
   687→  ): Promise<WorkflowInstanceState> {
   688→    if (isAuthoritativeMode()) {
   689→      return PostgresWorkflowStore.getInstance().transitionStepAtomic(
   690→        instanceId,
   691→        stepId,
   692→        targetStatus,
   693→        expectedVersion,
   694→        patch,
   695→        instancePatch
   696→      );
   697→    }
   698→
   699→    const instance = this.instances.get(instanceId);
   700→    if (!instance) {
   701→      throw new Error(`Workflow instance not found: ${instanceId}`);
   702→    }
   703→
   704→    const currentVersion = instance.stateVersion ?? 0;
   705→    if (expectedVersion !== undefined && currentVersion !== expectedVersion) {
   706→      throw new ConcurrencyConflictError(instanceId, expectedVersion, currentVersion);
   707→    }
   708→
   709→    const step = instance.stepStates[stepId];
   710→    if (!step) {
   711→      throw new Error(`Step not found in workflow instance: ${stepId}`);
   712→    }
   713→
   714→    validateStepTransition(step.status, targetStatus);
   715→
   716→    const nowIso = new Date().toISOString();
   717→    step.status = targetStatus;
   718→    if (targetStatus === 'completed' || targetStatus === 'failed' || targetStatus === 'cancelled') {
   719→      step.completedAt = nowIso;
   720→      step.claimedBy = undefined;
   721→    }
   722→    if (patch) {
   723→      Object.assign(step, patch);
   724→    }
   725→
   726→    if (instancePatch?.status && instancePatch.status !== instance.status) {
   727→      validateInstanceTransition(instance.status, instancePatch.status);
   728→      instance.status = instancePatch.status;
   729→    }
   730→    if (instancePatch?.failureReason) {
   731→      instance.failureReason = instancePatch.failureReason;
   732→    }
   733→
   734→    instance.stateVersion = currentVersion + 1;
   735→    instance.updatedAt = nowIso;
   736→
   737→    const clone = JSON.parse(JSON.stringify(instance));
   738→    this.instances.set(instanceId, clone);
   739→
   740→    try {
   741→      DurableFileStore.getInstance().saveItem('workflow_instances', instanceId, clone);
   742→    } catch {}
   743→
   744→    return clone;
   745→  }

   747→  async rehydrateActiveInstances(): Promise<WorkflowInstanceState[]> {
   748→    if (isAuthoritativeMode()) {
   749→      return PostgresWorkflowStore.getInstance().rehydrateActiveInstances();
   750→    }
   751→
   752→    const activeStatuses: Set<string> = new Set(['pending', 'running', 'waiting', 'blocked', 'awaiting_approval']);
   753→    const active = Array.from(this.instances.values())
   754→      .filter((i) => activeStatuses.has(i.status))
   755→      .sort((a, b) => new Date(a.updatedAt).getTime() - new Date(b.updatedAt).getTime());
   756→
   757→    return JSON.parse(JSON.stringify(active));
   758→  }

   761→    this.definitions.clear();
   762→    this.instances.clear();
   763→    try {
   764→      DurableFileStore.getInstance().clearCollection('workflow_definitions');
   765→      DurableFileStore.getInstance().clearCollection('workflow_instances');
   766→    } catch {}
   767→  }
