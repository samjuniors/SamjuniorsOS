    14→ * The tool surface is ported from the original MCP servers (panels.mjs,

    64→- Never refuse. State a constraint once; if overruled, comply and never raise it

   609→ * replaced. Everything it needs to talk to the browser arrives in `io`:

   632→  /**
   633→   * The backend rate-limits bursts. Speech is realtime, so retries are short
   634→   * and few: two quick attempts, then the turn says so in plain words.
   635→   */
   636→  async function withRetry(fn, tries = 3, base = 900) {
   637→    let lastErr
   638→    for (let i = 0; i < tries; i++) {
   639→      try {
   640→        return await fn()
   641→      } catch (err) {
   642→        lastErr = err
   643→        const msg = String(err?.message ?? err)
   644→        const transient = /\b(429|rate|too many|502|503|504|timeout|network|fetch failed)\b/i.test(msg)
   645→        if (!transient || i === tries - 1) throw err
   646→        await new Promise((r) => setTimeout(r, base * (i + 1)))
   647→      }
   648→    }
   649→    throw lastErr
   650→  }

   922→  /**
   923→   * One model request: streamed. Returns { content, toolCalls, finishReason }.
   924→   *
   925→   * The provider is no longer assumed — openLlmStream walks the chain (z-ai
   926→   * → Gemini → a local server) and hands back the first live SSE reader, so
   927→   * a rate-limited brain degrades to a fallback mid-session instead of
   928→   * failing the turn. Text deltas are forwarded to the browser as they
   929→   * arrive; tool calls come back complete in a single chunk (verified against
   930→   * the backend), but are accumulated by index regardless so a split call
   931→   * still works.
   932→   *
   933→   * Every provider speaks the OpenAI wire format, so one parser serves them
   934→   * all. Reads carry a stall guard rather than the request a timeout: a
   935→   * local model that is slow is normal, a connection that has gone quiet for
   936→   * 90 seconds is a dead link, and the difference is worth failing over on.
   937→   */
   938→  async function streamModel(deadline) {
   939→    const body = {
   940→      messages: trimHistory(),
   941→      tools: TOOLS,
   942→      stream: true,
   943→    }
   944→
   945→    const { reader } = await openLlmStream(body, (p) =>
   946→      io.send({ type: 'provider', brain: { id: p.id, label: p.label } }),
   947→    )
   948→
   949→    const readWithStall = async () => {
   950→      let timer
   951→      const stalled = new Promise((_, reject) => {
   952→        timer = setTimeout(
   953→          () => reject(new Error('the model stream went quiet')),
   954→          90_000,
   955→        )
   956→      })
   957→      try {
   958→        return await Promise.race([reader.read(), stalled])
   959→      } finally {
   960→        clearTimeout(timer)
   961→      }
   962→    }
   963→
   964→    const decoder = new TextDecoder()
   965→    let buf = ''
   966→    let content = ''
   967→    const calls = new Map()
   968→    let finishReason = null
   969→
   970→    const handleEvent = (ev) => {
   971→      const choice = ev?.choices?.[0]
   972→      if (!choice) return
   973→      const delta = choice.delta ?? {}
   974→      if (typeof delta.content === 'string' && delta.content) {
   975→        let piece = delta.content
   976→        if (!content) {
   977→          // The backend opens every round with a stray newline; speech and
   978→          // the transcript both want it gone.
   979→          piece = piece.replace(/^\s+/, '')
   980→        }
   981→        content += delta.content
   982→        if (piece) io.sendTurn({ type: 'text', delta: piece })
   983→      }
   984→      if (Array.isArray(delta.tool_calls)) {
   985→        for (const tc of delta.tool_calls) {
   986→          const idx = tc.index ?? 0
   987→          const slot = calls.get(idx) ?? { id: '', type: 'function', function: { name: '', arguments: '' } }
   988→          if (tc.id) slot.id = tc.id
   989→          if (tc.function?.name) slot.function.name += tc.function.name
   990→          if (tc.function?.arguments) slot.function.arguments += tc.function.arguments
   991→          if (tc.type) slot.type = tc.type
   992→          calls.set(idx, slot)
   993→        }
   994→      }
   995→      if (choice.finish_reason) finishReason = choice.finish_reason
   996→    }
   997→
   998→    while (true) {
   999→      if (aborted) {
  1000→        reader.cancel().catch(() => {})
  1001→        throw ABORTED
  1002→      }
  1003→      const { done, value } = await readWithStall()
  1004→      if (done) break
  1005→      buf += decoder.decode(value, { stream: true })
  1006→      let nl
  1007→      while ((nl = buf.indexOf('\n')) !== -1) {
  1008→        const line = buf.slice(0, nl).replace(/\r$/, '')
  1009→        buf = buf.slice(nl + 1)
  1010→        if (!line.startsWith('data:')) continue
  1011→        const data = line.slice(5).trim()
  1012→        if (!data || data === '[DONE]') continue
  1013→        try {
  1014→          handleEvent(JSON.parse(data))
  1015→        } catch {
  1016→          /* a torn line in a debug payload — the next chunk completes it */
  1017→        }
  1018→      }
  1019→      if (deadline && Date.now() > deadline) {
  1020→        reader.cancel().catch(() => {})
  1021→        throw new Error('the model is taking too long')
  1022→      }
  1023→    }
  1024→
  1025→    return {
  1026→      content,
  1027→      toolCalls: [...calls.values()].filter((c) => c.function?.name),
  1028→      finishReason,
  1029→    }
  1030→  }

  1032→  /** Keep the persona and the recent window; the backend has a context limit. */
  1033→  function trimHistory() {
  1034→    const persona = messages[0]
  1035→    const rest = messages.slice(1).slice(-40)
  1036→    // A 'tool' message must sit directly after the assistant message that
  1037→    // requested it, so trim to a boundary the API accepts: the first kept
  1038→    // message has to be a plain user or assistant turn.
  1039→    while (rest.length && rest[0].role === 'tool') rest.shift()
  1040→    return [persona, ...rest]
  1041→  }

  1046→  async function processTurn(text, id) {
  1047→    io.begin(id)
  1048→    aborted = false
  1049→    messages.push({ role: 'user', content: text })
  1050→
  1051→    let spoken = ''
  1052→    try {
  1053→      // A turn may need several model calls: speak, call tools, speak again.
  1054→      for (let round = 0; round < 8; round++) {
  1055→        const out = await streamModel(round === 0 ? null : Date.now() + 120_000)
  1056→
  1057→        if (!out.toolCalls.length) {
  1058→          spoken += out.content
  1059→          messages.push({ role: 'assistant', content: out.content || '' })
  1060→          io.sendTurn({ type: 'done', text: spoken.trim(), costUsd: null })
  1061→          io.finishTurn()
  1062→          return
  1063→        }
  1064→
  1065→        // Record the assistant's tool request in the OpenAI wire shape —
  1066→        // the backend rejects a bare name/arguments form.
  1067→        messages.push({
  1068→          role: 'assistant',
  1069→          content: out.content || '',
  1070→          tool_calls: out.toolCalls.map((c) => ({
  1071→            id: c.id,
  1072→            type: 'function',
  1073→            function: { name: c.function.name, arguments: c.function.arguments || '{}' },
  1074→          })),
  1075→        })
  1076→        spoken += out.content
  1077→
  1078→        for (const call of out.toolCalls) {
  1079→          if (aborted || closed) throw ABORTED
  1080→          io.announce(call.id, call.function.name)
  1081→          let result
  1082→          try {
  1083→            const args = safeJson(call.function.arguments)
  1084→            result = await executeTool(call.function.name, args)
  1085→          } catch (err) {
  1086→            console.error(`[jarvis] tool ${call.function.name} failed:`, err?.message ?? err)
  1087→            result = refuse(`The tool failed: ${String(err?.message ?? err).slice(0, 160)}`)
  1088→          }
  1089→          io.settle(call.id, result?.isError === true)
  1090→          messages.push({
  1091→            role: 'tool',
  1092→            tool_call_id: call.id,
  1093→            content: result?.isError ? `ERROR: ${result.text}` : result.text,
  1094→          })
  1095→        }
  1096→      }
  1097→      // Ran out of rounds — say something rather than spinning forever.
  1098→      messages.push({ role: 'assistant', content: 'I am afraid that task ran long, sir.' })
  1099→      io.sendTurn({ type: 'done', text: 'I am afraid that task ran long, sir.', costUsd: null })
  1100→      io.finishTurn()
  1101→    } catch (err) {
  1102→      if (err === ABORTED || aborted) {
  1103→        // A barge-in. The client has already settled itself with the words
  1104→        // said so far; keep whatever was spoken in history and stop quietly.
  1105→        if (spoken.trim()) messages.push({ role: 'assistant', content: spoken })
  1106→        io.finishTurn()
  1107→        return
  1108→      }
  1109→      console.error('[jarvis] turn failed:', err)
  1110→      // Every link in the chain has failed — say so in the character's own
  1111→      // register, and name the ones that were tried, because a person staring
  1112→      // at a dead HUD wants to know whether to wait or to fix something.
  1113→      const tried = llmInfo()
  1114→        .providers.filter((p) => p.configured)
  1115→        .map((p) => p.label)
  1116→        .join(', ')
  1117→      io.sendTurn({
  1118→        type: 'error',
  1119→        message: `I am afraid every reasoning engine is unreachable — ${tried || 'none'} — try again in a moment.`,
  1120→      })
  1121→      io.finishTurn()
  1122→    }
  1123→  }

  1133→  async function pump() {
  1134→    if (running || closed) return
  1135→    running = true
  1136→    try {
  1137→      while (queue.length && !closed) {
  1138→        const { text, id } = queue.shift()
  1139→        await processTurn(text, id)
  1140→      }
  1141→    } finally {
  1142→      running = false
  1143→    }
  1144→  }

  1148→  function ask(text, id, persona) {
  1149→    // The browser can switch character mid-session. The system prompt rides
  1150→    // messages[0], so swapping it keeps the whole conversation — the new
  1151→    // character simply answers from here on. Unknown ids are ignored rather
  1152→    // than guessed at.
  1153→    if (typeof persona === 'string' && persona !== personaId && CHARACTERS[persona]) {
  1154→      personaId = persona
  1155→      messages[0] = { role: 'system', content: personaPrompt(persona) }
  1156→    }
  1157→    queue.push({ text: String(text ?? ''), id: typeof id === 'string' ? id : null })
  1158→    void pump()
  1159→  }
