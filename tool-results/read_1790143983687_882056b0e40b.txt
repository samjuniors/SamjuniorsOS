    25→/**
    26→ * ============================================================================
    27→ * M4-A — SOPHIA PERSONAL MEMORY CAPTURE + MEMORYGATE SUITE
    28→ * ============================================================================
    29→ *
    30→ * M4-A state machine under test (nothing more):
    31→ *
    32→ *   LLM extraction (untrusted proposals)
    33→ *     -> deterministic MemoryGate
    34→ *     -> REJECT            : nothing persisted
    35→ *     -> NEEDS_REVIEW      : INACTIVE SophiaMemory candidate (pending)
    36→ *     -> Founder confirmation via the governed PATCH active:true
    37→ *     -> active personal memory (renders in PERSONAL_MIND_MEMORY)
    38→ *
    39→ * There is NO automatic activation: no test may assert (and no code may
    40→ * produce) an automatically-activated captured memory. ACCEPT exists only in
    41→ * the gate's type for future compatibility.
    42→ *
    43→ * Coverage map:
    44→ *   G1-G12  MemoryGate (valid/malformed/type/confidence/size/instruction/
    45→ *           secret/company-domain/transient/duplicate/no-ACCEPT)
    46→ *   C1-C6   Capture pipeline (candidate persists inactive; extraction/gate
    47→ *           failures never fail the conversation; malformed LLM output never
    48→ *           persists; mixed proposals partially persist)
    49→ *   I1-I4   Idempotency (replayed turn, deterministic keys, store replay,
    50→ *           per-candidate suffixes)
    51→ *   P1      Provenance + capture metadata preserved
    52→ *   R1-R4   Founder review (queue visibility, governed activation, cross-
    53→ *           founder 403, invisible-until-confirmed)
    54→ *   X1-X5   Security (prompt-boundary structural delimiting + escaping,
    55→ *           company-brain isolation, source-level import bans, smuggled LLM
    56→ *           identity fields ignored, cross-founder context isolation)
    57→ *   S1-S3   Store extensions (default active, capture-key prefix scoping,
    58→ *           confirmation stamping)
    59→ *   E1-E4   Execution-path integration (executor trivial-skip + replay skip,
    60→ *           agent-chat Sophia path, non-Sophia persona never captures)
    61→ *
    62→ * The production 401 contract of the governed /api/sofia/memory route
    63→ * (including the PATCH activation path) is pinned by the K-2 suite's
    64→ * route-unauth child, which is re-run as part of the regression stage.
    65→ *
    66→ * Deterministic by construction: every pipeline test injects the extraction
    67→ * step (options.extract) — no live LLM call is required anywhere.
    68→ */

    91→function readCollectionFile<T>(name: string): Record<string, T> {
    92→  const p = path.join(DATA_DIR, `${name}.json`);
    93→  if (!fs.existsSync(p)) return {};
    94→  try {
    95→    return JSON.parse(fs.readFileSync(p, 'utf-8')) as Record<string, T>;
    96→  } catch {
    97→    return {};
    98→  }
    99→}

   101→function runChild(args: string[], env: Record<string, string> = {}): Promise<any> {
   102→  return new Promise((resolve, reject) => {
   103→    // @ts-ignore — Bun global exists when the suite runs under bun
   104→    const proc = Bun.spawn(['bun', 'tests/sophia/k2-memory-child.ts', ...args], {
   105→      stdout: 'pipe',
   106→      stderr: 'pipe',
   107→      env: { ...process.env, ...env },
   108→    });
   109→    new Response(proc.stdout)
   110→      .text()
   111→      .then((out) =>
   112→        proc.exited.then((code) => {
   113→          if (code !== 0) {
   114→            return new Response(proc.stderr)
   115→              .text()
   116→              .then((errText) =>
   117→                reject(new Error(`child (${args.join(' ')}) exited ${code}; stderr: ${errText.slice(0, 500)}`))
   118→              );
   119→          }
   120→          const lines = out.trim().split('\n').filter((l: string) => l.trim());
   121→          const last = lines[lines.length - 1];
   122→          try {
   123→            resolve(JSON.parse(last));
   124→          } catch {
   125→            reject(new Error(`child (${args.join(' ')}) emitted no JSON line; stdout: ${out.slice(0, 500)}`));
   126→          }
   127→        })
   128→      );
   129→  });
   130→}
