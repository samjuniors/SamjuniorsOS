    77→/**
    78→ * Central Side-Effect Authorization Gate.
    79→ * Enforces company safety, Founder approval boundaries, audit trail generation,
    80→ * and single server-side authorization enforcement before executing side effects.
    81→ */
    82→export class SideEffectAuthorizationGate {
    83→  private static instance: SideEffectAuthorizationGate;
    84→  private approvalStore: InMemoryApprovalStore;
    85→  private auditStore: InMemoryAuditStore;
    86→  private evaluator: SideEffectPolicyEvaluator;
    87→
    88→  constructor(
    89→    approvalStore?: InMemoryApprovalStore,
    90→    auditStore?: InMemoryAuditStore,
    91→    evaluator?: SideEffectPolicyEvaluator
    92→  ) {
    93→    this.approvalStore = approvalStore || InMemoryApprovalStore.getInstance();
    94→    this.auditStore = auditStore || InMemoryAuditStore.getInstance();
    95→    this.evaluator = evaluator || new SideEffectPolicyEvaluator(this.approvalStore);
    96→  }
    97→
    98→  public static getInstance(): SideEffectAuthorizationGate {
    99→    if (!SideEffectAuthorizationGate.instance) {
   100→      SideEffectAuthorizationGate.instance = new SideEffectAuthorizationGate();
   101→    }
   102→    return SideEffectAuthorizationGate.instance;
   103→  }
   104→
   105→  /**
   106→   * Evaluate whether an intended action is allowed, requires approval, or is denied.
   107→   */
   108→  public async evaluateAuthorization(
   109→    request: AuthorizationEvaluationRequest
   110→  ): Promise<AuthorizationDecision> {
   111→    return this.evaluator.evaluate(request);
   112→  }
   113→
   114→  /**
   115→   * Request a new Founder Approval for an intended side effect.
   116→   */
   117→  public async requestApproval(params: RequestApprovalParams): Promise<FounderApprovalRecord> {
   118→    const existing = await this.approvalStore.findActiveMatching({
   119→      workflowInstanceId: params.workflowInstanceId,
   120→      stepId: params.stepId,
   121→      employeeRole: params.employeeRole,
   122→      actionName: params.actionName,
   123→      classification: params.classification,
   124→    });
   125→
   126→    // If there is already a pending or approved approval, return it
   127→    if (existing && (existing.decision === 'pending' || existing.decision === 'approved')) {
   128→      return existing;
   129→    }
   130→
   131→    const id = `appr-${uuidv4()}`;
   132→    const now = new Date().toISOString();
   133→
   134→    const scope: ApprovalScope = {
   135→      scopeType: params.scope?.scopeType || 'step',
   136→      workflowInstanceId: params.workflowInstanceId,
   137→      stepId: params.stepId,
   138→      campaignId: params.scope?.campaignId,
   139→      targetSystem: params.target?.targetSystem || params.scope?.targetSystem,
   140→      operationPattern: params.scope?.operationPattern,
   141→      maxUses: params.scope?.maxUses || 1,
   142→      usedCount: 0,
   143→    };
   144→
   145→    const payloadHash = computeApprovalPayloadHash(
   146→      params.actionName,
   147→      params.target,
   148→      params.payload
   149→    );
   150→
   151→    const record: FounderApprovalRecord = {
   152→      id,
   153→      decision: 'pending',
   154→      actionName: params.actionName,
   155→      classification: params.classification,
   156→      workflowInstanceId: params.workflowInstanceId,
   157→      stepId: params.stepId,
   158→      employeeRole: params.employeeRole,
   159→      scope,
   160→      requestedAt: now,
   161→      notes: params.notes,
   162→      target: params.target,
   163→      payload: params.payload,
   164→      payloadHash,
   165→      isConsumed: false,
   166→    };
   167→
   168→    return this.approvalStore.save(record);
   169→  }
   170→
   171→  /**
   172→   * Founder Decision Handler (Approve / Reject).
   173→   * Strictly enforces that ONLY the Founder can approve or reject requests.
   174→   * AI Employees (researcher, pm, coo, finance, advisor) cannot self-approve or approve others.
   175→   */
   176→  public async decideApproval(params: DecideApprovalParams): Promise<FounderApprovalRecord> {
   177→    const ALLOWED_FOUNDER_IDENTITIES = [
   178→      'founder',
   179→      'founder-001',
   180→      'founder-local-session',
   181→      'executive_founder',
   182→      'executive-founder',
   183→    ];
   184→    const actor = params.decidedBy?.toLowerCase()?.trim() || '';
   185→    const isAllowedFounder =
   186→      ALLOWED_FOUNDER_IDENTITIES.includes(actor) ||
   187→      (actor.startsWith('founder-') && !actor.includes('agent')) ||
   188→      params.userContext?.role === 'FOUNDER';
   189→
   190→    if (!isAllowedFounder) {
   191→      throw new Error(
   192→        `Permission denied: Identity "${params.decidedBy}" is not an authorized Founder. Only verified Founder identities can approve or reject side-effect requests.`
   193→      );
   194→    }
   195→
   196→    return this.approvalStore.decide(
   197→      params.approvalId,
   198→      params.decision,
   199→      params.decidedBy,
   200→      params.reason,
   201→      params.expiresAt
   202→    );
   203→  }
   204→
   205→  /**
   206→   * Founder Revocation Handler.
   207→   * Immediately revokes a previously granted approval.
   208→   */
   209→  public async revokeApproval(params: RevokeApprovalParams): Promise<FounderApprovalRecord> {
   210→    const prohibitedRoles = ['researcher', 'pm', 'coo', 'finance', 'advisor', 'agent', 'system', 'developer', 'guest'];
   211→    const actor = params.revokedBy?.toLowerCase()?.trim() || '';
   212→    if (!actor || prohibitedRoles.includes(actor)) {
   213→      throw new Error(
   214→        `Permission denied: Only the Founder can revoke approvals. Attempted by "${params.revokedBy}".`
   215→      );
   216→    }
   217→
   218→    return this.approvalStore.revoke(params.approvalId, params.revokedBy, params.reason);
   219→  }
   220→
   221→  /**
   222→   * Execute an operation behind the centralized Side-Effect Authorization Gate.
   223→   * 1. Evaluates policy.
   224→   * 2. If approval is required or denied, refuses execution and returns the decision.
   225→   * 3. If allowed, verifies cryptographic payload binding and consumes single-use approval (if applicable).
   226→   * 4. Executes executeFn.
   227→   * 5. Emits a deterministic, unforgeable audit record.
   228→   */
   229→  public async executeWithGate<T>(params: ExecuteWithGateParams<T>): Promise<ExecuteWithGateResult<T>> {
   230→    const { request, executeFn, executionRef } = params;
   231→    const auditId = `audit-${uuidv4()}`;
   232→    const timestamp = new Date().toISOString();
   233→    const requestId = `req-${uuidv4()}`;
   234→
   235→    // 0. Pre-Execution Idempotency Check (completed replay, in-progress conflict, ambiguous recovery, payload tampering)
   236→    const shouldEnforceIdempotency = Boolean(
   237→      params.idempotency ||
   238→      request.classification === 'external_communication' ||
   239→      request.classification === 'financial_action'
   240→    );
   241→
   242→    let idempotencyKey: string | undefined;
   243→    let payloadHash: string | undefined;
   244→
   245→    if (shouldEnforceIdempotency) {
   246→      idempotencyKey = params.idempotency?.key || generateLogicalIdempotencyKey({
   247→        actionName: request.actionName,
   248→        targetSystem: params.idempotency?.targetSystem || request.target?.targetSystem || 'default',
   249→        logicalOpId: params.idempotency?.logicalOpId || (request.workflowContext?.workflowInstanceId ? `${request.workflowContext.workflowInstanceId}-${request.workflowContext.stepId}` : (executionRef || requestId)),
   250→      });
   251→
   252→      const effectivePayload = params.idempotency?.payload !== undefined ? params.idempotency.payload : request.payload;
   253→      payloadHash = computeApprovalPayloadHash(request.actionName, request.target, effectivePayload ?? {});
   254→
   255→      const existing = await getIdempotencyStore().get(idempotencyKey);
   256→      if (existing) {
   257→        if (payloadHash && existing.payloadHash && existing.payloadHash !== payloadHash) {
   258→          const mismatchErr = new IdempotencyPayloadMismatchError(
   259→            idempotencyKey,
   260→            existing.payloadHash,
   261→            payloadHash
   262→          );
   263→          const auditRecord: SideEffectAuditRecord = {
   264→            id: auditId,
   265→            timestamp,
   266→            requestId,
   267→            employeeRole: request.employeeRole,
   268→            requestedBy: request.requestedBy || request.employeeRole,
   269→            skillId: typeof request.skillId === 'string' ? request.skillId : undefined,
   270→            workflowInstanceId: request.workflowContext?.workflowInstanceId,
   271→            stepId: request.workflowContext?.stepId,
   272→            actionClassification: request.classification,
   273→            actionName: request.actionName,
   274→            target: request.target,
   275→            decision: 'denied',
   276→            reasonCode: 'IDEMPOTENCY_PAYLOAD_MISMATCH',
   277→            reason: mismatchErr.message,
   278→            approvalId: request.approvalId,
   279→            executionReference: executionRef,
   280→            executed: false,
   281→          };
   282→          await this.auditStore.record(auditRecord);
   283→          throw mismatchErr;
   284→        }
   285→
   286→        if (existing.status === 'completed') {
   287→          const auditRecord: SideEffectAuditRecord = {
   288→            id: auditId,
   289→            timestamp,
   290→            requestId,
   291→            employeeRole: request.employeeRole,
   292→            requestedBy: request.requestedBy || request.employeeRole,
   293→            skillId: typeof request.skillId === 'string' ? request.skillId : undefined,
   294→            workflowInstanceId: request.workflowContext?.workflowInstanceId,
   295→            stepId: request.workflowContext?.stepId,
   296→            actionClassification: request.classification,
   297→            actionName: request.actionName,
   298→            target: request.target,
   299→            decision: 'allowed',
   300→            reasonCode: 'IDEMPOTENT_REPLAY',
   301→            reason: `Operation already executed successfully with idempotency key ${idempotencyKey}. Replaying cached response without re-invoking side effect.`,
   302→            approvalId: request.approvalId,
   303→            executionReference: existing.executionRef || executionRef,
   304→            executed: false,
   305→          };
   306→          await this.auditStore.record(auditRecord);
   307→
   308→          return {
   309→            allowed: true,
   310→            executed: false,
   311→            decision: {
   312→              effect: 'allowed',
   313→              reasonCode: 'IDEMPOTENT_REPLAY',
   314→              reason: `Operation already executed successfully with idempotency key ${idempotencyKey}. Replaying cached response without re-invoking side effect.`,
   315→              approvalId: request.approvalId,
   316→              evaluatedAt: timestamp,
   317→              evaluator: 'central_side_effect_gate',
   318→            },
   319→            result: existing.response as T,
   320→            auditId,
   321→            idempotencyKey,
   322→            idempotentReplay: true,
   323→          };
   324→        }
   325→
   326→        if (existing.status === 'in_progress') {
   327→          const inProgressErr = new OperationInProgressError(idempotencyKey, existing.executionRef || undefined, existing.createdAt);
   328→          const auditRecord: SideEffectAuditRecord = {
   329→            id: auditId,
   330→            timestamp,
   331→            requestId,
   332→            employeeRole: request.employeeRole,
   333→            requestedBy: request.requestedBy || request.employeeRole,
   334→            skillId: typeof request.skillId === 'string' ? request.skillId : undefined,
   335→            workflowInstanceId: request.workflowContext?.workflowInstanceId,
   336→            stepId: request.workflowContext?.stepId,
   337→            actionClassification: request.classification,
   338→            actionName: request.actionName,
   339→            target: request.target,
   340→            decision: 'denied',
   341→            reasonCode: 'OPERATION_IN_PROGRESS',
   342→            reason: inProgressErr.message,
   343→            approvalId: request.approvalId,
   344→            executionReference: executionRef,
   345→            executed: false,
   346→          };
   347→          await this.auditStore.record(auditRecord);
   348→          throw inProgressErr;
   349→        }
   350→
   351→        if (existing.status === 'unknown') {
   352→          const unknownErr = new UnknownExternalResultError(
   353→            idempotencyKey,
   354→            existing.executionRef || undefined,
   355→            'Operation has ambiguous prior execution status (crash/timeout recovery state). Manual or reconciliatory inspection required before retrying.'
   356→          );
   357→          const auditRecord: SideEffectAuditRecord = {
   358→            id: auditId,
   359→            timestamp,
   360→            requestId,
   361→            employeeRole: request.employeeRole,
   362→            requestedBy: request.requestedBy || request.employeeRole,
   363→            skillId: typeof request.skillId === 'string' ? request.skillId : undefined,
   364→            workflowInstanceId: request.workflowContext?.workflowInstanceId,
   365→            stepId: request.workflowContext?.stepId,
   366→            actionClassification: request.classification,
   367→            actionName: request.actionName,
   368→            target: request.target,
   369→            decision: 'denied',
   370→            reasonCode: 'UNKNOWN_EXTERNAL_RESULT',
   371→            reason: unknownErr.message,
   372→            approvalId: request.approvalId,
   373→            executionReference: executionRef,
   374→            executed: false,
   375→          };
   376→          await this.auditStore.record(auditRecord);
   377→          throw unknownErr;
   378→        }
   379→      }
   380→    }
   381→
   382→    // 1. Evaluate authorization
   383→    const decision = await this.evaluateAuthorization(request);
   384→
   385→    // 2. If not allowed, record audit and return blocked result
   386→    if (decision.effect !== 'allowed') {
   387→      const auditRecord: SideEffectAuditRecord = {
   388→        id: auditId,
   389→        timestamp,
   390→        requestId,
   391→        employeeRole: request.employeeRole,
   392→        requestedBy: request.requestedBy || request.employeeRole,
   393→        skillId: typeof request.skillId === 'string' ? request.skillId : undefined,
   394→        workflowInstanceId: request.workflowContext?.workflowInstanceId,
   395→        stepId: request.workflowContext?.stepId,
   396→        actionClassification: request.classification,
   397→        actionName: request.actionName,
   398→        target: request.target,
   399→        decision: decision.effect,
   400→        reasonCode: decision.reasonCode,
   401→        reason: decision.reason,
   402→        approvalId: decision.approvalId,
   403→        executionReference: undefined,
   404→        executed: false,
   405→      };
   406→
   407→      await this.auditStore.record(auditRecord);
   408→
   409→      return {
   410→        allowed: false,
   411→        executed: false,
   412→        decision,
   413→        auditId,
   414→        error: decision.reason,
   415→      };
   416→    }
   417→
   418→    // 3. If allowed, verify cryptographic approval-payload binding
   419→    let approval: FounderApprovalRecord | null = null;
   420→    if (decision.approvalId) {
   421→      approval = await this.approvalStore.get(decision.approvalId);
   422→      if (!approval) {
   423→        const missingApprovalReason = `Approval record (${decision.approvalId}) not found in store. Execution blocked.`;
   424→        const auditRecord: SideEffectAuditRecord = {
   425→          id: auditId,
   426→          timestamp,
   427→          requestId,
   428→          employeeRole: request.employeeRole,
   429→          requestedBy: request.requestedBy || request.employeeRole,
   430→          skillId: typeof request.skillId === 'string' ? request.skillId : undefined,
   431→          workflowInstanceId: request.workflowContext?.workflowInstanceId,
   432→          stepId: request.workflowContext?.stepId,
   433→          actionClassification: request.classification,
   434→          actionName: request.actionName,
   435→          target: request.target,
   436→          decision: 'denied',
   437→          reasonCode: 'APPROVAL_NOT_FOUND',
   438→          reason: missingApprovalReason,
   439→          approvalId: decision.approvalId,
   440→          executionReference: executionRef,
   441→          executed: false,
   442→        };
   443→
   444→        await this.auditStore.record(auditRecord);
   445→
   446→        return {
   447→          allowed: false,
   448→          executed: false,
   449→          decision: {
   450→            effect: 'denied',
   451→            reasonCode: 'APPROVAL_NOT_FOUND',
   452→            reason: missingApprovalReason,
   453→            approvalId: decision.approvalId,
   454→            evaluatedAt: timestamp,
   455→            evaluator: 'central_side_effect_gate',
   456→          },
   457→          auditId,
   458→          error: missingApprovalReason,
   459→        };
   460→      }
   461→
   462→      // Enforce mandatory cryptographic payload binding:
   463→      // Any approval-backed consequential execution MUST have a payload hash.
   464→      if (!approval.payloadHash) {
   465→        const missingHashReason = `Cryptographic payload binding missing: approval record (${approval.id}) does not contain a mandatory payloadHash. Consequential execution denied.`;
   466→        const auditRecord: SideEffectAuditRecord = {
   467→          id: auditId,
   468→          timestamp,
   469→          requestId,
   470→          employeeRole: request.employeeRole,
   471→          requestedBy: request.requestedBy || request.employeeRole,
   472→          skillId: typeof request.skillId === 'string' ? request.skillId : undefined,
   473→          workflowInstanceId: request.workflowContext?.workflowInstanceId,
   474→          stepId: request.workflowContext?.stepId,
   475→          actionClassification: request.classification,
   476→          actionName: request.actionName,
   477→          target: request.target,
   478→          decision: 'denied',
   479→          reasonCode: 'APPROVAL_PAYLOAD_HASH_MISSING',
   480→          reason: missingHashReason,
   481→          approvalId: decision.approvalId,
   482→          executionReference: executionRef,
   483→          executed: false,
   484→        };
   485→
   486→        await this.auditStore.record(auditRecord);
   487→
   488→        return {
   489→          allowed: false,
   490→          executed: false,
   491→          decision: {
   492→            effect: 'denied',
   493→            reasonCode: 'APPROVAL_PAYLOAD_HASH_MISSING',
   494→            reason: missingHashReason,
   495→            approvalId: decision.approvalId,
   496→            evaluatedAt: timestamp,
   497→            evaluator: 'central_side_effect_gate',
   498→          },
   499→          auditId,
   500→          error: missingHashReason,
   501→        };
   502→      }
   503→
   504→      // Enforce deterministic payload binding verification
   505→      const bindingCheck = verifyApprovalPayloadBinding(approval, {
   506→        actionName: request.actionName,
   507→        target: request.target,
   508→        payload: request.payload,
   509→      });
   510→
   511→      if (!bindingCheck.isMatch) {
   512→        const mismatchReason = `Cryptographic payload binding mismatch: payload or target was altered after Founder approval was granted (expected: ${bindingCheck.expectedHash}, got: ${bindingCheck.actualHash}). Execution blocked.`;
   513→        const auditRecord: SideEffectAuditRecord = {
   514→          id: auditId,
   515→          timestamp,
   516→          requestId,
   517→          employeeRole: request.employeeRole,
   518→          requestedBy: request.requestedBy || request.employeeRole,
   519→          skillId: typeof request.skillId === 'string' ? request.skillId : undefined,
   520→          workflowInstanceId: request.workflowContext?.workflowInstanceId,
   521→          stepId: request.workflowContext?.stepId,
   522→          actionClassification: request.classification,
   523→          actionName: request.actionName,
   524→          target: request.target,
   525→          decision: 'denied',
   526→          reasonCode: 'APPROVAL_PAYLOAD_HASH_MISMATCH',
   527→          reason: mismatchReason,
   528→          approvalId: decision.approvalId,
   529→          executionReference: executionRef,
   530→          executed: false,
   531→        };
   532→
   533→        await this.auditStore.record(auditRecord);
   534→
   535→        return {
   536→          allowed: false,
   537→          executed: false,
   538→          decision: {
   539→            effect: 'denied',
   540→            reasonCode: 'APPROVAL_PAYLOAD_HASH_MISMATCH',
   541→            reason: mismatchReason,
   542→            approvalId: decision.approvalId,
   543→            evaluatedAt: timestamp,
   544→            evaluator: 'central_side_effect_gate',
   545→          },
   546→          auditId,
   547→          error: mismatchReason,
   548→        };
   549→      }
   550→    }
   551→
   552→    // 4. Atomic Idempotency Claiming
   553→    if (shouldEnforceIdempotency && idempotencyKey && payloadHash) {
   554→      try {
   555→        await getIdempotencyStore().claim({
   556→          key: idempotencyKey,
   557→          actionName: request.actionName,
   558→          payloadHash,
   559→          executionRef: executionRef || auditId,
   560→          ttlMs: params.idempotency?.ttlMs,
   561→        });
   562→      } catch (claimErr: any) {
   563→        if (claimErr instanceof IdempotencyPayloadMismatchError) {
   564→          const auditRecord: SideEffectAuditRecord = {
   565→            id: auditId,
   566→            timestamp,
   567→            requestId,
   568→            employeeRole: request.employeeRole,
   569→            requestedBy: request.requestedBy || request.employeeRole,
   570→            skillId: typeof request.skillId === 'string' ? request.skillId : undefined,
   571→            workflowInstanceId: request.workflowContext?.workflowInstanceId,
   572→            stepId: request.workflowContext?.stepId,
   573→            actionClassification: request.classification,
   574→            actionName: request.actionName,
   575→            target: request.target,
   576→            decision: 'denied',
   577→            reasonCode: 'IDEMPOTENCY_PAYLOAD_MISMATCH',
   578→            reason: claimErr.message,
   579→            approvalId: decision.approvalId,
   580→            executionReference: executionRef,
   581→            executed: false,
   582→          };
   583→          await this.auditStore.record(auditRecord);
   584→        } else if (claimErr instanceof OperationInProgressError) {
   585→          const auditRecord: SideEffectAuditRecord = {
   586→            id: auditId,
   587→            timestamp,
   588→            requestId,
   589→            employeeRole: request.employeeRole,
   590→            requestedBy: request.requestedBy || request.employeeRole,
   591→            skillId: typeof request.skillId === 'string' ? request.skillId : undefined,
   592→            workflowInstanceId: request.workflowContext?.workflowInstanceId,
   593→            stepId: request.workflowContext?.stepId,
   594→            actionClassification: request.classification,
   595→            actionName: request.actionName,
   596→            target: request.target,
   597→            decision: 'denied',
   598→            reasonCode: 'OPERATION_IN_PROGRESS',
   599→            reason: claimErr.message,
   600→            approvalId: decision.approvalId,
   601→            executionReference: executionRef,
   602→            executed: false,
   603→          };
   604→          await this.auditStore.record(auditRecord);
   605→        } else if (claimErr instanceof UnknownExternalResultError) {
   606→          const auditRecord: SideEffectAuditRecord = {
   607→            id: auditId,
   608→            timestamp,
   609→            requestId,
   610→            employeeRole: request.employeeRole,
   611→            requestedBy: request.requestedBy || request.employeeRole,
   612→            skillId: typeof request.skillId === 'string' ? request.skillId : undefined,
   613→            workflowInstanceId: request.workflowContext?.workflowInstanceId,
   614→            stepId: request.workflowContext?.stepId,
   615→            actionClassification: request.classification,
   616→            actionName: request.actionName,
   617→            target: request.target,
   618→            decision: 'denied',
   619→            reasonCode: 'UNKNOWN_EXTERNAL_RESULT',
   620→            reason: claimErr.message,
   621→            approvalId: decision.approvalId,
   622→            executionReference: executionRef,
   623→            executed: false,
   624→          };
   625→          await this.auditStore.record(auditRecord);
   626→        }
   627→        throw claimErr;
   628→      }
   629→    }
   630→
   631→    // 5. Consume single-use approval only after authorization + idempotency claim pass
   632→    if (decision.approvalId && approval && approval.scope.scopeType === 'single_action') {
   633→      await this.approvalStore.consume(decision.approvalId);
   634→    }
   635→
   636→    // 6. Execute the authorized operation
   637→    let result: T;
   638→    try {
   639→      result = await executeFn();
   640→    } catch (err: any) {
   641→      const isAmbiguousTimeout =
   642→        err?.name === 'TimeoutError' ||
   643→        err?.code === 'ETIMEDOUT' ||
   644→        err?.code === 'ECONNRESET' ||
   645→        err?.isAmbiguous === true ||
   646→        /timeout|timed out|abort/i.test(err?.message || '');
   647→
   648→      if (shouldEnforceIdempotency && idempotencyKey) {
   649→        if (isAmbiguousTimeout) {
   650→          await getIdempotencyStore().markUnknown(
   651→            idempotencyKey,
   652→            err?.message || 'Execution timed out or aborted with ambiguous provider state'
   653→          );
   654→        } else {
   655→          await getIdempotencyStore().fail(
   656→            idempotencyKey,
   657→            err?.message || 'Execution failed'
   658→          );
   659→        }
   660→      }
   661→
   662→      const auditRecord: SideEffectAuditRecord = {
   663→        id: auditId,
   664→        timestamp,
   665→        requestId,
   666→        employeeRole: request.employeeRole,
   667→        requestedBy: request.requestedBy || request.employeeRole,
   668→        skillId: typeof request.skillId === 'string' ? request.skillId : undefined,
   669→        workflowInstanceId: request.workflowContext?.workflowInstanceId,
   670→        stepId: request.workflowContext?.stepId,
   671→        actionClassification: request.classification,
   672→        actionName: request.actionName,
   673→        target: request.target,
   674→        decision: 'allowed',
   675→        reasonCode: decision.reasonCode,
   676→        reason: decision.reason,
   677→        approvalId: decision.approvalId,
   678→        executionReference: executionRef,
   679→        executed: false, // Execution threw error
   680→      };
   681→
   682→      await this.auditStore.record(auditRecord);
   683→
   684→      if (isAmbiguousTimeout) {
   685→        throw new UnknownExternalResultError(idempotencyKey || 'unknown', executionRef, err?.message || 'Operation timed out; external state is unknown');
   686→      }
   687→      throw err;
   688→    }
   689→
   690→    // 7. Record completion in idempotency store
   691→    if (shouldEnforceIdempotency && idempotencyKey) {
   692→      await getIdempotencyStore().complete(idempotencyKey, result, executionRef || auditId);
   693→    }
   694→
   695→    // 8. Record successful execution in audit trail
   696→    const auditRecord: SideEffectAuditRecord = {
   697→      id: auditId,
   698→      timestamp,
   699→      requestId,
   700→      employeeRole: request.employeeRole,
   701→      requestedBy: request.requestedBy || request.employeeRole,
   702→      skillId: typeof request.skillId === 'string' ? request.skillId : undefined,
   703→      workflowInstanceId: request.workflowContext?.workflowInstanceId,
   704→      stepId: request.workflowContext?.stepId,
   705→      actionClassification: request.classification,
   706→      actionName: request.actionName,
   707→      target: request.target,
   708→      decision: 'allowed',
   709→      reasonCode: decision.reasonCode,
   710→      reason: decision.reason,
   711→      approvalId: decision.approvalId,
   712→      executionReference: executionRef || `exec-${Date.now()}`,
   713→      executed: true,
   714→    };
   715→
   716→    await this.auditStore.record(auditRecord);
   717→
   718→    return {
   719→      allowed: true,
   720→      executed: true,
   721→      decision,
   722→      result,
   723→      auditId,
   724→      idempotencyKey,
   725→      idempotentReplay: false,
   726→    };
   727→  }

   731→  }
