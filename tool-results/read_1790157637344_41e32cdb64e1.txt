    25→/**
    26→ * ============================================================================
    27→ * M4-A — SOPHIA PERSONAL MEMORY CAPTURE + MEMORYGATE SUITE
    28→ * ============================================================================
    29→ *
    30→ * M4-A state machine under test (nothing more):
    31→ *
    32→ *   LLM extraction (untrusted proposals)
    33→ *     -> deterministic MemoryGate
    34→ *     -> REJECT            : nothing persisted
    35→ *     -> NEEDS_REVIEW      : INACTIVE SophiaMemory candidate (pending)
    36→ *     -> Founder confirmation via the governed PATCH active:true
    37→ *     -> active personal memory (renders in PERSONAL_MIND_MEMORY)
    38→ *
    39→ * There is NO automatic activation: no test may assert (and no code may
    40→ * produce) an automatically-activated captured memory. ACCEPT exists only in
    41→ * the gate's type for future compatibility.
    42→ *
    43→ * Coverage map:
    44→ *   G1-G12  MemoryGate (valid/malformed/type/confidence/size/instruction/
    45→ *           secret/company-domain/transient/duplicate/no-ACCEPT)
    46→ *   C1-C8   Capture pipeline (candidate persists inactive; extraction/gate
    47→ *           failures never fail the conversation; malformed LLM output never
    48→ *           persists; mixed proposals partially persist; C8 exercises the
    49→ *           DEFAULT production extractor seam with class context intact)
    50→ *   I1-I4   Idempotency (replayed turn, deterministic keys, store replay,
    51→ *           per-candidate suffixes)
    52→ *   P1      Provenance + capture metadata preserved
    53→ *   R1-R4   Founder review (queue visibility, governed activation, cross-
    54→ *           founder 403, invisible-until-confirmed)
    55→ *   X1-X5   Security (prompt-boundary structural delimiting + escaping,
    56→ *           company-brain isolation, source-level import bans, smuggled LLM
    57→ *           identity fields ignored, cross-founder context isolation)
    58→ *   S1-S3   Store extensions (default active, capture-key prefix scoping,
    59→ *           confirmation stamping)
    60→ *   E1-E4   Execution-path integration (executor trivial-skip + replay skip,
    61→ *           agent-chat Sophia path, non-Sophia persona never captures)
    62→ *
    63→ * The production 401 contract of the governed /api/sofia/memory route
    64→ * (including the PATCH activation path) is pinned by the K-2 suite's
    65→ * route-unauth child, which is re-run as part of the regression stage.
    66→ *
    67→ * Deterministic by construction: every pipeline test injects the extraction
    68→ * step (options.extract) — no live LLM call is required anywhere. The single
    69→ * exception is C8, which deliberately runs the DEFAULT extractor path: its
    70→ * child mocks only the z-ai SDK chat completion (module-level, registered
    71→ * before any src import), so it stays deterministic while executing the real
    72→ * static extractor end-to-end.
    73→ */

    96→function readCollectionFile<T>(name: string): Record<string, T> {
    97→  const p = path.join(DATA_DIR, `${name}.json`);
    98→  if (!fs.existsSync(p)) return {};
    99→  try {
   100→    return JSON.parse(fs.readFileSync(p, 'utf-8')) as Record<string, T>;
   101→  } catch {
   102→    return {};
   103→  }
   104→}

   106→function runChild(
   107→  args: string[],
   108→  env: Record<string, string> = {},
   109→  script = 'tests/sophia/k2-memory-child.ts'
   110→): Promise<any> {
   111→  return new Promise((resolve, reject) => {
   112→    // @ts-ignore — Bun global exists when the suite runs under bun
   113→    const proc = Bun.spawn(['bun', script, ...args], {
   114→      stdout: 'pipe',
   115→      stderr: 'pipe',
   116→      env: { ...process.env, ...env },
   117→    });
   118→    new Response(proc.stdout)
   119→      .text()
   120→      .then((out) =>
   121→        proc.exited.then((code) => {
   122→          if (code !== 0) {
   123→            return new Response(proc.stderr)
   124→              .text()
   125→              .then((errText) =>
   126→                reject(new Error(`child (${args.join(' ')}) exited ${code}; stderr: ${errText.slice(0, 500)}`))
   127→              );
   128→          }
   129→          const lines = out.trim().split('\n').filter((l: string) => l.trim());
   130→          const last = lines[lines.length - 1];
   131→          try {
   132→            resolve(JSON.parse(last));
   133→          } catch {
   134→            reject(new Error(`child (${args.join(' ')}) emitted no JSON line; stdout: ${out.slice(0, 500)}`));
   135→          }
   136→        })
   137→      );
   138→  });
   139→}

   145→function routeReq(method: string, founderId: string | null, body?: Record<string, unknown>, query = ''): NextRequest {
   146→  const headers: Record<string, string> = { 'content-type': 'application/json' };
   147→  if (founderId) {
   148→    headers['x-samjuniors-user-id'] = founderId;
   149→    headers['x-samjuniors-role'] = 'FOUNDER';
   150→  }
   151→  return new NextRequest(`http://localhost:3000/api/sofia/memory${query}`, {
   152→    method,
   153→    headers,
   154→    body: body ? JSON.stringify(body) : undefined,
   155→  });
