   136→/**
   137→ * Independent re-derivation of the pre-M4-C policy (same mechanism family as
   138→ * the m4c suite, kept so fallback equality can be asserted without trusting
   139→ * the implementation under test).
   140→ */
   141→function expectedPreM4cOrder(records: EvalMem[]): string[] {
   142→  const byType = new Map<string, EvalMem[]>();
   143→  for (const r of records) {
   144→    byType.set(r.memoryType, [...(byType.get(r.memoryType) ?? []), r]);
   145→  }
   146→  for (const list of byType.values()) {
   147→    list.sort((a, b) => {
   148→      if (b.confidence !== a.confidence) return b.confidence - a.confidence;
   149→      const ta = new Date(a.updatedAt).getTime();
   150→      const tb = new Date(b.updatedAt).getTime();
   151→      if (tb !== ta) return tb - ta;
   152→      return a.id < b.id ? -1 : a.id > b.id ? 1 : 0;
   153→    });
   154→  }
   155→  const order: string[] = [];
   156→  for (let round = 0; order.length < 20; round++) {
   157→    let advanced = false;
   158→    for (const t of SOPHIA_MEMORY_TYPES) {
   159→      if (order.length >= 20) break;
   160→      const list = byType.get(t);
   161→      if (list && list.length > round) {
   162→        order.push(list[round].id);
   163→        advanced = true;
   164→      }
   165→    }
   166→    if (!advanced) break;
   167→  }
   168→  return order;
   169→}

   171→/** Lexical score re-derivation for evaluation (same shared exported tokenizer). */
   172→function lexicalScore(m: EvalMem, query: string): number {
   173→  const queryTokens = new Set(extractTokens(query));
   174→  const contentTokens = new Set(extractTokens(m.content));
   175→  let score = 0;
   176→  for (const t of queryTokens) if (contentTokens.has(t)) score++;
   177→  return score;
   178→}

   180→interface QueryMetrics {
   181→  queryId: string;
   182→  label: string;
   183→  query: string;
   184→  corpusSize: number;
   185→  goldSize: number;
   186→  goldRanks: Array<{ id: string; rank: number }>;
   187→  tier1Size: number;
   188→  recall5: number;
   189→  recall10: number;
   190→  recall20: number;
   191→  precision5: number;
   192→  precision10: number;
   193→  precision20: number;
   194→  firstHitRank: number | null;
   195→}
   196→
   197→function measureQuery(
   198→  queryId: string,
   199→  label: string,
   200→  query: string,
   201→  corpus: EvalMem[],
   202→  gold: string[]
   203→): QueryMetrics {
   204→  const ranked = ids(selectPersonalMindMemories(corpus, query));
   205→  const goldSet = new Set(gold);
   206→  const recallAt = (k: number) =>
   207→    gold.filter((g) => ranked.slice(0, k).includes(g)).length / gold.length;
   208→  const precisionAt = (k: number) =>
   209→    ranked.slice(0, k).filter((id) => goldSet.has(id)).length / Math.min(k, ranked.length);
   210→  const goldRanks = gold.map((g) => ({ id: g, rank: ranked.indexOf(g) + 1 }));
   211→  const tier1Size = corpus.filter((m) => lexicalScore(m, query) > 0).length;
   212→  const firstHit = goldRanks.find((r) => r.rank > 0)?.rank ?? null;
   213→  return {
   214→    queryId,
   215→    label,
   216→    query,
   217→    corpusSize: corpus.length,
   218→    goldSize: gold.length,
   219→    goldRanks,
   220→    tier1Size,
   221→    recall5: recallAt(5),
   222→    recall10: recallAt(10),
   223→    recall20: recallAt(20),
   224→    precision5: precisionAt(5),
   225→    precision10: precisionAt(10),
   226→    precision20: precisionAt(20),
   227→    firstHitRank: firstHit,
   228→  };
   229→}

   434→// ============================================================================
   435→// DETERMINISTIC SYNTHETIC CORPUS GENERATOR (scale runs, section F)
   436→// ============================================================================
   437→
   438→function mulberry32(seed: number): () => number {
   439→  let a = seed >>> 0;
   440→  return () => {
   441→    a = (a + 0x6d2b79f5) | 0;
   442→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
   443→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
   444→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
   445→  };
   446→}

   448→const SCALE_TOPICS = ['sailing', 'photography', 'ceramics', 'beekeeping', 'archery', 'pottery',
   449→  'astronomy', 'birdwatching', 'chess', 'hiking', 'kayaking', 'woodworking', 'sketching',
   450→  'gardening', 'meditation', 'calligraphy', 'origami', 'baking', 'climbing', 'snorkeling'];
   451→const SCALE_VERBS = ['enjoys', 'practises', 'studies', 'teaches', 'explores', 'collects',
   452→  'discusses', 'sketches', 'photographs', 'reviews'];
   453→const SCALE_ADJ = ['quiet', 'annual', 'local', 'traditional', 'modern', 'detailed', 'casual'];
   454→const SCALE_WHEN = ['spring', 'summer', 'autumn', 'winter', 'holiday', 'weekend'];
   455→
   456→/**
   457→ * Deterministic scale corpus: n ACTIVE memories, one planted 4-token match
   458→ * ('matcha tea ceremony kyoto'), ~10% single-generic-token partial matches
   459→ * ('club'/'team'), remainder lexically unrelated to the scale query.
   460→ */
   461→function buildScaleCorpus(n: number, seed: number): EvalMem[] {
   462→  const rand = mulberry32(seed);
   463→  const records: EvalMem[] = [];
   464→  const partialCount = Math.max(2, Math.floor(n / 10));
   465→  for (let i = 0; i < n; i++) {
   466→    const type = SOPHIA_MEMORY_TYPES[i % SOPHIA_MEMORY_TYPES.length];
   467→    const confidence = Math.round((0.3 + rand() * 0.65) * 100) / 100;
   468→    const day = 1 + Math.floor(rand() * 28);
   469→    const month = 1 + Math.floor(rand() * 9);
   470→    const updatedAt = `2026-${String(month).padStart(2, '0')}-${String(day).padStart(2, '0')}T09:00:00.000Z`;
   471→    const topic = SCALE_TOPICS[Math.floor(rand() * SCALE_TOPICS.length)];
   472→    const verb = SCALE_VERBS[Math.floor(rand() * SCALE_VERBS.length)];
   473→    const adj = SCALE_ADJ[Math.floor(rand() * SCALE_ADJ.length)];
   474→    const when = SCALE_WHEN[Math.floor(rand() * SCALE_WHEN.length)];
   475→    let content: string;
   476→    if (i === 0) {
   477→      // PLANTED MATCH — 4 distinct query-token overlap, low confidence + old:
   478→      // must still rank first-tier regardless of secondary policy.
   479→      records.push(mem('ev-scale-plant', 'INTERACTION_OBSERVATION', 0.3, '2025-10-01T09:00:00.000Z',
   480→        'Attends the annual matcha tea ceremony near Kyoto every spring.'));
   481→      continue;
   482→    } else if (i <= partialCount) {
   483→      content = `Joins ${adj} ${topic} clubs with the local club team during ${when} meetups.`;
   484→    } else {
   485→      content = `${verb.charAt(0).toUpperCase()}${verb.slice(1)} ${adj} ${topic} during ${when} afternoons at home.`;
   486→    }
   487→    records.push(mem(`ev-scale-${i}`, type, confidence, updatedAt, content));
   488→  }
   489→  return records;
   490→}
