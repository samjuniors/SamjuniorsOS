     1→/**
     2→ * The resilience layer — every provider chain in one place.
     3→ *
     4→ * The assistant is a conversation, and a conversation dies the moment one
     5→ * sentence goes unanswered. Every external service it depends on — the model
     6→ * that thinks, the transcriber that hears, the voice that speaks — has a
     7→ * moment where it rate-limits, times out, or simply goes down. So none of
     8→ * them is load-bearing alone: each is one link in a chain, and the chain is
     9→ * walked on failure, automatically, mid-turn.
    10→ *
    11→ *   LLM   z-ai (built-in, no key) → Gemini → any local OpenAI-compatible
    12→ *         server (Ollama, LM Studio, vLLM, llama.cpp)
    13→ *   STT   Deepgram → ElevenLabs Scribe → z-ai ASR (no key) → a local
    14→ *         OpenAI-compatible transcription server (faster-whisper, LocalAI)
    15→ *   TTS   ElevenLabs → z-ai neural → a local OpenAI-compatible speech server
    16→ *         (kokoro-fastapi, LocalAI, openedai-speech) → the browser's own voice
    17→ *
    18→ * Three ideas make the chains safe rather than just long:
    19→ *
    20→ *   Circuit breakers. A provider that fails twice in a row (LLM) or three
    21→ *   times (STT) is skipped for a cooldown — 90s / 60s — so a dead provider
    22→ *   costs one failed attempt, not one per sentence. When every link is
    23→ *   cooling the chain is walked anyway, in order: a retry is better than a
    24→ *   shrug. Any success heals the provider immediately.
    25→ *
    26→ *   Pins. SOFIA_LLM_PROVIDER / SOFIA_STT_PROVIDER / SOFIA_TTS_PROVIDER
    27→ *   collapse a chain to one link on purpose — for testing, or because a
    28→ *   person has decided they know better than the ladder. (The JARVIS_*
    29→ *   spellings from the bridge era still work as aliases.)
    30→ *
    31→ *   Honesty. llmInfo() / sttInfo() are the single source of truth for
    32→ *   /api/sofia/health, which is what the settings panel shows the user:
    33→ *   which links exist, which are configured, which one is answering right
    34→ *   now. A fallback you can't see is a fallback you can't trust.
    35→ *
    36→ * Everything speaks the OpenAI wire format — the request bodies and the
    37→ * SSE event stream — which is why one parser (in brain.ts) serves every
    38→ * link: the z-ai SDK, Gemini's OpenAI-compat endpoint and a local server
    39→ * all say `data: {"choices":[{"delta":{...}}]}`.
    40→ *
    41→ * Ported from bridge/providers.mjs (Vite era) — behaviour identical; the env
    42→ * vars gained SOFIA_* aliases and Next.js loads .env.local itself.
    43→ */

    60→/** The one SDK instance for this process. Throws when the machine has no z-ai

    62→ *  treat that as "provider unavailable", never as a crash. */
    63→export async function zaiClient(): Promise<ZaiClient> {
    64→  if (zaiShared) return zaiShared
    65→  if (zaiBroken) throw new Error('z-ai sdk is not available on this machine')

    67→    zaiShared = (await ZAI.create()) as ZaiClient
    68→    return zaiShared
    69→  } catch (err) {
    70→    zaiBroken = true
    71→    console.error('[sofia] z-ai sdk unavailable:', errText(err))
    72→    throw new Error('z-ai sdk is not available on this machine')
    73→  }
    74→}

    86→type GuardState = { fails: number; until: number }
    87→
    88→function makeGuard(threshold: number, cooldownMs: number) {
    89→  const state = new Map<string, GuardState>()
    90→
    91→  const trip = (id: string) => {
    92→    const s = state.get(id) ?? { fails: 0, until: 0 }
    93→    s.fails += 1
    94→    // Only a run of failures opens the breaker, and opening it resets the
    95→    // count, so the cooldown that follows is a fixed window, not a ratchet.
    96→    if (s.fails >= threshold) {
    97→      s.until = Date.now() + cooldownMs
    98→      s.fails = 0
    99→    }
   100→    state.set(id, s)
   101→  }
   102→  return {
   103→    trip,
   104→    heal: (id: string) => state.set(id, { fails: 0, until: 0 }),
   105→    cooling: (id: string) => {
   106→      const s = state.get(id)
   107→      return Boolean(s && s.until > Date.now())
   108→    },
   109→  }
   110→}

   126→/** A provider that refused our tool schema (small local models often do)
   127→ *  gets one retry without tools — a plain answer beats a failed turn. A
   128→ *  missing endpoint (404) is NOT a schema refusal, or every typo would
   129→ *  silently strip the tools for good. Latched so the retry is paid once per
   130→ *  provider, not once per turn. */
   131→function looksLikeToolRejection(err: unknown): boolean {
   132→  const t = errText(err)
   133→  return /(\b400\b|\b422\b|unsupported|not support|tool|function)/i.test(t)
   134→}

   188→/** What the whole server sends the model. Kept loose on purpose: tool
   189→ *  definitions and messages come straight from brain.ts's own shapes. */
   190→export type LlmBody = {
   191→  messages: Array<Record<string, unknown>>
   192→  tools?: unknown[]
   193→  stream?: boolean
   194→  [key: string]: unknown
   195→}

   197→/** One SSE reader from any OpenAI-compatible chat endpoint. The shape —
   198→ *  `data: {...}\n\n` lines ending in `data: [DONE]` — is exactly what
   199→ *  brain.ts already parses for the z-ai SDK's streams. */
   200→async function openAiCompatReader(
   201→  url: string,
   202→  key: string,
   203→  body: LlmBody,
   204→): Promise<ReadableStreamDefaultReader<Uint8Array>> {
   205→  const res = await fetch(url, {
   206→    method: 'POST',
   207→    headers: {
   208→      'content-type': 'application/json',
   209→      ...(key ? { authorization: `Bearer ${key}` } : {}),
   210→    },
   211→    body: JSON.stringify(body),
   212→    // No abort signal: it would kill the body mid-stream, and local models
   213→    // legitimately take their time. The brain's own stall guard owns hangs.
   214→  })
   215→  if (!res.ok) {
   216→    const detail = await res.text().catch(() => '')
   217→    throw new Error(`${res.status} ${res.statusText} ${String(detail).slice(0, 200)}`)
   218→  }
   219→  const reader = res.body?.getReader()
   220→  if (!reader) throw new Error('the provider returned no stream')
   221→  return reader
   222→}

   225→  id: string
   226→  label: string

   233→/** Each link adapts the shared body to its own dialect. */
   234→function buildLlmBody(p: { id: string; model?: string }, body: LlmBody, withTools: boolean): LlmBody {
   235→  const tools =
   236→    withTools && Array.isArray(body.tools) && body.tools.length ? body.tools : undefined
   237→  if (p.id === 'zai') {
   238→    // `thinking` is a z-ai extension; harmless nowhere else but not theirs
   239→    // to carry, so it stays on the one link that reads it.
   240→    return { ...body, tools, thinking: { type: 'disabled' } }
   241→  }
   242→  const out: LlmBody = { ...body, tools, model: p.model }
   243→  delete out.thinking
   244→  return out
   245→}

   328→/**
   329→ * Walk the chain and return the first live SSE reader.
   330→ *
   331→ * `announce(provider)` fires exactly when the active link *changes* — the
   332→ * browser turns that into the BRAIN rail line, so a person can see the
   333→ * moment Gemini takes over. Failures are announced the loud way instead: a
   334→ * warning on the server log naming the link and the error.
   335→ */
   336→export async function openLlmStream(
   337→  body: LlmBody,
   338→  announce?: (p: LlmProvider) => void,
   339→): Promise<{ reader: ReadableStreamDefaultReader<Uint8Array>; provider: LlmProvider }> {
   340→  const configured = LLM_CHAIN.filter((p) => p.available())
   341→  if (!configured.length) throw new Error('no llm provider is configured')
   342→
   343→  const pinId = llmPinId()
   344→  const pin = pinId ? configured.find((p) => p.id === pinId) : null
   345→  if (pinId && !pin) {
   346→    console.warn(`[sofia] SOFIA_LLM_PROVIDER="${pinId}" matches nothing — using the chain`)
   347→  }
   348→  const ordered = pin ? [pin] : configured
   349→
   350→  let awake = ordered.filter((p) => !LLM_GUARD.cooling(p.id))
   351→  if (!awake.length) {
   352→    // Every link cooling. Try them in order anyway — a chain that has given
   353→    // up is worse than one that keeps failing loudly.
   354→    awake = ordered
   355→  }
   356→
   357→  let lastErr: unknown
   358→  for (const p of awake) {
   359→    for (let attempt = 0; attempt < 2; attempt++) {
   360→      const withTools = !noTools.has(p.id)
   361→      try {
   362→        const reader = await p.start(buildLlmBody(p, body, withTools))
   363→        if (activeLlm !== p.id) {
   364→          activeLlm = p.id
   365→          announce?.(p)
   366→        }
   367→        llmHealth.set(p.id, 'ok')
   368→        llmErrors.delete(p.id)
   369→        LLM_GUARD.heal(p.id)
   370→        return { reader, provider: p }
   371→      } catch (err) {
   372→        lastErr = err
   373→        // First failure with tools + a rejection-shaped error + never latched:
   374→        // retry the same link once with the tools stripped.
   375→        if (attempt === 0 && withTools && !noTools.has(p.id) && looksLikeToolRejection(err)) {
   376→          noTools.add(p.id)
   377→          console.warn(`[sofia] llm "${p.id}" refused the tool schema — continuing without tools`)
   378→          continue
   379→        }
   380→        if (!isTransient(err) || attempt === 1) break
   381→        await sleep(600)
   382→      }
   383→    }
   384→    LLM_GUARD.trip(p.id)
   385→    llmHealth.set(p.id, 'err')
   386→    llmErrors.set(p.id, errText(lastErr))
   387→    console.warn(`[sofia] llm provider "${p.id}" failed: ${errText(lastErr)}`)
   388→  }
   389→  throw lastErr ?? new Error('every llm provider failed')
   390→}

   438→/** The extension is the only hint a transcription server gets about the
   439→ *  codec — derive it from the content-type the recorder reported. */
   440→function extFromType(type: string | undefined): string {
   441→  const t = String(type ?? '')
   442→  if (t.includes('ogg')) return 'ogg'
   443→  if (t.includes('mp4') || t.includes('mpeg')) return 'mp4'
   444→  if (t.includes('wav')) return 'wav'
   445→  return 'webm'
   446→}

   448→/**
   449→ * z-ai ASR is keyless, which makes it the difference between "server-side
   450→ * transcription needs a key" and "it works for everyone" — but on a machine
   451→ * without the sandbox's z-ai credentials it cannot work at all. So it is
   452→ * probed once at startup with half a second of silence: ok means live,
   453→ * rate-limited means live (a quota window closes on its own), and an auth
   454→ * error means honestly OFF.
   455→ */
   456→const asrProbe = { state: 'pending' as 'pending' | 'ok' | 'no', detail: '' }
   457→
   458→function silenceWav(): Buffer {
   459→  const samples = 8000 // half a second of 16kHz zeros
   460→  const buf = Buffer.alloc(44 + samples * 2)
   461→  buf.write('RIFF', 0)
   462→  buf.writeUInt32LE(36 + samples * 2, 4)
   463→  buf.write('WAVE', 8)
   464→  buf.write('fmt ', 12)
   465→  buf.writeUInt32LE(16, 16)
   466→  buf.writeUInt16LE(1, 20) // PCM
   467→  buf.writeUInt16LE(1, 22) // mono
   468→  buf.writeUInt32LE(16000, 24)
   469→  buf.writeUInt32LE(32000, 28)
   470→  buf.writeUInt16LE(2, 32)
   471→  buf.writeUInt16LE(16, 34)
   472→  buf.write('data', 36)
   473→  buf.writeUInt32LE(samples * 2, 40)
   474→  return buf
   475→}

   477→export async function probeZaiAsr(): Promise<'ok' | 'no' | 'pending'> {
   478→  try {
   479→    const api = await zaiClient()
   480→    const res = await (
   481→      api.audio.asr.create as (b: unknown) => Promise<{ text?: unknown }>
   482→    ).call(api.audio.asr, { file_base64: silenceWav().toString('base64') })
   483→    asrProbe.state = 'ok'
   484→    asrProbe.detail = typeof res?.text === 'string' ? '' : 'unexpected shape'
   485→  } catch (err) {
   486→    const text = errText(err)
   487→    if (/\b(429|rate|too many|502|503|504)\b/i.test(text)) {
   488→      // A quota window — the same one that trips the LLM chain — not an
   489→      // outage. Available, and the breaker will absorb the failures.
   490→      asrProbe.state = 'ok'
   491→      asrProbe.detail = 'rate-limited at probe time'
   492→    } else {
   493→      asrProbe.state = 'no'
   494→      asrProbe.detail = text
   495→    }
   496→  }
   497→  console.log(
   498→    `[sofia] z-ai asr: ${asrProbe.state}${asrProbe.detail ? ` (${asrProbe.detail})` : ''}`,
   499→  )
   500→  return asrProbe.state
   501→}

   607→const sttErrors = new Map<string, string>()
   608→const sttHealth = new Map<string, string>()

   611→/**
   612→ * Transcribe one segment through the first link that answers. Per-request
   613→ * fallthrough (not just breaker-driven): a segment is a few hundred
   614→ * milliseconds of speech, and the words are only useful now.
   615→ */
   616→export async function transcribeAnywhere(
   617→  buffer: Buffer,
   618→  type: string | undefined,
   619→): Promise<{ text: string; provider: string }> {
   620→  const configured = STT_CHAIN.filter((p) => p.available())
   621→  if (!configured.length) throw new Error('no transcription provider is configured')
   622→
   623→  const pinId = sttPinId()
   624→  const pin = pinId ? configured.find((p) => p.id === pinId) : null
   625→  if (pinId && !pin) {
   626→    console.warn(`[sofia] SOFIA_STT_PROVIDER="${pinId}" matches nothing — using the chain`)
   627→  }
   628→  const ordered = pin ? [pin] : configured
   629→
   630→  let awake = ordered.filter((p) => !STT_GUARD.cooling(p.id))
   631→  if (!awake.length) awake = ordered
   632→
   633→  let lastErr: unknown
   634→  for (const p of awake) {
   635→    try {
   636→      const text = await p.transcribe(buffer, type)
   637→      sttHealth.set(p.id, 'ok')
   638→      sttErrors.delete(p.id)
   639→      STT_GUARD.heal(p.id)
   640→      activeStt = p.id
   641→      return { text, provider: p.id }
   642→    } catch (err) {
   643→      lastErr = err
   644→      STT_GUARD.trip(p.id)
   645→      sttHealth.set(p.id, 'err')
   646→      sttErrors.set(p.id, errText(err))
   647→      console.warn(`[sofia] stt provider "${p.id}" failed: ${errText(err)}`)
   648→    }
   649→  }
   650→  throw lastErr ?? new Error('every transcription provider failed')
   651→}

   671→/** Chain status for /health. `available` decides the browser's engine pick. */
   672→export function sttInfo(): SttInfo {
   673→  const pinId = sttPinId()
   674→  const providers: SttLinkInfo[] = STT_CHAIN.map((p) => {
   675→    const state =
   676→      p.id === 'zai' && asrProbe.state === 'pending'
   677→        ? 'never'
   678→        : (sttHealth.get(p.id) ?? 'never')
   679→    return {
   680→      id: p.id,
   681→      label: p.label,
   682→      note: p.note,
   683→      configured: p.available(),
   684→      state,
   685→      healthy: state !== 'err',
   686→      cooling: STT_GUARD.cooling(p.id),
   687→      error: sttErrors.get(p.id) ?? null,
   688→      detail: p.id === 'zai' ? asrProbe.detail : '',
   689→    }
   690→  })
   691→  return {
   692→    pin: pinId || null,
   693→    active: activeStt || providers.find((p) => p.configured)?.id || '',
   694→    providers,
   695→  }
   696→}

   718→/** One utterance as an audio buffer from a local speech server
   719→ *  (kokoro-fastapi, LocalAI, openedai-speech — anything speaking the
   720→ *  OpenAI /v1/audio/speech shape). */
   721→export async function synthesizeLocalTts(text: string): Promise<Buffer> {
   722→  const base = localTtsBase()
   723→  const res = await fetch(`${base}/audio/speech`, {
   724→    method: 'POST',
   725→    headers: {
   726→      'content-type': 'application/json',

   729→        : {}),

   731→    body: JSON.stringify({
   732→      model: env('LOCAL_TTS_MODEL') || 'kokoro',
   733→      voice: env('LOCAL_TTS_VOICE') || 'af_sky',
   734→      input: text,
   735→      response_format: 'mp3',
   736→    }),
   737→    signal: AbortSignal.timeout(30_000),
   738→  })
   739→  if (!res.ok) throw new Error(`local tts ${res.status}: ${errText(await res.text())}`)
   740→  return Buffer.from(await res.arrayBuffer())
