    68→  try {

    81→function readCollectionFile<T>(name: string): Record<string, T> {
    82→  const p = path.join(DATA_DIR, `${name}.json`);
    83→  if (!fs.existsSync(p)) return {};
    84→  try {
    85→    return JSON.parse(fs.readFileSync(p, 'utf-8')) as Record<string, T>;
    86→  } catch {
    87→    return {};
    88→  }
    89→}

    91→function runChild(args: string[], env: Record<string, string> = {}): Promise<any> {
    92→  return new Promise((resolve, reject) => {
    93→    // @ts-ignore — Bun global exists when the suite runs under bun
    94→    const proc = Bun.spawn(['bun', 'tests/sophia/k2-memory-child.ts', ...args], {
    95→      stdout: 'pipe',
    96→      stderr: 'pipe',
    97→      env: { ...process.env, ...env },
    98→    });
    99→    new Response(proc.stdout)
   100→      .text()
   101→      .then((out) =>
   102→        proc.exited.then((code) => {
   103→          if (code !== 0) {
   104→            return new Response(proc.stderr)
   105→              .text()
   106→              .then((errText) =>
   107→                reject(new Error(`child (${args.join(' ')}) exited ${code}; stderr: ${errText.slice(0, 500)}`))
   108→              );
   109→          }
   110→          const lines = out.trim().split('\n').filter((l: string) => l.trim());
   111→          const last = lines[lines.length - 1];
   112→          try {
   113→            resolve(JSON.parse(last));
   114→          } catch {
   115→            reject(new Error(`child (${args.join(' ')}) emitted no JSON line; stdout: ${out.slice(0, 500)}`));
   116→          }
   117→        })
   118→      );
   119→  });
   120→}

   122→function routeReq(method: string, founderId: string | null, body?: Record<string, unknown>, query = ''): NextRequest {
   123→  const headers: Record<string, string> = { 'content-type': 'application/json' };
   124→  if (founderId) {
   125→    headers['x-samjuniors-user-id'] = founderId;
   126→    headers['x-samjuniors-role'] = 'FOUNDER';
   127→  }
   128→  return new NextRequest(`http://localhost:3000/api/sofia/memory${query}`, {
   129→    method,
   130→    headers,
   131→    body: body ? JSON.stringify(body) : undefined,
   132→  });
   133→}
