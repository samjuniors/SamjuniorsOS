    94→/**
    95→ * PHASE 4.4E — EVIDENCE LINEAGE (Source → Signal → Claim).
    96→ *
    97→ * Wires REAL gated tool execution output into the EXISTING epistemic pipeline
    98→ * (ingestSource → extractSignal → submitClaim). No second evidence store is
    99→ * created: the raw retrieved content becomes an EvidenceSource, each grounded
   100→ * research claim becomes an EpistemicSignal, and a PENDING claim is submitted
   101→ * referencing both — so the founder board can answer "what source produced
   102→ * this claim / what signal produced it / what evidence supports it".
   103→ *
   104→ * Honesty rules:
   105→ * - Only claims the tool path actually grounded ('claim_supported') receive
   106→ *   lineage. Unverified model output never gets a fabricated source.
   107→ * - Lineage persistence failure must NEVER break orchestration — the failure
   108→ *   is logged honestly and the run proceeds.
   109→ * - Bounded: at most 6 claims per execution (claim spam protection).
   110→ */
   111→const MAX_LINEAGE_CLAIMS_PER_EXECUTION = 6;
   112→
   113→async function persistResearchEvidenceLineage(params: {
   114→  directive: string;
   115→  sourceSystem: string;
   116→  sourceTitle: string;
   117→  uri?: string;
   118→  rawContent: string;
   119→  signalType: 'system_event' | 'code_analysis' | 'metric_observation';
   120→  groundedClaims: Array<{
   121→    statement: string;
   122→    supportingSourceUrls: string[];
   123→    evidenceExcerpt?: string;
   124→  }>;
   125→  subject: string;
   126→  category: 'market_research' | 'architectural';
   127→  agentRunId?: string;
   128→  metadata?: Record<string, any>;
   129→}): Promise<void> {
   130→  try {
   131→    const pipeline = EpistemicPipeline.getInstance();
   132→
   133→    const source = await pipeline.ingestSource({
   134→      sourceSystem: params.sourceSystem,
   135→      uri: params.uri,
   136→      title: params.sourceTitle,
   137→      rawContent: params.rawContent,
   138→      capturedBy: 'researcher',
   139→      provenanceKind: 'live_operational',
   140→      metadata: {
   141→        capturedVia: 'gated_orchestration_tool_execution',
   142→        directive: params.directive,
   143→        ...(params.metadata || {}),
   144→      },
   145→    });
   146→
   147→    for (const grounded of params.groundedClaims.slice(0, MAX_LINEAGE_CLAIMS_PER_EXECUTION)) {
   148→      const signal = await pipeline.extractSignal({
   149→        sourceId: source.id,
   150→        signalType: params.signalType,
   151→        extractedObservation: grounded.evidenceExcerpt || grounded.statement,
   152→        data: {
   153→          statement: grounded.statement,
   154→          supportingSourceUrls: grounded.supportingSourceUrls,
   155→          ...(grounded.evidenceExcerpt ? { evidenceExcerpt: grounded.evidenceExcerpt } : {}),
   156→        },
   157→        confidence: 'high_confidence',
   158→      });
   159→
   160→      await pipeline.submitClaim({
   161→        sourceId: source.id,
   162→        signalId: signal.id,
   163→        statement: grounded.statement,
   164→        subject: params.subject,
   165→        category: params.category,
   166→        proposedBy: 'researcher',
   167→        confidence: 'high_confidence',
   168→        evidenceReferences: grounded.supportingSourceUrls,
   169→        verificationNotes: `Source-backed observation persisted from gated ${params.sourceSystem} execution${params.agentRunId ? ` (agent run ${params.agentRunId})` : ''}. Pending founder verification — NOT company truth.`,
   170→        agentRunId: params.agentRunId,
   171→      });
   172→    }
   173→  } catch (lineageErr) {
   174→    console.error(
   175→      `[ORCHESTRATOR] Epistemic evidence lineage persistence failed (run continues honestly without lineage):`,
   176→      lineageErr
   177→    );
   178→  }
   179→}
