    16→ * ============================================================================
    17→ * M4-C — QUERY-CONDITIONED PERSONAL MIND RETRIEVAL TEST SUITE
    18→ * ============================================================================
    19→ *
    20→ * The M4-C contract: Personal Mind memory selection is conditioned on the
    21→ * CURRENT founder message instead of returning the same memories every turn,
    22→ * while every existing guarantee is preserved. Pinned here:
    23→ *
    24→ *   C1  A memory lexically matching the query outranks an unrelated memory —
    25→ *       even one the pre-M4-C policy would rank FIRST (higher confidence,
    26→ *       newer). The same pool WITHOUT the query keeps the old order.
    27→ *   C2  Identical inputs produce identical ordering (pure determinism).
    28→ *   C3  No-token-match (and no-usable-token) queries fall back to the EXACT
    29→ *       pre-M4-C deterministic selection — independently re-derived in this
    30→ *       suite — never an empty memory context.
    31→ *   C4  The 20-memory retrieval cap remains enforced (all-matching pool).
    32→ *   C5  Matched tier fills first, remaining capacity is filled by the
    33→ *       pre-M4-C policy over unmatched memories; combined cap stays 20.
    34→ *   C6  Within the matched tier, lexical score DESC is the primary order and
    35→ *       confidence/recency only break ties (secondary policy preserved).
    36→ *   C7  Type round-robin is preserved inside the matched tier.
    37→ *   C8  Tokenization is the SHARED CompanyKnowledgeStore pipeline
    38→ *       (extractTokens + stop-word filtering) — case/punctuation-insensitive,
    39→ *       no second retrieval framework.
    40→ *   C9  End-to-end: the matching memory renders BEFORE an unrelated,
    41→ *       higher-confidence, newer memory inside PERSONAL_MIND_MEMORY; a
    42→ *       no-match message restores the old order (message flips selection).
    43→ *   C10 End-to-end determinism: identical assemble() inputs yield identical
    44→ *       slice content.
    45→ *   C11 End-to-end fallback: a token-bearing message that matches nothing
    46→ *       still renders the personal-mind slice (existing behavior preserved).
    47→ *   C12 lifecycleState stays authoritative: PENDING_REVIEW / ARCHIVED /
    48→ *       SUPERSEDED / REJECTED records whose CONTENT matches the query are
    49→ *       never retrieved; only ACTIVE renders.
    50→ *   C13 The 1200-character Personal Mind container budget remains enforced
    51→ *       for query-matched memories (truncation, always-closed container).
    52→ *   C14 Personal Mind retrieval cannot cross into Company Brain authority:
    53→ *       company knowledge content never enters the personal slice, personal
    54→ *       content never enters a company slice or a company store, and Company
    55→ *       Brain retrieval itself is unchanged.
    56→ *   C15 The assembly call site threads the CURRENT founder message into the
    57→ *       selection path (source-level pin, repository convention).
    58→ *
    59→ * NOT UNDER TEST (out of M4-C scope by contract): embeddings/vectors/Redis/
    60→ * Graphiti, LLM-driven retrieval, consolidation, decay/TTL, older-conversation
    61→ * search, Company Brain scoring changes, lifecycle semantics changes.
    62→ */

    83→/** Minimal record shape the deterministic selection policy consumes. */
    84→interface SelMem {
    85→  memoryType: string;
    86→  confidence: number;
    87→  updatedAt: string;
    88→  id: string;
    89→  content: string;
    90→}
    91→
    92→function mem(id: string, memoryType: string, confidence: number, updatedAt: string, content: string): SelMem {
    93→  return { id, memoryType, confidence, updatedAt, content };
    94→}
    95→
    96→const ids = (selected: SelMem[]) => selected.map((m) => m.id);
    97→
    98→/**
    99→ * INDEPENDENT re-derivation of the pre-M4-C policy (P2 follow-up):
   100→ * within-type order = confidence DESC, updatedAt DESC, id ASC; round-robin
   101→ * across SOPHIA_MEMORY_TYPES allow-list order; capped at 20. Written with a
   102→ * different mechanism (index-walking rounds, not queue shifts) so this suite
   103→ * encodes the DOCUMENTED contract rather than mirroring the implementation.
   104→ */
   105→function expectedPreM4cOrder(records: SelMem[]): string[] {
   106→  const byType = new Map<string, SelMem[]>();
   107→  for (const r of records) {
   108→    byType.set(r.memoryType, [...(byType.get(r.memoryType) ?? []), r]);
   109→  }
   110→  for (const list of byType.values()) {
   111→    list.sort((a, b) => {
   112→      if (b.confidence !== a.confidence) return b.confidence - a.confidence;
   113→      const ta = new Date(a.updatedAt).getTime();
   114→      const tb = new Date(b.updatedAt).getTime();
   115→      if (tb !== ta) return tb - ta;
   116→      return a.id < b.id ? -1 : a.id > b.id ? 1 : 0;
   117→    });
   118→  }
   119→  const order: string[] = [];
   120→  for (let round = 0; order.length < 20; round++) {
   121→    let advanced = false;
   122→    for (const t of SOPHIA_MEMORY_TYPES) {
   123→      if (order.length >= 20) break;
   124→      const list = byType.get(t);
   125→      if (list && list.length > round) {
   126→        order.push(list[round].id);
   127→        advanced = true;
   128→      }
   129→    }
   130→    if (!advanced) break;
   131→  }
   132→  return order;
   133→}
