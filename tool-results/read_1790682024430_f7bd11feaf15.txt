    59→/**
    60→ * M5.2 — deterministic lexical fact scoring, shared by both store modes so
    61→ * local and authoritative retrieval rank identically. Uses the SAME shared
    62→ * extractTokens/stop-word pipeline as CompanyKnowledgeStore.queryKnowledge
    63→ * (M4-C convention — no second tokenizer). Subject identifiers tokenize
    64→ * with underscores as spaces ('lumora_pricing' → 'lumora pricing') so a
    65→ * stored subject remains reachable from natural-language queries.
    66→ * Pure function; zero-overlap facts score 0.
    67→ */
    68→export function scoreFactAgainstQuery(fact: CanonicalFact, queryText: string): number {
    69→  const queryTokens = new Set(extractTokens(queryText));
    70→  if (queryTokens.size === 0) return 0;
    71→  const factTokens = new Set([
    72→    ...extractTokens(fact.statement),
    73→    ...extractTokens((fact.subject || '').replace(/_/g, ' ')),
    74→  ]);
    75→  let matched = 0;
    76→  for (const token of queryTokens) {
    77→    if (factTokens.has(token)) matched++;
    78→  }
    79→  return matched;
    80→}
    81→
    82→/**
    83→ * M5.2 — deterministic query-conditioned fact ranking (shared by both
    84→ * modes): score DESC, then promotedAt DESC, then id ASC (total stable order).
    85→ * Zero-score facts are excluded (a fact that shares no token with the query
    86→ * is not query-relevant; the CALLER applies the documented never-empty
    87→ * fallback when every candidate scores zero).
    88→ */
    89→export function rankFactsForQuery(
    90→  facts: CanonicalFact[],
    91→  queryText: string,
    92→  limit: number
    93→): CanonicalFact[] {
    94→  return facts
    95→    .map((fact) => ({ fact, score: scoreFactAgainstQuery(fact, queryText) }))
    96→    .filter((entry) => entry.score > 0)
    97→    .sort((a, b) => {
    98→      if (b.score !== a.score) return b.score - a.score;
    99→      const ta = Date.parse(a.fact.promotedAt || '') || 0;
   100→      const tb = Date.parse(b.fact.promotedAt || '') || 0;
   101→      if (tb !== ta) return tb - ta;
   102→      return a.fact.id < b.fact.id ? -1 : a.fact.id > b.fact.id ? 1 : 0;
   103→    })
   104→    .slice(0, limit)
   105→    .map((entry) => entry.fact);
   106→}

   413→  /**
   414→   * M5.2 — query-conditioned fact retrieval (authoritative mode). Fetches
   415→   * the eligible rows (active, plus superseded when includeSuperseded) and
   416→   * applies the shared deterministic rankFactsForQuery — the SAME ranking
   417→   * the local mode uses, so retrieval semantics never diverge by mode.
   418→   */
   419→  public async queryFacts(params: FactQueryParams): Promise<CanonicalFact[]> {
   420→    const db = await requireAuthoritativeDatabase();
   421→    const where: any = params.includeSuperseded
   422→      ? { validityState: { in: ['active', 'superseded'] } }
   423→      : { validityState: 'active' };
   424→    const facts = await db.canonicalFact.findMany({ where });
   425→    return rankFactsForQuery(facts.map((f) => this.mapPrismaToFact(f)), params.queryText, params.limit ?? 5);
   426→  }

   477→/**
   478→ * Dual-Mode Epistemic Store.
   479→ * In Authoritative Mode: delegates directly to PostgresEpistemicStore (fail-closed).
   480→ * In Test/Local Mode: uses fast in-memory maps with local file persistence.
   481→ */
   482→export class EpistemicClaimStore implements IEpistemicClaimStore {
   483→  private static instance: EpistemicClaimStore | null = null;
   484→
   485→  public sources: Map<string, EvidenceSource> = new Map();
   486→  public signals: Map<string, EpistemicSignal> = new Map();
   487→  public claims: Map<string, EpistemicClaim> = new Map();
   488→  public facts: Map<string, CanonicalFact> = new Map();
   489→  public verifications: Map<string, VerificationPolicyResult> = new Map();
   490→
   491→  constructor() {
   492→    this.loadFromDurableStorage();
   493→  }
   494→
   495→  public static getInstance(): EpistemicClaimStore {
   496→    if (!EpistemicClaimStore.instance) {
   497→      EpistemicClaimStore.instance = new EpistemicClaimStore();
   498→    }
   499→    return EpistemicClaimStore.instance;
   500→  }
   501→
   502→  public loadFromDurableStorage(): void {
   503→    try {
   504→      const fileStore = DurableFileStore.getInstance();
   505→
   506→      const sourcesDisk = fileStore.readCollection<EvidenceSource>('epistemic_sources');
   507→      Object.values(sourcesDisk).forEach((s) => this.sources.set(s.id, s));
   508→
   509→      const signalsDisk = fileStore.readCollection<EpistemicSignal>('epistemic_signals');
   510→      Object.values(signalsDisk).forEach((sig) => this.signals.set(sig.id, sig));
   511→
   512→      const claimsDisk = fileStore.readCollection<EpistemicClaim>('epistemic_claims');
   513→      Object.values(claimsDisk).forEach((c) => this.claims.set(c.id, c));
   514→
   515→      const factsDisk = fileStore.readCollection<CanonicalFact>('canonical_facts');
   516→      Object.values(factsDisk).forEach((f) => this.facts.set(f.id, f));
   517→
   518→      const verificationsDisk = fileStore.readCollection<VerificationPolicyResult>('epistemic_verifications');
   519→      Object.values(verificationsDisk).forEach((v) => this.verifications.set(v.claimId, v));
   520→    } catch (err) {
   521→      console.warn('[EpistemicClaimStore] Failed to load from durable storage:', err);
   522→    }
   523→  }
   524→
   525→  public async saveSource(source: EvidenceSource): Promise<void> {
   526→    if (isAuthoritativeMode()) {
   527→      return PostgresEpistemicStore.getInstance().saveSource(source);
   528→    }
   529→    this.sources.set(source.id, { ...source });
   530→    try {
   531→      DurableFileStore.getInstance().saveItem('epistemic_sources', source.id, source);
   532→    } catch {}
   533→  }

   535→  public async getSource(id: string): Promise<EvidenceSource | null> {
   536→    if (isAuthoritativeMode()) {
   537→      return PostgresEpistemicStore.getInstance().getSource(id);
   538→    }
   539→    const s = this.sources.get(id);
   540→    return s ? { ...s } : null;
   541→  }

   543→  public async listSources(): Promise<EvidenceSource[]> {
   544→    if (isAuthoritativeMode()) {
   545→      return PostgresEpistemicStore.getInstance().listSources();

   547→    return Array.from(this.sources.values());

   643→  public async listActiveFacts(filter?: {
   644→    category?: CanonicalFact['category'];
   645→    subject?: string;
   646→  }): Promise<CanonicalFact[]> {
   647→    if (isAuthoritativeMode()) {
   648→      return PostgresEpistemicStore.getInstance().listActiveFacts(filter);
   649→    }
   650→    let result = Array.from(this.facts.values()).filter((f) => f.validityState === 'active');
   651→    if (filter?.category) {
   652→      result = result.filter((f) => f.category === filter.category);
   653→    }
   654→    if (filter?.subject) {
   655→      result = result.filter((f) => f.subject.toLowerCase() === filter.subject!.toLowerCase());
   656→    }
   657→    return result.sort((a, b) => b.promotedAt.localeCompare(a.promotedAt));
   658→  }

   660→  /**
   661→   * M5.2 — every fact regardless of validityState (local mode). Ordered
   662→   * promotedAt DESC, id ASC — identical order contract to the authoritative
   663→   * mode. Superseded facts retain their supersededById pointers.
   664→   */
   665→  public async listAllFacts(): Promise<CanonicalFact[]> {
   666→    if (isAuthoritativeMode()) {
   667→      return PostgresEpistemicStore.getInstance().listAllFacts();
   668→    }
   669→    return Array.from(this.facts.values())
   670→      .sort((a, b) => {
   671→        if (b.promotedAt !== a.promotedAt) return b.promotedAt.localeCompare(a.promotedAt);
   672→        return a.id < b.id ? -1 : a.id > b.id ? 1 : 0;
   673→      })
   674→      .map((f) => ({ ...f }));
   675→  }
   676→
   677→  /**
   678→   * M5.2 — query-conditioned canonical-fact retrieval (local mode).
   679→   * Deterministic lexical ranking over the shared tokenizer (see
   680→   * rankFactsForQuery). Zero-overlap facts are excluded; the CALLER applies
   681→   * the documented never-empty fallback (A0 top-3-newest) when nothing
   682→   * matches. includeSuperseded adds superseded facts to the candidate pool
   683→   * WITHOUT changing their lifecycle state or truth position.
   684→   */
   685→  public async queryFacts(params: FactQueryParams): Promise<CanonicalFact[]> {
   686→    if (isAuthoritativeMode()) {
   687→      return PostgresEpistemicStore.getInstance().queryFacts(params);
   688→    }
   689→    const pool = Array.from(this.facts.values()).filter((f) =>
   690→      params.includeSuperseded
   691→        ? f.validityState === 'active' || f.validityState === 'superseded'
   692→        : f.validityState === 'active'
   693→    );
   694→    return rankFactsForQuery(pool, params.queryText, params.limit ?? 5);
   695→  }

   697→  public async markFactSuperseded(factId: string, supersededById: string): Promise<void> {
   698→    if (isAuthoritativeMode()) {
   699→      return PostgresEpistemicStore.getInstance().markFactSuperseded(factId, supersededById);
   700→    }
   701→    const fact = this.facts.get(factId);
   702→    if (fact) {
   703→      fact.validityState = 'superseded';
   704→      fact.supersededById = supersededById;
   705→      await this.saveFact(fact);
   706→    }
   707→  }
