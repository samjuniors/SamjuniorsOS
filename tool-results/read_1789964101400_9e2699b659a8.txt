    43→/**
    44→ * Who is allowed to talk to this bridge.
    45→ *
    46→ * A WebSocket handshake is not subject to the same-origin policy: the browser
    47→ * sends it on behalf of whatever page asked, no preflight stands in the way,
    48→ * and the page reads every byte that comes back. Without a check here, any tab
    49→ * the user happens to have open could open a socket to ws://localhost:8787,
    50→ * drive the agent with every MCP server on this machine, and read back every
    51→ * token and panel. The Origin header is the only thing that separates our own
    52→ * dev server from someone else's page, so it is checked explicitly.
    53→ *
    54→ * A missing Origin means a non-browser client — curl, a script, a native app.
    55→ * That is also exactly what local malware looks like, so it is refused on the
    56→ * socket unless JARVIS_ALLOW_NO_ORIGIN=1 says otherwise.
    57→ */
    58→const EXTRA_ORIGINS = new Set(
    59→  (process.env.JARVIS_ALLOWED_ORIGINS ?? '')
    60→    .split(',')
    61→    .map((s) => s.trim().replace(/\/+$/, ''))
    62→    .filter(Boolean),
    63→)
    64→const ALLOW_NO_ORIGIN = process.env.JARVIS_ALLOW_NO_ORIGIN === '1'
    65→
    66→const LOCAL_HOSTS = new Set(['localhost', '127.0.0.1', '[::1]'])
    67→
    68→/**
    69→ * Vite takes the next free port when 5173 is busy and `vite preview` starts at
    70→ * 4173, so the dev ranges are allowed rather than two exact numbers. Anything
    71→ * else — including localhost on a port some other app is serving — has to be
    72→ * named in JARVIS_ALLOWED_ORIGINS.
    73→ */
    74→const isDevPort = (port) =>
    75→  (port >= 5173 && port <= 5199) || (port >= 4173 && port <= 4199)
    76→
    77→function originAllowed(origin) {
    78→  if (!origin) return ALLOW_NO_ORIGIN
    79→  if (EXTRA_ORIGINS.has(origin.replace(/\/+$/, ''))) return true
    80→  let url
    81→  try {
    82→    url = new URL(origin)
    83→  } catch {
    84→    return false
    85→  }
    86→  if (url.protocol !== 'http:') return false
    87→  if (!LOCAL_HOSTS.has(url.hostname)) return false
    88→  return isDevPort(Number(url.port))
    89→}

   147→ * This does two jobs. The HUD wants the names while the boot animation plays,
   148→ * and the agent doesn't emit its init message — and therefore its server

   178→/** MCP tools arrive as `mcp__<server>__<tool>`. */
   179→const mcpServerOf = (toolName) =>
   180→  toolName.startsWith('mcp__') ? toolName.split('__')[1] : null
   181→
   182→/** The tool half, which can itself contain underscores: `mcp__x__a__b` -> `a__b`. */
   183→const mcpToolOf = (toolName) => toolName.split('__').slice(2).join('__')
   184→
   185→/**
   186→ * MCP policy, and why it is shaped this way.
   187→ *
   188→ * A short list of "servers that can change things" is the wrong default,
   189→ * because it is a list of what we happened to think of. Every server not on it
   190→ * runs unconditionally — and on a real machine that quietly includes placing a
   191→ * phone call, spending an advertising budget, deleting a generated character
   192→ * and writing files to disk. A voice assistant cannot ask "are you sure", so
   193→ * the bridge has to be the one that is sure.
   194→ *
   195→ * So the default is deny, softened in two ways so the demo stays usable:
   196→ *
   197→ *   1. READ_ONLY_MCP is an explicit allowlist of servers whose whole surface is
   198→ *      lookups and generation — search, registries, analytics reads. Anything
   199→ *      there runs in read-only mode.
   200→ *   2. Everywhere else, the tool has to argue for itself: its own name must
   201→ *      begin with a read verb. `list_devices` runs; `install_apk` does not.
   202→ *
   203→ * On top of both sits a veto: a name containing a plainly effectful verb needs
   204→ * ALLOW_WRITES no matter which server it came from, which is what keeps
   205→ * `make_outbound_call` and `download_lottie` still until you ask for them.
   206→ */
   207→const READ_ONLY_MCP = new Set([
   208→  'exa', 'exa-code', 'serper', 'serpapi', 'lottie-search', 'mcp-registry',
   209→  'openrouter', 'openrouter-image', 'Microsoft_Clarity',
   210→  // The generation servers belong here too, and leaving them out was a real
   211→  // regression: `generate_image` begins with no read verb, so it fell to the
   212→  // deny branch and "generate an image of the Mark VII suit" — the headline
   213→  // demo — stopped working in the default mode.
   214→  //
   215→  // Putting them on the allowlist is safe because the veto below still applies
   216→  // to allowlisted servers: it is what continues to withhold
   217→  // make_outbound_call, delete_character, create_* and edit_image. Generation
   218→  // runs; acting on the world does not.
   219→  'higgsfield', 'heygen', 'elevenlabs',
   220→])
   221→
   222→/**
   223→ * Anchored on the tool name, so it reads the verb rather than the noun.
   224→ * `screenshot` is in here because it is a read that doesn't sound like one,
   225→ * and the persona is told in as many words to put screenshots on the display.
   226→ */
   227→const READ_VERB =
   228→  /^(get|list|read|search|find|query|fetch|check|describe|inspect|show|view|explain|screenshot)/i
   229→
   230→/**
   231→ * Unanchored on purpose — `make_outbound_call` and `Bulk-Edit-Events` both
   232→ * hide their verb in the middle. `download` is here because it writes a file
   233→ * even though it sounds like a read.
   234→ */
   235→const EFFECTFUL_VERB =
   236→  /(send|call|post|create|delete|remove|update|edit|write|install|launch|tap|swipe|press|type|buy|pay|charge|publish|deploy|outbound|download)/i
   237→
   238→/**
   239→ * Tools whose names trip the veto without deserving it.
   240→ *
   241→ * The veto reads verbs out of names, which is the right instinct and
   242→ * occasionally the wrong answer. `openrouter send-message` sends a prompt to a
   243→ * language model and gets text back — nothing in the world changes — but it is
   244→ * indistinguishable by name from sending mail. Asking a second model a question
   245→ * is one of the better things this assistant can do, so it is named here
   246→ * instead of being lost to a regex.
   247→ *
   248→ * Full `server__tool` keys, so an exemption can never leak across servers.
   249→ */
   250→const VETO_EXEMPT = new Set([
   251→  'openrouter__send-message',
   252→  'openrouter__send-feedback',
   253→])
   254→
   255→function decideTool(name) {
   256→  if (READ_ONLY_BUILTINS.has(name)) return true
   257→  if (WRITE_BUILTINS.has(name)) return ALLOW_WRITES
   258→
   259→  const server = mcpServerOf(name)
   260→  if (server) {
   261→    // The HUD, and the interface controls beside it. Both run in this process
   262→    // and draw on our own screen, so neither is something to withhold —
   263→    // without them JARVIS has no display at all. They also have to be named
   264→    // here rather than left to the verb rules below, which read `ui_theme` as
   265→    // a write and would hold the whole surface back behind ALLOW_WRITES.
   266→    if (server === 'jarvis' || server === 'jarvis_ui') return true
   267→
   268→    // The browser server gates itself, at construction: chromeServer() only
   269→    // builds the acting tools — click, type, form input, close tab — when
   270→    // ALLOW_WRITES is set, so anything that reaches here at all is something
   271→    // the same policy has already permitted. Deciding it a second time by
   272→    // reading verbs out of the name would only get it wrong: `chrome_navigate`
   273→    // begins with no read verb and would fall to the write branch, which would
   274→    // withhold the one tool the whole server is for.
   275→    if (server === 'jarvis_chrome') return true
   276→
   277→    // The camera. Not withheld behind ALLOW_WRITES: looking changes nothing,
   278→    // and the real gate is the browser's own camera permission plus an
   279→    // indicator the user can see for as long as it is live.
   280→    if (server === 'jarvis_eyes') return true
   281→
   282→    const tool = mcpToolOf(name)
   283→    if (EFFECTFUL_VERB.test(tool) && !VETO_EXEMPT.has(`${server}__${tool}`)) {
   284→      return ALLOW_WRITES
   285→    }
   286→    // The session tools this bridge is developed inside count as read-only too.
   287→    if (READ_ONLY_MCP.has(server) || server.startsWith('ccd_session')) return true
   288→    return READ_VERB.test(tool) ? true : ALLOW_WRITES
   289→  }
   290→  return ALLOW_WRITES
   291→}

   466→ * like with like — on macOS os.tmpdir() is a symlink into /private/var, and a
   467→ * string prefix test against the unresolved form would reject every screenshot.
   468→ */
   469→const IMAGE_TYPES = {
   470→  '.png': 'image/png',
   471→  '.jpg': 'image/jpeg',
   472→  '.jpeg': 'image/jpeg',
   473→  '.webp': 'image/webp',
   474→  '.gif': 'image/gif',
   475→  // .svg is deliberately absent. An SVG is a scriptable document, and this
   476→  // endpoint serves it from the bridge's own origin — the one origin allowed
   477→  // to open the agent socket. A picture is not worth that.
   478→}
   479→
   480→const MAX_FILE_BYTES = 25 * 1024 * 1024
   481→
   482→const FILE_ROOTS = [
   483→  homedir(),
   484→  // Both temp directories, because on macOS os.tmpdir() is the per-user
   485→  // $TMPDIR under /var/folders while half the tools that take a screenshot
   486→  // still write it to /tmp. Dropping one of them loses real panels.
   487→  tmpdir(),
   488→  '/tmp',
   489→  ...(process.env.JARVIS_FILE_ROOTS ?? '')
   490→    .split(',')
   491→    .map((s) => s.trim())
   492→    .filter(Boolean),
   493→].map((root) => {
   494→  try {
   495→    return realpathSync(root)
   496→  } catch {
   497→    return resolvePath(root)
   498→  }
   499→})
   500→
   501→/** True when `real` sits inside one of the roots, after both are resolved. */
   502→const withinRoots = (real) =>
   503→  FILE_ROOTS.some((root) => {
   504→    const rel = relative(root, real)
   505→    return rel !== '' && !rel.startsWith('..') && !isAbsolute(rel)
   506→  })

   509→
   510→/**
   511→ * Remote media, fetched by the bridge instead of by the page.
   512→ *
   513→ * JARVIS used to refuse to show anything he found on the web, and the refusal
   514→ * was not squeamishness — a bare <img src="https://some-cdn/..."> in a panel
   515→ * genuinely did not work. Three reasons, and all three are fixed by moving the
   516→ * fetch to this side of the wire:
   517→ *
   518→ *   1. Hotlink blocking. News sites and image CDNs check Referer and User-Agent
   519→ *      and hand a browser-that-isn't-their-page a 403 or a placeholder. That is
   520→ *      why thumbnails rendered as empty rectangles. A server-side fetch that
   521→ *      looks like an ordinary browser and sends no referrer gets the bytes.
   522→ *   2. Privacy. Panel HTML is authored by a model that has just been reading
   523→ *      untrusted web pages, so a remote URL in it is a prompt-injection beacon:
   524→ *      load it directly and the user's IP, and the fact they asked, go to a host
   525→ *      the page chose. Proxying means the browser only ever talks to localhost
   526→ *      and the page CSP can stay tight.
   527→ *   3. One place to cap size, set timeouts and insist the bytes really are the
   528→ *      media type they claim.
   529→ *
   530→ * The cost is that this process — unlike a browser tab — can reach the user's
   531→ * LAN, their router's admin page, and cloud metadata endpoints. So everything
   532→ * below is an SSRF gate first and a proxy second.
   533→ */
   534→
   535→const MAX_IMG_BYTES = 15 * 1024 * 1024
   536→const MAX_MEDIA_BYTES = 200 * 1024 * 1024
   537→const IMG_TIMEOUT_MS = 10_000
   538→const MEDIA_TIMEOUT_MS = 30_000
   539→
   540→// The SSRF gate and the guarded outbound clients now live in ./net.mjs, so the
   541→// media proxy below and the page proxy share one implementation of the rules
   542→// rather than two that can drift apart.
   543→
   544→/**
   545→ * The shared body of /img and /media.
   546→ *
   547→ * `kinds` is the list of content-type prefixes we are willing to hand back.
   548→ * That check is load-bearing: without it this is an open proxy that will serve
   549→ * an attacker's HTML from the bridge's own origin — the one origin allowed to
   550→ * open the agent socket — which is the same reason IMAGE_TYPES has no .svg.
   551→ */
   552→async function proxyRemote(req, res, cors, { kinds, maxBytes, timeoutMs, ranged }) {
   553→  const asked = new URL(req.url, 'http://x').searchParams.get('url') ?? ''
   554→  const target = vetTarget(asked)
   555→
   556→  const headers = {
   557→    'user-agent': PROXY_UA,
   558→    accept: ranged ? '*/*' : 'image/*,*/*;q=0.8',
   559→    // Identity encoding so the byte cap counts the bytes we actually stream and
   560→    // content-length means what it says. Media is already compressed anyway.
   561→    'accept-encoding': 'identity',
   562→  }
   563→  // Range is the difference between a <video> that seeks and one Safari refuses
   564→  // to play at all, so the browser's request is passed through verbatim.
   565→  if (ranged && typeof req.headers.range === 'string') {
   566→    headers.range = req.headers.range
   567→  }
   568→
   569→  const { res: upstream } = await openRemote(target, headers, timeoutMs)
   570→  const status = upstream.statusCode ?? 0
   571→
   572→  if (status !== 200 && status !== 206) {
   573→    upstream.resume()
   574→    throw proxyError(status === 404 ? 404 : 502, `upstream said ${status}`)
   575→  }
   576→
   577→  const type = String(upstream.headers['content-type'] ?? '')
   578→    .split(';')[0]
   579→    .trim()
   580→    .toLowerCase()
   581→  if (!kinds.some((kind) => type.startsWith(kind))) {
   582→    upstream.resume()
   583→    throw proxyError(415, `not ${kinds.join(' or ')} (got ${type || 'nothing'})`)
   584→  }
   585→
   586→  const declared = Number(upstream.headers['content-length'])
   587→  if (Number.isFinite(declared) && declared > maxBytes) {
   588→    upstream.resume()
   589→    throw proxyError(413, 'too large')
   590→  }
   591→
   592→  const out = {
   593→    ...cors,
   594→    'content-type': type,
   595→    'x-content-type-options': 'nosniff',
   596→    // Thumbnails get looked at, panelled again, and re-rendered on every HUD
   597→    // repaint; re-fetching from the CDN each time is slow and rude.
   598→    'cache-control': 'private, max-age=600',
   599→  }
   600→  if (Number.isFinite(declared)) out['content-length'] = String(declared)
   601→  if (ranged) {
   602→    // Only claim range support when the origin actually demonstrated it — a
   603→    // 206, or an explicit accept-ranges of its own. Plenty of hosts ignore the
   604→    // Range header and hand back the whole file with a 200; advertising
   605→    // accept-ranges on top of that tells the video element it may seek by
   606→    // issuing byte requests that will never be honoured, and the scrub bar
   607→    // then misbehaves in a way that looks like our bug rather than theirs.
   608→    if (status === 206 || upstream.headers['accept-ranges'] === 'bytes') {
   609→      out['accept-ranges'] = 'bytes'
   610→    }
   611→    if (upstream.headers['content-range']) {
   612→      out['content-range'] = upstream.headers['content-range']
   613→    }
   614→  }
   615→  res.writeHead(status, out)
   616→
   617→  // Stream with a running cap. Buffering a 200 MB video into this process
   618→  // would stall the token stream the voice is riding on, and trusting
   619→  // content-length would let a host that lies about it eat the heap.
   620→  let sent = 0
   621→  upstream.on('data', (chunk) => {
   622→    sent += chunk.length
   623→    if (sent > maxBytes) {
   624→      // Headers went out long ago, so a truncated body is the only way left to
   625→      // say no. The player sees a short read; we see this line in the log.
   626→      console.warn(`[jarvis] proxy cut ${target.href} at ${maxBytes} bytes`)
   627→      upstream.destroy()
   628→      res.destroy()
   629→      return
   630→    }
   631→    if (!res.write(chunk)) {
   632→      upstream.pause()
   633→      res.once('drain', () => upstream.resume())
   634→    }
   635→  })
   636→  upstream.on('end', () => res.end())
   637→  upstream.on('error', () => res.destroy())
   638→  req.on('close', () => upstream.destroy())
   639→}
   640→
   641→// ---------------------------------------------------------------------------
   642→
   643→/**
   644→ * CORS, reflected rather than wildcarded.
   645→ *
   646→ * `*` on this origin means any page on the internet can read whatever the
   647→ * bridge serves, so the same allowlist that guards the socket picks the
   648→ * header. A request carrying an Origin we don't know is refused outright —
   649→ * but a request with no Origin at all is served, because an <img src> load
   650→ * (which is how panels fetch screenshots) never sends one.
   651→ */
   652→function corsFor(req) {
   653→  const origin = req.headers.origin
   654→  const headers = { vary: 'origin' }
   655→  if (origin) {
   656→    headers['access-control-allow-origin'] = origin
   657→    headers['access-control-allow-headers'] = 'content-type'
   658→  }
   659→  return headers
   660→}
   661→
   662→// One HTTP server for both the speech proxy and the WebSocket upgrade.
   663→const http = await import('node:http')
   664→
   665→const handleRequest = async (req, res) => {
   666→  const origin = req.headers.origin
   667→  if (origin && !originAllowed(origin)) {
   668→    console.warn(`[jarvis] refused http request from origin ${origin}`)
   669→    res.writeHead(403, { vary: 'origin' })
   670→    return res.end('forbidden')
   671→  }
   672→  const cors = corsFor(req)
   673→
   674→  if (req.method === 'OPTIONS') {
   675→    res.writeHead(204, cors)
   676→    return res.end()
   677→  }
   678→
   679→  if (req.method === 'GET' && req.url === '/health') {
   680→    // The browser reads this once at boot to decide which voice engine to use.
   681→    // Both premium paths ride the same ElevenLabs key, so both flags track it:
   682→    // with a key the app transcribes with Scribe and speaks with ElevenLabs;
   683→    // without one it falls back to the browser's own recogniser and voice, so a
   684→    // student with nothing configured still has a working assistant.
   685→    const eleven = Boolean(elevenKey())
   686→    res.writeHead(200, { ...cors, 'content-type': 'application/json' })
   687→    return res.end(JSON.stringify({ ok: true, tts: eleven, stt: eleven }))
   688→  }
   689→
   690→  // Serve local image files to the page. Screenshots and generated art land on
   691→  // disk as absolute paths, and a page served over http can't read file:// —
   692→  // so the bridge, which can, hands them over.
   693→  if (req.method === 'GET' && req.url?.startsWith('/file?')) {
   694→    const asked = new URL(req.url, 'http://x').searchParams.get('path') ?? ''
   695→    // Resolve symlinks BEFORE judging anything. A name ending in .png can be a
   696→    // link pointing at /etc/hosts, and checking the suffix the caller supplied
   697→    // would wave that straight through — which is exactly how this endpoint
   698→    // used to serve the contents of arbitrary system files.
   699→    let real = null
   700→    try {
   701→      if (isAbsolute(asked)) real = await realpath(asked)
   702→    } catch {
   703→      real = null
   704→    }
   705→    const dot = real ? real.lastIndexOf('.') : -1
   706→    const ext = dot === -1 ? '' : real.slice(dot).toLowerCase()
   707→    // Images only, absolute paths only, and only under roots we expect things
   708→    // to be written to. This endpoint exists to show pictures, not to be a
   709→    // general file read for whatever the model — or another page — asks for.
   710→    if (!real || !Object.hasOwn(IMAGE_TYPES, ext) || !withinRoots(real)) {
   711→      res.writeHead(400, cors)
   712→      return res.end('images only')
   713→    }
   714→    try {
   715→      const info = await stat(real)
   716→      if (!info.isFile() || info.size > MAX_FILE_BYTES) {
   717→        res.writeHead(413, cors)
   718→        return res.end('too large')
   719→      }
   720→      // Asynchronous because this process is also pumping the agent's token
   721→      // stream; a synchronous read of a large screenshot stalls the voice.
   722→      const body = await readFile(real)
   723→      res.writeHead(200, {
   724→        ...cors,
   725→        'content-type': IMAGE_TYPES[ext],
   726→        'x-content-type-options': 'nosniff',
   727→      })
   728→      return res.end(body)
   729→    } catch {
   730→      res.writeHead(404, cors)
   731→      return res.end('not found')
   732→    }
   733→  }
   734→
   735→  // Remote images, fetched here so the page never talks to the wider web. The
   736→  // renderer rewrites every http(s) <img src> in a panel to this endpoint.
   737→  if (req.method === 'GET' && req.url?.startsWith('/img?')) {
   738→    try {
   739→      await proxyRemote(req, res, cors, {
   740→        kinds: ['image/'],
   741→        maxBytes: MAX_IMG_BYTES,
   742→        timeoutMs: IMG_TIMEOUT_MS,
   743→        ranged: false,
   744→      })
   745→    } catch (err) {
   746→      if (res.headersSent) return res.destroy()
   747→      res.writeHead(err.status ?? 502, cors)
   748→      return res.end(err.message ?? 'proxy failed')
   749→    }
   750→    return
   751→  }
   752→
   753→  // The same, for video and audio. Separate from /img because the limits and
   754→  // the Range handling are genuinely different, not because the code is.
   755→  if (req.method === 'GET' && req.url?.startsWith('/media?')) {
   756→    try {
   757→      await proxyRemote(req, res, cors, {
   758→        kinds: ['video/', 'audio/'],
   759→        maxBytes: MAX_MEDIA_BYTES,
   760→        timeoutMs: MEDIA_TIMEOUT_MS,
   761→        ranged: true,
   762→      })
   763→    } catch (err) {
   764→      if (res.headersSent) return res.destroy()
   765→      res.writeHead(err.status ?? 502, cors)
   766→      return res.end(err.message ?? 'proxy failed')
   767→    }
   768→    return
   769→  }
   770→
   771→  // A whole web page, fetched here and served from this origin so it can be
   772→  // framed. The publisher's X-Frame-Options and CORS rules are enforced against
   773→  // the browser, and from the browser's point of view this document is ours —
   774→  // so an article that refuses to be embedded anywhere still opens on the
   775→  // display. See page.mjs for what each mode does to the markup.
   776→  //
   777→  // No Origin header arrives on an iframe navigation, so this rides the same
   778→  // path as an <img> load through the check at the top of this handler.
   779→  if (req.method === 'GET' && req.url?.startsWith('/page?')) {
   780→    const asked = new URL(req.url, 'http://x')
   781→    const target = asked.searchParams.get('url') ?? ''
   782→    const mode = asked.searchParams.get('mode') === 'live' ? 'live' : 'reader'
   783→    try {
   784→      const page = await renderPage(target, mode, `http://localhost:${PORT}`)
   785→      res.writeHead(200, { ...cors, ...page.headers })
   786→      return res.end(page.body)
   787→    } catch (err) {
   788→      // Rendered as a page rather than returned as a status, because this lands
   789→      // inside an iframe: a bare 502 body is a blank rectangle on the display,
   790→      // which reads as the interface being broken rather than as the article
   791→      // being unavailable.
   792→      res.writeHead(err.status ?? 502, {
   793→        ...cors,
   794→        'content-type': 'text/html; charset=utf-8',
   795→        'content-security-policy': "default-src 'none'; style-src 'unsafe-inline'",
   796→      })
   797→      return res.end(
   798→        `<!doctype html><meta charset="utf-8"><style>
   799→           body{margin:0;padding:26px;background:transparent;color:#7fb6bf;
   800→                font:400 13px/1.6 ui-monospace,monospace}
   801→           b{color:#cfe9ee;font-weight:500;display:block;margin-bottom:6px}
   802→         </style><b>This page could not be opened.</b>${
   803→           String(err?.message ?? 'unknown error').replace(/[<&]/g, '')
   804→         }`,
   805→      )
   806→    }
   807→  }
   808→
   809→  if (req.method === 'POST' && req.url === '/tts') {
   810→    const key = elevenKey()
   811→    if (!key) {
   812→      res.writeHead(503, cors)
   813→      return res.end('no elevenlabs key')
   814→    }
   815→    // A spoken line is a few hundred bytes. Anything approaching this is not a
   816→    // sentence, and buffering it unbounded would let one request eat the heap.
   817→    let body = ''
   818→    let overflowed = false
   819→    for await (const chunk of req) {
   820→      body += chunk
   821→      if (body.length > 64 * 1024) {
   822→        overflowed = true
   823→        break
   824→      }
   825→    }
   826→    if (overflowed) {
   827→      req.destroy()
   828→      res.writeHead(400, cors)
   829→      return res.end('body too large')
   830→    }
   831→    // Inside a try: this handler is async with nothing catching its rejection,
   832→    // so a malformed body used to take the entire bridge down with it.
   833→    let text
   834→    try {
   835→      ;({ text } = JSON.parse(body || '{}'))
   836→    } catch {
   837→      res.writeHead(400, cors)
   838→      return res.end('bad json')
   839→    }
   840→    if (!text) {
   841→      res.writeHead(400, cors)
   842→      return res.end('no text')
   843→    }
   844→    try {
   845→      const upstream = await fetch(
   846→        `https://api.elevenlabs.io/v1/text-to-speech/${VOICE_ID}/stream` +
   847→          // 22kHz mono is half the bytes of 44kHz and indistinguishable through
   848→          // a laptop speaker; optimize_streaming_latency=3 trades a little
   849→          // prosody for a much earlier first byte.
   850→          `?output_format=mp3_22050_32&optimize_streaming_latency=3`,
   851→        {
   852→          method: 'POST',
   853→          headers: { 'xi-api-key': key, 'content-type': 'application/json' },
   854→          body: JSON.stringify({
   855→            text,
   856→            // Flash is the low-latency model — a conversation needs speed more
   857→            // than it needs the last few percent of quality.
   858→            model_id: 'eleven_flash_v2_5',
   859→            voice_settings: {
   860→              stability: 0.4,
   861→              similarity_boost: 0.75,
   862→              speed: 1.05,
   863→            },
   864→          }),
   865→        },
   866→      )
   867→      if (!upstream.ok) {
   868→        res.writeHead(upstream.status, cors)
   869→        return res.end(await upstream.text())
   870→      }
   871→
   872→      // Pipe it through rather than buffering. Waiting for the whole file here
   873→      // would throw away everything the streaming endpoint just bought us.
   874→      res.writeHead(200, {
   875→        ...cors,
   876→        'content-type': 'audio/mpeg',
   877→        'cache-control': 'no-cache',
   878→      })
   879→      for await (const chunk of upstream.body) res.write(Buffer.from(chunk))
   880→      return res.end()
   881→    } catch (err) {
   882→      res.writeHead(502, cors)
   883→      return res.end(String(err?.message ?? err))
   884→    }
   885→  }
   886→
   887→  // Speech to text. The browser captures one spoken segment as a compressed
   888→  // audio blob and posts the raw bytes here; the bridge hands them to
   889→  // ElevenLabs Scribe and returns the transcript. This is what replaced the
   890→  // browser's own SpeechRecognition — that API dies silently under always-on
   891→  // use, and a server-side transcriber cannot. Detecting that the user is
   892→  // speaking at all is done locally with voice-activity detection, which never
   893→  // touches this endpoint; this is only for the words.
   894→  if (req.method === 'POST' && req.url === '/stt') {
   895→    const key = elevenKey()
   896→    if (!key) {
   897→      res.writeHead(503, cors)
   898→      return res.end('no elevenlabs key')
   899→    }
   900→
   901→    const type = req.headers['content-type'] || 'audio/webm'
   902→    const chunks = []
   903→    let size = 0
   904→    let overflowed = false
   905→    // A few seconds of Opus is well under a megabyte; 25 MB is a generous
   906→    // ceiling that still refuses a runaway stream before it eats the heap.
   907→    for await (const chunk of req) {
   908→      chunks.push(chunk)
   909→      size += chunk.length
   910→      if (size > 25 * 1024 * 1024) {
   911→        overflowed = true
   912→        break
   913→      }
   914→    }
   915→    if (overflowed) {
   916→      req.destroy()
   917→      res.writeHead(413, cors)
   918→      return res.end('audio too large')
   919→    }
   920→    // Silence, or a click. Nothing to transcribe, and calling out to the API
   921→    // for it would only add latency to a non-answer.
   922→    if (size < 1200) {
   923→      res.writeHead(200, { ...cors, 'content-type': 'application/json' })
   924→      return res.end(JSON.stringify({ text: '' }))
   925→    }
   926→
   927→    try {
   928→      // The filename extension is the only hint Scribe gets about the codec, so
   929→      // derive it from the content-type the MediaRecorder reported rather than
   930→      // hard-coding one.
   931→      const ext = type.includes('ogg')
   932→        ? 'ogg'
   933→        : type.includes('mp4') || type.includes('mpeg')
   934→          ? 'mp4'
   935→          : type.includes('wav')
   936→            ? 'wav'
   937→            : 'webm'
   938→      const form = new FormData()
   939→      form.append('model_id', 'scribe_v1')
   940→      form.append(
   941→        'file',
   942→        new Blob([Buffer.concat(chunks)], { type }),
   943→        `speech.${ext}`,
   944→      )
   945→
   946→      const upstream = await fetch('https://api.elevenlabs.io/v1/speech-to-text', {
   947→        method: 'POST',
   948→        headers: { 'xi-api-key': key },
   949→        body: form,
   950→      })
   951→      if (!upstream.ok) {
   952→        res.writeHead(upstream.status, cors)
   953→        return res.end(await upstream.text())
   954→      }
   955→      const data = await upstream.json()
   956→      res.writeHead(200, { ...cors, 'content-type': 'application/json' })
   957→      return res.end(JSON.stringify({ text: (data.text ?? '').trim() }))
   958→    } catch (err) {
   959→      res.writeHead(502, cors)
   960→      return res.end(String(err?.message ?? err))
   961→    }
   962→  }
   963→
   964→  res.writeHead(404, cors)
   965→  res.end()
   966→}
   967→
   968→const server = http.createServer((req, res) => {
   969→  // The handler is async, so anything it throws would otherwise become an
   970→  // unhandled rejection and leave the browser waiting on a socket that is
   971→  // never going to answer.
   972→  handleRequest(req, res).catch((err) => {
   973→    console.error('[jarvis] request failed:', err)
   974→    if (!res.headersSent) res.writeHead(500)
   975→    res.end()
   976→  })
   977→})
   978→
   979→const wss = new WebSocketServer({
   980→  server,
   981→  // The handshake is the only place a page can be turned away, so it happens
   982→  // here rather than after the socket is open. Rejections are logged loudly:
   983→  // the likeliest cause is a dev server on an unexpected port, and a silent
   984→  // 403 would look like the bridge simply isn't running.
   985→  verifyClient: ({ origin, req }, done) => {
   986→    const path = (req.url ?? '/').split('?')[0]
   987→    if (path !== '/' && path !== '/ws') {

  1390→      // on a pump that has already stopped. Close it so it reconnects.

  1410→       * A barge-in is two messages in quick succession — interrupt, then the
  1411→       * new question — and session.interrupt() is asynchronous. Delivering the
  1412→       * question the instant it arrives means the agent can still be winding
  1413→       * down the previous turn, so its last tokens are emitted after the new
  1414→       * one has begun and land on the new turn's listener. Measured: ask "one",
  1415→       * interrupt, ask "two", and the answer to "two" comes back as "One."
  1416→       *
  1417→       * Waiting costs nothing when nothing is interrupting — the chain is an
  1418→       * already-resolved promise — and removes the cross-talk when there is.
  1419→       */
  1420→      const text = msg.text
  1421→      const id = typeof msg.id === 'string' ? msg.id : null

  1423→        answering = id
  1424→        if (deliver) {
  1425→          const resolve = deliver
  1426→          deliver = null
  1427→          resolve(text)
  1428→        } else {
  1429→          inbox.push(text)
  1430→        }
  1431→      })
  1432→    }
  1433→

  1436→      if (slot) {
  1437→        waiting.delete(msg.id)
  1438→        clearTimeout(slot.timer)
  1439→        slot.resolve(msg)
  1440→      }
  1441→    }
  1442→

  1445→      const stopped = turnFinished()

  1447→        .catch(() => {})
  1448→        .then(() =>
  1449→          Promise.race([
  1450→            stopped,
  1451→            new Promise((r) => setTimeout(r, SETTLE_CAP_MS)),
  1452→          ]),
  1453→        )
  1454→    }
  1455→  })
  1456→

  1458→    console.log('[jarvis] client disconnected')
  1459→    closed = true
  1460→    deliver?.(null)
  1461→    session.close?.()
  1462→  })
