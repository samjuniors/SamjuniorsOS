    98→function pruneDurableCollection(collection: string, ids: Set<string>): void {
    99→  const file = join(process.cwd(), ".data", `${collection}.json`);
   100→  try {
   101→    if (!existsSync(file) || ids.size === 0) return;
   102→    const raw = JSON.parse(readFileSync(file, "utf-8"));
   103→    let changed = false;
   104→    for (const id of ids) {
   105→      if (raw[id] !== undefined) {
   106→        delete raw[id];
   107→        changed = true;
   108→      }
   109→    }
   110→    if (changed) writeFileSync(file, JSON.stringify(raw), "utf-8");
   111→  } catch {
   112→    /* best-effort cleanup */
   113→  }
   114→}

   172→/**
   173→ * Implements the REAL ServerAgentExecutor contract AND persists runs through
   174→ * the REAL AgentRunStore (dev dual-mode: in-memory map + DurableFileStore +
   175→ * best-effort SQLite) — exactly the persistence behavior of the real

   178→class ContractExecutorDouble {
   179→  public calls = 0;
   180→  public mode: "success" | "fail" = "success";
   181→  private readonly prefix: string;
   182→
   183→  constructor(prefix = `run-44c-${uuidv4().slice(0, 6)}`) {
   184→    this.prefix = prefix;
   185→  }
   186→
   187→  async executeAgentTask(agentId: any, context: any, _prompt: string): Promise<any> {
   188→    this.calls += 1;
   189→    const runId = `${this.prefix}-${this.calls}`;
   190→    const timestamp = new Date().toISOString();
   191→    const agentName = "Sophia Vance";
   192→    // Mirrors the REAL executor's Phase 4.4C provenance stamping: the
   193→    // workflow execution context (instance/step/occurrence) supplied by the
   194→    // WorkflowRuntime is persisted on the run's provenance.
   195→    const wfProvenance = {
   196→      workflowInstanceId: context.workflowInstanceId,
   197→      stepId: context.stepId,
   198→      occurrenceId: context.occurrenceId,
   199→      occurrenceNumber: context.occurrenceNumber,
   200→    };
   201→    if (this.mode === "fail") {
   202→      await runStore.saveRun({
   203→        runId,
   204→        agentId,
   205→        agentName,
   206→        protocolStep: context.protocolStep,
   207→        taskTitle: context.taskTitle,
   208→        directive: context.directive,
   209→        status: "failed",
   210→        durationMs: 3,
   211→        outputContent: `**Execution Error**: Task "${context.taskTitle}" failed (provider unavailable).`,
   212→        provenance: { agentId, agentName, protocolStep: context.protocolStep, timestamp, isVerified: false, evidenceBasis: "unverified", ...wfProvenance },
   213→        error: "provider unavailable (44c double)",
   214→        timestamp,
   215→      });
   216→      createdRunIds.add(runId);
   217→      return {
   218→        success: false,
   219→        agentId,
   220→        agentName,
   221→        protocolStep: context.protocolStep,
   222→        statusMessage: "[COO] Task execution failed.",
   223→        outputContent: `**Execution Error**: provider unavailable.`,
   224→        runId,
   225→        provenance: { agentId, agentName, protocolStep: context.protocolStep, timestamp, isVerified: false, ...wfProvenance },
   226→        error: "provider unavailable (44c double)",
   227→      };
   228→    }
   229→    await runStore.saveRun({
   230→      runId,
   231→      agentId,
   232→      agentName,
   233→      protocolStep: context.protocolStep,
   234→      taskTitle: context.taskTitle,
   235→      directive: context.directive,
   236→      status: "completed",
   237→      durationMs: 12,
   238→      outputContent: `44c deterministic output for: ${context.directive}`,
   239→      provenance: { agentId, agentName, protocolStep: context.protocolStep, timestamp, isVerified: true, evidenceBasis: "model_reasoning", ...wfProvenance },
   240→      timestamp,
   241→    });
   242→    createdRunIds.add(runId);
   243→    return {
   244→      success: true,
   245→      agentId,
   246→      agentName,
   247→      protocolStep: context.protocolStep,
   248→      statusMessage: "[COO] Completed.",
   249→      outputContent: `44c deterministic output for: ${context.directive}`,
   250→      runId,
   251→      provenance: { agentId, agentName, protocolStep: context.protocolStep, timestamp, isVerified: true, ...wfProvenance },
   252→    };
   253→  }
   254→}
