     6→     6→/**
     7→     7→ * ============================================================================
     8→     8→ * SOPHIA PERSONAL MEMORY STORE (M3 K-2 — Personal Mind)
     9→     9→ * ============================================================================
    10→    10→ * Server-authoritative, founder-scoped personal/interaction memory store for
    11→    11→ * the Sophia Personal Mind (SOPHIA_MEMORY_ARCHITECTURE.md §1).
    12→    12→ *
    13→    13→ * TWO BRAINS, ONE GATE:
    14→    14→ *   This store is the Personal Mind's ONLY persistence boundary. It holds
    15→    15→ *   founder-specific interaction context (preferences, patterns, personal
    16→    16→ *   contextual notes). It is NOT the Company Brain:
    17→    17→ *     - it never stores company facts / strategy / financial truth
    18→    18→ *     - it never feeds CompanyKnowledge, CompanyMemory, or the epistemic
    19→    19→ *       claim/fact pipeline
    20→    20→ *     - it is NEVER an authorization source — personal memories cannot
    21→    21→ *       approve, bypass, or govern anything (the SideEffectAuthorizationGate
    22→    22→ *       and SophiaServerGateway read approval/policy state exclusively from
    23→    23→ *       their own governed stores)
    24→    24→ *     - there is NO promotion path out of this store; a personal memory
    25→    25→ *       stays personal/unverified context forever unless a Founder moves the
    26→    26→ *       underlying fact through the governed epistemic pipeline by hand.
    27→    27→ *
    28→    28→ * AUTHORITY SEMANTICS (four layers — keep these distinct in any claim):
    29→    29→ *
    30→    30→ *   [canonical abstraction]  This class IS the canonical Personal Mind
    31→    31→ *                            memory abstraction. Callers never read/write
    32→    32→ *                            `.data/sophia_memories.json` or the Prisma
    33→    33→ *                            SophiaMemory table directly.
    34→    34→ *
    35→    35→ *   [current local persistence — AUTHORITATIVE TODAY]
    36→    36→ *                            The DurableFileStore collection
    37→    37→ *                            `.data/sophia_memories.json` is the primary
    38→    38→ *                            write target and the authoritative read source
    39→    39→ *                            for every method (mirrors ConversationStore):
    40→    40→ *                              - listMemories()/findByIdempotencyKey() read
    41→    41→ *                                the file ONLY
    42→    42→ *                              - getMemory()/updateMemory()/deleteMemory()
    43→    43→ *                                resolve through the file ONLY
    44→    44→ *
    45→    45→ *   [current Prisma dual-write — opportunistic mirror, NOT authoritative]
    46→    46→ *                            createMemory()/updateMemory() upsert to the
    47→    47→ *                            Prisma SophiaMemory table best-effort AFTER
    48→    48→ *                            the durable file write; errors are swallowed
    49→    49→ *                            (never blocks the interaction).
    50→    50→ *                            deleteMemory() best-effort deletes the mirror
    51→    51→ *                            row. There is NO Prisma read fallback — Prisma
    52→    52→ *                            SophiaMemory rows are write-only shadows
    53→    53→ *                            (ADR 0002 §7 layering, applied to K-2).
    54→    54→ *
    55→    55→ *   [authoritative mode]     THIS STORE HAS NONE — deliberately, exactly
    56→    56→ *                            like ConversationStore. The same local-style
    57→    57→ *                            semantics (file primary, Prisma best-effort,
    58→    58→ *                            swallowed dual-write errors) apply in every
    59→    59→ *                            DATABASE_MODE, including production. The
    60→    60→ *                            M6 authoritative (PostgreSQL) migration moves
    61→    61→ *                            conversations and personal memories together;
    62→    62→ *                            until then Prisma must NOT be claimed as
    63→    63→ *                            authoritative for personal memory.
    64→    64→ *
    65→    65→ * INVARIANTS:
    66→    66→ * 1. Founder Ownership: every read/write/delete is bound to the caller's
    67→    67→ *    founderId. Cross-founder access FAILS CLOSED (403) — never a silent
    68→    68→ *    empty result when the record exists (existence is leaked only through
    69→    69→ *    the explicit 403; the victim's content is never returned).
    70→    70→ * 2. Process-Restart Durability: atomic DurableFileStore writes + best-effort
    71→    71→ *    Prisma dual-write.
    72→    72→ * 3. Deterministic Retrieval: listMemories is founder-scoped, filter-bounded,
    73→    73→ *    sorted (updatedAt DESC, id ASC tie-break), hard-capped — no vector
    74→    74→ *    search, no embeddings, no LLM-generated retrieval decisions.
    75→    75→ * 4. Idempotent Writes (optional): a founder-scoped idempotencyKey dedupes
    76→    76→ *    createMemory exactly like ChatMessage's idempotency convention
    77→    77→ *    (no key → no dedupe).
    78→    78→ * 5. No Fabrication: an empty store returns empty results — never seeded,
    79→    79→ *    never invented (bootstrap seeds are a CompanyKnowledge concept and do
    80→    80→ *    NOT apply to personal memory).
    81→    81→ */

   175→   175→/**
   176→   176→ * M4-A HARDENING: personal memory content that establishes or implies
   177→   177→ * AUTHORIZATION / PRIVILEGE / CONTROL / GOVERNANCE semantics is refused at
   178→   178→ * the STORE layer — the single persistence choke point. This applies to
   179→   179→ * BOTH ingress paths:
   180→   180→ *   - autonomous capture (the MemoryGate rejects it earlier with
   181→   181→ *     AUTHORITY_PRIVILEGE_CONTENT; the store check is defense-in-depth)
   182→   182→ *   - founder-direct authoring (/api/sofia/memory POST/PATCH): explicit
   183→   183→ *     Founder authoring is preserved for every legitimate personal memory,
   184→   184→ *     but authorization/privilege semantics are NOT personal preferences —
   185→   185→ *     they are policy, and policy belongs exclusively to the governed
   186→   186→ *     authorization / workflow system. A memory like "never ask me for
   187→   187→ *     confirmation" would render into every future prompt as founder data
   188→   188→ *     and launder an instruction through the personal-memory channel; the
   189→   189→ *     deterministic authority-content guard refuses it instead.

   206→   206→const SOPHIA_MEMORIES_COLLECTION = 'sophia_memories';
   207→   207→
   208→   208→export class SophiaMemoryStore {
   209→   209→  private static instance: SophiaMemoryStore;
   210→   210→  private fileStore: DurableFileStore;
   211→   211→
   212→   212→  private constructor() {
   213→   213→    this.fileStore = DurableFileStore.getInstance();
   214→   214→  }
   215→   215→
   216→   216→  public static getInstance(): SophiaMemoryStore {
   217→   217→    if (!SophiaMemoryStore.instance) {
   218→   218→      SophiaMemoryStore.instance = new SophiaMemoryStore();
   219→   219→    }
   220→   220→    return SophiaMemoryStore.instance;
   221→   221→  }
   222→   222→
   223→   223→  // --------------------------------------------------------------------------
   224→   224→  // Validation helpers (fail-closed; never silently coerce or truncate)
   225→   225→  // --------------------------------------------------------------------------
   226→   226→
   227→   227→  private requireFounderId(founderId: string): string {
   228→   228→    if (typeof founderId !== 'string' || !founderId.trim()) {
   229→   229→      throw new SophiaMemorySecurityError(
   230→   230→        'Authenticated founderId is required for every personal memory operation.'
   231→   231→      );
   232→   232→    }
   233→   233→    return founderId.trim();
   234→   234→  }
   235→   235→
   236→   236→  private validateMemoryType(memoryType: string): SophiaMemoryType {
   237→   237→    if (
   238→   238→      typeof memoryType !== 'string' ||
   239→   239→      !(SOPHIA_MEMORY_TYPES as readonly string[]).includes(memoryType)
   240→   240→    ) {
   241→   241→      throw new SophiaMemoryValidationError(
   242→   242→        `memoryType must be one of [${SOPHIA_MEMORY_TYPES.join(', ')}] (got: ${String(memoryType)}).`
   243→   243→      );
   244→   244→    }
   245→   245→    return memoryType as SophiaMemoryType;
   246→   246→  }
   247→   247→
   248→   248→  private validateContent(content: string): string {
   249→   249→    if (typeof content !== 'string' || !content.trim()) {
   250→   250→      throw new SophiaMemoryValidationError('content must be a non-empty string.');
   251→   251→    }
   252→   252→    if (content.length > SOPHIA_MEMORY_CONTENT_MAX_CHARS) {
   253→   253→      throw new SophiaMemoryValidationError(
   254→   254→        `content exceeds the ${SOPHIA_MEMORY_CONTENT_MAX_CHARS}-character bound (got ${content.length}).`
   255→   255→      );
   256→   256→    }
   257→   257→    return content.trim();
   258→   258→  }
   259→   259→
   260→   260→  private validateProvenance(provenance: string): string {
   261→   261→    if (typeof provenance !== 'string' || !provenance.trim()) {
   262→   262→      throw new SophiaMemoryValidationError('provenance must be a non-empty source reference.');
   263→   263→    }
   264→   264→    if (provenance.length > SOPHIA_MEMORY_PROVENANCE_MAX_CHARS) {
   265→   265→      throw new SophiaMemoryValidationError(
   266→   266→        `provenance exceeds the ${SOPHIA_MEMORY_PROVENANCE_MAX_CHARS}-character bound.`
   267→   267→      );
   268→   268→    }
   269→   269→    return provenance.trim();
   270→   270→  }
   271→   271→
   272→   272→  private validateConfidence(confidence: number): number {
   273→   273→    if (typeof confidence !== 'number' || !Number.isFinite(confidence) || confidence < 0 || confidence > 1) {
   274→   274→      throw new SophiaMemoryValidationError('confidence must be a finite number in [0, 1].');
   275→   275→    }
   276→   276→    return confidence;
   277→   277→  }
   278→   278→
   279→   279→  /**
   280→   280→   * M4-A HARDENING: deterministic fail-closed check on the AUTHORITY /
   281→   281→   * PRIVILEGE / CONTROL / GOVERNANCE semantic category (see
   282→   282→   * authority-content-guard.ts). Applied on create AND on content updates
   283→   283→   * so neither ingress path — capture or founder-direct authoring — can
   284→   284→   * persist authorization-bearing personal memory.
   285→   285→   */
   286→   286→  private validateNotAuthorityContent(content: string): string {
   287→   287→    const evaluation = evaluateAuthorityContent(content);
   288→   288→    if (evaluation.blocked) {
   289→   289→      throw new SophiaMemoryAuthorityError();
   290→   290→    }
   291→   291→    return content;
   292→   292→  }
   293→   293→
   294→   294→  /**
   295→   295→   * Resolves a memory from the durable file store (the ONLY read source today)
   296→   296→   * and enforces founder ownership STRICTLY. Existence + owner mismatch fails
   297→   297→   * closed (403); the victim's content is never returned to the caller.
   298→   298→   */
   299→   299→  private getOwnedMemory(founderId: string, memoryId: string): SophiaMemoryRecord {
   300→   300→    const record = this.fileStore.getItem<SophiaMemoryRecord>(SOPHIA_MEMORIES_COLLECTION, memoryId);
   301→   301→    if (!record) {
   302→   302→      throw new SophiaMemoryNotFoundError(
   303→   303→        `Personal memory "${memoryId}" not found for Founder "${founderId}".`
   304→   304→      );
   305→   305→    }
   306→   306→    if (record.founderId !== founderId) {
   307→   307→      throw new SophiaMemorySecurityError(
   308→   308→        `Principal "${founderId}" is not authorized to access personal memory "${memoryId}".`
   309→   309→      );
   310→   310→    }
   311→   311→    return record;
   312→   312→  }
   313→   313→
   314→   314→  // --------------------------------------------------------------------------
   315→   315→  // Writes
   316→   316→  // --------------------------------------------------------------------------
   317→   317→
   318→   318→  /**
   319→   319→   * Creates a personal memory bound to the authenticated founder.
   320→   320→   *
   321→   321→   * Idempotency (ChatMessage convention): when a non-empty idempotencyKey is
   322→   322→   * supplied and a memory with the SAME key already exists for the SAME
   323→   323→   * founder, the existing record is returned and nothing is written. Without
   324→   324→   * a key there is NO dedupe. The key is founder-scoped — Founder B may use
   325→   325→   * Founder A's key without collision.
   326→   326→   *
   327→   327→   * Lifecycle: created records are ACTIVE by default (founder-direct). The
   328→   328→   * M4-A capture path creates INACTIVE candidates (active: false) that stay
   329→   329→   * out of every context render until a Founder explicitly activates them.
   330→   330→   */
   331→   331→  public async createMemory(params: CreateSophiaMemoryParams): Promise<SophiaMemoryRecord> {
   332→   332→    const founderId = this.requireFounderId(params.founderId);
   333→   333→    const memoryType = this.validateMemoryType(params.memoryType);
   334→   334→    const content = this.validateNotAuthorityContent(this.validateContent(params.content));
   335→   335→    const provenance = this.validateProvenance(params.provenance || 'founder_direct');
   336→   336→    const confidence = this.validateConfidence(params.confidence ?? 0.8);
   337→   337→
   338→   338→    const cleanKey =
   339→   339→      typeof params.idempotencyKey === 'string' && params.idempotencyKey.trim()
   340→   340→        ? params.idempotencyKey.trim()
   341→   341→        : undefined;
   342→   342→
   343→   343→    const now = new Date().toISOString();
   344→   344→    const record: SophiaMemoryRecord = {
   345→   345→      id: `smem-${randomUUID()}`,
   346→   346→      founderId,
   347→   347→      memoryType,
   348→   348→      content,
   349→   349→      provenance,
   350→   350→      confidence,
   351→   351→      active: params.active ?? true,
   352→   352→      idempotencyKey: cleanKey,
   353→   353→      createdAt: now,
   354→   354→      updatedAt: now,
   355→   355→      metadata: params.metadata && typeof params.metadata === 'object' ? { ...params.metadata } : {},
   356→   356→    };
   357→   357→
   358→   358→    // 1. Atomic durable file write (authoritative today). The idempotency
   359→   359→    //    check and the save are ONE serialized cross-process read-check-write
   360→   360→    //    unit (M4-A hardening): two concurrent creators of the same key can
   361→   361→    //    no longer interleave the check and both persist, and concurrent
   362→   362→    //    writers can no longer silently erase each other's records. The lock
   363→   363→    //    is re-entrant, so the saveItem inside runs without re-acquiring.
   364→   364→    const written = this.fileStore.withCollectionLock(SOPHIA_MEMORIES_COLLECTION, () => {
   365→   365→      if (cleanKey) {
   366→   366→        const existing = this.findByIdempotencyKeySync(founderId, cleanKey);
   367→   367→        if (existing) {
   368→   368→          return existing;
   369→   369→        }
   370→   370→      }
   371→   371→      this.fileStore.saveItemStrict(SOPHIA_MEMORIES_COLLECTION, record.id, record);
   372→   372→      return record;
   373→   373→    });
   374→   374→
   375→   375→    // Idempotent replay: the existing record is returned and nothing else
   376→   376→    // (including the Prisma mirror) is touched.
   377→   377→    if (written !== record) {
   378→   378→      return written;
   379→   379→    }
   380→   380→
   381→   381→    // 2. Best-effort Prisma dual-write (opportunistic mirror, never authoritative)
   382→   382→    try {
   383→   383→      if ((await isDatabaseAvailable()) && (prisma as any).sophiaMemory) {
   384→   384→        await (prisma as any).sophiaMemory.upsert({
   385→   385→          where: { id: record.id },
   386→   386→          create: {
   387→   387→            id: record.id,
   388→   388→            founderId: record.founderId,
   389→   389→            memoryType: record.memoryType,
   390→   390→            content: record.content,
   391→   391→            provenance: record.provenance,
   392→   392→            confidence: record.confidence,
   393→   393→            active: record.active,
   394→   394→            idempotencyKey: record.idempotencyKey ?? null,
   395→   395→            metadata: record.metadata,
   396→   396→            createdAt: new Date(record.createdAt),
   397→   397→            updatedAt: new Date(record.updatedAt),
   398→   398→          },
   399→   399→          update: {
   400→   400→            content: record.content,
   401→   401→            confidence: record.confidence,
   402→   402→            active: record.active,
   403→   403→            metadata: record.metadata,
   404→   404→            updatedAt: new Date(record.updatedAt),
   405→   405→          },
   406→   406→        });
   407→   407→      }
   408→   408→    } catch {
   409→   409→      // Non-fatal if DB is offline (local durable mode)
   410→   410→    }
   411→   411→
   412→   412→    return record;
   413→   413→  }
   414→   414→
   415→   415→  /**
   416→   416→   * Updates a personal memory's mutable fields (content / confidence / active).
   417→   417→   * Ownership fails closed (403); unknown id → 404.
   418→   418→   */
   419→   419→  public async updateMemory(
   420→   420→    founderId: string,
   421→   421→    memoryId: string,
   422→   422→    patch: UpdateSophiaMemoryParams
   423→   423→  ): Promise<SophiaMemoryRecord> {
   424→   424→    const owner = this.requireFounderId(founderId);
   425→   425→    if (typeof memoryId !== 'string' || !memoryId.trim()) {
   426→   426→      throw new SophiaMemoryValidationError('memoryId is required.');
   427→   427→    }
   428→   428→    // Fail-closed validation BEFORE the lock (pure checks, no store state):
   429→   429→    // a content patch may not introduce authorization/privilege semantics
   430→   430→    // into an existing record (M4-A hardening — PATCH must not become the
   431→   431→    // laundering path around the create-time guard).
   432→   432→    if (patch.content !== undefined) {
   433→   433→      this.validateNotAuthorityContent(this.validateContent(patch.content));
   434→   434→    }
   435→   435→    if (patch.confidence !== undefined) {
   436→   436→      this.validateConfidence(patch.confidence);
   437→   437→    }
   438→   438→    if (patch.active !== undefined && typeof patch.active !== 'boolean') {
   439→   439→      throw new SophiaMemoryValidationError('active must be a boolean.');
   440→   440→    }
   441→   441→
   442→   442→    // Read-merge-write serialized on the collection's cross-process lock
   443→   443→    // (M4-A hardening): a concurrent update or create can no longer be
   444→   444→    // silently erased by this writer, and vice versa.
   445→   445→    const next = this.fileStore.withCollectionLock(SOPHIA_MEMORIES_COLLECTION, () => {
   446→   446→      const record = this.getOwnedMemory(owner, memoryId.trim());
   447→   447→
   448→   448→      const updated: SophiaMemoryRecord = { ...record };
   449→   449→
   450→   450→      if (patch.content !== undefined) {
   451→   451→        updated.content = this.validateContent(patch.content);
   452→   452→      }
   453→   453→      if (patch.confidence !== undefined) {
   454→   454→        updated.confidence = this.validateConfidence(patch.confidence);
   455→   455→      }
   456→   456→      if (patch.active !== undefined) {
   457→   457→        updated.active = patch.active;
   458→   458→        // M4-A founder-review confirmation stamp: activating a pending
   459→   459→        // captured candidate records WHEN the Founder confirmed it. This is
   460→   460→        // deterministic lifecycle metadata (server-side), not new authority —
   461→   461→        // the record only becomes visible in context because active is now
   462→   462→        // true, exactly like any founder-direct memory.
   463→   463→        if (patch.active === true && updated.metadata?.captureStatus === 'pending') {
   464→   464→          updated.metadata = {
   465→   465→            ...updated.metadata,
   466→   466→            captureStatus: 'confirmed',
   467→   467→            confirmedAt: new Date().toISOString(),
   468→   468→          };
   469→   469→        }
   470→   470→      }
   471→   471→      updated.updatedAt = new Date().toISOString();
   472→   472→
   473→   473→      // 1. Atomic durable file write (STRICT — M4-A hardening: a failed
   474→   474→      //    authoritative write THROWS instead of returning a phantom update)
   475→   475→      this.fileStore.saveItemStrict(SOPHIA_MEMORIES_COLLECTION, updated.id, updated);
   476→   476→      return updated;
   477→   477→    });
   478→   478→
   479→   479→    // 2. Best-effort Prisma dual-write
   480→   480→    try {
   481→   481→      if ((await isDatabaseAvailable()) && (prisma as any).sophiaMemory) {
   482→   482→        await (prisma as any).sophiaMemory.update({
   483→   483→          where: { id: next.id },
   484→   484→          data: {
   485→   485→            content: next.content,
   486→   486→            confidence: next.confidence,
   487→   487→            active: next.active,
   488→   488→            metadata: next.metadata as any,
   489→   489→            updatedAt: new Date(next.updatedAt),
   490→   490→          },
   491→   491→        });
   492→   492→      }
   493→   493→    } catch {
   494→   494→      // Non-fatal (mirror only)
   495→   495→    }
   496→   496→
   497→   497→    return next;
   498→   498→  }
   499→   499→
   500→   500→  /**
   501→   501→   * Deletes a personal memory. Ownership fails closed (403); unknown id →
   502→   502→   * returns false (idempotent delete for the OWNER — a second delete of a
   503→   503→   * memory the caller legitimately owned is a no-op, never an error).
   504→   504→   */
   505→   505→  public async deleteMemory(founderId: string, memoryId: string): Promise<boolean> {
   506→   506→    const owner = this.requireFounderId(founderId);
   507→   507→    if (typeof memoryId !== 'string' || !memoryId.trim()) {
   508→   508→      throw new SophiaMemoryValidationError('memoryId is required.');
   509→   509→    }
   510→   510→    const id = memoryId.trim();
   511→   511→
   512→   512→    // Read-check-delete serialized on the collection's cross-process lock
   513→   513→    // (M4-A hardening — same race class as create/update).
   514→   514→    const deleted = this.fileStore.withCollectionLock(SOPHIA_MEMORIES_COLLECTION, () => {
   515→   515→      const record = this.fileStore.getItem<SophiaMemoryRecord>(SOPHIA_MEMORIES_COLLECTION, id);
   516→   516→      if (!record) {
   517→   517→        return false;
   518→   518→      }
   519→   519→      if (record.founderId !== owner) {
   520→   520→        throw new SophiaMemorySecurityError(
   521→   521→          `Principal "${owner}" is not authorized to delete personal memory "${id}".`
   522→   522→        );
   523→   523→      }
   524→   524→      // 1. Authoritative durable file delete (STRICT — a failed delete-write
   525→   525→      //    throws instead of reporting a deletion that never reached disk)
   526→   526→      return this.fileStore.deleteItemStrict(SOPHIA_MEMORIES_COLLECTION, id);
   527→   527→    });
   528→   528→
   529→   529→    if (!deleted) {
   530→   530→      return false;
   531→   531→    }
   532→   532→
   533→   533→    // 2. Best-effort Prisma mirror delete
   534→   534→    try {
   535→   535→      if ((await isDatabaseAvailable()) && (prisma as any).sophiaMemory) {
   536→   536→        await (prisma as any).sophiaMemory.delete({ where: { id } }).catch(() => {});
   537→   537→      }
   538→   538→    } catch {
   539→   539→      // Non-fatal (mirror only)
   540→   540→    }
   541→   541→
   542→   542→    return true;
   543→   543→  }
   544→   544→
   545→   545→  // --------------------------------------------------------------------------
   546→   546→  // Reads (DurableFileStore ONLY — Prisma rows are write-only shadows)
   547→   547→  // --------------------------------------------------------------------------
   548→   548→
   549→   549→  /**
   550→   550→   * Retrieves one personal memory, strictly validating ownership.
   551→   551→   * Unknown id → null; owner mismatch → SophiaMemorySecurityError (403).
   552→   552→   */
   553→   553→  public async getMemory(founderId: string, memoryId: string): Promise<SophiaMemoryRecord | null> {
   554→   554→    const owner = this.requireFounderId(founderId);
   555→   555→    if (typeof memoryId !== 'string' || !memoryId.trim()) {
   556→   556→      return null;
   557→   557→    }
   558→   558→    const record = this.fileStore.getItem<SophiaMemoryRecord>(SOPHIA_MEMORIES_COLLECTION, memoryId.trim());
   559→   559→    if (!record) {
   560→   560→      return null;
   561→   561→    }
   562→   562→    if (record.founderId !== owner) {
   563→   563→      throw new SophiaMemorySecurityError(
   564→   564→        `Principal "${owner}" is not authorized to access personal memory "${memoryId.trim()}".`
   565→   565→      );
   566→   566→    }
   567→   567→    return { ...record };
   568→   568→  }
   569→   569→
   570→   570→  /**
   571→   571→   * Deterministic, founder-scoped, bounded retrieval.
   572→   572→   *
   573→   573→   * Ordering: updatedAt DESC, then id ASC (total, stable, explainable order).
   574→   574→   * Filters: memoryType (exact allow-list match), active flag. Limit defaults
   575→   575→   * to 20 and is hard-capped at 50. No scoring, no vectors, no LLM decisions.
   576→   576→   *
   577→   577→   * PAGINATION (P2 follow-up — queue >50 visibility): `offset` skips the
   578→   578→   * first N records of the SAME deterministic order (applied after sorting,
   579→   579→   * before the limit slice), so page k is `listMemories(f, { offset: k*50,
   580→   580→   * limit: 50 })`. Without it, the hard cap made the oldest pending review
   581→   581→   * candidates permanently invisible to both the API and the UI once the
   582→   582→   * founder-scoped set exceeded 50 records. Offset is clamped to
   583→   583→   * [0, SOPHIA_MEMORY_LIST_MAX_OFFSET] and a too-large offset yields an empty
   584→   584→   * page (a legitimate "past the end" result, not an error).
   585→   585→   */
   586→   586→  public async listMemories(
   587→   587→    founderId: string,
   588→   588→    opts?: { memoryType?: SophiaMemoryType; active?: boolean; limit?: number; offset?: number }
   589→   589→  ): Promise<SophiaMemoryRecord[]> {
   590→   590→    const owner = this.requireFounderId(founderId);
   591→   591→
   592→   592→    const items = this.sortedFounderMemories(owner, opts);
   593→   593→
   594→   594→    const requested = opts?.limit ?? SOPHIA_MEMORY_LIST_DEFAULT_LIMIT;
   595→   595→    const limit = Math.max(1, Math.min(SOPHIA_MEMORY_LIST_MAX_LIMIT, Math.floor(requested)));
   596→   596→    const offset = Math.max(0, Math.min(SOPHIA_MEMORY_LIST_MAX_OFFSET, Math.floor(opts?.offset ?? 0)));
   597→   597→    return items.slice(offset, offset + limit).map((m) => ({ ...m }));
   598→   598→  }
   599→   599→
   600→   600→  /**
   601→   601→   * TOTAL record count for the same founder-scoped filter listMemories uses
   602→   602→   * (P2 follow-up): callers paginating the review queue need the authoritative
   603→   603→   * total to compute hasMore / badge counts — the page length alone cannot
   604→   604→   * distinguish "exactly one page" from "one visible page of many".
   605→   605→   */
   606→   606→  public async countMemories(
   607→   607→    founderId: string,
   608→   608→    opts?: { memoryType?: SophiaMemoryType; active?: boolean }
   609→   609→  ): Promise<number> {
   610→   610→    const owner = this.requireFounderId(founderId);
   611→   611→    return this.sortedFounderMemories(owner, opts).length;
   612→   612→  }
   613→   613→
   614→   614→  /**
   615→   615→   * AUTHORITATIVE COLLECTION READ (P2 follow-up — dedupe must not read the
   616→   616→   * visible page): UNBOUNDED founder-scoped read used ONLY by internal
   617→   617→   * deterministic consumers that must compare against the whole collection:
   618→   618→   *   - the capture stage's duplicate detection (a duplicate of a record
   619→   619→   *     outside the newest-50 window must still be rejected — the phase-2
   620→   620→   *     observation proved the paginated read re-persisted such duplicates);
   621→   621→   *   - the review-queue annotation pool (duplicateOf/similarTo/contradicts
   622→   622→   *     must consider the founder's full set, active + pending).
   623→   623→   * This is the SAME full-collection read every store mutation already
   624→   624→   * performs internally; it is not a new search surface (no filters beyond
   625→   625→   * founder scope / active / type, same total order, no vectors). Route-level
   626→   626→   * pagination (limit/offset) governs what a CLIENT sees; dedupe correctness
   627→   627→   * is a store-internal concern and must never depend on page visibility.
   628→   628→   */
   629→   629→  public async listAllMemories(
   630→   630→    founderId: string,
   631→   631→    opts?: { memoryType?: SophiaMemoryType; active?: boolean }
   632→   632→  ): Promise<SophiaMemoryRecord[]> {
   633→   633→    const owner = this.requireFounderId(founderId);
   634→   634→    return this.sortedFounderMemories(owner, opts).map((m) => ({ ...m }));
   635→   635→  }
   636→   636→
   637→   637→  /**
   638→   638→   * Shared deterministic read: full founder-scoped collection, optional exact
   639→   639→   * type/active filters, total order (updatedAt DESC, id ASC).
   640→   640→   */
   641→   641→  private sortedFounderMemories(
   642→   642→    owner: string,
   643→   643→    opts?: { memoryType?: SophiaMemoryType; active?: boolean }
   644→   644→  ): SophiaMemoryRecord[] {
   645→   645→    const all = this.fileStore.readCollection<SophiaMemoryRecord>(SOPHIA_MEMORIES_COLLECTION);
   646→   646→    let items = Object.values(all).filter((m) => m && m.founderId === owner);
   647→   647→
   648→   648→    if (opts?.memoryType !== undefined) {
   649→   649→      const type = this.validateMemoryType(opts.memoryType);
   650→   650→      items = items.filter((m) => m.memoryType === type);
   651→   651→    }
   652→   652→    if (opts?.active !== undefined) {
   653→   653→      items = items.filter((m) => m.active === opts.active);
   654→   654→    }
   655→   655→
   656→   656→    items.sort((a, b) => {
   657→   657→      const ta = new Date(a.updatedAt).getTime();
   658→   658→      const tb = new Date(b.updatedAt).getTime();
   659→   659→      if (tb !== ta) return tb - ta;
   660→   660→      return a.id < b.id ? -1 : a.id > b.id ? 1 : 0;
   661→   661→    });
   662→   662→    return items;
   663→   663→  }

   668→   668→  public async findByIdempotencyKey(
   669→   669→    founderId: string,
   670→   670→    idempotencyKey: string
   671→   671→  ): Promise<SophiaMemoryRecord | null> {
   672→   672→    return this.findByIdempotencyKeySync(founderId, idempotencyKey);
   673→   673→  }
   674→   674→
   675→   675→  private findByIdempotencyKeySync(founderId: string, idempotencyKey: string): SophiaMemoryRecord | null {
   676→   676→    const owner = this.requireFounderId(founderId);
   677→   677→    if (typeof idempotencyKey !== 'string' || !idempotencyKey.trim()) {
   678→   678→      return null;
   679→   679→    }
   680→   680→    const key = idempotencyKey.trim();
   681→   681→    const all = this.fileStore.readCollection<SophiaMemoryRecord>(SOPHIA_MEMORIES_COLLECTION);
   682→   682→    for (const m of Object.values(all)) {
   683→   683→      if (m && m.founderId === owner && m.idempotencyKey === key) {
   684→   684→        return m;
   685→   685→      }
   686→   686→    }
   687→   687→    return null;
   688→   688→  }
   689→   689→
   690→   690→  /**
   691→   691→   * M4-A capture-replay guard: reports whether this founder already has any
   692→   692→   * memory whose idempotencyKey starts with the given prefix. The capture
   693→   693→   * stage derives deterministic keys ("m4cap:<conversationId>:<turnId>:<n>")
   694→   694→   * from authoritative turn identity; a REPLAYED turn must never capture
   695→   695→   * again — the prefix check short-circuits the whole capture stage before
   696→   696→   * any extraction or persistence happens. Unbounded by list limits (reads
   697→   697→   * the whole founder-scoped collection, which is the same read the store
   698→   698→   * already performs for dedupe).
   699→   699→   */
   700→   700→  public async hasIdempotencyKeyPrefix(founderId: string, prefix: string): Promise<boolean> {
   701→   701→    const owner = this.requireFounderId(founderId);
   702→   702→    if (typeof prefix !== 'string' || !prefix.trim()) {
   703→   703→      return false;
   704→   704→    }
   705→   705→    const all = this.fileStore.readCollection<SophiaMemoryRecord>(SOPHIA_MEMORIES_COLLECTION);
   706→   706→    return Object.values(all).some(
   707→   707→      (m) =>
   708→   708→        m &&
   709→   709→        m.founderId === owner &&
   710→   710→        typeof m.idempotencyKey === 'string' &&
   711→   711→        m.idempotencyKey.startsWith(prefix)
   712→   712→    );
   713→   713→  }
