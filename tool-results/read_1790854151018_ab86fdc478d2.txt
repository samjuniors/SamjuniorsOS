   151→  async saveInstance(instance: WorkflowInstanceState): Promise<void> {
   152→    const db = await requireAuthoritativeDatabase();
   153→
   154→    // PHASE 4.4B.1 — founder attribution write-guard: `initiatedById` mirrors
   155→    // the authoritative Prisma column (FK → User). The column is written ONLY
   156→    // when the attributed identity resolves to a REAL User row; otherwise it
   157→    // is omitted rather than fabricating an identity that would violate the
   158→    // foreign key. Attribution is set at creation and never overwritten by
   159→    // later status updates.
   160→    let initiatedById: string | null = null;
   161→    if (instance.initiatedById) {
   162→      try {
   163→        const user = await db.user.findUnique({
   164→          where: { id: instance.initiatedById },
   165→          select: { id: true },
   166→        });
   167→        initiatedById = user ? user.id : null;
   168→      } catch {
   169→        initiatedById = null;
   170→      }
   171→    }
   172→
   173→    await db.workflowInstance.upsert({
   174→      where: { id: instance.instanceId },
   175→      create: {
   176→        id: instance.instanceId,
   177→        definitionId: instance.workflowId,
   178→        version: instance.version || '1.0.0',
   179→        stateVersion: instance.stateVersion ?? 0,
   180→        claimedBy: instance.claimedBy || null,
   181→        claimedAt: instance.claimedAt ? new Date(instance.claimedAt) : null,
   182→        status: instance.status,
   183→        objective: instance.objective || '',
   184→        stepStates: (instance.stepStates as any) ?? {},
   185→        context: {},
   186→        error: instance.failureReason || null,
   187→        startedAt: instance.createdAt ? new Date(instance.createdAt) : null,
   188→        completedAt: instance.status === 'completed' && instance.updatedAt ? new Date(instance.updatedAt) : null,
   189→        initiatedById,
   190→      },
   191→      update: {
   192→        status: instance.status,
   193→        stateVersion: instance.stateVersion ?? 0,
   194→        claimedBy: instance.claimedBy || null,
   195→        claimedAt: instance.claimedAt ? new Date(instance.claimedAt) : null,
   196→        stepStates: (instance.stepStates as any) ?? {},
   197→        context: {},
   198→        error: instance.failureReason || null,
   199→        completedAt: instance.status === 'completed' && instance.updatedAt ? new Date(instance.updatedAt) : null,
   200→      },
   201→    });
   202→  }

   224→  async claimStepAtomic(
   225→    instanceId: string,
   226→    stepId: string,
   227→    workerId: string,
   228→    expectedVersion?: number
   229→  ): Promise<{ instance: WorkflowInstanceState; step: WorkflowStepState }> {
   230→    const db = await requireAuthoritativeDatabase();
   231→    return await db.$transaction(async (tx) => {
   232→      const dbInst = await tx.workflowInstance.findUnique({
   233→        where: { id: instanceId },
   234→      });
   235→      if (!dbInst) {
   236→        throw new Error(`Workflow instance not found: ${instanceId}`);
   237→      }
   238→
   239→      if (expectedVersion !== undefined && dbInst.stateVersion !== expectedVersion) {
   240→        throw new ConcurrencyConflictError(instanceId, expectedVersion, dbInst.stateVersion);
   241→      }
   242→
   243→      const stepStates: Record<string, WorkflowStepState> = (dbInst.stepStates as any) || {};
   244→      const step = stepStates[stepId];
   245→      if (!step) {
   246→        throw new Error(`Step not found in workflow instance: ${stepId}`);
   247→      }
   248→
   249→      if (step.status !== 'ready') {
   250→        throw new StepClaimError(instanceId, stepId, step.status, step.claimedBy);
   251→      }
   252→
   253→      if (step.claimedBy && step.claimedBy !== workerId) {
   254→        throw new StepClaimError(instanceId, stepId, step.status, step.claimedBy);
   255→      }
   256→
   257→      validateStepTransition(step.status, 'running');
   258→
   259→      const nowIso = new Date().toISOString();
   260→      step.status = 'running';
   261→      step.claimedBy = workerId;
   262→      step.claimedAt = nowIso;
   263→      if (!step.startedAt) {
   264→        step.startedAt = nowIso;
   265→      }
   266→      stepStates[stepId] = step;
   267→
   268→      let instanceStatus = dbInst.status;
   269→      if (instanceStatus === 'pending' || instanceStatus === 'waiting') {
   270→        validateInstanceTransition(instanceStatus as any, 'running');
   271→        instanceStatus = 'running';
   272→      }
   273→
   274→      const nextVersion = (dbInst.stateVersion || 0) + 1;
   275→
   276→      const updated = await tx.workflowInstance.update({
   277→        where: { id: instanceId },
   278→        data: {
   279→          status: instanceStatus,
   280→          stepStates: stepStates as any,
   281→          stateVersion: nextVersion,
   282→          claimedBy: workerId,
   283→          claimedAt: new Date(),
   284→          updatedAt: new Date(),
   285→        },
   286→      });
   287→
   288→      return {
   289→        instance: this.mapPrismaToInstance(updated),
   290→        step,
   291→      };
   292→    });
   293→  }

   295→  async transitionStepAtomic(
   296→    instanceId: string,
   297→    stepId: string,
   298→    targetStatus: WorkflowStepStatus,
   299→    expectedVersion?: number,
   300→    patch?: Partial<WorkflowStepState>,
   301→    instancePatch?: Partial<WorkflowInstanceState>
   302→  ): Promise<WorkflowInstanceState> {
   303→    const db = await requireAuthoritativeDatabase();
   304→    return await db.$transaction(async (tx) => {
   305→      const dbInst = await tx.workflowInstance.findUnique({
   306→        where: { id: instanceId },
   307→      });
   308→      if (!dbInst) {
   309→        throw new Error(`Workflow instance not found: ${instanceId}`);
   310→      }
   311→
   312→      if (expectedVersion !== undefined && dbInst.stateVersion !== expectedVersion) {
   313→        throw new ConcurrencyConflictError(instanceId, expectedVersion, dbInst.stateVersion);
   314→      }
   315→
   316→      const stepStates: Record<string, WorkflowStepState> = (dbInst.stepStates as any) || {};
   317→      const step = stepStates[stepId];
   318→      if (!step) {
   319→        throw new Error(`Step not found in workflow instance: ${stepId}`);
   320→      }
   321→
   322→      validateStepTransition(step.status, targetStatus);
   323→
   324→      const nowIso = new Date().toISOString();
   325→      step.status = targetStatus;
   326→      if (targetStatus === 'completed' || targetStatus === 'failed' || targetStatus === 'cancelled') {
   327→        step.completedAt = nowIso;
   328→        step.claimedBy = undefined;
   329→      }
   330→      if (patch) {
   331→        Object.assign(step, patch);
   332→      }
   333→      stepStates[stepId] = step;
   334→
   335→      let nextInstanceStatus = instancePatch?.status ?? dbInst.status;
   336→      if (instancePatch?.status && instancePatch.status !== dbInst.status) {
   337→        validateInstanceTransition(dbInst.status as any, instancePatch.status);
   338→      }
   339→
   340→      const nextVersion = (dbInst.stateVersion || 0) + 1;
   341→
   342→      const updated = await tx.workflowInstance.update({
   343→        where: { id: instanceId },
   344→        data: {
   345→          status: nextInstanceStatus,
   346→          stepStates: stepStates as any,
   347→          stateVersion: nextVersion,
   348→          error: instancePatch?.failureReason ?? (targetStatus === 'failed' ? (step.error || 'Step failed') : dbInst.error),
   349→          completedAt: (nextInstanceStatus === 'completed' || targetStatus === 'completed') ? new Date() : dbInst.completedAt,
   350→          updatedAt: new Date(),
   351→        },
   352→      });
   353→
   354→      return this.mapPrismaToInstance(updated);
   355→    });
   356→  }
   357→
   358→  async rehydrateActiveInstances(): Promise<WorkflowInstanceState[]> {
   359→    const db = await requireAuthoritativeDatabase();
   360→    const activeStatuses = ['pending', 'running', 'waiting', 'blocked', 'awaiting_approval'];
   361→    const dbInsts = await db.workflowInstance.findMany({
   362→      where: {
   363→        status: { in: activeStatuses },
   364→      },
   365→      orderBy: { updatedAt: 'asc' },
   366→    });
   367→    return dbInsts.map((inst) => this.mapPrismaToInstance(inst));
   368→  }
   369→
   370→  private mapPrismaToInstance(dbInst: any): WorkflowInstanceState {
   371→    return {
   372→      instanceId: dbInst.id,
   373→      workflowId: dbInst.definitionId,
   374→      version: dbInst.version,
   375→      stateVersion: dbInst.stateVersion ?? 0,
   376→      claimedBy: dbInst.claimedBy || undefined,
   377→      claimedAt: dbInst.claimedAt ? dbInst.claimedAt.toISOString() : undefined,
   378→      status: dbInst.status as any,
   379→      objective: dbInst.objective,
   380→      stepStates: (dbInst.stepStates as any) || {},
   381→      createdAt: dbInst.createdAt.toISOString(),
   382→      updatedAt: dbInst.updatedAt.toISOString(),
   383→      outputs: {},
   384→      evidenceReferences: [],
   385→      failureReason: dbInst.error || undefined,
   386→      // PHASE 4.4B.1 — authoritative founder attribution round-trip.
   387→      initiatedById: dbInst.initiatedById || undefined,
   388→    };
   389→  }
   390→
   391→  clear(): void {
   392→    if (process.env.NODE_ENV === 'production') {
   393→      throw new Error('Workflow records cannot be cleared in production.');
   394→    }
   395→  }

   420→  public loadFromDurableStorage(): void {
   421→    try {
   422→      const persistedDefs = DurableFileStore.getInstance().readCollection<WorkflowDefinition>('workflow_definitions');
   423→      for (const [key, def] of Object.entries(persistedDefs)) {
   424→        this.definitions.set(key, def);
   425→      }
   426→      const persistedInsts = DurableFileStore.getInstance().readCollection<WorkflowInstanceState>('workflow_instances');
   427→      for (const [id, inst] of Object.entries(persistedInsts)) {
   428→        this.instances.set(id, inst);
   429→      }
   430→    } catch {
   431→      // fallback
   432→    }
   433→  }

   486→    const key = version ? `${id}@${version}` : id;
   487→    const cached = this.definitions.get(key);
   488→    if (cached) return cached;
   489→
   490→    if (process.env.DATABASE_URL) {
   491→      try {
   492→        const dbDef = await prisma.workflowDefinition.findFirst({
   493→          where: {
   494→            id,
   495→            ...(version ? { version } : {}),
   496→          },
   497→          orderBy: { updatedAt: 'desc' },
   498→        });
   499→        if (dbDef) {
   500→          const mapped: WorkflowDefinition = {
   501→            id: dbDef.id,
   502→            version: dbDef.version,
   503→            name: dbDef.name,
   504→            description: dbDef.description || '',
   505→            objective: '',
   506→            steps: (dbDef.steps as any) || [],
   507→            dependencies: [],
   508→            requiredApprovals: 0,
   509→            allowedRoles: [],
   510→            allowedSkills: [],
   511→            expectedOutputs: [],
   512→          };
   513→          this.definitions.set(key, mapped);
   514→          return mapped;
   515→        }
   516→      } catch {
   517→        // Fallback
   518→      }
   519→    }
   520→
   521→    return null;
   522→  }

   526→      return PostgresWorkflowStore.getInstance().listDefinitions();

   617→            evidenceReferences: [],
   618→            failureReason: dbInst.error || undefined,
   619→          };
   620→          this.instances.set(mapped.instanceId, mapped);
   621→          return mapped;
   622→        }
   623→      } catch {
   624→        // Fallback
   625→      }
   626→    }
   627→
   628→    return null;
   629→  }

   631→  async listInstances(status?: string): Promise<WorkflowInstanceState[]> {

   643→  async claimStepAtomic(
   644→    instanceId: string,
   645→    stepId: string,
   646→    workerId: string,
   647→    expectedVersion?: number
   648→  ): Promise<{ instance: WorkflowInstanceState; step: WorkflowStepState }> {
   649→    if (isAuthoritativeMode()) {
   650→      return PostgresWorkflowStore.getInstance().claimStepAtomic(instanceId, stepId, workerId, expectedVersion);
   651→    }
   652→
   653→    const instance = this.instances.get(instanceId);
   654→    if (!instance) {
   655→      throw new Error(`Workflow instance not found: ${instanceId}`);
   656→    }
   657→
   658→    const currentVersion = instance.stateVersion ?? 0;
   659→    if (expectedVersion !== undefined && currentVersion !== expectedVersion) {
   660→      throw new ConcurrencyConflictError(instanceId, expectedVersion, currentVersion);
   661→    }
   662→
   663→    const step = instance.stepStates[stepId];
   664→    if (!step) {
   665→      throw new Error(`Step not found in workflow instance: ${stepId}`);
   666→    }
   667→
   668→    if (step.status !== 'ready') {
   669→      throw new StepClaimError(instanceId, stepId, step.status, step.claimedBy);
   670→    }
   671→
   672→    if (step.claimedBy && step.claimedBy !== workerId) {
   673→      throw new StepClaimError(instanceId, stepId, step.status, step.claimedBy);
   674→    }
   675→
   676→    validateStepTransition(step.status, 'running');
   677→
   678→    const nowIso = new Date().toISOString();
   679→    step.status = 'running';
   680→    step.claimedBy = workerId;
   681→    step.claimedAt = nowIso;
   682→    if (!step.startedAt) {
   683→      step.startedAt = nowIso;
   684→    }
   685→
   686→    if (instance.status === 'pending' || instance.status === 'waiting') {
   687→      validateInstanceTransition(instance.status, 'running');
   688→      instance.status = 'running';
   689→    }
   690→
   691→    instance.stateVersion = currentVersion + 1;
   692→    instance.claimedBy = workerId;
   693→    instance.claimedAt = nowIso;
   694→    instance.updatedAt = nowIso;
   695→
   696→    const clone = JSON.parse(JSON.stringify(instance));
   697→    this.instances.set(instanceId, clone);
   698→
   699→    try {
   700→      DurableFileStore.getInstance().saveItem('workflow_instances', instanceId, clone);
   701→    } catch {}
   702→
   703→    return {
   704→      instance: clone,
   705→      step: JSON.parse(JSON.stringify(step)),
   706→    };
   707→  }

   709→  async transitionStepAtomic(
   710→    instanceId: string,
   711→    stepId: string,
   712→    targetStatus: WorkflowStepStatus,
   713→    expectedVersion?: number,
   714→    patch?: Partial<WorkflowStepState>,
   715→    instancePatch?: Partial<WorkflowInstanceState>
   716→  ): Promise<WorkflowInstanceState> {
   717→    if (isAuthoritativeMode()) {
   718→      return PostgresWorkflowStore.getInstance().transitionStepAtomic(
   719→        instanceId,
   720→        stepId,
   721→        targetStatus,
   722→        expectedVersion,
   723→        patch,
   724→        instancePatch
   725→      );
   726→    }
   727→
   728→    const instance = this.instances.get(instanceId);
   729→    if (!instance) {
   730→      throw new Error(`Workflow instance not found: ${instanceId}`);
   731→    }
   732→
   733→    const currentVersion = instance.stateVersion ?? 0;
   734→    if (expectedVersion !== undefined && currentVersion !== expectedVersion) {
   735→      throw new ConcurrencyConflictError(instanceId, expectedVersion, currentVersion);
   736→    }
   737→
   738→    const step = instance.stepStates[stepId];
   739→    if (!step) {
   740→      throw new Error(`Step not found in workflow instance: ${stepId}`);
   741→    }
   742→
   743→    validateStepTransition(step.status, targetStatus);
   744→
   745→    const nowIso = new Date().toISOString();
   746→    step.status = targetStatus;
   747→    if (targetStatus === 'completed' || targetStatus === 'failed' || targetStatus === 'cancelled') {
   748→      step.completedAt = nowIso;
   749→      step.claimedBy = undefined;
   750→    }
   751→    if (patch) {
   752→      Object.assign(step, patch);
   753→    }
   754→
   755→    if (instancePatch?.status && instancePatch.status !== instance.status) {
   756→      validateInstanceTransition(instance.status, instancePatch.status);
   757→      instance.status = instancePatch.status;
   758→    }
   759→    if (instancePatch?.failureReason) {
   760→      instance.failureReason = instancePatch.failureReason;
   761→    }
   762→
   763→    instance.stateVersion = currentVersion + 1;
   764→    instance.updatedAt = nowIso;
   765→
   766→    const clone = JSON.parse(JSON.stringify(instance));
   767→    this.instances.set(instanceId, clone);
   768→
   769→    try {
   770→      DurableFileStore.getInstance().saveItem('workflow_instances', instanceId, clone);
   771→    } catch {}
   772→
   773→    return clone;
   774→  }

   776→  async rehydrateActiveInstances(): Promise<WorkflowInstanceState[]> {
   777→    if (isAuthoritativeMode()) {
   778→      return PostgresWorkflowStore.getInstance().rehydrateActiveInstances();
   779→    }
   780→
   781→    const activeStatuses: Set<string> = new Set(['pending', 'running', 'waiting', 'blocked', 'awaiting_approval']);
   782→    const active = Array.from(this.instances.values())
   783→      .filter((i) => activeStatuses.has(i.status))
   784→      .sort((a, b) => new Date(a.updatedAt).getTime() - new Date(b.updatedAt).getTime());
   785→
   786→    return JSON.parse(JSON.stringify(active));
   787→  }
