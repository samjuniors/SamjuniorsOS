   124→   124→function layoutColumn(
   125→   125→  nodes: GraphNodeDTO[],
   126→   126→  centerY: number,
   127→   127→  gap: number
   128→   128→): void {
   129→   129→  if (nodes.length === 0) return;
   130→   130→  const totalH =
   131→   131→    nodes.reduce((acc, n) => acc + n.geometry.height, 0) + (nodes.length - 1) * gap;
   132→   132→  let curY = centerY - totalH / 2;
   133→   133→
   134→   134→  for (const n of nodes) {
   135→   135→    n.geometry.y = Math.round(curY + n.geometry.height / 2);
   136→   136→    curY += n.geometry.height + gap;
   137→   137→  }
   138→   138→}

   140→   140→/**
   141→   141→ * Deterministic identity hash for derived graph entities (workstreams).
   142→   142→ * Exported since Phase 4.4C: the authoritative Activity projection reuses the
   143→   143→ * EXACT same derivation so Activity provenance traces to the same workstream
   144→   144→ * identities the canvas read-model projects — no parallel identity scheme.
   145→   145→ */
   146→   146→export function deterministicIdHash(prefix: string, text: string): string {
   147→   147→  let h = 5381;
   148→   148→  for (let i = 0; i < text.length; i++) {
   149→   149→    h = ((h << 5) + h + text.charCodeAt(i)) | 0;
   150→   150→  }
   151→   151→  return `${prefix}-${(h >>> 0).toString(36)}`;
   152→   152→}

   154→   154→function truncateTitle(text: string, max = 50): string {
   155→   155→  if (text.length <= max) return text;
   156→   156→  return `${text.slice(0, max - 1).trimEnd()}…`;
   157→   157→}

   185→   185→function protocolStepLabel(step: string): string {
   186→   186→  const canonical = CANONICAL_PROTOCOL_STEPS.find((c) => c.step === step);
   187→   187→  if (canonical) return canonical.label;
   188→   188→  return step.replace(/_/g, ' ').replace(/^\w/, (c) => c.toUpperCase());
   189→   189→}

   191→   191→function runStepStatus(runs: AgentRunRecord[]): GraphExecutionStepDTO['status'] {
   192→   192→  if (runs.length === 0) return 'pending';
   193→   193→  if (runs.some((r) => r.status === 'failed' || !!r.error)) return 'failed';
   194→   194→  if (runs.some((r) => r.status === 'running')) return 'current';
   195→   195→  if (runs.some((r) => r.status === 'completed')) return 'done';
   196→   196→  return 'pending';
   197→   197→}
   198→   198→
   199→   199→/**
   200→   200→ * Derives the deterministic execution trail of a workstream from its
   201→   201→ * chronological run group. Unknown protocol steps are appended after the
   202→   202→ * canonical sequence in first-occurrence order (never dropped, never invented).
   203→   203→ */
   204→   204→function deriveExecutionStepsFromRuns(sortedRuns: AgentRunRecord[]): GraphExecutionStepDTO[] {
   205→   205→  const byStep = new Map<string, AgentRunRecord[]>();
   206→   206→  for (const r of sortedRuns) {
   207→   207→    const key = r.protocolStep;
   208→   208→    if (!key) continue;
   209→   209→    const list = byStep.get(key) ?? [];
   210→   210→    list.push(r);
   211→   211→    byStep.set(key, list);
   212→   212→  }
   213→   213→
   214→   214→  const unknownSteps = [...byStep.keys()].filter(
   215→   215→    (k) => !CANONICAL_PROTOCOL_STEPS.some((c) => c.step === k)
   216→   216→  );
   217→   217→
   218→   218→  const steps: GraphExecutionStepDTO[] = CANONICAL_PROTOCOL_STEPS.map(({ step, label }) => {
   219→   219→    const runs = byStep.get(step) ?? [];
   220→   220→    const latest = runs[runs.length - 1];
   221→   221→    return {
   222→   222→      step,
   223→   223→      label,
   224→   224→      status: runStepStatus(runs),
   225→   225→      ownerAgentId: latest?.agentId,
   226→   226→      durationMs: latest?.durationMs,
   227→   227→    };
   228→   228→  });
   229→   229→
   230→   230→  for (const step of unknownSteps) {
   231→   231→    const runs = byStep.get(step) ?? [];
   232→   232→    const latest = runs[runs.length - 1];
   233→   233→    steps.push({
   234→   234→      step,
   235→   235→      label: protocolStepLabel(step),
   236→   236→      status: runStepStatus(runs),
   237→   237→      ownerAgentId: latest?.agentId,
   238→   238→      durationMs: latest?.durationMs,
   239→   239→    });
   240→   240→  }
   241→   241→
   242→   242→  return steps;
   243→   243→}

   245→   245→/** Maps durable WorkflowStepState status onto the execution trail status. */
   246→   246→function workflowStepStatus(status: string): GraphExecutionStepDTO['status'] {
   247→   247→  switch (status) {
   248→   248→    case 'running':
   249→   249→    case 'ready':
   250→   250→      return 'current';
   251→   251→    case 'waiting':
   252→   252→    case 'blocked':
   253→   253→    case 'awaiting_approval':
   254→   254→      return 'waiting';
   255→   255→    case 'completed':
   256→   256→      return 'done';
   257→   257→    case 'failed':
   258→   258→    case 'cancelled':
   259→   259→      return 'failed';
   260→   260→    default:
   261→   261→      return 'pending';
   262→   262→  }
   263→   263→}

   265→   265→/**
   266→   266→ * Derives the execution trail of a workstream backed by a durable
   267→   267→ * WorkflowInstance from its authoritative stepStates.
   268→   268→ */
   269→   269→function deriveExecutionStepsFromInstance(wf: WorkflowInstanceState): GraphExecutionStepDTO[] {
   270→   270→  const skillOrder = (skill: string) => {
   271→   271→    const idx = CANONICAL_PROTOCOL_STEPS.findIndex((c) => c.step === skill);
   272→   272→    return idx === -1 ? CANONICAL_PROTOCOL_STEPS.length : idx;
   273→   273→  };
   274→   274→  return Object.values(wf.stepStates ?? {})
   275→   275→    .sort((a, b) => {
   276→   276→      const sa = skillOrder(a.skill);
   277→   277→      const sb = skillOrder(b.skill);
   278→   278→      if (sa !== sb) return sa - sb;
   279→   279→      return a.stepId.localeCompare(b.stepId);
   280→   280→    })
   281→   281→    .map((s) => ({
   282→   282→      step: s.stepId,
   283→   283→      label: protocolStepLabel(s.skill),
   284→   284→      status: workflowStepStatus(s.status),
   285→   285→      ownerAgentId: s.assignedRole,
   286→   286→    }));
   287→   287→}

   293→   293→export function deriveGraphProjection(inputs: AuthoritativeGraphInputs): GraphDTO {
   294→   294→  const {
   295→   295→    agentRuns,
   296→   296→    approvals,
   297→   297→    workflows,
   298→   298→    claimsPendingCount,
   299→   299→    factsActiveCount,
   300→   300→    company,
   301→   301→  } = inputs;
   302→   302→
   303→   303→  const nodes: GraphNodeDTO[] = [];
   304→   304→  const edges: GraphEdgeDTO[] = [];
   305→   305→  const spatialCards: GraphSpatialCardDTO[] = [];
   306→   306→
   307→   307→  // Filter pending approvals (consequential escalation boundary)
   308→   308→  const pendingApprovals = approvals.filter((a) => a.decision === 'pending');
   309→   309→  const hasOpenApprovals = pendingApprovals.length > 0;
   310→   310→
   311→   311→  // Group durable agent runs by directive (canonical workstreams)
   312→   312→  const directiveGroups = new Map<string, AgentRunRecord[]>();
   313→   313→  for (const r of agentRuns) {
   314→   314→    const key = r.directive?.trim();
   315→   315→    if (!key) continue;
   316→   316→    const list = directiveGroups.get(key) ?? [];
   317→   317→    list.push(r);
   318→   318→    directiveGroups.set(key, list);
   319→   319→  }
   320→   320→
   321→   321→  // Derive workstreams from groups
   322→   322→  interface DerivedWorkstream {
   323→   323→    id: string;
   324→   324→    directive: string;
   325→   325→    title: string;
   326→   326→    latestRun: AgentRunRecord;
   327→   327→    owner: string;
   328→   328→    stepStage: 'discovery' | 'build' | 'review' | 'ship' | 'done';
   329→   329→    runtimeState: GraphRuntimeState;
   330→   330→    hasFailed: boolean;
   331→   331→    hasReport: boolean;
   332→   332→    latestRunTime: number;
   333→   333→    executionSteps: GraphExecutionStepDTO[];
   334→   334→  }
   335→   335→
   336→   336→  const workstreams: DerivedWorkstream[] = [];
   337→   337→  for (const [directive, group] of directiveGroups) {
   338→   338→    const sorted = [...group].sort((a, b) => {
   339→   339→      const ta = (a as any).timestamp || (a as any).createdAt ? new Date((a as any).timestamp || (a as any).createdAt).getTime() : 0;
   340→   340→      const tb = (b as any).timestamp || (b as any).createdAt ? new Date((b as any).timestamp || (b as any).createdAt).getTime() : 0;
   341→   341→      return ta - tb;
   342→   342→    });
   343→   343→    const latest = sorted[sorted.length - 1];
   344→   344→    const hasReport = sorted.some((r) => r.protocolStep === 'report');
   345→   345→    const hasFailed = sorted.some((r) => r.status === 'failed' || !!r.error);
   346→   346→    const hasRunning = sorted.some((r) => r.status === 'running');
   347→   347→
   348→   348→    const specialistRuns = sorted.filter((r) => r.agentId !== 'coo');
   349→   349→    const latestSpecialist = specialistRuns[specialistRuns.length - 1];
   350→   350→    const owner = latestSpecialist ? latestSpecialist.agentId : 'researcher';
   351→   351→
   352→   352→    let stepStage: DerivedWorkstream['stepStage'] = 'discovery';
   353→   353→    if (hasReport) {
   354→   354→      stepStage = 'done';
   355→   355→    } else if (latest.protocolStep === 'review' || latest.protocolStep === 'verify') {
   356→   356→      stepStage = 'review';
   357→   357→    } else if (latest.protocolStep === 'plan' || latest.protocolStep === 'build_execute' || latest.protocolStep === 'test') {
   358→   358→      stepStage = 'build';
   359→   359→    } else if (latest.protocolStep === 'report') {
   360→   360→      stepStage = 'ship';
   361→   361→    }
   362→   362→
   363→   363→    const runtimeState: GraphRuntimeState = hasFailed
   364→   364→      ? 'failed'
   365→   365→      : hasReport
   366→   366→      ? 'completed'
   367→   367→      : hasRunning
   368→   368→      ? 'running'
   369→   369→      : 'paused';
   370→   370→
   371→   371→    workstreams.push({
   372→   372→      id: deterministicIdHash('ws', directive),
   373→   373→      directive,
   374→   374→      title: truncateTitle(directive),
   375→   375→      latestRun: latest,
   376→   376→      owner,
   377→   377→      stepStage,
   378→   378→      runtimeState,
   379→   379→      hasFailed,
   380→   380→      hasReport,
   381→   381→      latestRunTime: (() => {
   382→   382→        const t = (latest as any).timestamp || (latest as any).createdAt;
   383→   383→        return t ? new Date(t).getTime() : 0;
   384→   384→      })(),
   385→   385→      executionSteps: deriveExecutionStepsFromRuns(sorted),
   386→   386→    });
   387→   387→  }
   388→   388→
   389→   389→  // Cross-reference durable WorkflowInstances
   390→   390→  for (const wf of workflows) {
   391→   391→    if (workstreams.some((w) => w.directive === wf.objective)) continue;
   392→   392→    const isDone = wf.status === 'completed';
   393→   393→    const isFailed = wf.status === 'failed';
   394→   394→    const isRunning = wf.status === 'running';
   395→   395→    workstreams.push({
   396→   396→      id: deterministicIdHash('wf', wf.instanceId),
   397→   397→      directive: wf.objective,
   398→   398→      title: truncateTitle(wf.objective),
   399→   399→      latestRun: {
   400→   400→        runId: wf.instanceId,
   401→   401→        agentId: 'coo',
   402→   402→        agentName: 'Sophia Vance',
   403→   403→        protocolStep: 'plan',
   404→   404→        taskTitle: wf.objective,
   405→   405→        directive: wf.objective,
   406→   406→        status: wf.status as any,
   407→   407→        durationMs: 0,
   408→   408→        outputContent: '',
   409→   409→        timestamp: wf.createdAt || new Date().toISOString(),
   410→   410→        provenance: {
   411→   411→          agentId: 'coo',
   412→   412→          agentName: 'Sophia Vance',
   413→   413→          taskId: wf.instanceId,
   414→   414→          protocolStep: 'plan',
   415→   415→          timestamp: wf.createdAt || new Date().toISOString(),
   416→   416→          isVerified: true,
   417→   417→          evidenceBasis: 'calculation',
   418→   418→        },
   419→   419→      },
   420→   420→      owner: 'coo',
   421→   421→      stepStage: isDone ? 'done' : 'build',
   422→   422→      runtimeState: isFailed ? 'failed' : isDone ? 'completed' : isRunning ? 'running' : 'idle',
   423→   423→      hasFailed: isFailed,
   424→   424→      hasReport: isDone,
   425→   425→      latestRunTime: wf.createdAt ? new Date(wf.createdAt).getTime() : 0,
   426→   426→      executionSteps: deriveExecutionStepsFromInstance(wf),
   427→   427→    });
   428→   428→  }
   429→   429→
   430→   430→  const activeWorkstreams = workstreams.filter((w) => w.runtimeState === 'running');
   431→   431→  const failedWorkstreams = workstreams.filter((w) => w.runtimeState === 'failed');
   432→   432→  const completedWorkstreams = workstreams.filter((w) => w.runtimeState === 'completed');
   433→   433→
   434→   434→  const hasActiveWork = activeWorkstreams.length > 0;
   435→   435→  const hasFailedWork = failedWorkstreams.length > 0;
   436→   436→  const hasCompletedWork = completedWorkstreams.length > 0;
   437→   437→  const hasReviewWork = workstreams.some(
   438→   438→    (w) => (w.stepStage === 'review' || w.stepStage === 'ship') && w.runtimeState === 'running'
   439→   439→  );
   440→   440→
   441→   441→  // -------------------------------------------------------------------------
   442→   442→  // Column 0: Founder / Inputs (Human Authority Boundary)
   443→   443→  // -------------------------------------------------------------------------
   444→   444→  const founderGovernance: GraphGovernanceState = hasOpenApprovals
   445→   445→    ? 'awaiting_founder_approval'
   446→   446→    : 'none';
   447→   447→  const founderNode: GraphNodeDTO = {
   448→   448→    id: 'founder',
   449→   449→    type: 'founder',
   450→   450→    role: 'founder',
   451→   451→    title: 'Founder / Inputs',
   452→   452→    subtitle: company.focus ? `Focus: ${company.focus}` : 'Directives & Authority',
   453→   453→    activity: hasOpenApprovals
   454→   454→      ? `${pendingApprovals.length} Approval${pendingApprovals.length > 1 ? 's' : ''} Awaiting Review`
   455→   455→      : undefined,
   456→   456→    runtimeState: 'idle',
   457→   457→    governanceState: founderGovernance,
   458→   458→    epistemicValidity: 'not_applicable',
   459→   459→    presentationState: mapSemanticToPresentationState('idle', founderGovernance, 'not_applicable'),
   460→   460→    geometry: { shape: 'rectangle', width: 124, height: 80, x: 180, y: 440 },
   461→   461→    relevance: hasOpenApprovals ? 1.0 : 0.6,
   462→   462→    metadata: { actionCount: pendingApprovals.length },
   463→   463→  };
   464→   464→  nodes.push(founderNode);
   465→   465→
   466→   466→  // -------------------------------------------------------------------------
   467→   467→  // Column 1: Sophia Vance (COO & Master Orchestrator)
   468→   468→  // -------------------------------------------------------------------------
   469→   469→  const sophiaRuntime: GraphRuntimeState = hasFailedWork
   470→   470→    ? 'failed'
   471→   471→    : hasActiveWork
