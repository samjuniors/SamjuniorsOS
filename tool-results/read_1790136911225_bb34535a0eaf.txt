    91→    91→function runChild(args: string[], env: Record<string, string> = {}): Promise<any> {
    92→    92→  return new Promise((resolve, reject) => {
    93→    93→    // @ts-ignore — Bun global exists when the suite runs under bun
    94→    94→    const proc = Bun.spawn(['bun', 'tests/sophia/k2-memory-child.ts', ...args], {
    95→    95→      stdout: 'pipe',
    96→    96→      stderr: 'pipe',
    97→    97→      env: { ...process.env, ...env },
    98→    98→    });
    99→    99→    new Response(proc.stdout)
   100→   100→      .text()
   101→   101→      .then((out) =>
   102→   102→        proc.exited.then((code) => {
   103→   103→          if (code !== 0) {
   104→   104→            return new Response(proc.stderr)
   105→   105→              .text()
   106→   106→              .then((errText) =>
   107→   107→                reject(new Error(`child (${args.join(' ')}) exited ${code}; stderr: ${errText.slice(0, 500)}`))
   108→   108→              );
   109→   109→          }
   110→   110→          const lines = out.trim().split('\n').filter((l: string) => l.trim());
   111→   111→          const last = lines[lines.length - 1];
   112→   112→          try {
   113→   113→            resolve(JSON.parse(last));
   114→   114→          } catch {
   115→   115→            reject(new Error(`child (${args.join(' ')}) emitted no JSON line; stdout: ${out.slice(0, 500)}`));
   116→   116→          }
   117→   117→        })
   118→   118→      );
   119→   119→  });
   120→   120→}
