   202→/**
   203→ * Computes the SHA-256 content hash used by the `CompanyKnowledge.hash`
   204→ * column for change detection (M2: the ingestion path respects this column).
   205→ */
   206→export function computeKnowledgeContentHash(content: string): string {
   207→  return createHash('sha256').update(content || '', 'utf-8').digest('hex');
   208→}

   210→/**
   211→ * ============================================================================
   212→ * COMPANY KNOWLEDGE STORE — M2: CANONICAL PERSISTENCE ACTIVATED
   213→ * ============================================================================
   214→ *
   215→ * Implements ICompanyKnowledgeStore with the SAME dual-mode persistence
   216→ * pattern as its sibling stores (memory-store, conversation store, etc.):
   217→ *
   218→ *  - LOCAL MODE: in-memory cache (constructor-hydrated from DurableFileStore)
   219→ *    + .data/company_knowledge.json durability + best-effort Prisma dual-write.
   220→ *  - AUTHORITATIVE MODE: Prisma fail-closed (requireAuthoritativeDatabase);
   221→ *    reads and writes go to the CompanyKnowledge table (target: PostgreSQL;
   222→ *    currently the SQLite sandbox port — see the M0 naming note).
   223→ *
   224→ * Seed migration (one-time, idempotent): the 8 canonical documents that
   225→ * previously existed ONLY as code constants are persisted to DurableFileStore
   226→ * on first boot (empty durable collection), and mirrored into the Prisma
   227→ * table lazily on first store usage (upsert keyed by item id, hash-respecting).
   228→ * The `hash` column (SHA-256 of content) provides change detection: an
   229→ * addKnowledge with the same id but different content updates the row under a
   230→ * new hash; a different id with byte-identical content collides on the unique
   231→ * hash index and is reported as a duplicate in authoritative mode (silently
   232→ * skipped in best-effort local dual-writes).
   233→ *
   234→ * Designed so the target PostgreSQL / Supabase deployment can replace the
   235→ * persistence internals without altering retrieval or business logic.
   236→ */
   237→export class CompanyKnowledgeStore implements ICompanyKnowledgeStore {
   238→  private static instance: CompanyKnowledgeStore | null = null;
   239→
   240→  private knowledgeItems: CompanyKnowledgeItem[] = [...CANONICAL_COMPANY_KNOWLEDGE];
   241→
   242→  /** One-shot guard for the lazy Prisma seed mirror (M2 seed migration). */
   243→  private static prismaSeedPromise: Promise<void> | null = null;
   244→
   245→  private constructor() {
   246→    this.loadFromDurableStorage();
   247→  }
   248→
   249→  public static getInstance(): CompanyKnowledgeStore {
   250→    if (!CompanyKnowledgeStore.instance) {
   251→      CompanyKnowledgeStore.instance = new CompanyKnowledgeStore();
   252→    }
   253→    return CompanyKnowledgeStore.instance;
   254→  }
   255→
   256→  /**
   257→   * Hydrates the in-memory cache from DurableFileStore, merging over the
   258→   * canonical seed documents. On first boot (no durable collection), the
   259→   * canonical seeds are themselves persisted — the M2 one-time seed
   260→   * migration of the 8 canonical documents.
   261→   */
   262→  private loadFromDurableStorage(): void {
   263→    try {
   264→      const persisted = DurableFileStore.getInstance().readCollection<CompanyKnowledgeItem>('company_knowledge');
   265→      const persistedItems = Object.values(persisted);
   266→      if (persistedItems.length === 0) {
   267→        // First boot: persist the canonical seed set (one-time migration).
   268→        const seedRecord: Record<string, CompanyKnowledgeItem> = {};
   269→        for (const item of CANONICAL_COMPANY_KNOWLEDGE) {
   270→          seedRecord[item.id] = item;
   271→        }
   272→        DurableFileStore.getInstance().writeCollection('company_knowledge', seedRecord);
   273→        this.knowledgeItems = [...CANONICAL_COMPANY_KNOWLEDGE];
   274→        return;
   275→      }
   276→      // Merge: canonical seeds first (stable order), persisted items overlay
   277→      // by id (edits win), extras appended.
   278→      const byId = new Map<string, CompanyKnowledgeItem>();
   279→      for (const item of CANONICAL_COMPANY_KNOWLEDGE) byId.set(item.id, item);
   280→      for (const item of persistedItems) byId.set(item.id, item);
   281→      this.knowledgeItems = [...byId.values()];
   282→    } catch {
   283→      // fallback to canonical seeds
   284→    }
   285→  }
   286→
   287→  /**
   288→   * Lazily mirrors the canonical seed documents into the Prisma table once
   289→   * per process (idempotent upserts keyed by item id). Local mode:
   290→   * best-effort. Authoritative mode: fail-closed via the DB authority.
   291→   */
   292→  private ensurePrismaSeeded(): Promise<void> {
   293→    if (!CompanyKnowledgeStore.prismaSeedPromise) {
   294→      CompanyKnowledgeStore.prismaSeedPromise = (async () => {
   295→        try {
   296→          const db = isAuthoritativeMode()
   297→            ? await requireAuthoritativeDatabase()
   298→            : prisma;
   299→          for (const item of CANONICAL_COMPANY_KNOWLEDGE) {
   300→            await db.companyKnowledge.upsert({
   301→              where: { id: item.id },
   302→              create: {
   303→                id: item.id,
   304→                category: item.category,
   305→                title: item.title,
   306→                content: item.content,
   307→                hash: computeKnowledgeContentHash(item.content),
   308→                metadata: {
   309→                  documentId: item.documentId,
   310→                  version: item.version,
   311→                  summary: item.summary,
   312→                  tags: item.tags,
   313→                  applicableDepartments: item.applicableDepartments,
   314→                  authorAuthority: item.authorAuthority,
   315→                  lastVerifiedDate: item.lastVerifiedDate,
   316→                  isDurableReference: item.isDurableReference,
   317→                },
   318→              },
   319→              update: {
   320→                category: item.category,
   321→                title: item.title,
   322→                content: item.content,
   323→                hash: computeKnowledgeContentHash(item.content),
   324→                metadata: {
   325→                  documentId: item.documentId,
   326→                  version: item.version,
   327→                  summary: item.summary,
   328→                  tags: item.tags,
   329→                  applicableDepartments: item.applicableDepartments,
   330→                  authorAuthority: item.authorAuthority,
   331→                  lastVerifiedDate: item.lastVerifiedDate,
   332→                  isDurableReference: item.isDurableReference,
   333→                },
   334→              },
   335→            });
   336→          }
   337→        } catch (err: any) {
   338→          if (isAuthoritativeMode()) throw err;
   339→          // Local mode: best-effort only — the durable file store remains the
   340→          // local-mode primary per ADR 0002.
   341→        }
   342→      })().catch(() => {
   343→        // Do not cache a failed best-effort seed; retry on next usage.
   344→        CompanyKnowledgeStore.prismaSeedPromise = null;
   345→      });
   346→    }
   347→    return CompanyKnowledgeStore.prismaSeedPromise;
   348→  }
   349→
   350→  public async getAllKnowledge(): Promise<CompanyKnowledgeItem[]> {
   351→    if (isAuthoritativeMode()) {
   352→      await this.ensurePrismaSeeded();
   353→      const db = await requireAuthoritativeDatabase();
   354→      const rows = await db.companyKnowledge.findMany({ orderBy: { updatedAt: 'desc' } });
   355→      return rows.map((r: any) => this.mapRowToItem(r));
   356→    }
   357→    return [...this.knowledgeItems];
   358→  }
   359→
   360→  public async getKnowledgeById(id: string): Promise<CompanyKnowledgeItem | null> {
   361→    if (isAuthoritativeMode()) {
   362→      await this.ensurePrismaSeeded();
   363→      const db = await requireAuthoritativeDatabase();
   364→      const row = await db.companyKnowledge.findFirst({
   365→        where: { OR: [{ id }, { title: { contains: id } }] },
   366→      });
   367→      if (row) return this.mapRowToItem(row);
   368→      return null;
   369→    }
   370→
   371→    const item = this.knowledgeItems.find((k) => k.id === id || k.documentId === id);
   372→    return item ? { ...item } : null;
   373→  }

   375→  public async addKnowledge(item: CompanyKnowledgeItem): Promise<void> {
   376→    if (isAuthoritativeMode()) {
   377→      const db = await requireAuthoritativeDatabase();
   378→      await db.companyKnowledge.upsert({
   379→        where: { id: item.id },
   380→        create: {
   381→          id: item.id,
   382→          category: item.category,
   383→          title: item.title,
   384→          content: item.content,
   385→          hash: computeKnowledgeContentHash(item.content),
   386→          metadata: this.buildRowMetadata(item),
   387→        },
   388→        update: {
   389→          category: item.category,
   390→          title: item.title,
   391→          content: item.content,
   392→          hash: computeKnowledgeContentHash(item.content),
   393→          metadata: this.buildRowMetadata(item),
   394→        },
   395→      });
   396→      // Keep the in-process cache coherent for queryKnowledge.
   397→      this.knowledgeItems = [
   398→        item,
   399→        ...this.knowledgeItems.filter((k) => k.id !== item.id),
   400→      ];
   401→      return;
   402→    }
   403→
   404→    this.knowledgeItems.unshift(item);
   405→    try {
   406→      DurableFileStore.getInstance().saveItem('company_knowledge', item.id, item);
   407→    } catch {}
   408→
   409→    if (process.env.DATABASE_URL) {
   410→      try {
   411→        await prisma.companyKnowledge.upsert({
   412→          where: { id: item.id },
   413→          create: {
   414→            id: item.id,
   415→            category: item.category,
   416→            title: item.title,
   417→            content: item.content,
   418→            hash: computeKnowledgeContentHash(item.content),
   419→            metadata: this.buildRowMetadata(item),
   420→          },
   421→          update: {
   422→            category: item.category,
   423→            title: item.title,
   424→            content: item.content,
   425→            hash: computeKnowledgeContentHash(item.content),
   426→            metadata: this.buildRowMetadata(item),
   427→          },
   428→        });
   429→      } catch {
   430→        // Best-effort dual-write (e.g. unique-hash duplicate collision or
   431→        // offline DB): the durable file store remains the local-mode primary.
   432→      }
   433→    }
   434→  }

   436→  public async setKnowledge(items: CompanyKnowledgeItem[]): Promise<void> {
   437→    this.knowledgeItems = [...items];
   438→
   439→    if (isAuthoritativeMode()) {
   440→      const db = await requireAuthoritativeDatabase();
   441→      await db.$transaction(async (tx: any) => {
   442→        await tx.companyKnowledge.deleteMany({});
   443→        if (items.length > 0) {
   444→          await tx.companyKnowledge.createMany({
   445→            data: items.map((item) => ({
   446→              id: item.id,
   447→              category: item.category,
   448→              title: item.title,
   449→              content: item.content,
   450→              hash: computeKnowledgeContentHash(item.content),
   451→              metadata: this.buildRowMetadata(item),
   452→            })) as any,
   453→          });
   454→        }
   455→      });
   456→      return;
   457→    }
   458→
   459→    try {
   460→      const record: Record<string, CompanyKnowledgeItem> = {};
   461→      for (const item of items) record[item.id] = item;
   462→      DurableFileStore.getInstance().writeCollection('company_knowledge', record);
   463→    } catch {}
   464→
   465→    if (process.env.DATABASE_URL) {
   466→      try {
   467→        await prisma.$transaction([
   468→          prisma.companyKnowledge.deleteMany({}),
   469→          ...(items.length > 0
   470→            ? [prisma.companyKnowledge.createMany({
   471→                data: items.map((item) => ({
   472→                  id: item.id,
   473→                  category: item.category,
   474→                  title: item.title,
   475→                  content: item.content,
   476→                  hash: computeKnowledgeContentHash(item.content),
   477→                  metadata: this.buildRowMetadata(item),
   478→                })) as any,
   479→              })]
   480→            : []),
   481→        ]);
   482→      } catch {
   483→        // Best-effort dual-write; durable file remains local-mode primary.
   484→      }
   485→    }
   486→  }

   488→  /**

   490→   * Attaches epistemic label: 'durable_reference' and provenance metadata.
   491→   */
   492→  public async queryKnowledge(params: KnowledgeQueryParams): Promise<RetrievedKnowledgeItem[]> {
   493→    const queryText = (params.queryText || '').toLowerCase();
   494→    const queryTokens = new Set([
   495→      ...extractTokens(queryText),
   496→      ...(params.keywords ? params.keywords.flatMap(extractTokens) : []),
   497→      ...(params.tags ? params.tags.map((t) => t.toLowerCase()) : []),
   498→    ]);
   499→
   500→    const targetRole = params.role;
   501→    const nowIso = new Date().toISOString();
   502→    const results: RetrievedKnowledgeItem[] = [];
   503→
   504→    for (const item of this.knowledgeItems) {
   505→      // Role match check
   506→      const isRoleApplicable = !targetRole || targetRole === 'orchestrator' || targetRole === 'council' || targetRole === 'advisor'
   507→        || item.applicableDepartments.includes(targetRole)
   508→        || item.applicableDepartments.includes('council');
   509→
   510→      const itemTokens = [
   511→        ...extractTokens(item.title),
   512→        ...extractTokens(item.summary),
   513→        ...extractTokens(item.content),
   514→        ...item.tags.flatMap(extractTokens),
   515→        item.documentId.toLowerCase(),
   516→      ];
   517→
   518→      const matchedTerms = itemTokens.filter((t) => queryTokens.has(t));
   519→      const hasTermMatch = matchedTerms.length > 0;
   520→
   521→      // Check category match if provided
   522→      const categoryMatch = params.category && item.category.toLowerCase().includes(params.category.toLowerCase());
   523→
   524→      if (hasTermMatch || categoryMatch) {
   525→        const relevanceScore = (hasTermMatch ? matchedTerms.length * 2 : 0)
   526→          + (categoryMatch ? 3 : 0)
   527→          + (isRoleApplicable ? 1 : 0);
   528→
   529→        results.push({
   530→          knowledgeId: item.id,
   531→          documentId: item.documentId,
   532→          title: item.title,
   533→          category: item.category,
   534→          version: item.version,
   535→          summary: item.summary,
   536→          contentSnippet: item.summary,
   537→          fullContent: item.content,
   538→          applicableDepartments: item.applicableDepartments,
   539→          relevanceScore,
   540→          matchReason: matchedTerms.length > 0
   541→            ? `Matched knowledge terms: ${[...new Set(matchedTerms)].join(', ')}`
   542→            : `Category match (${item.category})`,
   543→          provenance: {
   544→            sourceSystem: 'company_knowledge',
   545→            sourceId: `knowledge:${item.documentId}`,
   546→            sourceTitle: item.title,
   547→            epistemicType: 'durable_reference',
   548→            authority: item.authorAuthority,
   549→            timestamp: item.lastVerifiedDate || nowIso,
   550→            confidence: 'reference_standard',
   551→            notes: `Durable company reference documentation (v${item.version})`,
   552→          },
   553→        });
   554→      }
   555→    }
   556→
   557→    // Sort by relevance score descending
   558→    results.sort((a, b) => b.relevanceScore - a.relevanceScore);
   559→    const limit = params.limit || 5;
   560→    return results.slice(0, limit);
   561→  }

   563→  private buildRowMetadata(item: CompanyKnowledgeItem): any {
   564→    return {
   565→      documentId: item.documentId,
   566→      version: item.version,
   567→      summary: item.summary,
   568→      tags: item.tags,
   569→      applicableDepartments: item.applicableDepartments,
   570→      authorAuthority: item.authorAuthority,
   571→      lastVerifiedDate: item.lastVerifiedDate,
   572→      isDurableReference: item.isDurableReference,
   573→    } as any;
   574→  }

   576→  private mapRowToItem(r: any): CompanyKnowledgeItem {
   577→    const meta = (r.metadata && typeof r.metadata === 'object') ? r.metadata : {};
   578→    return {
   579→      id: r.id,
   580→      documentId: (meta.documentId as string) || r.id,
   581→      title: r.title,
   582→      category: r.category,
   583→      version: (meta.version as string) || '1.0.0',
   584→      summary: (meta.summary as string) || r.content?.slice(0, 240) || '',
   585→      content: r.content,
   586→      tags: (meta.tags as string[]) || [],
   587→      applicableDepartments: ((meta.applicableDepartments as CompanyKnowledgeItem['applicableDepartments']) || ['council']) as any,
   588→      authorAuthority: (meta.authorAuthority as string) || 'Unknown authority',
   589→      lastVerifiedDate: (meta.lastVerifiedDate as string) || r.updatedAt?.toISOString?.() || new Date().toISOString(),
   590→      isDurableReference: true as const,
   591→    };
   592→  }
