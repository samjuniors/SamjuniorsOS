    49→class Edge(BaseModel, ABC):
    50→    uuid: str = Field(default_factory=lambda: str(uuid4()))
    51→    group_id: str = Field(description='partition of the graph')
    52→    source_node_uuid: str
    53→    target_node_uuid: str
    54→    created_at: datetime
    55→
    56→    @abstractmethod
    57→    async def save(self, driver: GraphDriver): ...
    58→
    59→    async def delete(self, driver: GraphDriver):
    60→        if driver.graph_operations_interface:
    61→            try:
    62→                return await driver.graph_operations_interface.edge_delete(self, driver)
    63→            except NotImplementedError:
    64→                pass
    65→
    66→        if driver.provider == GraphProvider.KUZU:
    67→            await driver.execute_query(
    68→                """
    69→                MATCH (n)-[e:MENTIONS|HAS_MEMBER {uuid: $uuid}]->(m)
    70→                DELETE e
    71→                """,
    72→                uuid=self.uuid,
    73→            )
    74→            await driver.execute_query(
    75→                """
    76→                MATCH (e:RelatesToNode_ {uuid: $uuid})
    77→                DETACH DELETE e
    78→                """,
    79→                uuid=self.uuid,
    80→            )
    81→        else:
    82→            await driver.execute_query(
    83→                """
    84→                MATCH (n)-[e:MENTIONS|RELATES_TO|HAS_MEMBER {uuid: $uuid}]->(m)
    85→                DELETE e
    86→                """,
    87→                uuid=self.uuid,
    88→            )
    89→
    90→        logger.debug(f'Deleted Edge: {self.uuid}')
    91→

   106→                WHERE e.uuid IN $uuids
   107→                DELETE e
   108→                """,
   109→                uuids=uuids,
   110→            )
   111→            await driver.execute_query(
   112→                """
   113→                MATCH (e:RelatesToNode_)
   114→                WHERE e.uuid IN $uuids
   115→                DETACH DELETE e
   116→                """,
   117→                uuids=uuids,
   118→            )
   119→        else:
   120→            await driver.execute_query(
   121→                """
   122→                MATCH (n)-[e:MENTIONS|RELATES_TO|HAS_MEMBER]->(m)
   123→                WHERE e.uuid IN $uuids
   124→                DELETE e
   125→                """,
   126→                uuids=uuids,
   127→            )
   128→
   129→        logger.debug(f'Deleted Edges: {uuids}')

   263→class EntityEdge(Edge):
   264→    name: str = Field(description='name of the edge, relation name')
   265→    fact: str = Field(description='fact representing the edge and nodes that it connects')
   266→    fact_embedding: list[float] | None = Field(default=None, description='embedding of the fact')
   267→    episodes: list[str] = Field(
   268→        default=[],
   269→        description='list of episode ids that reference these entity edges',
   270→    )
   271→    expired_at: datetime | None = Field(
   272→        default=None, description='datetime of when the node was invalidated'
   273→    )
   274→    valid_at: datetime | None = Field(
   275→        default=None, description='datetime of when the fact became true'
   276→    )
   277→    invalid_at: datetime | None = Field(
   278→        default=None, description='datetime of when the fact stopped being true'
   279→    )
   280→    reference_time: datetime | None = Field(
   281→        default=None, description='reference timestamp from the episode that produced this edge'
   282→    )
   283→    attributes: dict[str, Any] = Field(
   284→        default={}, description='Additional attributes of the edge. Dependent on edge name'
   285→    )
   286→
   287→    async def generate_embedding(self, embedder: EmbedderClient):
   288→        start = time()
   289→
   290→        text = self.fact.replace('\n', ' ')
   291→        self.fact_embedding = await embedder.create(input_data=[text])
   292→
   293→        end = time()
   294→        logger.debug(
   295→            f'embedded edge {self.uuid} fact ({len(text)} chars) in {(end - start) * 1000} ms'
   296→        )
   297→
   298→        return self.fact_embedding
   299→
   300→    async def load_fact_embedding(self, driver: GraphDriver):
   301→        if driver.graph_operations_interface:
   302→            try:
   303→                return await driver.graph_operations_interface.edge_load_embeddings(self, driver)
   304→            except NotImplementedError:
   305→                pass
   306→
   307→        query = """
   308→            MATCH (n:Entity)-[e:RELATES_TO {uuid: $uuid}]->(m:Entity)
   309→            RETURN e.fact_embedding AS fact_embedding
   310→        """
   311→
   312→        if driver.provider == GraphProvider.NEPTUNE:
   313→            query = """
   314→                MATCH (n:Entity)-[e:RELATES_TO {uuid: $uuid}]->(m:Entity)
   315→                RETURN [x IN split(e.fact_embedding, ",") | toFloat(x)] as fact_embedding
   316→            """
   317→
   318→        if driver.provider == GraphProvider.KUZU:
   319→            query = """
   320→                MATCH (n:Entity)-[:RELATES_TO]->(e:RelatesToNode_ {uuid: $uuid})-[:RELATES_TO]->(m:Entity)
   321→                RETURN e.fact_embedding AS fact_embedding
   322→            """
   323→
   324→        records, _, _ = await driver.execute_query(
   325→            query,
   326→            uuid=self.uuid,
   327→            routing_='r',
   328→        )
   329→
   330→        if len(records) == 0:
   331→            raise EdgeNotFoundError(self.uuid)
   332→
   333→        self.fact_embedding = records[0]['fact_embedding']
   334→
   335→    async def save(self, driver: GraphDriver):
   336→        if driver.graph_operations_interface:
   337→            try:
   338→                return await driver.graph_operations_interface.edge_save(self, driver)
   339→            except NotImplementedError:
   340→                pass
   341→
   342→        edge_data: dict[str, Any] = {
   343→            'source_uuid': self.source_node_uuid,
   344→            'target_uuid': self.target_node_uuid,
   345→            'uuid': self.uuid,
   346→            'name': self.name,
   347→            'group_id': self.group_id,
   348→            'fact': self.fact,
   349→            'fact_embedding': self.fact_embedding,
   350→            'episodes': self.episodes,
   351→            'created_at': self.created_at,
   352→            'expired_at': self.expired_at,
   353→            'valid_at': self.valid_at,
   354→            'invalid_at': self.invalid_at,
   355→            'reference_time': self.reference_time,
   356→        }
   357→
   358→        if driver.provider == GraphProvider.KUZU:
   359→            edge_data['attributes'] = json.dumps(self.attributes)
   360→            result = await driver.execute_query(
   361→                get_entity_edge_save_query(driver.provider),
   362→                **edge_data,
   363→            )
   364→        else:
   365→            for k, v in (self.attributes or {}).items():
   366→                if k not in edge_data:
   367→                    edge_data[k] = v
   368→            result = await driver.execute_query(
   369→                get_entity_edge_save_query(driver.provider),
   370→                edge_data=edge_data,
   371→            )
   372→
   373→        logger.debug(f'Saved edge to Graph: {self.uuid}')
   374→
   375→        return result

   985→        attributes.pop('invalid_at', None)

   988→    edge = EntityEdge(

  1038→async def create_entity_edge_embeddings(embedder: EmbedderClient, edges: list[EntityEdge]):
  1039→    # filter out falsey values from edges
  1040→    filtered_edges = [edge for edge in edges if edge.fact]
  1041→
  1042→    if len(filtered_edges) == 0:
  1043→        return
  1044→    fact_embeddings = await embedder.create_batch([edge.fact for edge in filtered_edges])
  1045→    for edge, fact_embedding in zip(filtered_edges, fact_embeddings, strict=True):
  1046→        edge.fact_embedding = fact_embedding
