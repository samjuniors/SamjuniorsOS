     9→/**
    10→ * The blades.
    11→ *
    12→ * This file used to be six slivers of light raking across the frame while a
    13→ * tool ran — pure atmosphere, nothing you could read. That effect is still
    14→ * here, at the bottom, because it is still the right answer to "something is
    15→ * happening". But the name now belongs to the surface it was decorating.
    16→ *
    17→ * A panel is a card you glance at while listening: a figure, three headlines, a
    18→ * status line. A blade is the thing you actually look at. The distinction is
    19→ * not styling, it is geometry — an article you are meant to READ needs a column
    20→ * of a particular width and a height you can scroll, and no amount of care
    21→ * makes that work inside a 320px card stacked beside the reactor. So blades own
    22→ * their size, they stack instead of replacing one another, and the user can
    23→ * pull an older one forward or throw one to full screen.
    24→ *
    25→ * The hard problem a blade solves is that most of the web refuses to be shown.
    26→ * X-Frame-Options and frame-ancestors stop an article being framed, CORS stops
    27→ * the page fetching it, and hotlink protection stops even its images loading.
    28→ * All three are rules the origin server enforces against the *browser*, so the
    29→ * bridge takes the browser out of it: it fetches server-side and serves the
    30→ * result from localhost, and at that point the document in the iframe is ours.
    31→ *
    32→ * Nothing in this file is a special case for a particular site, and nothing
    33→ * here sniffs a file extension. The model asks `probe_url` what a thing is and
    34→ * says what it wants shown; this only knows how to show it.
    35→ */
    36→
    37→/* ------------------------------------------------------------------ sources */
    38→
    39→/**
    40→ * Paths that are genuinely on this machine's disk, as opposed to app-relative
    41→ * URLs that happen to start with a slash. Mirrors the test in sanitise.ts and
    42→ * Orbits.tsx — the list of root directories is the sort of thing that should be
    43→ * changed in each place deliberately.
    44→ */
    45→const DISK_PATH =
    46→  /^\/(Users|home|root|Volumes|Applications|System|Library|private|tmp|var|opt|mnt|media|srv|data)\//
    47→
    48→/** Route a source through the bridge, which is the only origin that can
    49→ *  actually fetch it — and the only one the page CSP will load from. */
    50→function viaBridge(raw: string, route: 'img' | 'media'): string {
    51→  const src = String(raw ?? '').trim()
    52→  if (!src) return ''
    53→  const path = src.replace(/^file:\/\//, '')
    54→  if (DISK_PATH.test(path)) {
    55→    return `${BRIDGE_HTTP_URL}/file?path=${encodeURIComponent(path)}`
    56→  }
    57→  if (!/^https?:\/\//i.test(src)) return src
    58→  if (src.startsWith(`${BRIDGE_HTTP_URL}/`)) return src
    59→  return `${BRIDGE_HTTP_URL}/${route}?url=${encodeURIComponent(src)}`
    60→}
    61→
    62→/** A whole document, rendered by the bridge so it can be framed at all. */
    63→const pageUrl = (url: string, mode: 'reader' | 'live') =>
    64→  `${BRIDGE_HTTP_URL}/page?mode=${mode}&url=${encodeURIComponent(url)}`
    65→
    66→/**
    67→ * The three embed hosts, and only these.
    68→ *
    69→ * Same closed list as the panel sanitiser, for the same reason: YouTube and
    70→ * Vimeo will not hand over the media file, so an iframe is the only way to play
    71→ * a result inline, and the trade for that is that the host list does not grow.
    72→ */
    73→function embedUrl(raw: string): string | null {
    74→  let url: URL
    75→  try {
    76→    url = new URL(raw)
    77→  } catch {
    78→    return null
    79→  }
    80→  const host = url.hostname.toLowerCase().replace(/^(?:www|m|music)\./, '')
    81→  const id = (s: string) => (/^[\w-]{6,20}$/.test(s) ? s : null)
    82→
    83→  if (host === 'youtube.com' && url.pathname === '/watch') {
    84→    const v = id(url.searchParams.get('v') ?? '')
    85→    return v && `https://www.youtube-nocookie.com/embed/${v}`
    86→  }
    87→  if (host === 'youtu.be') {
    88→    const v = id(url.pathname.slice(1))
    89→    return v && `https://www.youtube-nocookie.com/embed/${v}`
    90→  }
    91→  if (host === 'youtube-nocookie.com' && /^\/embed\/[\w-]+/.test(url.pathname)) return url.href
    92→  if (host === 'vimeo.com') {
    93→    const v = url.pathname.split('/').filter(Boolean)[0] ?? ''
    94→    return /^\d+$/.test(v) ? `https://player.vimeo.com/video/${v}` : null
    95→  }
    96→  if (host === 'player.vimeo.com' && /^\/video\/\d+/.test(url.pathname)) return url.href
    97→  return null
    98→}

   102→/**
   103→ * What goes inside a blade.
   104→ *
   105→ * Every iframe is sandboxed. `allow-same-origin` is deliberately absent from
   106→ * the proxied-page case: that document is served from the bridge's own origin —
   107→ * the one origin permitted to open the agent socket — so granting it
   108→ * same-origin would let a page JARVIS found on the web reach that socket. It
   109→ * does not need it. It is being read, not run.
   110→ */
   111→/**
   112→ * The live camera, on screen.
   113→ *
   114→ * Holding the camera for as long as the blade is open does two jobs. It shows
   115→ * the user what JARVIS can see, which is the honest way to run a camera; and it
   116→ * starts the rolling buffer, which is the only reason "what did I just do" can
   117→ * ever be answered — a question that cannot be satisfied by starting to record
   118→ * at the moment it is asked.
   119→ *
   120→ * Mirrored here and only here. A person expects their own image to behave like
   121→ * a reflection, so the preview is flipped for them; the frames handed to the
   122→ * model are not, because a label held up to the lens has to arrive the right
   123→ * way round.
   124→ */
   125→const CameraView = memo(function CameraView() {
   126→  const el = useRef<HTMLVideoElement>(null)
   127→  const [failed, failed_] = useState<string | null>(null)
   128→
   129→  useEffect(() => {
   130→    let held = false
   131→    let gone = false
   132→    void camera
   133→      .holdCamera()
   134→      .then((source) => {
   135→        if (gone) {
   136→          camera.releaseCamera()
   137→          return
   138→        }
   139→        held = true
   140→        camera.startBuffer()
   141→        if (el.current && source.srcObject) el.current.srcObject = source.srcObject
   142→      })
   143→      .catch((err: DOMException) =>
   144→        failed_(
   145→          err?.name === 'NotAllowedError'
   146→            ? 'Camera access is not permitted.'
   147→            : `The camera could not be opened: ${err?.message ?? err}`,
   148→        ),
   149→      )
   150→    return () => {
   151→      gone = true
   152→      if (held) camera.releaseCamera()
   153→    }
   154→  }, [])
   155→
   156→  if (failed) return <p className="bl-note">{failed}</p>
   157→  return <video ref={el} className="bl-camera" autoPlay playsInline muted />
   158→})
   159→
   160→const Body = memo(function Body({ blade }: { blade: Blade }) {
   161→  if (blade.kind === 'camera') return <CameraView />
   162→
   163→  if (blade.kind === 'article' && blade.url) {
   164→    return (
   165→      <iframe
   166→        className="bl-frame"
   167→        src={pageUrl(blade.url, blade.mode ?? 'reader')}
   168→        // allow-scripts WITHOUT allow-same-origin. That combination is the
   169→        // point: the page runs in an opaque origin, so the one script the
   170→        // bridge injects can move the document's own scroll position and can
   171→        // reach nothing of ours — not this origin, not the agent socket. Adding
   172→        // allow-same-origin would hand a page found on the web the keys.
   173→        sandbox="allow-scripts"
   174→        referrerPolicy="no-referrer"
   175→        title={blade.title}
   176→      />
   177→    )
   178→  }
   179→
   180→  if (blade.kind === 'embed' && blade.url) {
   181→    const embed = embedUrl(blade.url)
   182→    if (!embed) return <p className="bl-note">That video link could not be played.</p>
   183→    return (
   184→      <iframe
   185→        className="bl-frame"
   186→        src={embed}
   187→        // A player genuinely needs scripts and its own origin. Nothing that
   188→        // reaches the room the user is sitting in is granted.
   189→        sandbox="allow-scripts allow-same-origin allow-presentation"
   190→        allow="accelerometer; encrypted-media; picture-in-picture; fullscreen"
   191→        referrerPolicy="no-referrer"
   192→        allowFullScreen
   193→        title={blade.title}
   194→      />
   195→    )
   196→  }
   197→
   198→  if (blade.kind === 'video' && blade.url) {
   199→    return (
   200→      <video
   201→        className="bl-video"
   202→        src={viaBridge(blade.url, 'media')}
   203→        controls
   204→        playsInline
   205→        preload="metadata"
   206→      />
   207→    )
   208→  }
   209→
   210→  if (blade.kind === 'image' && blade.url) {
   211→    return <img className="bl-image" src={viaBridge(blade.url, 'img')} alt={blade.title} />
   212→  }
   213→
   214→  if (blade.kind === 'gallery') {
   215→    return (
   216→      <div className="bl-gallery">
   217→        {(blade.images ?? []).map((src, i) => (
   218→          <img key={`${src}-${i}`} className="bl-thumb" src={viaBridge(src, 'img')} alt="" />
   219→        ))}
   220→      </div>
   221→    )
   222→  }
   223→
   224→  if (blade.kind === 'markup' && blade.html) {
   225→    // Model-authored markup gets exactly the treatment panel markup gets.
   226→    // There is one sanitiser, and this is it.
   227→    return (
   228→      <div
   229→        className="bl-markup p-body"
   230→        dangerouslySetInnerHTML={{ __html: sanitisePanelHtml(blade.html) }}
   231→      />
   232→    )
   233→  }
   234→
   235→  return <p className="bl-note">Nothing to show.</p>
   236→})

   239→
   240→function Card({
   241→  blade,
   242→  depth,
   243→  focused,
   244→  expanded,
   245→  onFocus,
   246→  onExpand,
   247→  onClose,
   248→}: {
   249→  blade: Blade
   250→  /** 0 is front-most. Drives the offset and the dimming behind it. */
   251→  depth: number
   252→  focused: boolean
   253→  expanded: boolean
   254→  onFocus: () => void
   255→  onExpand: () => void
   256→  onClose: () => void
   257→}) {
   258→  /** Size the user has dragged this blade to, overriding the class preset. */
   259→  const [size, setSize] = useState<{ w: number; h: number } | null>(null)
   260→  /** Where the user has dragged it, relative to its slot. */
   261→  const [pos, setPos] = useState({ x: 0, y: 0 })
   262→  const shell = useRef<HTMLDivElement>(null)
   263→  const body = useRef<HTMLDivElement>(null)
   264→
   265→  /**
   266→   * Scroll whatever this blade is showing.
   267→   *
   268→   * Two destinations, because a blade holds two different kinds of thing. Its
   269→   * own overflow for markup and galleries; a postMessage for an article, since
   270→   * an iframe is a separate document that the parent cannot scroll directly —
   271→   * see the shim in bridge/page.mjs.
   272→   */
   273→  const scrollContent = (dy: number) => {
   274→    const el = body.current
   275→    if (!el) return
   276→    const frame = el.querySelector('iframe')
   277→    if (frame?.contentWindow) {
   278→      frame.contentWindow.postMessage({ jarvis: 'scroll', dy }, '*')
   279→    } else {
   280→      el.scrollTop += dy
   281→    }
   282→  }
   283→
   284→  /**
   285→   * Drag and resize both listen on `window`, and that is the whole trick.
   286→   *
   287→   * The obvious implementations do not work by hand. framer-motion's own drag
   288→   * tracks the pointer through internals we cannot reach, and the resize grip
   289→   * originally listened on the grip element — but the hand controller aims its
   290→   * synthetic events with elementFromPoint, and one pixel into a drag the
   291→   * element under the cursor is no longer the grip. So resizing by hand died on
   292→   * the first frame, and dragging never started at all.
   293→   *
   294→   * Listening on window fixes both for free: a synthetic event dispatched at
   295→   * whatever is under the cursor still bubbles to window, so these handlers see
   296→   * a hand and a mouse identically. Which is the property the gesture layer was
   297→   * designed around — one interaction, not two implementations of it.
   298→   */
   299→  const grab = (
   300→    e: React.PointerEvent,
   301→    onMove: (dx: number, dy: number) => void,
   302→  ) => {
   303→    e.preventDefault()
   304→    e.stopPropagation()
   305→    const sx = e.clientX
   306→    const sy = e.clientY
   307→    let baseX = sx
   308→    let baseY = sy
   309→    let dx = 0
   310→    let dy = 0
   311→
   312→    const move = (ev: PointerEvent) => {
   313→      /**
   314→       * Both hands pinching means this is not a drag.
   315→       *
   316→       * One pinch is a grab. Two is somebody doing something two-handed, and
   317→       * whichever hand happened to press first should not be hauling the blade
   318→       * around underneath it — the result is a blade that lurches away while
   319→       * you are trying to do something else with both hands.
   320→       *
   321→       * Suppressed by re-anchoring rather than by returning early. A plain
   322→       * return would leave the origin where the press began, so the moment one
   323→       * hand released, the blade would leap by however far the other hand had
   324→       * travelled in the meantime. Moving the origin with the hand keeps the
   325→       * offset constant, so letting go of one hand simply resumes from here.
   326→       */
   327→      if (ev.pointerType === 'touch' && pinchCount() > 1) {
   328→        baseX = ev.clientX - dx
   329→        baseY = ev.clientY - dy
   330→        return
   331→      }
   332→      dx = ev.clientX - baseX
   333→      dy = ev.clientY - baseY
   334→      onMove(dx, dy)
   335→    }
   336→    const done = () => {
   337→      window.removeEventListener('pointermove', move)
   338→      window.removeEventListener('pointerup', done)
   339→      window.removeEventListener('pointercancel', done)
   340→    }
   341→    window.addEventListener('pointermove', move)
   342→    window.addEventListener('pointerup', done)
   343→    window.addEventListener('pointercancel', done)
   344→  }
   345→
   346→  const onHeadDown = (e: React.PointerEvent) => {
   347→    // Buttons live in the header too; starting a drag from one would mean the
   348→    // click never lands.
   349→    if ((e.target as HTMLElement).closest('button')) return
   350→    if (!focused) onFocus()
   351→    if (expanded) return
   352→    const from = { ...pos }
   353→    grab(e, (dx, dy) => setPos({ x: from.x + dx, y: from.y + dy }))
   354→  }
   355→
   356→  /**
   357→   * Pinch anywhere on a blade to grab it.
   358→   *
   359→   * This used to scroll, and moving a blade was possible only by hitting the
   360→   * header — a strip 31 pixels tall. Asking someone to land a hand cursor on 31
   361→   * pixels is not an interaction, and since a pinch on the body scrolled
   362→   * instead, there was in practice no way to move a blade by hand at all.
   363→   *
   364→   * Grabbing is also what people try first: you see a thing and reach for it.
   365→   * So a pinch anywhere picks the blade up, and scrolling moves to a pose that
   366→   * is deliberate and hard to make by accident — two fingers, see the effect
   367→   * below. Only for 'touch', which is what the gesture layer dispatches; a mouse
   368→   * keeps its wheel and its ability to select text.
   369→   */
   370→  const onBodyDown = (e: React.PointerEvent) => {
   371→    if (e.pointerType !== 'touch') return
   372→    if (!focused) onFocus()
   373→    if (expanded) return
   374→    const from = { ...pos }
   375→    grab(e, (dx, dy) => setPos({ x: from.x + dx, y: from.y + dy }))
   376→  }
   377→
   378→  /**
   379→   * Two fingers up, moved up or down, scrolls the front blade.
   380→   *
   381→   * Reads a distance from hands.ts and decides here that it means scrolling —
   382→   * the tracker publishes the pose, not the consequence.
   383→   */
   384→  useEffect(() => {
   385→    if (!focused) return
   386→    let raf = 0
   387→    let last: number | null = null
   388→    const tick = () => {
   389→      raf = requestAnimationFrame(tick)
   390→      const travelled = peaceScroll()
   391→      if (travelled === null) {
   392→        last = null
   393→        return
   394→      }
   395→      if (last === null) {
   396→        last = travelled
   397→        return
   398→      }
   399→      // Inverted and amplified: pulling your hand up moves you down the page,
   400→      // and a hand does not have the travel a scroll wheel does.
   401→      scrollContent((last - travelled) * 2.4)
   402→      last = travelled
   403→    }
   404→    tick()
   405→    return () => cancelAnimationFrame(raf)
   406→  }, [focused])
   407→
   408→  /**
   409→   * Frame the blade with both hands to resize it.
   410→   *
   411→   * Index up, thumb out, one hand either side — the rectangle people already
   412→   * mime when they frame a shot. Pull the corners apart and the blade grows;
   413→   * bring them together and it shrinks; lean toward the camera and it grows
   414→   * too, because leaning in enlarges everything about the hands including the
   415→   * gap between them.
   416→   *
   417→   * This replaced a two-handed pinch, which read well on paper and collided
   418→   * badly in practice: a pinch is how you GRAB a blade, so two of them meant
   419→   * two hands each trying to pick something up while also asking to resize it.
   420→   * The framing pose collides with nothing, which is most of why it is right.
   421→   *
   422→   * The measurement arrives as a plain distance; that it means a resize is
   423→   * decided here. Only the focused blade, and never while expanded, where the
   424→   * size is the entire point of the state.
   425→   */
   426→  useEffect(() => {
   427→    if (!focused || expanded) return
   428→    let raf = 0
   429→    let from: { span: number; w: number; h: number } | null = null
   430→    const tick = () => {
   431→      raf = requestAnimationFrame(tick)
   432→      const span = frameSpan()
   433→      if (span === null) {
   434→        from = null
   435→        return
   436→      }
   437→      const box = shell.current?.getBoundingClientRect()
   438→      if (!box) return
   439→      if (!from) {
   440→        // Both hands have just closed. Anchor on the size as it is now.
   441→        from = { span, w: box.width, h: box.height }
   442→        return
   443→      }
   444→      // Guard the divisor: hands almost touching would send the scale to
   445→      // infinity and the blade off the screen in one frame.
   446→      const k = span / Math.max(from.span, 40)
   447→      setSize({
   448→        w: Math.max(280, Math.min(window.innerWidth * 0.96, from.w * k)),
   449→        h: Math.max(180, Math.min(window.innerHeight * 0.94, from.h * k)),
   450→      })
   451→    }
   452→    tick()
   453→    return () => cancelAnimationFrame(raf)
   454→  }, [focused, expanded])
   455→
   456→  const onGrip = (e: React.PointerEvent) => {
   457→    const box = shell.current?.getBoundingClientRect()
   458→    if (!box) return
   459→    const from = { w: box.width, h: box.height }
   460→    grab(e, (dx, dy) =>
   461→      setSize({
   462→        w: Math.max(280, Math.min(window.innerWidth * 0.96, from.w + dx)),
   463→        h: Math.max(180, Math.min(window.innerHeight * 0.94, from.h + dy)),
   464→      }),
   465→    )
   466→  }
   467→
   468→  /**
   469→   * Resize from the bottom-right grip.
   470→   *
   471→   * Pointer capture rather than window listeners: the pointer spends most of a
   472→   * resize over an <iframe>, and an iframe swallows mousemove from the parent
   473→   * document entirely. Without capture the blade stops resizing the instant the
   474→   * cursor crosses into the article it is showing, which is precisely where it
   475→   * always crosses.
   476→   */
   477→  return (
   478→    /**
   479→     * Two elements, because two different things want the transform.
   480→     *
   481→     * The outer one carries the depth offset — the small lift and scale that
   482→     * makes the stack read as objects rather than as a list. The inner one is
   483→     * what the user drags. Framer owns `transform` on anything it animates, so
   484→     * with both jobs on one element the drag and the stack animation overwrite
   485→     * each other every frame and the blade jitters back to its slot.
   486→     */
   487→    <motion.div
   488→      className="bl-slot"
   489→      initial={{ opacity: 0, y: 26, scale: 0.96, filter: 'blur(6px)' }}
   490→      animate={{
   491→        opacity: expanded || depth === 0 ? 1 : Math.max(0.3, 1 - depth * 0.24),
   492→        y: expanded ? 0 : depth * -13,
   493→        x: expanded ? 0 : depth * 15,
   494→        scale: expanded ? 1 : 1 - depth * 0.035,
   495→        filter: depth === 0 || expanded ? 'blur(0px)' : `blur(${depth * 0.7}px)`,
   496→      }}
   497→      exit={{ opacity: 0, y: 18, filter: 'blur(8px)', transition: { duration: 0.28 } }}
   498→      transition={{ type: 'spring', stiffness: 260, damping: 30 }}
   499→      style={{ zIndex: expanded ? 60 : 40 - depth }}
   500→    >
   501→      <motion.section
   502→        ref={shell}
   503→        className={
   504→          `bl bl-${blade.size}` +
   505→          (expanded ? ' bl-expanded' : '') +
   506→          (focused ? ' bl-front' : '')
   507→        }
   508→        // Position and size are ours rather than framer's — see `grab` above for
   509→        // why. Applied as a plain transform because the depth animation lives on
   510→        // the slot wrapper, so nothing is competing for this element's own one.
   511→        style={{
   512→          ...(size && !expanded ? { width: size.w, height: size.h } : null),
   513→          transform: expanded ? undefined : `translate(${pos.x}px, ${pos.y}px)`,
   514→        }}
   515→        // pointerdown, not mousedown: a hand dispatches PointerEvents, and a
   516→        // mousedown handler simply never hears them. Focusing a blade by pinch
   517→        // was silently impossible until this changed.
   518→        onPointerDown={() => {
   519→          if (!focused) onFocus()
   520→        }}
   521→      >
   522→        <span className="pk pk-tl" />
   523→        <span className="pk pk-tr" />
   524→        <span className="pk pk-bl" />
   525→        <span className="pk pk-br" />
   526→
   527→        <header className="bl-head" onPointerDown={onHeadDown}>
   528→          <span className="bl-title">{blade.title}</span>
   529→          <span className="bl-kind">{blade.kind}</span>
   530→          <span className="bl-acts">
   531→            {(size || pos.x || pos.y) && !expanded && (
   532→              <button
   533→                className="bl-btn"
   534→                onClick={(e) => {
   535→                  e.stopPropagation()
   536→                  setSize(null)
   537→                  setPos({ x: 0, y: 0 })
   538→                }}
   539→                title="Back where it started"
   540→              >
   541→                ⤾
   542→              </button>
   543→            )}
   544→            <button
   545→              className="bl-btn"
   546→              onClick={(e) => {
   547→                e.stopPropagation()
   548→                onExpand()
   549→              }}
   550→              title={expanded ? 'Shrink (E)' : 'Full screen (E)'}
   551→            >
   552→              {expanded ? '⤡' : '⤢'}
   553→            </button>
   554→            <button
   555→              className="bl-btn"
   556→              onClick={(e) => {
   557→                e.stopPropagation()
   558→                onClose()
   559→              }}
   560→              title="Close (X)"
   561→            >
   562→              ✕
   563→            </button>
   564→          </span>
   565→        </header>
   566→
   567→        <div className="bl-body" ref={body} onPointerDown={onBodyDown}>
   568→          <Body blade={blade} />
   569→        </div>
   570→
   571→        {/* Resize grip. Absent while expanded, where the size is the point. */}
   572→        {!expanded && <span className="bl-grip" onPointerDown={onGrip} title="Drag to resize" />}
   573→      </motion.section>
   574→    </motion.div>
   575→  )
   576→}

   578→/* ------------------------------------------------------------------- stack */
   579→
   580→export function Blades() {
   581→  const blades = useStore((s) => s.blades)
   582→  const focusedBlade = useStore((s) => s.focusedBlade)
   583→  const expandedBlade = useStore((s) => s.expandedBlade)
   584→  const focusBlade = useStore((s) => s.focusBlade)
   585→  const expandBlade = useStore((s) => s.expandBlade)
   586→  const closeBlade = useStore((s) => s.closeBlade)
   587→
   588→  /**
   589→   * Newest first, then whichever the user pulled forward lifted to the front.
   590→   *
   591→   * Ordered here rather than in the store because it is a view concern, and the
   592→   * store's array order is the history — which is what makes "the one before
   593→   * that" a meaningful thing to ask for.
   594→   */
   595→  const ordered = useMemo(() => {
   596→    const newestFirst = [...blades].reverse()
   597→    if (!focusedBlade) return newestFirst
   598→    const hit = newestFirst.findIndex((b) => b.id === focusedBlade)
   599→    if (hit <= 0) return newestFirst
   600→    const copy = [...newestFirst]
   601→    const [lifted] = copy.splice(hit, 1)
   602→    return [lifted, ...copy]
   603→  }, [blades, focusedBlade])
   604→
   605→  const front = ordered[0]
   606→
   607→  const cycle = useCallback(
   608→    (by: number) => {
   609→      if (ordered.length < 2) return
   610→      const at = ordered.findIndex((b) => b.id === front?.id)
   611→      const next = ordered[(at + by + ordered.length) % ordered.length]
   612→      if (next) focusBlade(next.id)
   613→    },
   614→    [ordered, front, focusBlade],
   615→  )
   616→
   617→  // Bound here rather than in App, and only while something is open, so E and X
   618→  // are free for anything else the moment the last blade closes.
   619→  const live = useRef(false)
   620→  live.current = blades.length > 0
   621→
   622→  useEffect(() => {
   623→    const onKey = (e: KeyboardEvent) => {
   624→      if (!live.current) return
   625→      const tag = (e.target as HTMLElement)?.tagName
   626→      if (tag === 'INPUT' || tag === 'TEXTAREA') return
   627→      if (e.metaKey || e.ctrlKey || e.altKey || e.repeat) return
   628→
   629→      if (e.key === 'e') {
   630→        e.preventDefault()
   631→        expandBlade(expandedBlade ? null : (front?.id ?? null))
   632→      } else if (e.key === 'x') {
   633→        e.preventDefault()
   634→        if (front) closeBlade(front.id)
   635→      } else if (e.key === ']') {
   636→        e.preventDefault()
   637→        cycle(1)
   638→      } else if (e.key === '[') {
   639→        e.preventDefault()
   640→        cycle(-1)
   641→      }
   642→    }
   643→    window.addEventListener('keydown', onKey)
   644→    return () => window.removeEventListener('keydown', onKey)
   645→  }, [front, expandedBlade, expandBlade, closeBlade, cycle])
   646→
   647→  if (!blades.length) return null
   648→
   649→  return (
   650→    <div className={`blades-stack${expandedBlade ? ' blades-stack-full' : ''}`}>
   651→      <AnimatePresence>
   652→        {ordered.map((blade, i) => {
   653→          const expanded = expandedBlade === blade.id
   654→          // While one is expanded it is the only thing on screen; the rest are
   655→          // unmounted rather than hidden so their iframes stop loading.
   656→          if (expandedBlade && !expanded) return null
   657→          return (
   658→            <Card
   659→              key={blade.id}
   660→              blade={blade}
   661→              depth={expanded ? 0 : i}
   662→              focused={blade.id === front?.id}
   663→              expanded={expanded}
   664→              onFocus={() => focusBlade(blade.id)}
   665→              onExpand={() => expandBlade(expanded ? null : blade.id)}
   666→              onClose={() => closeBlade(blade.id)}
   667→            />
   668→          )
   669→        })}
   670→      </AnimatePresence>
   671→
   672→      {blades.length > 1 && !expandedBlade && (
   673→        <div className="bl-hint">
   674→          <kbd>[</kbd> <kbd>]</kbd> cycle · <kbd>E</kbd> full · <kbd>X</kbd> close
   675→        </div>
   676→      )}
   677→    </div>
   678→  )
   679→}
   680→
   681→/* --------------------------------------------------------------- the sweep */
   682→
   683→/**
   684→ * The original blades: slivers of light raking across the frame while a tool
   685→ * runs. Unchanged, because it is still the right answer to "something is
   686→ * happening" — a tool call is the one moment the interface stops being a face
   687→ * and becomes machinery, and the reactor cannot carry that on its own.
   688→ *
   689→ * Deliberately CSS rather than three.js: the scene is bloomed and tone-mapped,
   690→ * which is exactly wrong for a 1px edge. Kept in the DOM it stays a blade.
   691→ */
   692→const SWEEP = [1, 2, 3, 4, 5, 6]
   693→
   694→export function BladeSweep() {
   695→  const phase = useStore((s) => s.phase)
   696→  const activeTool = useStore((s) => s.activeTool)
   697→
   698→  return (
   699→    <AnimatePresence>
   700→      {phase === 'tooling' && (
   701→        <motion.div
   702→          className="blades"
   703→          // Only opacity is animated here. The sweeps are CSS keyframes on the
   704→          // children, and framer writes `transform` inline on anything it
   705→          // animates — one transform prop in this list and every blade would be
   706→          // sliding inside an element that is itself sliding.
   707→          initial={{ opacity: 0 }}
   708→          animate={{ opacity: 1 }}
   709→          exit={{ opacity: 0 }}
   710→          transition={{ duration: 0.3, ease: 'easeOut' }}
   711→        >
   712→          <div className="blade-field">
   713→            {SWEEP.map((n) => (
   714→              <span key={n} className={`blade blade-${n}`} />
   715→            ))}
   716→          </div>
   717→
   718→          {activeTool && (
   719→            // Keyed on the name so a chain of tools re-runs the ride-in for
   720→            // each one rather than silently swapping the text mid-sweep.
   721→            <div className="blade-carrier">
   722→              <span key={activeTool} className="blade-tool">
   723→                {activeTool}
   724→              </span>
   725→            </div>
   726→          )}
   727→        </motion.div>
   728→      )}
   729→    </AnimatePresence>
   730→  )
   731→}
