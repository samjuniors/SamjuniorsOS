    27→ * Deterministic tokenization shared with queryKnowledge (lowercase, strip
    28→ * non-alphanumeric except _-, drop stop words and <=2-char fragments).
    29→ *
    30→ * M4-C: exported so the Sophia Personal Mind's context retrieval reuses the
    31→ * EXACT same deterministic tokenizer/stop-word pipeline Company Brain
    32→ * knowledge retrieval uses — one tokenizer, no second retrieval framework,
    33→ * no drift between the two lexical scorers. Pure function; no behavior change
    34→ * to queryKnowledge itself.

    36→export function extractTokens(text?: string | null): string[] {
    37→  if (!text) return [];
    38→  return text
    39→    .toLowerCase()
    40→    .replace(/[^a-z0-9\s_-]/g, ' ')
    41→    .split(/\s+/)
    42→    .filter((w) => w.length > 2 && !STOP_WORDS.has(w));
    43→}

   212→/**
   213→ * Computes the SHA-256 content hash stored in the `CompanyKnowledge.hash`
   214→ * column (M2: the ingestion path records it on every explicit write).
   215→ *
   216→ * Exact guarantee: it fingerprints ONLY the `content` field.
   217→ *  - Identical content ⇒ identical hash (deterministic, byte-for-byte).
   218→ *  - Changed content ⇒ different hash (SHA-256 collision resistance).
   219→ *  - Title/category/version/metadata changes alone do NOT change the hash.
   220→ *
   221→ * See isKnowledgeContentUnchanged() for the comparison contract.
   222→ */
   223→export function computeKnowledgeContentHash(content: string): string {
   224→  return createHash('sha256').update(content || '', 'utf-8').digest('hex');
   225→}
   226→
   227→/**
   228→ * Minimal change-detection contract (no vectors/embeddings/FTS): compares a
   229→ * persisted hash against the hash of candidate content.
   230→ *  - equal  ⇒ content unchanged since the hash was recorded
   231→ *  - different ⇒ content changed
   232→ * A missing/blank stored hash never claims "unchanged".
   233→ */
   234→export function isKnowledgeContentUnchanged(
   235→  storedHash: string | null | undefined,
   236→  content: string
   237→): boolean {
   238→  if (!storedHash) return false;
   239→  return storedHash === computeKnowledgeContentHash(content);
   240→}

   242→/**
   243→ * ============================================================================
   244→ * COMPANY KNOWLEDGE STORE — M2: CANONICAL PERSISTENCE ACTIVATED
   245→ * ============================================================================
   246→ *
   247→ * Implements ICompanyKnowledgeStore with the SAME dual-mode persistence
   248→ * pattern as its sibling stores (memory-store, conversation store, etc.):
   249→ *
   250→ *  - LOCAL MODE: in-memory cache (constructor-hydrated from DurableFileStore)
   251→ *    + .data/company_knowledge.json durability + best-effort Prisma dual-write.
   252→ *  - AUTHORITATIVE MODE: Prisma fail-closed (requireAuthoritativeDatabase);
   253→ *    reads and writes go to the CompanyKnowledge table (target: PostgreSQL;
   254→ *    currently the SQLite sandbox port — see the M0 naming note).
   255→ *
   256→ * Seed migration (one-time, bootstrap-only): the 8 canonical documents that
   257→ * previously existed ONLY as code constants are persisted to DurableFileStore
   258→ * on first boot (empty durable collection), and bootstrapped into the Prisma
   259→ * table lazily on first store usage — CREATE-ONLY: a canonical document is
   260→ * seeded only when no row exists for its id; an existing persisted row is
   261→ * NEVER overwritten by the code constants on restart (Founder directive,
   262→ * pre-M3 correction pass). Explicit writes (addKnowledge / setKnowledge)
   263→ * remain the only mutation paths after boot.
   264→ *
   265→ * The `hash` column (SHA-256 of the `content` field ONLY) provides change
   266→ * detection: comparing a stored hash against
   267→ * computeKnowledgeContentHash(candidateContent) distinguishes unchanged
   268→ * content (equal) from changed content (different) — see
   269→ * isKnowledgeContentUnchanged(). It does NOT fingerprint title/category/
   270→ * metadata. A different id with byte-identical content collides on the
   271→ * unique hash index and is treated as a duplicate (seed skipped; explicit
   272→ * writes still dual-write best-effort in local mode).
   273→ *
   274→ * Designed so the target PostgreSQL / Supabase deployment can replace the
   275→ * persistence internals without altering retrieval or business logic.
   276→ */
   277→export class CompanyKnowledgeStore implements ICompanyKnowledgeStore {
   278→  private static instance: CompanyKnowledgeStore | null = null;
   279→
   280→  private knowledgeItems: CompanyKnowledgeItem[] = [...CANONICAL_COMPANY_KNOWLEDGE];
   281→
   282→  /** One-shot guard for the lazy Prisma seed mirror (M2 seed migration). */
   283→  private static prismaSeedPromise: Promise<void> | null = null;
   284→
   285→  private constructor() {
   286→    this.loadFromDurableStorage();
   287→  }
   288→
   289→  public static getInstance(): CompanyKnowledgeStore {
   290→    if (!CompanyKnowledgeStore.instance) {
   291→      CompanyKnowledgeStore.instance = new CompanyKnowledgeStore();
   292→    }
   293→    return CompanyKnowledgeStore.instance;
   294→  }
   295→
   296→  /**
   297→   * Hydrates the in-memory cache from DurableFileStore, merging over the
   298→   * canonical seed documents. On first boot (no durable collection), the
   299→   * canonical seeds are themselves persisted — the M2 one-time seed
   300→   * migration of the 8 canonical documents.
   301→   */
   302→  private loadFromDurableStorage(): void {
   303→    try {
   304→      const persisted = DurableFileStore.getInstance().readCollection<CompanyKnowledgeItem>('company_knowledge');
   305→      const persistedItems = Object.values(persisted);
   306→      if (persistedItems.length === 0) {
   307→        // First boot: persist the canonical seed set (one-time migration).
   308→        const seedRecord: Record<string, CompanyKnowledgeItem> = {};
   309→        for (const item of CANONICAL_COMPANY_KNOWLEDGE) {
   310→          seedRecord[item.id] = item;
   311→        }
   312→        DurableFileStore.getInstance().writeCollection('company_knowledge', seedRecord);
   313→        this.knowledgeItems = [...CANONICAL_COMPANY_KNOWLEDGE];
   314→        return;
   315→      }
   316→      // Merge: canonical seeds first (stable order), persisted items overlay
   317→      // by id (edits win), extras appended.
   318→      const byId = new Map<string, CompanyKnowledgeItem>();
   319→      for (const item of CANONICAL_COMPANY_KNOWLEDGE) byId.set(item.id, item);
   320→      for (const item of persistedItems) byId.set(item.id, item);
   321→      this.knowledgeItems = [...byId.values()];
   322→    } catch {
   323→      // fallback to canonical seeds
   324→    }
   325→  }
   326→
   327→  /**
   328→   * Lazily bootstraps the canonical seed documents into the Prisma table
   329→   * once per process. BOOTSTRAP-ONLY (Founder directive, pre-M3 correction):
   330→   * a canonical document is created ONLY when no row exists for its id — an
   331→   * existing persisted row is preserved as-is, so a restart can never
   332→   * overwrite authoritative persisted knowledge with the code constants.
   333→   * Explicit writes (addKnowledge / setKnowledge) remain the only mutation
   334→   * paths after boot. A unique-hash collision on create means byte-identical
   335→   * content is already persisted under another id — the seed is skipped (the
   336→   * hash column is the duplicate-content guard); any other error propagates
   337→   * (fail-closed in authoritative mode). Local mode: best-effort.
   338→   */
   339→  private ensurePrismaSeeded(): Promise<void> {
   340→    if (!CompanyKnowledgeStore.prismaSeedPromise) {
   341→      CompanyKnowledgeStore.prismaSeedPromise = (async () => {
   342→        try {
   343→          const db = isAuthoritativeMode()
   344→            ? await requireAuthoritativeDatabase()
   345→            : prisma;
   346→          for (const item of CANONICAL_COMPANY_KNOWLEDGE) {
   347→            const existing = await db.companyKnowledge.findUnique({
   348→              where: { id: item.id },
   349→            });
   350→            // Persisted document wins over the code constant — never overwrite.
   351→            if (existing) continue;
   352→            try {
   353→              await db.companyKnowledge.create({
   354→                data: {
   355→                  id: item.id,
   356→                  category: item.category,
   357→                  title: item.title,
   358→                  content: item.content,
   359→                  hash: computeKnowledgeContentHash(item.content),
   360→                  metadata: {
   361→                    documentId: item.documentId,
   362→                    version: item.version,
   363→                    summary: item.summary,
   364→                    tags: item.tags,
   365→                    applicableDepartments: item.applicableDepartments,
   366→                    authorAuthority: item.authorAuthority,
   367→                    lastVerifiedDate: item.lastVerifiedDate,
   368→                    isDurableReference: item.isDurableReference,
   369→                  },
   370→                },
   371→              });
   372→            } catch (createErr: any) {
   373→              const msg = String(createErr?.message || createErr);
   374→              const isUniqueCollision =
   375→                createErr?.code === 'P2002' || /unique constraint/i.test(msg);
   376→              if (!isUniqueCollision) throw createErr;
   377→              // Byte-identical content already persisted under another id:
   378→              // skip seeding a duplicate (unique hash = duplicate guard).
   379→            }
   380→          }
   381→        } catch (err: any) {
   382→          if (isAuthoritativeMode()) throw err;
   383→          // Local mode: best-effort only — the durable file store remains the
   384→          // local-mode primary per ADR 0002.
   385→        }
   386→      })().catch(() => {
   387→        // Do not cache a failed best-effort seed; retry on next usage.
   388→        CompanyKnowledgeStore.prismaSeedPromise = null;
   389→      });
   390→    }
   391→    return CompanyKnowledgeStore.prismaSeedPromise;
   392→  }
   393→
   394→  public async getAllKnowledge(): Promise<CompanyKnowledgeItem[]> {
   395→    if (isAuthoritativeMode()) {
   396→      await this.ensurePrismaSeeded();
   397→      const db = await requireAuthoritativeDatabase();
   398→      const rows = await db.companyKnowledge.findMany({ orderBy: { updatedAt: 'desc' } });
   399→      return rows.map((r: any) => this.mapRowToItem(r));
   400→    }
   401→    return [...this.knowledgeItems];
   402→  }
   403→
   404→  public async getKnowledgeById(id: string): Promise<CompanyKnowledgeItem | null> {
   405→    if (isAuthoritativeMode()) {
   406→      await this.ensurePrismaSeeded();
   407→      const db = await requireAuthoritativeDatabase();
   408→      const row = await db.companyKnowledge.findFirst({
   409→        where: { OR: [{ id }, { title: { contains: id } }] },
   410→      });
   411→      if (row) return this.mapRowToItem(row);
   412→      return null;
   413→    }
   414→
   415→    const item = this.knowledgeItems.find((k) => k.id === id || k.documentId === id);
   416→    return item ? { ...item } : null;
   417→  }
   418→
   419→  public async addKnowledge(item: CompanyKnowledgeItem): Promise<void> {
   420→    if (isAuthoritativeMode()) {
   421→      const db = await requireAuthoritativeDatabase();
   422→      await db.companyKnowledge.upsert({
   423→        where: { id: item.id },
   424→        create: {
   425→          id: item.id,
   426→          category: item.category,
   427→          title: item.title,
   428→          content: item.content,
   429→          hash: computeKnowledgeContentHash(item.content),
   430→          metadata: this.buildRowMetadata(item),
   431→        },
   432→        update: {
   433→          category: item.category,
   434→          title: item.title,
   435→          content: item.content,
   436→          hash: computeKnowledgeContentHash(item.content),
   437→          metadata: this.buildRowMetadata(item),
   438→        },
   439→      });
   440→      // Keep the in-process cache coherent for queryKnowledge.
   441→      this.knowledgeItems = [
   442→        item,
   443→        ...this.knowledgeItems.filter((k) => k.id !== item.id),
   444→      ];
   445→      return;
   446→    }
   447→
   448→    // Local mode: explicit write — REPLACES any existing item with the same
   449→    // id (no duplicate-id entries in the in-process cache, mirroring the
   450→    // authoritative branch's replace-by-id semantics).
   451→    this.knowledgeItems = [
   452→      item,
   453→      ...this.knowledgeItems.filter((k) => k.id !== item.id),
   454→    ];
   455→    try {
   456→      DurableFileStore.getInstance().saveItem('company_knowledge', item.id, item);
   457→    } catch {}
   458→
   459→    if (process.env.DATABASE_URL) {
   460→      try {
   461→        await prisma.companyKnowledge.upsert({
   462→          where: { id: item.id },
   463→          create: {
   464→            id: item.id,
   465→            category: item.category,
   466→            title: item.title,
   467→            content: item.content,
   468→            hash: computeKnowledgeContentHash(item.content),
   469→            metadata: this.buildRowMetadata(item),
   470→          },
   471→          update: {
   472→            category: item.category,

   548→   *   Prisma table in authoritative mode — so knowledge edited or added in
   549→   *   another process (or before a restart) is invisible to RETRIEVAL while
   550→   *   being visible to getAllKnowledge(). Callers are context-assembly hot
   551→   *   paths (SophiaContextAssembler, context-retrieval, context-assembly),

   554→   *   dependency and fail-closed 503 propagation to every Sophia turn and
   555→   *   needs its own authoritative-mode verification matrix. Fix requires a
   556→   *   dedicated pass with Founder approval. In the current deployment
   557→   *   (DATABASE_MODE=local) the gap is latent, not active.
   558→   */
   559→  public async queryKnowledge(params: KnowledgeQueryParams): Promise<RetrievedKnowledgeItem[]> {
   560→    const queryText = (params.queryText || '').toLowerCase();
   561→    const queryTokens = new Set([
   562→      ...extractTokens(queryText),
   563→      ...(params.keywords ? params.keywords.flatMap(extractTokens) : []),
   564→      ...(params.tags ? params.tags.map((t) => t.toLowerCase()) : []),
   565→    ]);
   566→
   567→    const targetRole = params.role;
   568→    const nowIso = new Date().toISOString();
   569→    const results: RetrievedKnowledgeItem[] = [];
   570→
   571→    for (const item of this.knowledgeItems) {
   572→      // Role match check
   573→      const isRoleApplicable = !targetRole || targetRole === 'orchestrator' || targetRole === 'council' || targetRole === 'advisor'
   574→        || item.applicableDepartments.includes(targetRole)
   575→        || item.applicableDepartments.includes('council');
   576→
   577→      const itemTokens = [
   578→        ...extractTokens(item.title),
   579→        ...extractTokens(item.summary),
   580→        ...extractTokens(item.content),
   581→        ...item.tags.flatMap(extractTokens),
   582→        item.documentId.toLowerCase(),
   583→      ];
   584→
   585→      const matchedTerms = itemTokens.filter((t) => queryTokens.has(t));
   586→      const hasTermMatch = matchedTerms.length > 0;
   587→
   588→      // Check category match if provided
   589→      const categoryMatch = params.category && item.category.toLowerCase().includes(params.category.toLowerCase());
   590→
   591→      if (hasTermMatch || categoryMatch) {
   592→        const relevanceScore = (hasTermMatch ? matchedTerms.length * 2 : 0)
   593→          + (categoryMatch ? 3 : 0)
   594→          + (isRoleApplicable ? 1 : 0);
   595→
   596→        results.push({
   597→          knowledgeId: item.id,
   598→          documentId: item.documentId,
   599→          title: item.title,
   600→          category: item.category,
   601→          version: item.version,
   602→          summary: item.summary,
   603→          contentSnippet: item.summary,
   604→          fullContent: item.content,
   605→          applicableDepartments: item.applicableDepartments,
   606→          relevanceScore,
   607→          matchReason: matchedTerms.length > 0
   608→            ? `Matched knowledge terms: ${[...new Set(matchedTerms)].join(', ')}`
   609→            : `Category match (${item.category})`,
   610→          provenance: {
   611→            sourceSystem: 'company_knowledge',
   612→            sourceId: `knowledge:${item.documentId}`,
   613→            sourceTitle: item.title,
   614→            epistemicType: 'durable_reference',
   615→            authority: item.authorAuthority,
   616→            timestamp: item.lastVerifiedDate || nowIso,
   617→            confidence: 'reference_standard',
   618→            notes: `Durable company reference documentation (v${item.version})`,
   619→          },
   620→        });
   621→      }
   622→    }
   623→
   624→    // Sort by relevance score descending
   625→    results.sort((a, b) => b.relevanceScore - a.relevanceScore);
   626→    const limit = params.limit || 5;
   627→    return results.slice(0, limit);
   628→  }
