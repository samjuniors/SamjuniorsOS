    40→function createAuthenticatedRequest(
    41→  method: 'POST' | 'GET',
    42→  url: string,
    43→  body?: Record<string, any>,
    44→  options: {
    45→    role?: 'FOUNDER' | 'EXECUTIVE' | 'AUDITOR';
    46→    founderId?: string;
    47→    email?: string;
    48→  } = {}
    49→): NextRequest {
    50→  const role = options.role || 'FOUNDER';
    51→  const founderId = options.founderId || (role === 'FOUNDER' ? 'founder_primary_001' : 'auditor_user_002');
    52→  const email = options.email || (role === 'FOUNDER' ? 'founder@samjuniors.os' : 'auditor@samjuniors.os');
    53→
    54→  const payload = JSON.stringify({
    55→    founderId,
    56→    email,
    57→    role,
    58→    createdAt: Date.now(),
    59→  });
    60→  const encoded = Buffer.from(payload).toString('base64');
    61→  const signature = Buffer.from(TEST_SECRET).toString('base64');
    62→  const sessionToken = `${encoded}.${signature}`;
    63→
    64→  const headers: Record<string, string> = {
    65→    'Content-Type': 'application/json',
    66→    'x-samjuniors-dev-as': sessionToken,
    67→    'x-samjuniors-dev-secret': TEST_SECRET,
    68→    'x-samjuniors-user-id': founderId,
    69→    'x-samjuniors-role': role,
    70→    cookie: `samjuniors_dev_session=${sessionToken}; samjuniors-role=${role}`,
    71→  };
    72→
    73→  return new NextRequest(url, {
    74→    method,
    75→    headers,
    76→    body: body ? JSON.stringify(body) : undefined,
    77→  });
    78→}
