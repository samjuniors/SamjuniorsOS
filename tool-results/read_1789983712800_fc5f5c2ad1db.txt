    38→/**
    39→ * The conversation.
    40→ *
    41→ * This used to be a sequential loop — greet, await a capture, await an answer,
    42→ * repeat — with the microphone opened and closed around each step. That shape
    43→ * cannot be interrupted: while it is awaiting the answer, nothing is listening,
    44→ * so there is no way for the user to get a word in.
    45→ *
    46→ * It is an event machine now. The voice loop runs continuously and pushes
    47→ * events at us; every one of them is legal in every phase. Saying anything at
    48→ * all stops him talking, and whatever you say next becomes the new turn.
    49→ */
    50→
    51→/** How long to wait for someone to start speaking after he wakes. Generous:
    52→ *  people say his name and *then* think about what they wanted. */
    53→const AWAIT_SPEECH_MS = 14000
    54→
    55→/** After an answer, how long the mic stays open for a follow-up before he
    56→ *  drops back to standby. Long enough that you don't have to say the name
    57→ *  again to continue a thought. */
    58→const FOLLOW_UP_MS = 11000
    59→
    60→/** crypto.randomUUID needs a secure context, which a LAN address over plain
    61→ *  http is not. Not worth failing a whole turn over an id. */
    62→const newId = () =>
    63→  globalThis.crypto?.randomUUID?.() ??
    64→  `id${Date.now().toString(36)}${Math.random().toString(36).slice(2, 10)}`
    65→
    66→/** The same mishearings voice.ts accepts for the wake word — otherwise a turn
    67→ *  that woke the machine as “travis” gets that word sent on to the model as a
    68→ *  question. Read fresh from the persona each time: the character can change
    69→ *  under a long-lived page, and a stale name would strip the wrong word. */
    70→const namePattern = () => personaById(useStore.getState().persona).heardAs
    71→/** A bare vocative — “Sofia”, “hey sofia” — with nothing asked. */
    72→const bareName = () =>
    73→  new RegExp(`^(?:hey|hi|ok|okay|yo)?\\s*(?:${namePattern()})[\\s,.!?]*$`, 'i')
    74→/** A leading vocative on a real command: “Sofia, what's the weather”. */
    75→const leadingName = () =>
    76→  new RegExp(`^(?:hey|hi|ok|okay|yo)?\\s*(?:${namePattern()})\\b[\\s,.:!?-]*`, 'i')

    78→export default function App() {
    79→  const store = useStore
    80→  const phase = useStore((s) => s.phase)
    81→  const persona = useStore((s) => s.persona)
    82→  const history = useRef<Msg[]>([])
    83→  const speaker = useRef<ReturnType<typeof createSpeaker> | null>(null)
    84→  const voice = useRef<Voice | null>(null)
    85→
    86→  // -- persona ------------------------------------------------------------
    87→
    88→  /**
    89→   * Everything that has to follow the character, gathered in one place so a
    90→   * switch is atomic: the wake word the recogniser hunts for, the filler
    91→   * lines it speaks, and the document title. The HUD, the dial and the boot
    92→   * screen subscribe to the store themselves.
    93→   */
    94→  useEffect(() => {
    95→    const p = personaById(persona)
    96→    setWakePersona(persona)
    97→    setFillerPersona(persona)
    98→    document.title = p.mark
    99→  }, [persona])
   100→
   101→  /**
   102→   * Monotonic turn counter. Every await in a turn checks it on the way out:
   103→   * if it has moved, that turn was superseded by a barge-in and must not touch
   104→   * the phase, the speaker, or the busy state on its way to the floor.
   105→   */
   106→  const turn = useRef(0)
   107→  const booting = useRef(false)
   108→  const idleTimer = useRef<ReturnType<typeof setTimeout> | null>(null)
   109→  const voicePoll = useRef<ReturnType<typeof setInterval> | null>(null)
   110→  /** Set by the voice loop's mic failure — the one error that means voice is
   111→   *  off for good, and so the one that must outlive startVoice returning. */
   112→  const micDead = useRef(false)
   113→
   114→  // -- helpers --------------------------------------------------------------
   115→
   116→  const clearIdle = () => {
   117→    if (idleTimer.current) clearTimeout(idleTimer.current)
   118→    idleTimer.current = null
   119→  }
   120→
   121→  const silence = () => {
   122→    speaker.current?.cancel()
   123→    speaker.current = null
   124→  }
   125→
   126→  const goDormant = () => {
   127→    clearIdle()
   128→    silence()
   129→    turn.current++
   130→    const s = store.getState()
   131→    s.setCaption('')
   132→    s.setActiveTool(null)
   133→    music.working(false)
   134→    music.duck(false)
   135→    sfx.duck(false)
   136→    s.setPhase('dormant')
   137→  }
   138→
   139→  /** Open the mic and wait. `window` is how long before he gives up. */
   140→  const listen = (window: number) => {
   141→    clearIdle()
   142→    const s = store.getState()
   143→    s.setCaption('')
   144→    s.setPhase('listening')
   145→    sfx.play('listen')
   146→    idleTimer.current = setTimeout(goDormant, window)
   147→  }
   148→
   149→  // -- one turn -------------------------------------------------------------
   150→
   151→  const respond = async (said: string): Promise<void> => {
   152→    const mine = ++turn.current
   153→    const stale = () => mine !== turn.current
   154→
   155→    clearIdle()
   156→    const s = store.getState()
   157→    // Last turn's panels and blades go now, before the new answer starts
   158→    // putting its own up. Anything the model marked sticky survives.
   159→    s.clearPanels()
   160→    s.clearBlades()
   161→    s.setCaption('')
   162→    s.pushTurn({ id: newId(), role: 'user', text: said })
   163→    s.setPhase('thinking')
   164→
   165→    const spk = createSpeaker()
   166→    speaker.current = spk
   167→    sfx.duck(true)
   168→    music.duck(true)
   169→
   170→    const turnId = newId()
   171→    let started = false
   172→    let filled = false
   173→
   174→    try {
   175→      const { text } = await ask(said, history.current, {
   176→        onText: (delta) => {
   177→          if (stale()) return
   178→          if (!started) {
   179→            started = true
   180→            store.getState().setPhase('speaking')
   181→            // The answer arriving is what ends the tool phase — a timer would
   182→            // clear the readout while a slow tool was still running.
   183→            store.getState().setActiveTool(null)
   184→            music.working(false)
   185→            store.getState().pushTurn({ id: turnId, role: 'jarvis', text: '' })
   186→          }
   187→          store.getState().appendToLastTurn(delta)
   188→          spk.push(delta)
   189→        },
   190→        onTool: (name) => {
   191→          if (stale()) return
   192→          // Only claim the tooling phase while he has nothing to say yet.
   193→          // Setting it unconditionally pinned the machine in 'tooling' for the
   194→          // rest of any answer that called a tool after it started talking,
   195→          // which also broke the reactor's lip-sync for the remainder.
   196→          if (!started) store.getState().setPhase('tooling')
   197→          store.getState().setActiveTool(name)
   198→          sfx.play('tool')
   199→          music.working(true)
   200→          // Say something the moment work starts — a tool can take ten seconds
   201→          // and silence that long reads as a crash. Once per turn only; a
   202→          // chain of five tools shouldn't produce five apologies.
   203→          if (!filled && !started) {
   204→            filled = true
   205→            spk.say(forTool(name))
   206→          }
   207→        },
   208→        // The character the brain should answer as — re-read per turn so a
   209→        // mid-session persona switch takes effect on the very next question.
   210→      }, store.getState().persona)
   211→
   212→      if (stale()) return
   213→
   214→      // The bridge keeps conversation state in its own session, so history is
   215→      // only threaded through on the direct path.
   216→      if (!usingBridge) {
   217→        history.current.push({ role: 'user', content: said })
   218→        history.current.push({ role: 'assistant', content: text || '…' })
   219→        if (history.current.length > 16) {
   220→          history.current = history.current.slice(-16)
   221→        }
   222→      }
   223→
   224→      await spk.end()
   225→      if (stale()) return
   226→      sfx.play('done')
   227→    } catch (err) {
   228→      if (stale()) return
   229→      console.error(err)
   230→      sfx.play('error')
   231→      store
   232→        .getState()
   233→        .setError(err instanceof Error ? err.message : 'Something went wrong.')
   234→    } finally {
   235→      if (!stale()) {
   236→        speaker.current = null
   237→        sfx.duck(false)
   238→        music.duck(false)
   239→        store.getState().setActiveTool(null)
   240→        music.working(false)
   241→        // Stay open. Having to say his name again to add one more sentence is
   242→        // the difference between a conversation and a vending machine.
   243→        listen(FOLLOW_UP_MS)
   244→      }
   245→    }
   246→  }
   247→
   248→  // -- voice events ---------------------------------------------------------
   249→
   250→  /** What the voice loop should do with what it hears, derived from phase. */
   251→  const mode = (): VoiceMode => {
   252→    switch (store.getState().phase) {
   253→      case 'offline':
   254→      case 'boot':
   255→        return 'deaf'
   256→      case 'dormant':
   257→        return 'wake'
   258→      case 'waking':
   259→      case 'listening':
   260→        return 'command'
   261→      default:
   262→        return 'guard' // thinking, tooling, speaking
   263→    }
   264→  }
   265→
   266→  const onWake = (trailing: string) => {
   267→    const phase = store.getState().phase
   268→    if (phase === 'offline' || phase === 'boot') return
   269→
   270→    store.getState().setError(null)
   271→    sfx.play('wake')
   272→
   273→    // "Jarvis, what's happening in AI this week" in one breath. Waiting for a
   274→    // greeting he didn't need is the most common way an assistant wastes time.
   275→    if (trailing) {
   276→      void respond(trailing)
   277→      return
   278→    }
   279→
   280→    store.getState().setPhase('waking')
   281→
   282→    // Answer to his name. Deliberately NOT awaited any more: the microphone is
   283→    // already open and the echo filter knows his voice, so the user can talk
   284→    // straight over the greeting instead of waiting it out.
   285→    const greeting = createSpeaker()
   286→    speaker.current = greeting
   287→    greeting.say(attention())
   288→    void greeting.end()
   289→
   290→    listen(AWAIT_SPEECH_MS)
   291→  }
   292→
   293→  /**
   294→   * Someone started talking. This is the whole point of the rewrite: he stops,
   295→   * immediately, whatever he was doing.
   296→   */
   297→  const onSpeechStart = () => {
   298→    clearIdle()
   299→    const phase = store.getState().phase
   300→    if (phase === 'offline' || phase === 'boot' || phase === 'dormant') return
   301→
   302→    const wasBusy =
   303→      phase === 'thinking' || phase === 'tooling' || phase === 'speaking'
   304→
   305→    silence()
   306→    if (wasBusy) {
   307→      // Abandon the answer in flight. The turn counter moves in respond()'s
   308→      // replacement; bumping it here covers the case where nothing replaces it.
   309→      turn.current++
   310→      interrupt()
   311→      store.getState().setActiveTool(null)
   312→      music.working(false)
   313→      sfx.duck(false)
   314→      music.duck(false)
   315→    }
   316→    store.getState().setPhase('listening')
   317→  }
   318→
   319→  const onUtterance = (text: string) => {
   320→    const phase = store.getState().phase
   321→    if (phase === 'offline' || phase === 'boot' || phase === 'dormant') return
   322→
   323→    // People keep using the name as a vocative once they're already talking to
   324→    // the machine. Strip it rather than sending "sofia" to the model as a
   325→    // question.
   326→    if (bareName().test(text)) {
   327→      listen(AWAIT_SPEECH_MS)
   328→      return
   329→    }
   330→    const said = text.replace(leadingName(), '').trim()
   331→    if (!said) {
   332→      listen(AWAIT_SPEECH_MS)
   333→      return
   334→    }
   335→
   336→    void respond(said)
   337→  }
   338→
   339→  /**
   340→   * The typed command line. A `jarvis:say` event is the keyboard's version of
   341→   * a recognised utterance: same road, same rules. Idle, it starts a turn;
   342→   * busy, it barge-ins exactly like speaking over him does — the in-flight
   343→   * answer is interrupted, its turn goes stale, and this one replaces it.
   344→   */
   345→  const onSay = (e: Event) => {
   346→    const said = String((e as CustomEvent<string>).detail ?? '').trim()
   347→    if (!said) return
   348→
   349→    const phase = store.getState().phase
   350→    if (phase === 'boot') return
   351→
   352→    if (phase === 'offline') {
   353→      // The ignition screen is still up; boot first, then ask the moment the
   354→      // interface lands.
   355→      void powerOn().then(() => respond(said))
   356→      return
   357→    }
   358→
   359→    if (phase === 'thinking' || phase === 'tooling' || phase === 'speaking') {
   360→      silence()
   361→      interrupt()
   362→    }
   363→    void respond(said)
   364→  }
   365→
   366→  const onPartial = (text: string) => {
   367→    store.getState().setCaption(text)
   368→  }
   369→
   370→  const onVoiceError = (message: string) => {
   371→    store.getState().setError(message)
   372→    // The two mic failures are the only ones that mean voice is off for good
   373→    // — transient recogniser hiccups fire this too, and must not flip the
   374→    // standby hint to “type a command”. The ref survives startVoice returning
   375→    // its dead object, so the optimistic setVoiceLive(true) after it cannot
   376→    // overwrite what happened here.
   377→    if (/microphone/i.test(message)) {
   378→      micDead.current = true
   379→      store.getState().setVoiceLive(false)
   380→    }
   381→  }
   382→
   383→  // -- power on -------------------------------------------------------------
   384→
   385→  const powerOn = async () => {
   386→    // The ignition button and the space bar can both land here, and the phase
   387→    // only moves after the first await — so without this a double press boots
   388→    // twice, arming two voice loops and two download polls.
   389→    if (booting.current) return
   390→    booting.current = true
   391→
   392→    try {
   393→      await ignite()
   394→    } catch (err) {
   395→      // The guard must not outlive a failed boot. Audio unlock can be refused,
   396→      // the microphone prompt dismissed, the bridge unreachable at the wrong
   397→      // moment — and with the flag still latched the ignition button was dead
   398→      // for the rest of the page, recoverable only by reloading. Reset it and
   399→      // put the button back so the user can simply press it again.
   400→      booting.current = false
   401→      console.error('[jarvis] power-up failed:', err)
   402→      store.getState().setPhase('offline')
   403→      store
   404→        .getState()
   405→        .setError(
   406→          err instanceof Error
   407→            ? `Power-up failed: ${err.message}`
   408→            : 'Power-up failed. Click to try again.',
   409→        )
   410→    }
   411→  }
   412→
   413→  const ignite = async () => {
   414→    const s = store.getState()
   415→
   416→    // Must happen inside the click handler — browsers won't start an
   417→    // AudioContext or speech synthesis without a user gesture.
   418→    await sfx.unlockAudio()
   419→    sfx.play('boot')
   420→    // The score. Must be started from inside this click handler for the same
   421→    // reason as the rest of the audio.
   422→    music.enable()
   423→    music.playBoot()
   424→    music.startAmbient()
   425→
   426→    s.setPhase('boot')
   427→
   428→    watchServers((servers) => store.getState().setConnected(servers))
   429→    // Which link of the chain is answering — announced on connect and on
   430→    // every mid-session failover, so the rail's BRAIN line is always the
   431→    // brain that is actually speaking.
   432→    watchBrain((b) => store.getState().setBrain(b))
   433→    watchPanels((panel) => store.getState().pushPanel(panel))
   434→    watchBlades((blade) => store.getState().pushBlade(blade))
   435→
   436→    /**
   437→     * JARVIS asking to see something.
   438→     *
   439→     * Announced on screen for as long as it takes, with whatever he said he was
   440→     * looking for. The camera's own light is on too, but a hardware light that
   441→     * appears with no explanation is exactly the thing that makes people
   442→     * distrust an assistant — so the interface says it before they have to ask.
   443→     */
   444→    watchCapture(async (req) => {
   445→      const note =
   446→        req.mode === 'watch'
   447→          ? req.when === 'past'
   448→            ? req.reason || 'reviewing the last few seconds'
   449→            : `${req.reason || 'watching'} · ${req.seconds}s`
   450→          : req.reason || 'taking a look'
   451→      store.getState().setLooking(note)
   452→
   453→      // The past is only available if something has been remembering it, and
   454→      // that only happens while the camera is on screen. Answering plainly
   455→      // beats opening the camera and recording the next few seconds instead,
   456→      // which is a different question from the one that was asked.
   457→      if (req.mode === 'watch' && req.when === 'past' && camera.bufferedSeconds() < 1) {
   458→        store.getState().setLooking(null)
   459→        return {
   460→          error:
   461→            'There is no recent footage — the camera has to be open on screen ' +
   462→            'for me to remember what just happened. Ask me to open the camera, ' +
   463→            'and I can watch from then on.',
   464→        }
   465→      }
   466→
   467→      // Held for the whole capture. Without this the stream can be torn down by
   468→      // whoever else was using it half way through a six-second watch.
   469→      let held = false
   470→      try {
   471→        await camera.holdCamera()
   472→        held = true
   473→        if (req.mode === 'look') return camera.grabFrame()
   474→        if (req.when === 'past') {
   475→          const grid = camera.recentGrid(req.seconds, 9)
   476→          return grid ?? { error: 'There is not enough recent footage to review.' }
   477→        }
   478→        return await camera.watchAhead(req.seconds, 9)
   479→      } catch (err) {
   480→        return {
   481→          error:
   482→            (err as DOMException)?.name === 'NotAllowedError'
   483→              ? 'The camera is not permitted, so I cannot see anything.'
   484→              : `The camera could not be read: ${(err as Error)?.message ?? err}`,
   485→        }
   486→      } finally {
   487→        if (held) camera.releaseCamera()
   488→        store.getState().setLooking(null)
   489→      }
   490→    })
   491→
   492→    // The interface is JARVIS's to drive. These arrive out of band, pushed
   493→    // mid-turn the way panels are, so a command can retint the reactor or put
   494→    // something into orbit while he is still speaking the sentence about it.
   495→    watchUi((op, args) => {
   496→      const s = store.getState()
   497→      const a = (args ?? {}) as Record<string, never>
   498→      switch (op) {
   499→        case 'patch':
   500→          s.applyUi(args)
   501→          break
   502→        case 'orbit':
   503→          if (a.action === 'add') s.addOrbit(args)
   504→          else if (a.action === 'remove') s.removeOrbit(String(a.id))
   505→          else s.clearOrbits()
   506→          break
   507→        case 'effect':
   508→          s.fireEffect(a.kind)
   509→          break
   510→        case 'reset':
   511→          s.resetUi()
   512→          break
   513→        case 'screen':
   514→          s.clearScreen(a.what ?? 'all')
   515→          break
   516→        default:
   517→          console.warn('[jarvis] unknown ui op:', op, args)
   518→      }
   519→    })
   520→    // In bridge mode the conversation lives in the agent session, which is tied
   521→    // to the socket — so a drop silently wipes his memory while the transcript
   522→    // on screen still shows it. Better to say so than to let him quietly forget.
   523→    watchConnection((state) => {
   524→      if (state === 'lost') {
   525→        store.getState().setError('Bridge connection lost — reconnecting.')
   526→      } else if (state === 'reconnected') {
   527→        store
   528→          .getState()
   529→          .setError('Bridge reconnected. The previous conversation was not kept.')
   530→      }
   531→    })
   532→    const warming = warm().catch((err: Error) => s.setError(err.message))
   533→
   534→    if (!usingBridge && !env.anthropicKey) {
   535→      s.setError(
   536→        'No Anthropic API key — copy .env.example to .env.local and set VITE_ANTHROPIC_API_KEY.',
   537→      )
   538→    }
   539→
   540→    // Pull the neural voice down during the boot sequence so the first
   541→    // "Hey Jarvis" isn't waiting on an 86MB download. Deliberately not awaited
   542→    // — if it's slow, JARVIS comes up on the system voice and swaps over the
   543→    // moment the model is ready.
   544→    if (TTS_ENGINE === 'kokoro') {
   545→      void kokoro.load()
   546→      voicePoll.current = setInterval(() => {
   547→        const p = kokoro.loadProgress()
   548→        if (kokoro.isReady() || kokoro.isUnavailable()) {
   549→          store.getState().setBootNote('')
   550→          if (voicePoll.current) clearInterval(voicePoll.current)
   551→          voicePoll.current = null
   552→        } else if (p > 0 && p < 1) {
   553→          store.getState().setBootNote(`voice ${Math.round(p * 100)}%`)
   554→        }
   555→      }, 200)
   556→    }
   557→
   558→    // No boot screen any more — the initializing screen was the last gate, and
   559→    // the interface lands the moment the click lands. Everything below is the
   560→    // live machine coming up: the warm-up, the mic analyser, the speech probe
   561→    // and the voice loop, in that order because each is cheap enough not to be
   562→    // worth a screen of its own.
   563→    await warming
   564→    store.getState().setConnected(connectedLabels())
   565→
   566→    // The analyser is what makes the reactor pulse with your voice. It needs a
   567→    // getUserMedia stream; speech recognition does not, and gets its own. So a
   568→    // failure here costs the animation and nothing else — saying "voice input
   569→    // is unavailable" was both alarming and untrue.
   570→    try {
   571→      await startAnalyser()
   572→    } catch {
   573→      console.warn(
   574→        '[jarvis] no microphone stream — the reactor will not pulse with your ' +
   575→          'voice. Speech recognition is unaffected.',
   576→      )
   577→    }
   578→
   579→    // Ask the bridge which speech engines exist before the loop starts, so the
   580→    // first turn already uses ElevenLabs when a key is present and the browser
   581→    // fallback when it is not — no flag, no reload.
   582→    await probeCapabilities()
   583→
   584→    // Only now can the HUD name the voice honestly: the label depends on the
   585→    // probe (neural catalogue vs system inventory). Reading it before the
   586→    // probe landed showed "default" for the whole session.
   587→    store.getState().setVoice(currentVoiceName())
   588→
   589→    // One voice loop, started once, running until the page closes.
   590→    voice.current = await startVoice({
   591→      mode,
   592→      onWake,
   593→      onSpeechStart,
   594→      onPartial,
   595→      onUtterance,
   596→      onError: onVoiceError,
   597→    })
   598→
   599→    // The loop is live unless the mic failed on the way up — a denied mic
   600→    // reports itself through onError before this line runs, and that verdict
   601→    // is the one that must stick.
   602→    if (!micDead.current) store.getState().setVoiceLive(true)
   603→
   604→    store.getState().setPhase('dormant')
   605→  }
   606→
   607→  // -- clap to wake ------------------------------------------------------------
   608→
   609→  /**
   610→   * A clap brings her up — from the ignition screen, and from standby.
   611→   *
   612→   * Two places it can land, and both are "the machine is idle and dark": while
   613→   * the initializing screen is showing (the click alternative), and once the
   614→   * HUD is standing by (the "hey sofia" alternative). Any busier than that —
   615→   * waking, listening, thinking, speaking — and a clap is either redundant or
   616→   * a stray interruption, so the listener is stood down instead.
   617→   *
   618→   * The microphone stream is shared with the voice loop, so arming here can
   619→   * never steal the assistant's hearing; it is torn down on every phase change
   620→   * and re-armed when the machine goes idle again.
   621→   *
   622→   * clapLive tracks whether the detector actually exists — a pane that blocks
   623→   * getUserMedia gets an honest hint instead of a promise a clap can't keep.
   624→   */
   625→  useEffect(() => {
   626→    if (phase !== 'offline' && phase !== 'dormant') return
   627→    let live: { stop: () => void } | null = null
   628→    let gone = false
   629→    void listenForClap(() => {
   630→      if (gone) return
   631→      const now = store.getState().phase
   632→      if (now === 'offline') {
   633→        void powerOn()
   634→      } else if (now === 'dormant') {
   635→        onWake('')
   636→      }
   637→    }).then((l) => {
   638→      if (gone) {
   639→        l?.stop()
   640→      } else {
   641→        live = l
   642→        if (l) store.getState().setClapLive(true)
   643→      }
   644→    })
   645→    return () => {
   646→      gone = true
   647→      live?.stop()
   648→      store.getState().setClapLive(false)
   649→    }
   650→    // eslint-disable-next-line react-hooks/exhaustive-deps
   651→  }, [phase])
   652→
   653→  // -- level pump + keys ----------------------------------------------------
   654→
   655→  useEffect(() => {
   656→    let raf = 0
   657→
   658→    const pump = () => {
   659→      const st = store.getState()
   660→      // While speaking, follow JARVIS's own output rather than the mic, so the
   661→      // orb lip-syncs instead of reacting to room noise.
   662→      const lvl =
   663→        st.phase === 'speaking' && speaker.current
   664→          ? speaker.current.level()
   665→          : micLevel()
   666→      st.setLevel(lvl)
   667→      raf = requestAnimationFrame(pump)
   668→    }
   669→    pump()
   670→
   671→    const onKey = (e: KeyboardEvent) => {
   672→      const tag = (e.target as HTMLElement)?.tagName
   673→      if (tag === 'INPUT' || tag === 'TEXTAREA') return
   674→
   675→      // V auditions the next British voice installed on this machine. Which
   676→      // ones exist varies per Mac, so hearing them beats trusting a ranking.
   677→      // Bare V only — ⌘V and ⌃V are paste, and swallowing those was rude.
   678→      if (
   679→        e.key === 'v' &&
   680→        !e.repeat &&
   681→        !e.metaKey &&
   682→        !e.ctrlKey &&
   683→        !e.altKey
   684→      ) {
   685→        e.preventDefault()
   686→        const name = cycleVoice()
   687→        store.getState().setVoice(name)
   688→        silence()
   689→        const demo = createSpeaker()
   690→        speaker.current = demo
   691→        demo.say(personaById(store.getState().persona).fillers.voiceSet)
   692→        void demo.end()
   693→        return
   694→      }
   695→
   696→      // G puts the camera on and starts tracking hands. Off by default and
   697→      // never implicit: a webcam that turns itself on because an interface
   698→      // thought it might be useful is not a trade anyone agreed to.
   699→      if (e.key === 'g' && !e.repeat && !e.metaKey && !e.ctrlKey && !e.altKey) {
   700→        e.preventDefault()
   701→        const on = store.getState().gestures
   702→        if (on) {
   703→          hands.disableHands()
   704→          store.getState().setGestures(false)
   705→        } else {
   706→          store.getState().setError(null)
   707→          void hands
   708→            .enableHands()
   709→            .then(() => store.getState().setGestures(true))
   710→            .catch((err: Error) => {
   711→              store.getState().setGestures(false)
   712→              store
   713→                .getState()
   714→                .setError(
   715→                  err?.name === 'NotAllowedError'
   716→                    ? 'Camera access denied — gesture control is unavailable.'
   717→                    : `Gesture control failed to start: ${err?.message ?? err}`,
   718→                )
   719→            })
   720→        }
   721→        return
   722→      }
   723→
   724→      // T speaks a fixed line, bypassing the wake word, the recogniser and the
   725→      // model entirely. When "I can't hear him" is the report, this is the one
   726→      // keypress that separates a broken voice engine from a broken voice loop
   727→      // — and it prints the verdict rather than making you infer it.
   728→      if (e.key === 't' && !e.repeat && !e.metaKey && !e.ctrlKey && !e.altKey) {
   729→        e.preventDefault()
   730→        silence()
   731→        const t = createSpeaker()
   732→        speaker.current = t
   733→        t.say('Audio test. If you can hear this, speech output is working, sir.')
   734→        void t.end().then(() => {
   735→          const d = (window as unknown as Record<string, Record<string, unknown>>).__tts
   736→          console.info('[jarvis] audio test →', d)
   737→          if (d && d.started === 0 && d.rescued === 0) {
   738→            store.getState().setError(
   739→              `No sound produced. engine=${d.engine} voice=${d.voice} error=${d.lastError || 'none'}`,
   740→            )
   741→          }
   742→        })
   743→        return
   744→      }
   745→
   746→      // Escape stands the whole thing down — the one thing the old build had
   747→      // no key for at all.
   748→      if (e.key === 'Escape') {
   749→        e.preventDefault()
   750→        if (store.getState().phase !== 'offline') goDormant()
   751→        return
   752→      }
   753→
   754→      // Space starts a turn without the wake word. Worth using while filming so
   755→      // a missed wake word doesn't cost a take.
   756→      if (e.code !== 'Space' || e.repeat) return
   757→      e.preventDefault()
   758→
   759→      const phase = store.getState().phase
   760→      if (phase === 'offline') {
   761→        void powerOn()
   762→      } else if (phase === 'boot') {
   763→        /* ignore — the boot sequence owns the phase until it finishes */
   764→      } else if (
   765→        phase === 'thinking' ||
   766→        phase === 'tooling' ||
   767→        phase === 'speaking'
   768→      ) {
   769→        onSpeechStart()
   770→        listen(AWAIT_SPEECH_MS)
   771→      } else {
   772→        onWake('')
   773→      }
   774→    }
   775→    window.addEventListener('keydown', onKey)
   776→    window.addEventListener('jarvis:say', onSay as EventListener)
   777→
   778→    return () => {
   779→      cancelAnimationFrame(raf)
   780→      window.removeEventListener('keydown', onKey)
   781→      window.removeEventListener('jarvis:say', onSay as EventListener)
   782→      clearIdle()
   783→      if (voicePoll.current) clearInterval(voicePoll.current)
   784→      voice.current?.stop()
   785→      speaker.current?.cancel()
   786→      // The camera must not outlive the page that turned it on.
   787→      hands.disableHands()
   788→    }
   789→    // eslint-disable-next-line react-hooks/exhaustive-deps
   790→  }, [])
   791→
   792→  return (
   793→    <>
   794→      <Scene />
   795→      <Hud />
   796→      <Diagnostics />
   797→      <CommandLine />
   798→      <Ignition onStart={() => void powerOn()} />
   799→    </>
   800→  )
   801→}
