    59→    59→/**
    60→    60→ * M5.2 — deterministic lexical fact scoring, shared by both store modes so
    61→    61→ * local and authoritative retrieval rank identically. Uses the SAME shared
    62→    62→ * extractTokens/stop-word pipeline as CompanyKnowledgeStore.queryKnowledge
    63→    63→ * (M4-C convention — no second tokenizer). Subject identifiers tokenize
    64→    64→ * with underscores as spaces ('lumora_pricing' → 'lumora pricing') so a
    65→    65→ * stored subject remains reachable from natural-language queries.
    66→    66→ * Pure function; zero-overlap facts score 0.
    67→    67→ */
    68→    68→export function scoreFactAgainstQuery(fact: CanonicalFact, queryText: string): number {
    69→    69→  const queryTokens = new Set(extractTokens(queryText));
    70→    70→  if (queryTokens.size === 0) return 0;
    71→    71→  const factTokens = new Set([
    72→    72→    ...extractTokens(fact.statement),
    73→    73→    ...extractTokens((fact.subject || '').replace(/_/g, ' ')),
    74→    74→  ]);
    75→    75→  let matched = 0;
    76→    76→  for (const token of queryTokens) {
    77→    77→    if (factTokens.has(token)) matched++;
    78→    78→  }
    79→    79→  return matched;
    80→    80→}
    81→    81→
    82→    82→/**
    83→    83→ * M5.2 — deterministic query-conditioned fact ranking (shared by both
    84→    84→ * modes): score DESC, then promotedAt DESC, then id ASC (total stable order).
    85→    85→ * Zero-score facts are excluded (a fact that shares no token with the query
    86→    86→ * is not query-relevant; the CALLER applies the documented never-empty
    87→    87→ * fallback when every candidate scores zero).
    88→    88→ */
    89→    89→export function rankFactsForQuery(
    90→    90→  facts: CanonicalFact[],
    91→    91→  queryText: string,
    92→    92→  limit: number
    93→    93→): CanonicalFact[] {
    94→    94→  return facts
    95→    95→    .map((fact) => ({ fact, score: scoreFactAgainstQuery(fact, queryText) }))
    96→    96→    .filter((entry) => entry.score > 0)
    97→    97→    .sort((a, b) => {
    98→    98→      if (b.score !== a.score) return b.score - a.score;
    99→    99→      const ta = Date.parse(a.fact.promotedAt || '') || 0;
   100→   100→      const tb = Date.parse(b.fact.promotedAt || '') || 0;
   101→   101→      if (tb !== ta) return tb - ta;
   102→   102→      return a.fact.id < b.fact.id ? -1 : a.fact.id > b.fact.id ? 1 : 0;
   103→   103→    })
   104→   104→    .slice(0, limit)
   105→   105→    .map((entry) => entry.fact);
   106→   106→}

   465→   465→      subject: f.subject,
   466→   466→      category: f.category as any,
   467→   467→      validityState: f.validityState as any,
   468→   468→      supersededById: f.supersededById || undefined,
   469→   469→      confidence: f.confidence as any,
   470→   470→      promotedAt: f.promotedAt.toISOString(),
   471→   471→      promotedBy: f.promotedBy,
   472→   472→      provenance: (f.provenance as any) || {},
   473→   473→    };
   474→   474→  }

   506→   506→      const sourcesDisk = fileStore.readCollection<EvidenceSource>('epistemic_sources');

   514→   514→

   651→   651→    if (filter?.category) {

   654→   654→    if (filter?.subject) {
   655→   655→      result = result.filter((f) => f.subject.toLowerCase() === filter.subject!.toLowerCase());
   656→   656→    }

   658→   658→  }
   659→   659→
   660→   660→  /**
   661→   661→   * M5.2 — every fact regardless of validityState (local mode). Ordered
   662→   662→   * promotedAt DESC, id ASC — identical order contract to the authoritative
   663→   663→   * mode. Superseded facts retain their supersededById pointers.
   664→   664→   */
   665→   665→  public async listAllFacts(): Promise<CanonicalFact[]> {
   666→   666→    if (isAuthoritativeMode()) {
   667→   667→      return PostgresEpistemicStore.getInstance().listAllFacts();
   668→   668→    }
   669→   669→    return Array.from(this.facts.values())
   670→   670→      .sort((a, b) => {
   671→   671→        if (b.promotedAt !== a.promotedAt) return b.promotedAt.localeCompare(a.promotedAt);
   672→   672→        return a.id < b.id ? -1 : a.id > b.id ? 1 : 0;
   673→   673→      })
   674→   674→      .map((f) => ({ ...f }));
   675→   675→  }
   676→   676→
   677→   677→  /**
   678→   678→   * M5.2 — query-conditioned canonical-fact retrieval (local mode).
   679→   679→   * Deterministic lexical ranking over the shared tokenizer (see
   680→   680→   * rankFactsForQuery). Zero-overlap facts are excluded; the CALLER applies
   681→   681→   * the documented never-empty fallback (A0 top-3-newest) when nothing
   682→   682→   * matches. includeSuperseded adds superseded facts to the candidate pool
   683→   683→   * WITHOUT changing their lifecycle state or truth position.
   684→   684→   */
   685→   685→  public async queryFacts(params: FactQueryParams): Promise<CanonicalFact[]> {
   686→   686→    if (isAuthoritativeMode()) {
   687→   687→      return PostgresEpistemicStore.getInstance().queryFacts(params);
   688→   688→    }
   689→   689→    const pool = Array.from(this.facts.values()).filter((f) =>
   690→   690→      params.includeSuperseded
   691→   691→        ? f.validityState === 'active' || f.validityState === 'superseded'
   692→   692→        : f.validityState === 'active'
   693→   693→    );
   694→   694→    return rankFactsForQuery(pool, params.queryText, params.limit ?? 5);
   695→   695→  }

   697→   697→  public async markFactSuperseded(factId: string, supersededById: string): Promise<void> {
   698→   698→    if (isAuthoritativeMode()) {
   699→   699→      return PostgresEpistemicStore.getInstance().markFactSuperseded(factId, supersededById);
   700→   700→    }
   701→   701→    const fact = this.facts.get(factId);
   702→   702→    if (fact) {
   703→   703→      fact.validityState = 'superseded';
   704→   704→      fact.supersededById = supersededById;
   705→   705→      await this.saveFact(fact);
   706→   706→    }
   707→   707→  }
