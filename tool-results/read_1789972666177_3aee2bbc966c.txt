    38→    38→/**
    39→    39→ * The conversation.
    40→    40→ *
    41→    41→ * This used to be a sequential loop — greet, await a capture, await an answer,
    42→    42→ * repeat — with the microphone opened and closed around each step. That shape
    43→    43→ * cannot be interrupted: while it is awaiting the answer, nothing is listening,
    44→    44→ * so there is no way for the user to get a word in.
    45→    45→ *
    46→    46→ * It is an event machine now. The voice loop runs continuously and pushes
    47→    47→ * events at us; every one of them is legal in every phase. Saying anything at
    48→    48→ * all stops him talking, and whatever you say next becomes the new turn.
    49→    49→ */
    50→    50→
    51→    51→/** How long to wait for someone to start speaking after he wakes. Generous:
    52→    52→ *  people say his name and *then* think about what they wanted. */
    53→    53→const AWAIT_SPEECH_MS = 14000
    54→    54→
    55→    55→/** After an answer, how long the mic stays open for a follow-up before he
    56→    56→ *  drops back to standby. Long enough that you don't have to say the name
    57→    57→ *  again to continue a thought. */
    58→    58→const FOLLOW_UP_MS = 11000
    59→    59→
    60→    60→/** crypto.randomUUID needs a secure context, which a LAN address over plain
    61→    61→ *  http is not. Not worth failing a whole turn over an id. */
    62→    62→const newId = () =>
    63→    63→  globalThis.crypto?.randomUUID?.() ??
    64→    64→  `id${Date.now().toString(36)}${Math.random().toString(36).slice(2, 10)}`
    65→    65→
    66→    66→/** The same mishearings voice.ts accepts for the wake word — otherwise a turn
    67→    67→ *  that woke the machine as “travis” gets that word sent on to the model as a
    68→    68→ *  question. Read fresh from the persona each time: the character can change
    69→    69→ *  under a long-lived page, and a stale name would strip the wrong word. */
    70→    70→const namePattern = () => personaById(useStore.getState().persona).heardAs
    71→    71→/** A bare vocative — “Sofia”, “hey sofia” — with nothing asked. */
    72→    72→const bareName = () =>
    73→    73→  new RegExp(`^(?:hey|hi|ok|okay|yo)?\\s*(?:${namePattern()})[\\s,.!?]*$`, 'i')
    74→    74→/** A leading vocative on a real command: “Sofia, what's the weather”. */
    75→    75→const leadingName = () =>
    76→    76→  new RegExp(`^(?:hey|hi|ok|okay|yo)?\\s*(?:${namePattern()})\\b[\\s,.:!?-]*`, 'i')

   139→   139→  /** Open the mic and wait. `window` is how long before he gives up. */

   142→   142→    const s = store.getState()

   144→   144→    s.setPhase('listening')

   149→   149→  // -- one turn -------------------------------------------------------------
   150→   150→
   151→   151→  const respond = async (said: string): Promise<void> => {
   152→   152→    const mine = ++turn.current
   153→   153→    const stale = () => mine !== turn.current
   154→   154→
   155→   155→    clearIdle()
   156→   156→    const s = store.getState()
   157→   157→    // Last turn's panels and blades go now, before the new answer starts
   158→   158→    // putting its own up. Anything the model marked sticky survives.
   159→   159→    s.clearPanels()
   160→   160→    s.clearBlades()
   161→   161→    s.setCaption('')
   162→   162→    s.pushTurn({ id: newId(), role: 'user', text: said })
   163→   163→    s.setPhase('thinking')
   164→   164→
   165→   165→    const spk = createSpeaker()
   166→   166→    speaker.current = spk
   167→   167→    sfx.duck(true)
   168→   168→    music.duck(true)
   169→   169→
   170→   170→    const turnId = newId()
   171→   171→    let started = false
   172→   172→    let filled = false
   173→   173→
   174→   174→    try {
   175→   175→      const { text } = await ask(said, history.current, {
   176→   176→        onText: (delta) => {
   177→   177→          if (stale()) return
   178→   178→          if (!started) {
   179→   179→            started = true
   180→   180→            store.getState().setPhase('speaking')
   181→   181→            // The answer arriving is what ends the tool phase — a timer would
   182→   182→            // clear the readout while a slow tool was still running.
   183→   183→            store.getState().setActiveTool(null)
   184→   184→            music.working(false)
   185→   185→            store.getState().pushTurn({ id: turnId, role: 'jarvis', text: '' })
   186→   186→          }
   187→   187→          store.getState().appendToLastTurn(delta)
   188→   188→          spk.push(delta)
   189→   189→        },
   190→   190→        onTool: (name) => {
   191→   191→          if (stale()) return
   192→   192→          // Only claim the tooling phase while he has nothing to say yet.
   193→   193→          // Setting it unconditionally pinned the machine in 'tooling' for the
   194→   194→          // rest of any answer that called a tool after it started talking,
   195→   195→          // which also broke the reactor's lip-sync for the remainder.
   196→   196→          if (!started) store.getState().setPhase('tooling')
   197→   197→          store.getState().setActiveTool(name)
   198→   198→          sfx.play('tool')
   199→   199→          music.working(true)
   200→   200→          // Say something the moment work starts — a tool can take ten seconds
   201→   201→          // and silence that long reads as a crash. Once per turn only; a
   202→   202→          // chain of five tools shouldn't produce five apologies.
   203→   203→          if (!filled && !started) {
   204→   204→            filled = true
   205→   205→            spk.say(forTool(name))
   206→   206→          }
   207→   207→        },
   208→   208→        // The character the brain should answer as — re-read per turn so a
   209→   209→        // mid-session persona switch takes effect on the very next question.
   210→   210→      }, store.getState().persona)
   211→   211→
   212→   212→      if (stale()) return
   213→   213→
   214→   214→      // The bridge keeps conversation state in its own session, so history is
   215→   215→      // only threaded through on the direct path.
   216→   216→      if (!usingBridge) {
   217→   217→        history.current.push({ role: 'user', content: said })
   218→   218→        history.current.push({ role: 'assistant', content: text || '…' })
   219→   219→        if (history.current.length > 16) {
   220→   220→          history.current = history.current.slice(-16)
   221→   221→        }
   222→   222→      }
   223→   223→
   224→   224→      await spk.end()
   225→   225→      if (stale()) return
   226→   226→      sfx.play('done')
   227→   227→    } catch (err) {
   228→   228→      if (stale()) return
   229→   229→      console.error(err)
   230→   230→      sfx.play('error')
   231→   231→      store
   232→   232→        .getState()
   233→   233→        .setError(err instanceof Error ? err.message : 'Something went wrong.')
   234→   234→    } finally {
   235→   235→      if (!stale()) {
   236→   236→        speaker.current = null
   237→   237→        sfx.duck(false)
   238→   238→        music.duck(false)
   239→   239→        store.getState().setActiveTool(null)
   240→   240→        music.working(false)
   241→   241→        // Stay open. Having to say his name again to add one more sentence is
   242→   242→        // the difference between a conversation and a vending machine.
   243→   243→        listen(FOLLOW_UP_MS)
   244→   244→      }
   245→   245→    }
   246→   246→  }
   247→   247→
   248→   248→  // -- voice events ---------------------------------------------------------
   249→   249→
   250→   250→  /** What the voice loop should do with what it hears, derived from phase. */
   251→   251→  const mode = (): VoiceMode => {
   252→   252→    switch (store.getState().phase) {
   253→   253→      case 'offline':
   254→   254→      case 'boot':
   255→   255→        return 'deaf'
   256→   256→      case 'dormant':
   257→   257→        return 'wake'
   258→   258→      case 'waking':
   259→   259→      case 'listening':
   260→   260→        return 'command'
   261→   261→      default:
   262→   262→        return 'guard' // thinking, tooling, speaking
   263→   263→    }
   264→   264→  }
   265→   265→
   266→   266→  const onWake = (trailing: string) => {
   267→   267→    const phase = store.getState().phase
   268→   268→    if (phase === 'offline' || phase === 'boot') return
   269→   269→
   270→   270→    store.getState().setError(null)
   271→   271→    sfx.play('wake')
   272→   272→
   273→   273→    // "Jarvis, what's happening in AI this week" in one breath. Waiting for a
   274→   274→    // greeting he didn't need is the most common way an assistant wastes time.
   275→   275→    if (trailing) {
   276→   276→      void respond(trailing)
   277→   277→      return
   278→   278→    }
   279→   279→
   280→   280→    store.getState().setPhase('waking')
   281→   281→
   282→   282→    // Answer to his name. Deliberately NOT awaited any more: the microphone is
   283→   283→    // already open and the echo filter knows his voice, so the user can talk
   284→   284→    // straight over the greeting instead of waiting it out.
   285→   285→    const greeting = createSpeaker()
   286→   286→    speaker.current = greeting
   287→   287→    greeting.say(attention())
   288→   288→    void greeting.end()
   289→   289→
   290→   290→    listen(AWAIT_SPEECH_MS)
   291→   291→  }
   292→   292→
   293→   293→  /**
   294→   294→   * Someone started talking. This is the whole point of the rewrite: he stops,
   295→   295→   * immediately, whatever he was doing.
   296→   296→   */
   297→   297→  const onSpeechStart = () => {
   298→   298→    clearIdle()
   299→   299→    const phase = store.getState().phase
   300→   300→    if (phase === 'offline' || phase === 'boot' || phase === 'dormant') return
   301→   301→
   302→   302→    const wasBusy =
   303→   303→      phase === 'thinking' || phase === 'tooling' || phase === 'speaking'
   304→   304→
   305→   305→    silence()
   306→   306→    if (wasBusy) {
   307→   307→      // Abandon the answer in flight. The turn counter moves in respond()'s
   308→   308→      // replacement; bumping it here covers the case where nothing replaces it.
   309→   309→      turn.current++
   310→   310→      interrupt()
   311→   311→      store.getState().setActiveTool(null)
   312→   312→      music.working(false)
   313→   313→      sfx.duck(false)
   314→   314→      music.duck(false)
   315→   315→    }
   316→   316→    store.getState().setPhase('listening')
   317→   317→  }
   318→   318→
   319→   319→  const onUtterance = (text: string) => {
   320→   320→    const phase = store.getState().phase
   321→   321→    if (phase === 'offline' || phase === 'boot' || phase === 'dormant') return
   322→   322→
   323→   323→    // People keep using the name as a vocative once they're already talking to
   324→   324→    // the machine. Strip it rather than sending "sofia" to the model as a
   325→   325→    // question.
   326→   326→    if (bareName().test(text)) {
   327→   327→      listen(AWAIT_SPEECH_MS)
   328→   328→      return
   329→   329→    }
   330→   330→    const said = text.replace(leadingName(), '').trim()
   331→   331→    if (!said) {
   332→   332→      listen(AWAIT_SPEECH_MS)
   333→   333→      return
   334→   334→    }
   335→   335→
   336→   336→    void respond(said)
   337→   337→  }
   338→   338→
   339→   339→  /**
   340→   340→   * The typed command line. A `jarvis:say` event is the keyboard's version of
   341→   341→   * a recognised utterance: same road, same rules. Idle, it starts a turn;
   342→   342→   * busy, it barge-ins exactly like speaking over him does — the in-flight
   343→   343→   * answer is interrupted, its turn goes stale, and this one replaces it.
   344→   344→   */
   345→   345→  const onSay = (e: Event) => {
   346→   346→    const said = String((e as CustomEvent<string>).detail ?? '').trim()
   347→   347→    if (!said) return
   348→   348→
   349→   349→    const phase = store.getState().phase
   350→   350→    if (phase === 'boot') return
   351→   351→
   352→   352→    if (phase === 'offline') {
   353→   353→      // The ignition screen is still up; boot first, then ask the moment the
   354→   354→      // interface lands.
   355→   355→      void powerOn().then(() => respond(said))
   356→   356→      return
   357→   357→    }
   358→   358→
   359→   359→    if (phase === 'thinking' || phase === 'tooling' || phase === 'speaking') {
   360→   360→      silence()
   361→   361→      interrupt()
   362→   362→    }
   363→   363→    void respond(said)
   364→   364→  }
   365→   365→
   366→   366→  const onPartial = (text: string) => {
   367→   367→    store.getState().setCaption(text)
   368→   368→  }
   369→   369→
   370→   370→  const onVoiceError = (message: string) => {
   371→   371→    store.getState().setError(message)
   372→   372→    // The two mic failures are the only ones that mean voice is off for good
   373→   373→    // — transient recogniser hiccups fire this too, and must not flip the
   374→   374→    // standby hint to “type a command”. The ref survives startVoice returning
   375→   375→    // its dead object, so the optimistic setVoiceLive(true) after it cannot
   376→   376→    // overwrite what happened here.
   377→   377→    if (/microphone/i.test(message)) {
   378→   378→      micDead.current = true
   379→   379→      store.getState().setVoiceLive(false)
   380→   380→    }
   381→   381→  }
   382→   382→
   383→   383→  // -- power on -------------------------------------------------------------
   384→   384→
   385→   385→  const powerOn = async () => {
   386→   386→    // The ignition button and the space bar can both land here, and the phase
   387→   387→    // only moves after the first await — so without this a double press boots
   388→   388→    // twice, arming two voice loops and two download polls.
   389→   389→    if (booting.current) return
   390→   390→    booting.current = true
   391→   391→
   392→   392→    try {
   393→   393→      await ignite()
   394→   394→    } catch (err) {
   395→   395→      // The guard must not outlive a failed boot. Audio unlock can be refused,
   396→   396→      // the microphone prompt dismissed, the bridge unreachable at the wrong
   397→   397→      // moment — and with the flag still latched the ignition button was dead
   398→   398→      // for the rest of the page, recoverable only by reloading. Reset it and
   399→   399→      // put the button back so the user can simply press it again.
   400→   400→      booting.current = false
   401→   401→      console.error('[jarvis] power-up failed:', err)
   402→   402→      store.getState().setPhase('offline')
   403→   403→      store
   404→   404→        .getState()
   405→   405→        .setError(
   406→   406→          err instanceof Error
   407→   407→            ? `Power-up failed: ${err.message}`
   408→   408→            : 'Power-up failed. Click to try again.',
   409→   409→        )
   410→   410→    }
   411→   411→  }
   412→   412→
   413→   413→  const ignite = async () => {
   414→   414→    const s = store.getState()
   415→   415→
   416→   416→    // Must happen inside the click handler — browsers won't start an
   417→   417→    // AudioContext or speech synthesis without a user gesture.
   418→   418→    await sfx.unlockAudio()
   419→   419→    sfx.play('boot')
   420→   420→    // The score. Must be started from inside this click handler for the same
   421→   421→    // reason as the rest of the audio.
   422→   422→    music.enable()
   423→   423→    music.playBoot()
   424→   424→    music.startAmbient()
   425→   425→
   426→   426→    s.setPhase('boot')
   427→   427→
   428→   428→    watchServers((servers) => store.getState().setConnected(servers))
   429→   429→    watchPanels((panel) => store.getState().pushPanel(panel))
   430→   430→    watchBlades((blade) => store.getState().pushBlade(blade))
   431→   431→
   432→   432→    /**
   433→   433→     * JARVIS asking to see something.
   434→   434→     *
   435→   435→     * Announced on screen for as long as it takes, with whatever he said he was
   436→   436→     * looking for. The camera's own light is on too, but a hardware light that
   437→   437→     * appears with no explanation is exactly the thing that makes people
   438→   438→     * distrust an assistant — so the interface says it before they have to ask.
   439→   439→     */
   440→   440→    watchCapture(async (req) => {
   441→   441→      const note =
   442→   442→        req.mode === 'watch'
   443→   443→          ? req.when === 'past'
   444→   444→            ? req.reason || 'reviewing the last few seconds'
   445→   445→            : `${req.reason || 'watching'} · ${req.seconds}s`
   446→   446→          : req.reason || 'taking a look'
   447→   447→      store.getState().setLooking(note)
   448→   448→
   449→   449→      // The past is only available if something has been remembering it, and
   450→   450→      // that only happens while the camera is on screen. Answering plainly
   451→   451→      // beats opening the camera and recording the next few seconds instead,
   452→   452→      // which is a different question from the one that was asked.
   453→   453→      if (req.mode === 'watch' && req.when === 'past' && camera.bufferedSeconds() < 1) {
   454→   454→        store.getState().setLooking(null)
   455→   455→        return {
   456→   456→          error:
   457→   457→            'There is no recent footage — the camera has to be open on screen ' +
   458→   458→            'for me to remember what just happened. Ask me to open the camera, ' +
   459→   459→            'and I can watch from then on.',
   460→   460→        }
   461→   461→      }
   462→   462→
   463→   463→      // Held for the whole capture. Without this the stream can be torn down by
   464→   464→      // whoever else was using it half way through a six-second watch.
   465→   465→      let held = false
   466→   466→      try {
   467→   467→        await camera.holdCamera()
   468→   468→        held = true
   469→   469→        if (req.mode === 'look') return camera.grabFrame()
   470→   470→        if (req.when === 'past') {
   471→   471→          const grid = camera.recentGrid(req.seconds, 9)
   472→   472→          return grid ?? { error: 'There is not enough recent footage to review.' }
   473→   473→        }
   474→   474→        return await camera.watchAhead(req.seconds, 9)
   475→   475→      } catch (err) {
   476→   476→        return {
   477→   477→          error:
   478→   478→            (err as DOMException)?.name === 'NotAllowedError'
   479→   479→              ? 'The camera is not permitted, so I cannot see anything.'
   480→   480→              : `The camera could not be read: ${(err as Error)?.message ?? err}`,
   481→   481→        }
   482→   482→      } finally {
   483→   483→        if (held) camera.releaseCamera()
   484→   484→        store.getState().setLooking(null)
   485→   485→      }
   486→   486→    })
   487→   487→
   488→   488→    // The interface is JARVIS's to drive. These arrive out of band, pushed
   489→   489→    // mid-turn the way panels are, so a command can retint the reactor or put
   490→   490→    // something into orbit while he is still speaking the sentence about it.
   491→   491→    watchUi((op, args) => {
   492→   492→      const s = store.getState()
   493→   493→      const a = (args ?? {}) as Record<string, never>
   494→   494→      switch (op) {
   495→   495→        case 'patch':
   496→   496→          s.applyUi(args)
   497→   497→          break
   498→   498→        case 'orbit':
   499→   499→          if (a.action === 'add') s.addOrbit(args)
   500→   500→          else if (a.action === 'remove') s.removeOrbit(String(a.id))
   501→   501→          else s.clearOrbits()
   502→   502→          break
   503→   503→        case 'effect':
   504→   504→          s.fireEffect(a.kind)
   505→   505→          break
   506→   506→        case 'reset':
   507→   507→          s.resetUi()
   508→   508→          break
   509→   509→        case 'screen':
   510→   510→          s.clearScreen(a.what ?? 'all')
   511→   511→          break
   512→   512→        default:
   513→   513→          console.warn('[jarvis] unknown ui op:', op, args)
   514→   514→      }
   515→   515→    })
   516→   516→    // In bridge mode the conversation lives in the agent session, which is tied
   517→   517→    // to the socket — so a drop silently wipes his memory while the transcript
   518→   518→    // on screen still shows it. Better to say so than to let him quietly forget.
   519→   519→    watchConnection((state) => {
   520→   520→      if (state === 'lost') {
   521→   521→        store.getState().setError('Bridge connection lost — reconnecting.')
   522→   522→      } else if (state === 'reconnected') {
   523→   523→        store
   524→   524→          .getState()
   525→   525→          .setError('Bridge reconnected. The previous conversation was not kept.')
   526→   526→      }
   527→   527→    })
   528→   528→    const warming = warm().catch((err: Error) => s.setError(err.message))
   529→   529→
   530→   530→    if (!usingBridge && !env.anthropicKey) {
   531→   531→      s.setError(
   532→   532→        'No Anthropic API key — copy .env.example to .env.local and set VITE_ANTHROPIC_API_KEY.',
   533→   533→      )
   534→   534→    }
   535→   535→
   536→   536→    // Pull the neural voice down during the boot sequence so the first
   537→   537→    // "Hey Jarvis" isn't waiting on an 86MB download. Deliberately not awaited
   538→   538→    // — if it's slow, JARVIS comes up on the system voice and swaps over the
   539→   539→    // moment the model is ready.
   540→   540→    if (TTS_ENGINE === 'kokoro') {
   541→   541→      void kokoro.load()
   542→   542→      voicePoll.current = setInterval(() => {
   543→   543→        const p = kokoro.loadProgress()
   544→   544→        if (kokoro.isReady() || kokoro.isUnavailable()) {
   545→   545→          store.getState().setBootNote('')
   546→   546→          if (voicePoll.current) clearInterval(voicePoll.current)
   547→   547→          voicePoll.current = null
   548→   548→        } else if (p > 0 && p < 1) {
   549→   549→          store.getState().setBootNote(`voice ${Math.round(p * 100)}%`)
   550→   550→        }
   551→   551→      }, 200)
   552→   552→    }
   553→   553→
   554→   554→    // Long enough for the four-beat start-up sequence in Boot.tsx to play —
   555→   555→    // status bar, rings, suit schematic, reactor power-up — before the live
   556→   556→    // interface takes over. Kept a touch under the boot cue so the music is
   557→   557→    // still rising as the reactor lands.
   558→   558→    await new Promise((r) => setTimeout(r, 9200)) // boot sequence
   559→   559→    await warming
   560→   560→    store.getState().setConnected(connectedLabels())
   561→   561→    store.getState().setVoice(currentVoiceName())
   562→   562→
   563→   563→    // The analyser is what makes the reactor pulse with your voice. It needs a
   564→   564→    // getUserMedia stream; speech recognition does not, and gets its own. So a
   565→   565→    // failure here costs the animation and nothing else — saying "voice input
   566→   566→    // is unavailable" was both alarming and untrue.
   567→   567→    try {
   568→   568→      await startAnalyser()
   569→   569→    } catch {
   570→   570→      console.warn(
   571→   571→        '[jarvis] no microphone stream — the reactor will not pulse with your ' +
   572→   572→          'voice. Speech recognition is unaffected.',
   573→   573→      )
   574→   574→    }
   575→   575→
   576→   576→    // Ask the bridge which speech engines exist before the loop starts, so the
   577→   577→    // first turn already uses ElevenLabs when a key is present and the browser
   578→   578→    // fallback when it is not — no flag, no reload.
   579→   579→    await probeCapabilities()
   580→   580→
   581→   581→    // One voice loop, started once, running until the page closes.
   582→   582→    voice.current = await startVoice({
   583→   583→      mode,
   584→   584→      onWake,
   585→   585→      onSpeechStart,
   586→   586→      onPartial,
   587→   587→      onUtterance,
   588→   588→      onError: onVoiceError,
   589→   589→    })
   590→   590→
   591→   591→    // The loop is live unless the mic failed on the way up — a denied mic
   592→   592→    // reports itself through onError before this line runs, and that verdict
   593→   593→    // is the one that must stick.
   594→   594→    if (!micDead.current) store.getState().setVoiceLive(true)
   595→   595→
   596→   596→    store.getState().setPhase('dormant')
   597→   597→  }
   598→   598→
   599→   599→  // -- clap to start --------------------------------------------------------
   600→   600→
   601→   601→  /**
   602→   602→   * A clap brings him up, as an alternative to the button.
   603→   603→   *
   604→   604→   * Only while the ignition screen is showing, and torn down the moment he
   605→   605→   * boots — the microphone is about to belong to the voice loop, and two
   606→   606→   * analysers arguing over the same stream is how you get an assistant that
   607→   607→   * hears half of what you say.
   608→   608→   *
   609→   609→   * Deliberately silent about failure. If the microphone is refused, or has not
   610→   610→   * been granted yet, the button is still right there; announcing an error
   611→   611→   * about a feature nobody asked for would be worse than quietly doing without.
   612→   612→   */
   613→   613→  useEffect(() => {
   614→   614→    if (phase !== 'offline') return
   615→   615→    let live: { stop: () => void } | null = null
   616→   616→    let gone = false
   617→   617→    void listenForClap(() => {
   618→   618→      if (!gone) void powerOn()
   619→   619→    }).then((l) => {
   620→   620→      if (gone) l.stop()
   621→   621→      else live = l
   622→   622→    })
   623→   623→    return () => {
   624→   624→      gone = true
   625→   625→      live?.stop()
   626→   626→    }
   627→   627→    // eslint-disable-next-line react-hooks/exhaustive-deps
   628→   628→  }, [phase])
   629→   629→
   630→   630→  // -- level pump + keys ----------------------------------------------------
   631→   631→
   632→   632→  useEffect(() => {
   633→   633→    let raf = 0
   634→   634→
   635→   635→    const pump = () => {
   636→   636→      const st = store.getState()
   637→   637→      // While speaking, follow JARVIS's own output rather than the mic, so the
   638→   638→      // orb lip-syncs instead of reacting to room noise.
   639→   639→      const lvl =
   640→   640→        st.phase === 'speaking' && speaker.current
   641→   641→          ? speaker.current.level()
   642→   642→          : micLevel()
   643→   643→      st.setLevel(lvl)
   644→   644→      raf = requestAnimationFrame(pump)
   645→   645→    }
   646→   646→    pump()
   647→   647→
   648→   648→    const onKey = (e: KeyboardEvent) => {
   649→   649→      const tag = (e.target as HTMLElement)?.tagName
   650→   650→      if (tag === 'INPUT' || tag === 'TEXTAREA') return
   651→   651→
   652→   652→      // V auditions the next British voice installed on this machine. Which
   653→   653→      // ones exist varies per Mac, so hearing them beats trusting a ranking.
   654→   654→      // Bare V only — ⌘V and ⌃V are paste, and swallowing those was rude.
   655→   655→      if (
   656→   656→        e.key === 'v' &&
   657→   657→        !e.repeat &&
   658→   658→        !e.metaKey &&
   659→   659→        !e.ctrlKey &&
   660→   660→        !e.altKey
   661→   661→      ) {
   662→   662→        e.preventDefault()
   663→   663→        const name = cycleVoice()
   664→   664→        store.getState().setVoice(name)
   665→   665→        silence()
   666→   666→        const demo = createSpeaker()
   667→   667→        speaker.current = demo
   668→   668→        demo.say(personaById(store.getState().persona).fillers.voiceSet)
   669→   669→        void demo.end()
   670→   670→        return
   671→   671→      }
   672→   672→
   673→   673→      // G puts the camera on and starts tracking hands. Off by default and
   674→   674→      // never implicit: a webcam that turns itself on because an interface
   675→   675→      // thought it might be useful is not a trade anyone agreed to.
   676→   676→      if (e.key === 'g' && !e.repeat && !e.metaKey && !e.ctrlKey && !e.altKey) {
   677→   677→        e.preventDefault()
   678→   678→        const on = store.getState().gestures
   679→   679→        if (on) {
   680→   680→          hands.disableHands()
   681→   681→          store.getState().setGestures(false)
   682→   682→        } else {
   683→   683→          store.getState().setError(null)
   684→   684→          void hands
   685→   685→            .enableHands()
   686→   686→            .then(() => store.getState().setGestures(true))
   687→   687→            .catch((err: Error) => {
   688→   688→              store.getState().setGestures(false)
   689→   689→              store
   690→   690→                .getState()
   691→   691→                .setError(
   692→   692→                  err?.name === 'NotAllowedError'
   693→   693→                    ? 'Camera access denied — gesture control is unavailable.'
   694→   694→                    : `Gesture control failed to start: ${err?.message ?? err}`,
   695→   695→                )
   696→   696→            })
   697→   697→        }
   698→   698→        return
   699→   699→      }
   700→   700→
   701→   701→      // T speaks a fixed line, bypassing the wake word, the recogniser and the
   702→   702→      // model entirely. When "I can't hear him" is the report, this is the one
   703→   703→      // keypress that separates a broken voice engine from a broken voice loop
   704→   704→      // — and it prints the verdict rather than making you infer it.
   705→   705→      if (e.key === 't' && !e.repeat && !e.metaKey && !e.ctrlKey && !e.altKey) {
   706→   706→        e.preventDefault()
   707→   707→        silence()
   708→   708→        const t = createSpeaker()
   709→   709→        speaker.current = t
   710→   710→        t.say('Audio test. If you can hear this, speech output is working, sir.')
   711→   711→        void t.end().then(() => {
   712→   712→          const d = (window as unknown as Record<string, Record<string, unknown>>).__tts
   713→   713→          console.info('[jarvis] audio test →', d)
   714→   714→          if (d && d.started === 0 && d.rescued === 0) {
   715→   715→            store.getState().setError(
   716→   716→              `No sound produced. engine=${d.engine} voice=${d.voice} error=${d.lastError || 'none'}`,
   717→   717→            )
   718→   718→          }
   719→   719→        })
   720→   720→        return
   721→   721→      }
   722→   722→
   723→   723→      // Escape stands the whole thing down — the one thing the old build had
   724→   724→      // no key for at all.
   725→   725→      if (e.key === 'Escape') {
   726→   726→        e.preventDefault()
   727→   727→        if (store.getState().phase !== 'offline') goDormant()
   728→   728→        return
   729→   729→      }
   730→   730→
   731→   731→      // Space starts a turn without the wake word. Worth using while filming so
   732→   732→      // a missed wake word doesn't cost a take.
   733→   733→      if (e.code !== 'Space' || e.repeat) return
   734→   734→      e.preventDefault()
   735→   735→
   736→   736→      const phase = store.getState().phase
   737→   737→      if (phase === 'offline') {
   738→   738→        void powerOn()
   739→   739→      } else if (phase === 'boot') {
   740→   740→        /* ignore — the boot sequence owns the phase until it finishes */
   741→   741→      } else if (
   742→   742→        phase === 'thinking' ||
   743→   743→        phase === 'tooling' ||
   744→   744→        phase === 'speaking'
   745→   745→      ) {
   746→   746→        onSpeechStart()
   747→   747→        listen(AWAIT_SPEECH_MS)
   748→   748→      } else {
   749→   749→        onWake('')
   750→   750→      }
   751→   751→    }
   752→   752→    window.addEventListener('keydown', onKey)
   753→   753→    window.addEventListener('jarvis:say', onSay as EventListener)
   754→   754→
   755→   755→    return () => {
   756→   756→      cancelAnimationFrame(raf)
   757→   757→      window.removeEventListener('keydown', onKey)
   758→   758→      window.removeEventListener('jarvis:say', onSay as EventListener)
   759→   759→      clearIdle()
   760→   760→      if (voicePoll.current) clearInterval(voicePoll.current)
   761→   761→      voice.current?.stop()
   762→   762→      speaker.current?.cancel()
   763→   763→      // The camera must not outlive the page that turned it on.
   764→   764→      hands.disableHands()
   765→   765→    }
   766→   766→    // eslint-disable-next-line react-hooks/exhaustive-deps
   767→   767→  }, [])
   768→   768→
   769→   769→  return (
   770→   770→    <>
   771→   771→      <Scene />
   772→   772→      <Hud />
   773→   773→      <Boot />
   774→   774→      <Diagnostics />
   775→   775→      <CommandLine />
   776→   776→      <Ignition onStart={() => void powerOn()} />
   777→   777→    </>
   778→   778→  )
   779→   779→}
