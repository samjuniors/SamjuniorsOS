   202→/**
   203→ * Computes the SHA-256 content hash stored in the `CompanyKnowledge.hash`
   204→ * column (M2: the ingestion path records it on every explicit write).
   205→ *
   206→ * Exact guarantee: it fingerprints ONLY the `content` field.
   207→ *  - Identical content ⇒ identical hash (deterministic, byte-for-byte).
   208→ *  - Changed content ⇒ different hash (SHA-256 collision resistance).
   209→ *  - Title/category/version/metadata changes alone do NOT change the hash.
   210→ *
   211→ * See isKnowledgeContentUnchanged() for the comparison contract.
   212→ */
   213→export function computeKnowledgeContentHash(content: string): string {
   214→  return createHash('sha256').update(content || '', 'utf-8').digest('hex');
   215→}
   216→
   217→/**
   218→ * Minimal change-detection contract (no vectors/embeddings/FTS): compares a
   219→ * persisted hash against the hash of candidate content.
   220→ *  - equal  ⇒ content unchanged since the hash was recorded
   221→ *  - different ⇒ content changed
   222→ * A missing/blank stored hash never claims "unchanged".
   223→ */
   224→export function isKnowledgeContentUnchanged(
   225→  storedHash: string | null | undefined,
   226→  content: string
   227→): boolean {
   228→  if (!storedHash) return false;
   229→  return storedHash === computeKnowledgeContentHash(content);
   230→}

   232→/**
   233→ * ============================================================================
   234→ * COMPANY KNOWLEDGE STORE — M2: CANONICAL PERSISTENCE ACTIVATED
   235→ * ============================================================================
   236→ *
   237→ * Implements ICompanyKnowledgeStore with the SAME dual-mode persistence
   238→ * pattern as its sibling stores (memory-store, conversation store, etc.):
   239→ *
   240→ *  - LOCAL MODE: in-memory cache (constructor-hydrated from DurableFileStore)
   241→ *    + .data/company_knowledge.json durability + best-effort Prisma dual-write.
   242→ *  - AUTHORITATIVE MODE: Prisma fail-closed (requireAuthoritativeDatabase);
   243→ *    reads and writes go to the CompanyKnowledge table (target: PostgreSQL;
   244→ *    currently the SQLite sandbox port — see the M0 naming note).
   245→ *
   246→ * Seed migration (one-time, bootstrap-only): the 8 canonical documents that
   247→ * previously existed ONLY as code constants are persisted to DurableFileStore
   248→ * on first boot (empty durable collection), and bootstrapped into the Prisma
   249→ * table lazily on first store usage — CREATE-ONLY: a canonical document is
   250→ * seeded only when no row exists for its id; an existing persisted row is
   251→ * NEVER overwritten by the code constants on restart (Founder directive,
   252→ * pre-M3 correction pass). Explicit writes (addKnowledge / setKnowledge)
   253→ * remain the only mutation paths after boot.
   254→ *
   255→ * The `hash` column (SHA-256 of the `content` field ONLY) provides change
   256→ * detection: comparing a stored hash against
   257→ * computeKnowledgeContentHash(candidateContent) distinguishes unchanged
   258→ * content (equal) from changed content (different) — see
   259→ * isKnowledgeContentUnchanged(). It does NOT fingerprint title/category/
   260→ * metadata. A different id with byte-identical content collides on the
   261→ * unique hash index and is treated as a duplicate (seed skipped; explicit
   262→ * writes still dual-write best-effort in local mode).
   263→ *
   264→ * Designed so the target PostgreSQL / Supabase deployment can replace the
   265→ * persistence internals without altering retrieval or business logic.
   266→ */
   267→export class CompanyKnowledgeStore implements ICompanyKnowledgeStore {
   268→  private static instance: CompanyKnowledgeStore | null = null;
   269→
   270→  private knowledgeItems: CompanyKnowledgeItem[] = [...CANONICAL_COMPANY_KNOWLEDGE];
   271→
   272→  /** One-shot guard for the lazy Prisma seed mirror (M2 seed migration). */
   273→  private static prismaSeedPromise: Promise<void> | null = null;
   274→
   275→  private constructor() {
   276→    this.loadFromDurableStorage();
   277→  }
   278→
   279→  public static getInstance(): CompanyKnowledgeStore {
   280→    if (!CompanyKnowledgeStore.instance) {
   281→      CompanyKnowledgeStore.instance = new CompanyKnowledgeStore();
   282→    }
   283→    return CompanyKnowledgeStore.instance;
   284→  }
   285→
   286→  /**
   287→   * Hydrates the in-memory cache from DurableFileStore, merging over the
   288→   * canonical seed documents. On first boot (no durable collection), the
   289→   * canonical seeds are themselves persisted — the M2 one-time seed
   290→   * migration of the 8 canonical documents.
   291→   */
   292→  private loadFromDurableStorage(): void {
   293→    try {
   294→      const persisted = DurableFileStore.getInstance().readCollection<CompanyKnowledgeItem>('company_knowledge');
   295→      const persistedItems = Object.values(persisted);
   296→      if (persistedItems.length === 0) {
   297→        // First boot: persist the canonical seed set (one-time migration).
   298→        const seedRecord: Record<string, CompanyKnowledgeItem> = {};
   299→        for (const item of CANONICAL_COMPANY_KNOWLEDGE) {
   300→          seedRecord[item.id] = item;
   301→        }
   302→        DurableFileStore.getInstance().writeCollection('company_knowledge', seedRecord);
   303→        this.knowledgeItems = [...CANONICAL_COMPANY_KNOWLEDGE];
   304→        return;
   305→      }
   306→      // Merge: canonical seeds first (stable order), persisted items overlay
   307→      // by id (edits win), extras appended.
   308→      const byId = new Map<string, CompanyKnowledgeItem>();
   309→      for (const item of CANONICAL_COMPANY_KNOWLEDGE) byId.set(item.id, item);
   310→      for (const item of persistedItems) byId.set(item.id, item);
   311→      this.knowledgeItems = [...byId.values()];
   312→    } catch {
   313→      // fallback to canonical seeds
   314→    }
   315→  }
   316→
   317→  /**
   318→   * Lazily bootstraps the canonical seed documents into the Prisma table
   319→   * once per process. BOOTSTRAP-ONLY (Founder directive, pre-M3 correction):
   320→   * a canonical document is created ONLY when no row exists for its id — an
   321→   * existing persisted row is preserved as-is, so a restart can never
   322→   * overwrite authoritative persisted knowledge with the code constants.
   323→   * Explicit writes (addKnowledge / setKnowledge) remain the only mutation
   324→   * paths after boot. A unique-hash collision on create means byte-identical
   325→   * content is already persisted under another id — the seed is skipped (the
   326→   * hash column is the duplicate-content guard); any other error propagates
   327→   * (fail-closed in authoritative mode). Local mode: best-effort.
   328→   */
   329→  private ensurePrismaSeeded(): Promise<void> {
   330→    if (!CompanyKnowledgeStore.prismaSeedPromise) {
   331→      CompanyKnowledgeStore.prismaSeedPromise = (async () => {
   332→        try {
   333→          const db = isAuthoritativeMode()
   334→            ? await requireAuthoritativeDatabase()
   335→            : prisma;
   336→          for (const item of CANONICAL_COMPANY_KNOWLEDGE) {
   337→            const existing = await db.companyKnowledge.findUnique({
   338→              where: { id: item.id },
   339→            });
   340→            // Persisted document wins over the code constant — never overwrite.
   341→            if (existing) continue;
   342→            try {
   343→              await db.companyKnowledge.create({
   344→                data: {
   345→                  id: item.id,
   346→                  category: item.category,
   347→                  title: item.title,
   348→                  content: item.content,
   349→                  hash: computeKnowledgeContentHash(item.content),
   350→                  metadata: {
   351→                    documentId: item.documentId,
   352→                    version: item.version,
   353→                    summary: item.summary,
   354→                    tags: item.tags,
   355→                    applicableDepartments: item.applicableDepartments,
   356→                    authorAuthority: item.authorAuthority,
   357→                    lastVerifiedDate: item.lastVerifiedDate,
   358→                    isDurableReference: item.isDurableReference,
   359→                  },
   360→                },
   361→              });
   362→            } catch (createErr: any) {
   363→              const msg = String(createErr?.message || createErr);
   364→              const isUniqueCollision =
   365→                createErr?.code === 'P2002' || /unique constraint/i.test(msg);
   366→              if (!isUniqueCollision) throw createErr;
   367→              // Byte-identical content already persisted under another id:
   368→              // skip seeding a duplicate (unique hash = duplicate guard).
   369→            }
   370→          }
   371→        } catch (err: any) {
   372→          if (isAuthoritativeMode()) throw err;
   373→          // Local mode: best-effort only — the durable file store remains the
   374→          // local-mode primary per ADR 0002.
   375→        }
   376→      })().catch(() => {
   377→        // Do not cache a failed best-effort seed; retry on next usage.
   378→        CompanyKnowledgeStore.prismaSeedPromise = null;
   379→      });
   380→    }
   381→    return CompanyKnowledgeStore.prismaSeedPromise;
   382→  }
   383→
   384→  public async getAllKnowledge(): Promise<CompanyKnowledgeItem[]> {
   385→    if (isAuthoritativeMode()) {
   386→      await this.ensurePrismaSeeded();
   387→      const db = await requireAuthoritativeDatabase();
   388→      const rows = await db.companyKnowledge.findMany({ orderBy: { updatedAt: 'desc' } });
   389→      return rows.map((r: any) => this.mapRowToItem(r));
   390→    }
   391→    return [...this.knowledgeItems];
   392→  }
   393→
   394→  public async getKnowledgeById(id: string): Promise<CompanyKnowledgeItem | null> {
   395→    if (isAuthoritativeMode()) {
   396→      await this.ensurePrismaSeeded();
   397→      const db = await requireAuthoritativeDatabase();
   398→      const row = await db.companyKnowledge.findFirst({
   399→        where: { OR: [{ id }, { title: { contains: id } }] },
   400→      });
   401→      if (row) return this.mapRowToItem(row);
   402→      return null;
   403→    }
   404→
   405→    const item = this.knowledgeItems.find((k) => k.id === id || k.documentId === id);
   406→    return item ? { ...item } : null;
   407→  }
   408→
   409→  public async addKnowledge(item: CompanyKnowledgeItem): Promise<void> {
   410→    if (isAuthoritativeMode()) {
   411→      const db = await requireAuthoritativeDatabase();
   412→      await db.companyKnowledge.upsert({
   413→        where: { id: item.id },
   414→        create: {
   415→          id: item.id,
   416→          category: item.category,
   417→          title: item.title,
   418→          content: item.content,
   419→          hash: computeKnowledgeContentHash(item.content),
   420→          metadata: this.buildRowMetadata(item),
   421→        },
   422→        update: {
   423→          category: item.category,
   424→          title: item.title,
   425→          content: item.content,
   426→          hash: computeKnowledgeContentHash(item.content),
   427→          metadata: this.buildRowMetadata(item),
   428→        },
   429→      });
   430→      // Keep the in-process cache coherent for queryKnowledge.
   431→      this.knowledgeItems = [
   432→        item,
   433→        ...this.knowledgeItems.filter((k) => k.id !== item.id),
   434→      ];
   435→      return;
   436→    }
   437→
   438→    // Local mode: explicit write — REPLACES any existing item with the same
   439→    // id (no duplicate-id entries in the in-process cache, mirroring the
   440→    // authoritative branch's replace-by-id semantics).
   441→    this.knowledgeItems = [
   442→      item,
   443→      ...this.knowledgeItems.filter((k) => k.id !== item.id),
   444→    ];
   445→    try {
   446→      DurableFileStore.getInstance().saveItem('company_knowledge', item.id, item);
   447→    } catch {}
   448→
   449→    if (process.env.DATABASE_URL) {
   450→      try {
   451→        await prisma.companyKnowledge.upsert({
   452→          where: { id: item.id },
   453→          create: {
   454→            id: item.id,
   455→            category: item.category,
   456→            title: item.title,
   457→            content: item.content,
   458→            hash: computeKnowledgeContentHash(item.content),
   459→            metadata: this.buildRowMetadata(item),
   460→          },
   461→          update: {
   462→            category: item.category,
   463→            title: item.title,
   464→            content: item.content,
   465→            hash: computeKnowledgeContentHash(item.content),
   466→            metadata: this.buildRowMetadata(item),
   467→          },
   468→        });
   469→      } catch {
   470→        // Best-effort dual-write (e.g. unique-hash duplicate collision or
   471→        // offline DB): the durable file store remains the local-mode primary.
   472→      }
   473→    }
   474→  }

   476→  public async setKnowledge(items: CompanyKnowledgeItem[]): Promise<void> {
   477→    this.knowledgeItems = [...items];
   478→
   479→    if (isAuthoritativeMode()) {
   480→      const db = await requireAuthoritativeDatabase();
   481→      await db.$transaction(async (tx: any) => {
   482→        await tx.companyKnowledge.deleteMany({});
   483→        if (items.length > 0) {
   484→          await tx.companyKnowledge.createMany({
   485→            data: items.map((item) => ({
   486→              id: item.id,
   487→              category: item.category,
   488→              title: item.title,
   489→              content: item.content,
   490→              hash: computeKnowledgeContentHash(item.content),
   491→              metadata: this.buildRowMetadata(item),
   492→            })) as any,
   493→          });
   494→        }
   495→      });
   496→      return;
   497→    }
   498→
   499→    try {
   500→      const record: Record<string, CompanyKnowledgeItem> = {};
   501→      for (const item of items) record[item.id] = item;
   502→      DurableFileStore.getInstance().writeCollection('company_knowledge', record);
   503→    } catch {}
   504→
   505→    if (process.env.DATABASE_URL) {
   506→      try {
   507→        await prisma.$transaction([
   508→          prisma.companyKnowledge.deleteMany({}),
   509→          ...(items.length > 0
   510→            ? [prisma.companyKnowledge.createMany({
   511→                data: items.map((item) => ({
   512→                  id: item.id,
   513→                  category: item.category,
   514→                  title: item.title,
   515→                  content: item.content,
   516→                  hash: computeKnowledgeContentHash(item.content),
   517→                  metadata: this.buildRowMetadata(item),
   518→                })) as any,
   519→              })]
   520→            : []),
   521→        ]);
   522→      } catch {
   523→        // Best-effort dual-write; durable file remains local-mode primary.
   524→      }
   525→    }
   526→  }

   528→  /**
   529→   * Deterministically queries durable reference information.
   530→   * Attaches epistemic label: 'durable_reference' and provenance metadata.
   531→   */
   532→  public async queryKnowledge(params: KnowledgeQueryParams): Promise<RetrievedKnowledgeItem[]> {
   533→    const queryText = (params.queryText || '').toLowerCase();
   534→    const queryTokens = new Set([
   535→      ...extractTokens(queryText),
   536→      ...(params.keywords ? params.keywords.flatMap(extractTokens) : []),
   537→      ...(params.tags ? params.tags.map((t) => t.toLowerCase()) : []),
   538→    ]);
   539→
   540→    const targetRole = params.role;
   541→    const nowIso = new Date().toISOString();
   542→    const results: RetrievedKnowledgeItem[] = [];
   543→
   544→    for (const item of this.knowledgeItems) {
   545→      // Role match check
   546→      const isRoleApplicable = !targetRole || targetRole === 'orchestrator' || targetRole === 'council' || targetRole === 'advisor'
   547→        || item.applicableDepartments.includes(targetRole)
   548→        || item.applicableDepartments.includes('council');
   549→
   550→      const itemTokens = [
   551→        ...extractTokens(item.title),
   552→        ...extractTokens(item.summary),
   553→        ...extractTokens(item.content),
   554→        ...item.tags.flatMap(extractTokens),
   555→        item.documentId.toLowerCase(),
   556→      ];
   557→
   558→      const matchedTerms = itemTokens.filter((t) => queryTokens.has(t));
   559→      const hasTermMatch = matchedTerms.length > 0;
   560→
   561→      // Check category match if provided
   562→      const categoryMatch = params.category && item.category.toLowerCase().includes(params.category.toLowerCase());
   563→
   564→      if (hasTermMatch || categoryMatch) {
   565→        const relevanceScore = (hasTermMatch ? matchedTerms.length * 2 : 0)
   566→          + (categoryMatch ? 3 : 0)
   567→          + (isRoleApplicable ? 1 : 0);
   568→
   569→        results.push({
   570→          knowledgeId: item.id,
   571→          documentId: item.documentId,
   572→          title: item.title,
   573→          category: item.category,
   574→          version: item.version,
   575→          summary: item.summary,
   576→          contentSnippet: item.summary,
   577→          fullContent: item.content,
   578→          applicableDepartments: item.applicableDepartments,
   579→          relevanceScore,
   580→          matchReason: matchedTerms.length > 0
   581→            ? `Matched knowledge terms: ${[...new Set(matchedTerms)].join(', ')}`
   582→            : `Category match (${item.category})`,
   583→          provenance: {
   584→            sourceSystem: 'company_knowledge',
   585→            sourceId: `knowledge:${item.documentId}`,
   586→            sourceTitle: item.title,
   587→            epistemicType: 'durable_reference',
   588→            authority: item.authorAuthority,
   589→            timestamp: item.lastVerifiedDate || nowIso,
   590→            confidence: 'reference_standard',
   591→            notes: `Durable company reference documentation (v${item.version})`,
   592→          },
   593→        });
   594→      }
   595→    }
   596→
   597→    // Sort by relevance score descending
   598→    results.sort((a, b) => b.relevanceScore - a.relevanceScore);
   599→    const limit = params.limit || 5;
   600→    return results.slice(0, limit);
   601→  }

   603→  private buildRowMetadata(item: CompanyKnowledgeItem): any {
   604→    return {
   605→      documentId: item.documentId,
   606→      version: item.version,
   607→      summary: item.summary,
   608→      tags: item.tags,
   609→      applicableDepartments: item.applicableDepartments,
   610→      authorAuthority: item.authorAuthority,
   611→      lastVerifiedDate: item.lastVerifiedDate,
   612→      isDurableReference: item.isDurableReference,
   613→    } as any;
   614→  }

   616→  private mapRowToItem(r: any): CompanyKnowledgeItem {
   617→    const meta = (r.metadata && typeof r.metadata === 'object') ? r.metadata : {};
   618→    return {
   619→      id: r.id,
   620→      documentId: (meta.documentId as string) || r.id,
   621→      title: r.title,
   622→      category: r.category,
   623→      version: (meta.version as string) || '1.0.0',
   624→      summary: (meta.summary as string) || r.content?.slice(0, 240) || '',
   625→      content: r.content,
   626→      tags: (meta.tags as string[]) || [],
   627→      applicableDepartments: ((meta.applicableDepartments as CompanyKnowledgeItem['applicableDepartments']) || ['council']) as any,
   628→      authorAuthority: (meta.authorAuthority as string) || 'Unknown authority',
   629→      lastVerifiedDate: (meta.lastVerifiedDate as string) || r.updatedAt?.toISOString?.() || new Date().toISOString(),
   630→      isDurableReference: true as const,
   631→    };
   632→  }
