    94→/** Number-word → digit mappings for the small expected value set. */
    95→const NUMBER_WORDS: Array<[RegExp, string]> = [
    96→  [/\bforty[- ]nine\b/g, '49'],
    97→  [/\btwenty[- ]nine\b/g, '29'],
    98→  [/\bthirty[- ]four thousand(?: dollars)?\b/g, '34000'],
    99→  [/\bthirty[- ]four k\b/g, '34000'],
   100→  [/\b34k\b/g, '34000'],
   101→  [/\beighteen thousand five hundred(?: dollars)?\b/g, '18500'],
   102→  [/\beighteen and a half (?:thousand|k)\b/g, '18500'],
   103→  [/\b18\.5k\b/g, '18500'],
   104→  [/\btwo hundred (?:and )?twenty[- ]two thousand(?: dollars)?\b/g, '222000'],
   105→  [/\btwo hundred twenty[- ]two thousand(?: dollars)?\b/g, '222000'],
   106→  [/\b222k\b/g, '222000'],
   107→  [/\bfourteen\b/g, '14'],
   108→];

   110→/**
   111→ * Deterministic answer normalization: lowercase, straight quotes, markdown
   112→ * emphasis stripped, currency variants unified to '$'-prefix form (postfix
   113→ * "49 usd"/"49 $" → "$49"), thousands separators removed, number-words for
   114→ * the expected value set mapped to digits, ".00" decimals dropped, whitespace
   115→ * collapsed.
   116→ */
   117→export function normalizeAnswer(raw: string): string {
   118→  let t = (raw || '')
   119→    .toLowerCase()
   120→    .replace(/[\u2018\u2019]/g, "'")
   121→    .replace(/[\u201C\u201D]/g, '"')
   122→    .replace(/[\u2013\u2014]/g, '-')
   123→    .replace(/[*_`]+/g, '')
   124→    // currency unification (order matters)
   125→    .replace(/\bus\s*\$\s*/g, '$')
   126→    .replace(/\busd\s*\$?\s*/g, '$')
   127→    .replace(/\b(\d+(?:\.\d+)?)\s*\$/g, '$$$1') // "49 $" → "$49"
   128→    .replace(/\$\s*(?=\d)/g, '$');
   129→  // thousands separators (only digit,comma,exactly-3-digits boundaries)
   130→  for (let i = 0; i < 3; i++) t = t.replace(/(\d),(\d{3})/g, '$1$2');
   131→  // number words for the expected value set
   132→  for (const [pattern, replacement] of NUMBER_WORDS) t = t.replace(pattern, replacement);
   133→  // "$49.00" → "$49" (exact-zero decimals only)
   134→  t = t.replace(/(\d)\.00(?=\D|$)/g, '$1');
   135→  return t.replace(/\s+/g, ' ').trim();
   136→}

   138→/**
   139→ * Deterministic sentence splitting on the RAW answer: sentence punctuation,
   140→ * line breaks, and list markers each start a new co-occurrence unit. The
   141→ * returned sentences are NOT normalized — apply normalizeAnswer per sentence.
   142→ */
   143→export function splitSentences(raw: string): string[] {
   144→  const out: string[] = [];
   145→  const lines = (raw || '').replace(/\r/g, '').split(/\n+/);
   146→  for (const line of lines) {
   147→    for (const part of line.split(/(?<=[.!?\u2026])\s+/)) {
   148→      // A colon introducing a list also separates co-occurrence units.
   149→      for (const piece of part.split(/(?<=:)\s+(?=[-\u2022*]\s)/)) {
   150→        const cleaned = piece.replace(/^[\s\-*\u2022]+/, '').trim();
   151→        if (cleaned) out.push(cleaned);
   152→      }
   153→    }
   154→  }
   155→  return out;
   156→}

   291→function anyMatch(text: string, patterns: RegExp[]): boolean {
   292→  return patterns.some((p) => p.test(text));
   293→}
   294→
   295→export function hasCurrentLanguage(sentence: string): boolean {
   296→  return anyMatch(sentence, CURRENT_LANGUAGE);
   297→}
   298→export function hasPastLanguage(sentence: string): boolean {
   299→  return anyMatch(sentence, PAST_LANGUAGE);
   300→}
   301→export function hasContrastLanguage(sentence: string): boolean {
   302→  return anyMatch(sentence, CONTRAST_LANGUAGE);
   303→}

   305→/**
   306→ * Value presence with strict token boundaries: the number must not have an
   307→ * adjacent digit or decimal-part ("449", "49.99" do not contain 49) — but a
   308→ * sentence-final period ("$49.") is not a decimal part and does match.
   309→ */
   310→export function hasValue(normalizedText: string, value: number): boolean {
   311→  const escaped = String(value).replace('.', '\\.');
   312→  return new RegExp(`(?<!\\d)(?<!\\d\\.)${escaped}(?!\\d)(?!\\.\\d)`).test(normalizedText);
   313→}

   315→function sentencesWith(normalizedSentences: string[], value: number): string[] {
   316→  return normalizedSentences.filter((s) => hasValue(s, value));
   317→}

   323→/** Result-of-change language: the value is the OUTCOME of the transition
   324→ * ("raised to $49", "became $49") — a correct current-value attribution for
   325→ * S3/S4 positives, and (for the old value) a stale-as-current signal. */
   326→export const RESULT_LANGUAGE: RegExp[] = [
   327→  /\braised to\b/,
   328→  /\bincreased to\b/,
   329→  /\bchanged to\b/,
   330→  /\bswitched to\b/,
   331→  /\bmoved to\b/,
   332→  /\brose to\b/,
   333→  /\bwent to\b/,
   334→  /\bbecame\b/,
   335→  /\bnow\b/,
   336→];

   338→function hasResultLanguage(sentence: string): boolean {
   339→  return anyMatch(sentence, RESULT_LANGUAGE);
   340→}
   341→
   342→/**
   343→ * STALE-AS-CURRENT (design S1/S3/S4/S10): a sentence presenting the OLD
   344→ * value as current — the value is present, the sentence has current-language
   345→ * or result-of-change language, and it has NEITHER past-language NOR
   346→ * contrast-language. This is exactly the design's "forbidden inversion
   347→ * patterns ('currently $29', …)" sentence-window form; the tight direct
   348→ * bindings are additionally pinned by findCurrentAsOld.
   349→ */
   350→function findStaleAsCurrent(normalizedSentences: string[], oldValue: number): string | null {
   351→  for (const s of sentencesWith(normalizedSentences, oldValue)) {
   352→    if (
   353→      (hasCurrentLanguage(s) || hasResultLanguage(s)) &&
   354→      !hasPastLanguage(s) &&
   355→      !hasContrastLanguage(s)
   356→    ) {
   357→      return s;
   358→    }
   359→  }
   360→  return null;
   361→}
   362→
   363→/**
   364→ * CURRENT-AS-OLD (design S3: "'was $49', 'previously $49'"): the NEW value
   365→ * directly bound to a past marker — "was $49", "was priced at $49",
   366→ * "previously $49" — OR named as the ORIGIN of the change ("changed from
   367→ * $49", "went from $49"). Movement-to constructions ("was raised to $49",
   368→ * "changed to $49") are deliberately NOT matched: they describe the change
   369→ * event's result, not the old value.
   370→ */
   371→function findCurrentAsOld(normalized: string, newValue: number): string | null {
   372→  const patterns = [
   373→    new RegExp(
   374→      `\\b(?:was|were|had been|previously|formerly|used to (?:be|cost)|stood at)\\s+` +
   375→        `(?:priced at\\s+|at\\s+|about\\s+|approximately\\s+|around\\s+|just\\s+|only\\s+)?` +
   376→        `\\$?${newValue}\\b`
   377→    ),
   378→    new RegExp(
   379→      `\\b(?:up from|went from|changed from|moved from|switched from|from)\\s+` +
   380→        `(?:about\\s+|approximately\\s+|around\\s+)?\\$?${newValue}\\b`
   381→    ),
   382→  ];
   383→  for (const p of patterns) {
   384→    const m = normalized.match(p);
   385→    if (m) return m[0];
   386→  }
   387→  return null;
   388→}
   389→
   390→/**
   391→ * TEMPORAL AS-OF VIOLATIONS (design S5): "$49 attributed to the as-of
   392→ * instant". Two deterministic forms:
   393→ *  (a) the new value directly bound to a past marker ("as of August 1 … was
   394→ *      $49") — the only past frame in S5 is the as-of instant;
   395→ *  (b) a sentence framing the as-of instant that states the NEW value without
   396→ *      the old one and without any later/newer framing.
   397→ */
   398→function findAsOfViolation(normalizedSentences: string[], newValue: number, oldValue: number): string | null {
   399→  // (a) tight past binding of the new value (movement verbs excluded by design)
   400→  const tight = findCurrentAsOld(normalizedSentences.join(' '), newValue);
   401→  if (tight) return tight;
   402→  // (b) as-of framed sentence stating the new value only
   403→  for (const s of sentencesWith(normalizedSentences, newValue)) {
   404→    if (anyMatch(s, AS_OF_LANGUAGE) && !hasValue(s, oldValue) && !anyMatch(s, LATER_LANGUAGE)) {
   405→      return s;
   406→    }
   407→  }
   408→  return null;
   409→}
   410→
   411→/**
   412→ * Supported-fact positive: the value stated as current in at least one
   413→ * sentence — current-language, contrast-language (before→after framing
   414→ * covers both sides), or result-of-change language ("raised to $49").
   415→ */
   416→function statedAsCurrent(normalizedSentences: string[], value: number): boolean {
   417→  return sentencesWith(normalizedSentences, value).some(
   418→    (s) => hasCurrentLanguage(s) || hasContrastLanguage(s) || hasResultLanguage(s)
   419→  );
   420→}

   440→/**
   441→ * A number "attributed" to a metric term: within the LOOKBACK window before
   442→ * the number there is the metric term and NO different-metric anchor.
   443→ */
   444→const LOOKBACK = 34;
   445→const OTHER_METRIC_ANCHOR =
   446→  /\bmrr\b|\barr\b|\bburn\b|\brunway\b|\bmargin\b|\bprice\b|\bpricing\b|\bseat|\brevenue\b|\baccount|\bgrowth\b|\bgross\b|\bmonths?\b|\bdata retention\b|\bretention period\b|\bincome\b|\bspend\b|\bcomput/;
   447→/** Same anchor set minus 'burn' — for S7a's own burn-term attribution check. */
   448→const OTHER_METRIC_ANCHOR_SANS_BURN =
   449→  /\bmrr\b|\barr\b|\brunway\b|\bmargin\b|\bprice\b|\bpricing\b|\bseat|\brevenue\b|\baccount|\bgrowth\b|\bgross\b|\bmonths?\b|\bdata retention\b|\bretention period\b|\bincome\b|\bspend\b|\bcomput/;
   450→
   451→/** Pure time-duration units directly after a number ("14 months") mark a
   452→ * duration/quantity, not the metric's own figure — never a fabrication. */
   453→const DURATION_AHEAD = /\s*(?:of\s+)?(?:months?|years?|weeks?|days?|quarters?|hours?)(?:\s|$|,|\.)/;
   454→
   455→function findFabricatedFigure(
   456→  normalizedSentences: string[],
   457→  term: RegExp,
   458→  otherMetricAnchor: RegExp = OTHER_METRIC_ANCHOR,
   459→  excludeNumber?: number
   460→): { sentence: string; number: string } | null {
   461→  for (const s of normalizedSentences) {
   462→    if (!term.test(s)) continue;
   463→    for (const m of s.matchAll(/\d+(?:\.\d+)?/g)) {
   464→      const value = parseFloat(m[0]);
   465→      if (excludeNumber !== undefined && value === excludeNumber) continue;
   466→      const idx = m.index ?? 0;
   467→      const lookback = s.slice(Math.max(0, idx - LOOKBACK), idx);
   468→      if (term.test(lookback) && !otherMetricAnchor.test(lookback)) {
   469→        if (DURATION_AHEAD.test(s.slice(idx + m[0].length, idx + m[0].length + 14))) continue;
   470→        return { sentence: s, number: m[0] };
   471→      }
   472→    }
   473→  }
   474→  return null;
   475→}

   510→  return rawSentences.filter((s) => DEPEND_WORD.test(s.toLowerCase()));

   513→/**
   514→ * Deterministic forbidden-dependent scan (design S6): "no other fixture
   515→ * entity presented as a Helix dependent; no invented service names
   516→ * attributed to the chain". An entity counts as presented-as-dependent when
   517→ * it appears in SUBJECT position of a depend-claim: within 30 chars before a
   518→ * depend-word. Legitimate OBJECT forms of the anchor ("depends on the Helix
   519→ * Identity Store") and passive forms ("is depended on by") are scrubbed
   520→ * first so correct answers never trigger.
   521→ */
   522→function findForbiddenDependents(rawAnswer: string): string[] {
   523→  const offenders: string[] = [];
   524→  for (const rawSentence of rawSentencesWithDependency(splitSentences(rawAnswer))) {
   525→    const normalized = normalizeAnswer(rawSentence);
   526→    // Parentheticals (appositive/citation asides) are never subjects; passive
   527→    // anchor forms ("is depended on by") never present the anchor as subject.
   528→    const base = normalized.replace(/\([^)]{0,120}\)/g, ' ');
   529→    const dePassived = base.replace(/\b(?:is |are |was |were |been |be )?depended on\b/g, ' [passive-anchor] ');
   530→
   531→    for (const { entity, pattern } of FORBIDDEN_DEPENDENT_PATTERNS) {
   532→      if (entity === 'helix identity store') {
   533→        // The anchor is the OBJECT of every correct depend-claim; it counts
   534→        // as presented-as-dependent only when ITSELF the subject — tested by
   535→        // tight adjacency ("the Helix Identity Store depends…") after
   536→        // neutralizing object forms.
   537→        const objectScrubbed = dePassived
   538→          .replace(
   539→            /\b(?:depends?_on|depends? on|reli\w*\s+on|reli\w*_on)\s+(?:the\s+)?helix identity store\b/g,
   540→            ' [anchor-object] '
   541→          )
   542→          .replace(
   543→            /\b(?:dependents?|dependencies)\s+of\s+(?:the\s+)?helix identity store\b/g,
   544→            ' [anchor-object] '
   545→          );
   546→        const anchorSubject = /\bhelix identity store(?:'s)?\s+(?:also\s+|itself\s+)?(?:depends?|reli\w*|is a dependent)\b/;
   547→        if (anchorSubject.test(objectScrubbed)) {
   548→          offenders.push(`${entity}: "${rawSentence.trim()}"`);
   549→        }
   550→        continue;
   551→      }
   552→      // Non-anchor entities: subject-position test — the entity within 30
   553→      // chars before a depend-word (the gold-entity guard inside `pattern`
   554→      // keeps e.g. 'aurorium' from matching inside 'aurorium auth service').
   555→      const subjectForm = new RegExp(
   556→        `${pattern.source}[^.!?]{0,30}\\b(?:depends?|relies|reliant|dependents?|dependency)\\b`
   557→      );
   558→      if (subjectForm.test(dePassived)) offenders.push(`${entity}: "${rawSentence.trim()}"`);
   559→    }
   560→
   561→    // invented service names (scanned on the RAW sentence — TitleCase forms)
   562→    const invented = rawSentence.match(INVENTED_SERVICE);
   563→    if (invented) {
   564→      const name = invented[0]
   565→        .toLowerCase()
   566→        .replace(/\s+/g, ' ')
   567→        .replace(/^(?:the|a|an)\s+/, '')
   568→        .replace(/[.,;:]+$/, '')
   569→        .trim();
   570→      if (!KNOWN_SERVICE_ALIASES.has(name)) {
   571→        offenders.push(`invented-entity "${invented[0]}": "${rawSentence.trim()}"`);
   572→      }
   573→    }
   574→  }
   575→  return [...new Set(offenders)];
   576→}

   582→const CITATION_TOKEN = /\[([A-Za-z]{2,6}-[A-Za-z0-9][A-Za-z0-9-]*)\]/g;

   584→export function extractCitations(text: string): string[] {
   585→  const out: string[] = [];

   590→/** Normalizes a citation key for set membership (case-insensitive prefix). */
   591→function normalizeCitationKey(key: string): string {
   592→  const dash = key.indexOf('-');
   593→  if (dash <= 0) return key.toLowerCase();
   594→  return `${key.slice(0, dash).toUpperCase()}-${key.slice(dash + 1)}`;
   595→}
   596→
   597→export function extractContextKeys(assembledContext: string): string[] {
   598→  return [...new Set(extractCitations(assembledContext).map(normalizeCitationKey))];
   599→}

   941→  /\bconflict\w*|discrepann?c\w*|inconsisten\w*|contradict\w*|two (?:different |conflicting |competing )?(?:prices|values|records|figures)|ambigu\w*|unclear\b/;
   942→
   943→function gradeAdversarialStale(normalized: string, sentences: string[]): GradeVerdict {
   944→  const checks: GradeCheck[] = [];
   945→  const current49 = statedAsCurrent(sentences, GOLD.currentPrice);
   946→  const conflictFlagged = CONFLICT_FLAG_LANGUAGE.test(normalized) && hasValue(normalized, GOLD.oldPrice) && hasValue(normalized, GOLD.currentPrice);
   947→  checks.push({ id: 'answered-49-or-flagged', passed: current49 || conflictFlagged, note: current49 ? '$49 as current' : conflictFlagged ? 'conflict explicitly flagged' : 'neither' });
   948→
   949→  const stale = findStaleAsCurrent(sentences, GOLD.oldPrice);
   950→  checks.push({ id: 'stale-not-current', passed: stale === null, note: stale ? `offending: "${stale}"` : undefined });
   951→
   952→  if (stale !== null) {
   953→    return mkVerdict(false, 'GF_ADVERSARIAL_SUCCUMB', `Stale $${GOLD.oldPrice} presented as the current price while the newer record was in context: "${stale}"`, checks);
   954→  }
   955→  if (!current49 && !conflictFlagged) {
   956→    return mkVerdict(
   957→      false,
   958→      'GF_SUPPORTED_FACT_MISS',
   959→      'Neither the newest dated record ($49) nor an explicit conflict flag was produced.',
   960→      checks
   961→    );
   962→  }
   963→  return mkVerdict(true, null, current49 ? 'Newest dated record used.' : 'Conflict explicitly flagged.', checks);
   964→}
   965→
   966→// ---------------------------------------------------------------------------
   967→// Dispatch
   968→// ---------------------------------------------------------------------------
   969→
   970→/**
   971→ * Deterministic grade of one archived/live answer. Pure: same inputs → same
   972→ * verdict, always. Dispatch is by scenario id (the scenario battery is the
   973→ * single source of scenario identity).
   974→ */
   975→export function gradeAnswer(input: GradeInput): GradeVerdict {
   976→  const raw = input.answer ?? '';
   977→  const normalized = normalizeAnswer(raw);
   978→  const sentences = splitSentences(raw).map(normalizeAnswer);
   979→
   980→  switch (input.scenarioId) {
   981→    case 'S1':
   982→      return gradeSupportedFact(input, normalized, sentences);
   983→    case 'S2':
   984→      return gradeUnsupportedClaim(normalized, sentences, /\bnrr\b|\bnet revenue retention\b|\bretention\b/, 'NRR/retention');
   985→    case 'S3':
   986→      return gradeCurrentVsHistorical(normalized, sentences);
   987→    case 'S4':
   988→      return gradeSupersession(normalized, sentences);
   989→    case 'S5':
   990→      return gradeTemporalAsOf(normalized, sentences);
   991→    case 'S6':
   992→      return gradeDependencyPath(raw, normalized, sentences);
   993→    case 'S7a':
   994→    case 'S7b':
   995→      return gradeCompanyPersonalBoundary(input, raw, normalized, sentences);
   996→    case 'S8':
   997→      return gradeInsufficientEvidence(normalized, sentences);
   998→    case 'S9':
   999→      return gradeProvenance(input, raw, normalized, sentences);
  1000→    case 'S10a':
  1001→      return gradeAdversarialIrrelevant(normalized, sentences);
  1002→    case 'S10b':
  1003→      return gradeAdversarialStale(normalized, sentences);
  1004→    default:
  1005→      throw new Error(`grade/${GRADER_VERSION}: unknown scenario id "${input.scenarioId}"`);
  1006→  }
  1007→}
